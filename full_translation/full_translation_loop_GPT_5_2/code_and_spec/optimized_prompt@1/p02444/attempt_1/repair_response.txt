```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let mut a: Vec<i32> = Vec::with_capacity(n);
    for _ in 0..n {
        a.push(it.next().unwrap().parse().unwrap());
    }

    let q: i32 = it.next().unwrap().parse().unwrap();

    // Auxiliary array of the same size (as in the C code).
    let mut tmp: Vec<i32> = vec![0; n];

    for _ in 0..q {
        let b_in: i32 = it.next().unwrap().parse().unwrap();
        let m_in: i32 = it.next().unwrap().parse().unwrap();
        let e_in: i32 = it.next().unwrap().parse().unwrap();

        // C code assumes valid indices; behavior is undefined otherwise (per spec).
        let b = b_in as usize;
        let mut m = m_in;
        let mut e = e_in;

        // Copy a[b..e) into tmp[b..e)
        for i in b..(e as usize) {
            tmp[i] = a[i];
        }

        // Transform to subarray-relative parameters (matches C exactly).
        e -= b_in; // length
        m -= b_in; // shift offset relative to b

        let len = e as usize;
        for i in 0..len {
            // In C, (i + m) % e uses C's % semantics; with valid inputs this is expected
            // to be non-negative. We mirror that by using rem_euclid to stay in-bounds.
            let src = ((i as i32) + m).rem_euclid(e) as usize;
            a[b + i] = tmp[b + src];
        }
    }

    // Output formatting: numbers separated by spaces, newline after last (matches C).
    let mut out = String::new();
    for i in 0..n {
        out.push_str(&a[i].to_string());
        out.push(if i + 1 == n { '\n' } else { ' ' });
    }
    print!("{out}");
}
```
