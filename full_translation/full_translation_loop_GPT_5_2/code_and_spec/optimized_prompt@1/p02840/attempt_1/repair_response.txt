```rust
use std::io::{self, Read};

const HASH: usize = 100_003;

fn gcd(mut a: i64, mut b: i64) -> i64 {
    a = a.abs();
    b = b.abs();
    while a != 0 {
        let t = b % a;
        b = a;
        a = t;
    }
    b
}

#[derive(Clone, Copy)]
struct Node {
    v: i64,
    l: i64,
    r: i64,
    next: Option<usize>,
}

fn main() {
    // Read all input
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_whitespace();
    let n: i64 = it.next().unwrap().parse().unwrap();
    let mut x: i64 = it.next().unwrap().parse().unwrap();
    let mut d: i64 = it.next().unwrap().parse().unwrap();

    // Special cases per spec (matches C behavior)
    if x == 0 || d == 0 {
        if x == 0 && d == 0 {
            print!("1\n");
        } else if d == 0 {
            print!("{}\n", n + 1);
        } else {
            // x == 0 && d != 0
            print!("{}\n", n * (n - 1) / 2 + 1);
        }
        return;
    }

    // Normalize by gcd(|X|, |D|)
    let g = gcd(x, d);
    x /= g;
    d /= g;

    // Compute Y and normalize D to be positive, matching C logic
    let y: i64;
    if d < 0 {
        y = x;
        x = x + d * (n - 1);
        d = -d;
    } else {
        y = x + d * (n - 1);
    }

    // Hash table of linked lists, nodes stored in a vector by index.
    // C uses a fixed array d[200001]; we allocate n+1 nodes (indices 0..=n).
    let nn = n as usize;
    let mut nodes: Vec<Node> = vec![
        Node {
            v: 0,
            l: 0,
            r: 0,
            next: None
        };
        nn + 1
    ];
    let mut head: Vec<Option<usize>> = vec![None; HASH];

    // Initial node (i=0)
    nodes[0] = Node {
        v: 0,
        l: 0,
        r: d,
        next: None,
    };
    head[0] = Some(0);

    for i in 1..=nn {
        let ii = i as i64;

        // L = (X*2 + D*(i-1)) * i / 2
        let l = (x * 2 + d * (ii - 1)) * ii / 2;

        // R = (Y*2 - D*(i-1)) * i / 2 + D
        let r = (y * 2 - d * (ii - 1)) * ii / 2 + d;

        // V = ((L % D) + D) % D
        let mut v = l % d;
        if v < 0 {
            v += d;
        }
        let h = (v as usize) % HASH;

        // Find an existing overlapping range with same v
        let mut cur = head[h];
        let mut found: Option<usize> = None;
        while let Some(idx) = cur {
            let p = nodes[idx];
            if p.v == v && p.l <= r && p.r >= l {
                found = Some(idx);
                break;
            }
            cur = p.next;
        }

        if let Some(idx) = found {
            // Merge ranges
            let p = &mut nodes[idx];
            if l < p.l {
                p.l = l;
            }
            if r > p.r {
                p.r = r;
            }
        } else {
            // Insert new node at head
            nodes[i] = Node {
                v,
                l,
                r,
                next: head[h],
            };
            head[h] = Some(i);
        }
    }

    // Sum contributions
    let mut ans: i64 = 0;
    for h in 0..HASH {
        let mut cur = head[h];
        while let Some(idx) = cur {
            let p = nodes[idx];
            ans += (p.r - p.l) / d;
            cur = p.next;
        }
    }

    print!("{}\n", ans);
}
```
