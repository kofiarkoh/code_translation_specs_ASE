```rust
use std::cmp::min;
use std::collections::VecDeque;
use std::io::{self, Read};

fn main() {
    // Tokenize like scanf with whitespace separation.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    loop {
        let w = match it.next() {
            Some(v) => v.parse::<usize>().unwrap(),
            None => break,
        };
        let h = match it.next() {
            Some(v) => v.parse::<usize>().unwrap(),
            None => break,
        };

        if w == 0 && h == 0 {
            break;
        }

        let mut m: Vec<Vec<u8>> = vec![vec![b'.'; w]; h];
        let mut dist: Vec<Vec<i32>> = vec![vec![0; w]; h];
        let mut start: Option<(usize, usize)> = None;

        // Grid characters may be given either space-separated or as contiguous strings per row.
        // scanf(" %c") reads a single non-whitespace character each time; to match that,
        // we consume tokens and, if a token has multiple characters, we use them sequentially.
        let mut pending: Vec<u8> = Vec::new(); // acts as a stack of remaining chars in current token

        for i in 0..h {
            for j in 0..w {
                let ch = loop {
                    if let Some(c) = pending.pop() {
                        break c;
                    }
                    let tok = it.next().unwrap(); // per spec, input is well-formed
                    // push in reverse so we can pop() in original order
                    pending = tok.as_bytes().iter().rev().copied().collect();
                };

                if ch == b'&' {
                    m[i][j] = b'.';
                    dist[i][j] = 1;
                    start = Some((i, j));
                } else {
                    m[i][j] = ch;
                }
            }
        }

        // Spec says start exists; if not, behavior undefined. Avoid panic in judge by defaulting.
        let (sy, sx) = start.unwrap_or((0, 0));

        // 0-1 BFS for minimal transition count (same char: 0, different: 1), matching C relaxations.
        let dirs = [(0isize, 1isize), (1, 0), (0, -1), (-1, 0)];
        let mut dq: VecDeque<(usize, usize)> = VecDeque::new();
        dq.push_back((sy, sx));

        while let Some((y, x)) = dq.pop_front() {
            let cur = dist[y][x];
            for (dy, dx) in dirs {
                let ny = y as isize + dy;
                let nx = x as isize + dx;
                if ny < 0 || nx < 0 || ny >= h as isize || nx >= w as isize {
                    continue;
                }
                let nyu = ny as usize;
                let nxu = nx as usize;

                let wgt = if m[y][x] == m[nyu][nxu] { 0 } else { 1 };
                let nd = cur + wgt;

                if dist[nyu][nxu] == 0 || nd < dist[nyu][nxu] {
                    dist[nyu][nxu] = nd;
                    if wgt == 0 {
                        dq.push_front((nyu, nxu));
                    } else {
                        dq.push_back((nyu, nxu));
                    }
                }
            }
        }

        // Preserve C's border-min logic: min initialized to w, and uses dist/2 (integer division).
        let mut ans: i32 = w as i32;
        for i in 0..w {
            ans = min(ans, dist[0][i] / 2);
            ans = min(ans, dist[h - 1][i] / 2);
        }
        for i in 0..h {
            ans = min(ans, dist[i][0] / 2);
            ans = min(ans, dist[i][w - 1] / 2);
        }

        println!("{}", ans);
    }
}
```
