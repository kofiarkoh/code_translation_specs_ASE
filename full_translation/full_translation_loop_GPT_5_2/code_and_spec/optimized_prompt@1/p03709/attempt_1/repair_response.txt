```rust
use std::cmp::{max, min};
use std::io::{self, Read};

const MOD: u64 = 1_000_000_007;
const VERYBIG: i64 = 20_000_000_000_000_007; // matches C's VERYBIG

#[derive(Clone, Copy, Debug)]
struct Pair {
    a: i64,
    b: i64,
}

fn mod_pow(mut a: u64, mut e: i64, m: u64) -> u64 {
    let mut r: u64 = 1;
    while e > 0 {
        if (e & 1) == 1 {
            r = (r * a) % m;
        }
        e >>= 1;
        a = (a * a) % m;
    }
    r
}

fn main() {
    // Input format in the C code is inconsistent:
    //   scanf("%llu", &n, &m);  // format has 1 specifier but 2 args (UB in C)
    // Spec says: reads n then n pairs. We'll follow the *intended* behavior:
    // read n, then read n pairs (a[i], b[i]). If an extra integer exists (m),
    // we ignore it (it is unused in solve()).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n_opt = it.next();
    if n_opt.is_none() {
        return;
    }
    let n: usize = n_opt.unwrap().parse().unwrap();

    // The original C seems to want to read m too, but m is unused.
    // If present, consume it; otherwise proceed.
    let _maybe_m: Option<u64> = it.next().and_then(|s| s.parse().ok());

    let mut a: Vec<i64> = vec![0; n];
    let mut b: Vec<i64> = vec![0; n];
    for i in 0..n {
        a[i] = it.next().unwrap().parse().unwrap();
        b[i] = it.next().unwrap().parse().unwrap();
    }

    // --- solve() ---
    let mut xy: Vec<Pair> = (0..n).map(|i| Pair { a: a[i], b: b[i] }).collect();
    xy.sort_by(|l, r| l.a.cmp(&r.a).then(l.b.cmp(&r.b)));
    for i in 0..n {
        a[i] = xy[i].a;
        b[i] = xy[i].b;
    }

    let mut c: Vec<i64> = vec![0; n];
    for i in 0..n {
        c[i] = max(if i > 0 { c[i - 1] } else { 0 }, b[i]);
    }

    let mut d: Vec<i64> = vec![0; n];
    for i in (0..n).rev() {
        d[i] = min(if i + 1 < n { d[i + 1] } else { VERYBIG }, b[i]);
    }

    // Reuse xy to store computed interval endpoints (as in C).
    for i in 0..n {
        // left endpoint
        if i == 0 || c[i - 1] <= b[i] {
            xy[i].a = i as i64;
        } else {
            let mut ok: i64 = -1;
            let mut ng: i64 = i as i64;
            while ok + 1 < ng {
                let med = (ok + ng) / 2;
                if c[med as usize] > b[i] {
                    ng = med;
                } else {
                    ok = med;
                }
            }
            xy[i].a = ng;
        }

        // right endpoint
        if i + 1 == n || d[i + 1] >= b[i] {
            xy[i].b = i as i64;
        } else {
            let mut ok: i64 = n as i64;
            let mut ng: i64 = i as i64;
            while ng + 1 < ok {
                let med = (ok + ng) / 2;
                if d[med as usize] < b[i] {
                    ng = med;
                } else {
                    ok = med;
                }
            }
            xy[i].b = ng;
        }
    }

    xy.sort_by(|l, r| l.a.cmp(&r.a).then(l.b.cmp(&r.b)));

    // e indexed 0..=n+1, initialized to -1
    let mut e: Vec<i64> = vec![-1; n + 2];
    for i in 0..n {
        // shift to 1-based indices
        xy[i].a += 1;
        xy[i].b += 1;
        let bi = xy[i].b as usize;
        e[bi] = xy[i].a;
    }

    // dp and prefix sums ss: (n+1) x 4
    let mut dp: Vec<[u64; 4]> = vec![[0; 4]; n + 1];
    let mut ss: Vec<[u64; 4]> = vec![[0; 4]; n + 1];

    let inv2: u64 = (MOD + 1) / 2;
    let pow2n = mod_pow(2, n as i64, MOD);
    dp[0][0] = pow2n;
    ss[0][0] = pow2n;
    dp[0][2] = pow2n;
    ss[0][2] = pow2n;

    for i in 1..=n {
        if e[i] >= 0 {
            let l = e[i] as usize;
            let r = i;

            for ki in 0..2usize {
                // dp[i][ki] = ss[l-1][1-ki] * inv2^(r-l+1)
                let term1 = ss[l - 1][1 - ki]
                    * mod_pow(inv2, (r as i64 - l as i64 + 1), MOD)
                    % MOD;

                // dp[i][ki] += (MOD + ss[r-1][3-ki] - ss[l-1][3-ki]) * inv2^r
                let diff = (MOD + ss[r - 1][3 - ki] + MOD - ss[l - 1][3 - ki]) % MOD;
                let term2 = diff * mod_pow(inv2, r as i64, MOD) % MOD;

                let val = (term1 + term2) % MOD;
                dp[i][ki] = val;
                dp[i][2 + ki] = val * mod_pow(2, i as i64, MOD) % MOD;
            }
        }

        for ki in 0..4usize {
            ss[i][ki] = (ss[i - 1][ki] + dp[i][ki]) % MOD;
        }
    }

    let result = (MOD + ss[n][0] + MOD - ss[n][1]) % MOD;

    // Output behavior in the provided C code: prints only the number and newline.
    // SPEC mentions printing "Yes"/"No" too, but the C code never reaches those labels.
    // We follow the actual implemented behavior.
    println!("{}", result);
}
```
