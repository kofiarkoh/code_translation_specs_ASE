```rust
use std::io::{self, Read};

fn main() {
    // C uses `scanf("%s", a)` which reads a single whitespace-delimited token.
    // We replicate that behavior by reading all stdin and taking the first token.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let token = match input.split_whitespace().next() {
        Some(t) => t,
        None => return, // No token => C scanf would fail; spec says no error handling.
    };

    // Spec: max length 999 (excluding null terminator). If exceeded, behavior undefined.
    // We do not enforce it to match "undefined" behavior rather than adding new handling.

    let mut out = String::with_capacity(token.len());
    for &b in token.as_bytes() {
        let transformed = if (b'D'..=b'Z').contains(&b) {
            b.wrapping_sub(3)
        } else {
            b.wrapping_add(23)
        };
        out.push(transformed as char);
    }

    // C prints "%s\n"
    print!("{out}\n");
}
```
