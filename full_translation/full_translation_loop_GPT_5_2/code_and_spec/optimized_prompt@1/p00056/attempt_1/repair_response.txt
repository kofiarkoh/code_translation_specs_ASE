use std::io::{self, Read};

const MAX: usize = 50001;

fn make_prime_table() -> Vec<u8> {
    // table[x] == 0 => prime (or 0/1), table[x] == 1 => composite
    let mut table = vec![0u8; MAX];

    for i in 2..MAX {
        if table[i] == 0 {
            let mut j = 2usize;
            while i * j < MAX {
                table[i * j] = 1;
                j += 1;
            }
        }
    }
    table
}

fn main() {
    let ptable = make_prime_table();

    let mut input = String::new();
    io::stdin().read_to_string(&mut input).ok();

    let mut out = String::new();

    for tok in input.split_whitespace() {
        // Spec: if input is not a valid integer, produce no output for that input.
        let n_i64 = match tok.parse::<i64>() {
            Ok(v) => v,
            Err(_) => continue,
        };

        // Spec precondition: non-negative integers. If violated, skip output.
        if n_i64 < 0 {
            continue;
        }

        let n = n_i64 as usize;

        if n == 0 {
            break;
        }

        // Spec precondition: maximum input value must not exceed the defined limit.
        // If violated, skip output (graceful behavior).
        if n >= MAX {
            continue;
        }

        let mut cnt = 0usize;
        for i in 2..=n / 2 {
            if ptable[i] == 0 && ptable[n - i] == 0 {
                cnt += 1;
            }
        }

        out.push_str(&format!("{cnt}\n"));
    }

    print!("{out}");
}
