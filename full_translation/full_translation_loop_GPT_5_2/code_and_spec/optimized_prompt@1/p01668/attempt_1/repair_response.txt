```rust
use std::io::{self, Read};

const MOD_BY: i32 = 1_000_000_007;

fn add(a: i32, b: i32) -> i32 {
    let mut r = a + b;
    if r >= MOD_BY {
        r -= MOD_BY;
    }
    r
}

struct Solver {
    n: usize,
    m: usize,
    exist_pos: usize,
    a: Vec<u8>, // digits 0..9, length n
    b: Vec<u8>, // digits 0..9, length n
    c: Vec<u8>, // digits 0..9, length m

    // memoization with "0 means uncomputed, else value+1" to match C behavior
    memo: Vec<i32>,            // [n+1][m+1][2][2]
    number_of_number: Vec<i32>, // [n+1][2][2]
}

impl Solver {
    fn idx_memo(&self, pos: usize, seq_pos: usize, ue_flag: usize, sita_flag: usize) -> usize {
        // layout: (((pos*(m+1) + seq_pos)*2 + ue_flag)*2 + sita_flag)
        (((pos * (self.m + 1) + seq_pos) * 2 + ue_flag) * 2 + sita_flag)
    }

    fn idx_non(&self, pos: usize, ue_flag: usize, sita_flag: usize) -> usize {
        ((pos * 2 + ue_flag) * 2 + sita_flag)
    }

    fn get_number_of_number(&mut self, pos: usize, ue_flag: usize, sita_flag: usize) -> i32 {
        if pos >= self.n {
            return 1;
        }
        let idx = self.idx_non(pos, ue_flag, sita_flag);
        let cached = self.number_of_number[idx];
        if cached > 0 {
            return cached - 1;
        }

        let imin = if sita_flag == 1 { self.a[pos] } else { 0 };
        let imax = if ue_flag == 1 { self.b[pos] } else { 9 };

        let mut ret: i32 = 0;
        for i in imin..=imax {
            let next_ue = if ue_flag == 1 && i == imax { 1 } else { 0 };
            let next_sita = if sita_flag == 1 && i == imin { 1 } else { 0 };
            ret = add(ret, self.get_number_of_number(pos + 1, next_ue, next_sita));
        }

        self.number_of_number[idx] = ret + 1;
        ret
    }

    fn search(&mut self, pos: usize, seq_pos: usize, ue_flag: usize, sita_flag: usize) -> i32 {
        if seq_pos >= self.m {
            return self.get_number_of_number(pos, ue_flag, sita_flag);
        }
        if pos >= self.n {
            return 0;
        }

        let idx = self.idx_memo(pos, seq_pos, ue_flag, sita_flag);
        let cached = self.memo[idx];
        if cached > 0 {
            return cached - 1;
        }

        let imin = if sita_flag == 1 { self.a[pos] } else { 0 };
        let imax = if ue_flag == 1 { self.b[pos] } else { 9 };

        let mut ret: i32 = 0;
        for i in imin..=imax {
            let next_ue = if ue_flag == 1 && i == imax { 1 } else { 0 };
            let next_sita = if sita_flag == 1 && i == imin { 1 } else { 0 };

            if seq_pos == 0 {
                ret = add(ret, self.search(pos + 1, 0, next_ue, next_sita));
            }

            // Condition mirrors C:
            // (!sitaFlag || i!=imin || existPos<=pos) && seqPos<M && C[seqPos]==i
            if (sita_flag == 0 || i != imin || self.exist_pos <= pos) && self.c[seq_pos] == i {
                ret = add(ret, self.search(pos + 1, seq_pos + 1, next_ue, next_sita));
            }
        }

        self.memo[idx] = ret + 1;
        ret
    }
}

fn main() {
    // Read all input, split by whitespace (matches scanf("%s%s%s"))
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let a_str = match it.next() {
        Some(s) => s.to_string(),
        None => std::process::exit(1),
    };
    let b_str = match it.next() {
        Some(s) => s.to_string(),
        None => std::process::exit(1),
    };
    let c_str = match it.next() {
        Some(s) => s.to_string(),
        None => std::process::exit(1),
    };

    let alen = a_str.len();
    let blen = b_str.len();

    // C code left-pads A with zeros to length of B if A is shorter.
    // If A is longer than B, C code effectively ignores extra leading digits of A
    // by setting N = blen and using A[0..N). We replicate that behavior.
    let mut a_padded = if alen < blen {
        let mut s = String::with_capacity(blen);
        for _ in 0..(blen - alen) {
            s.push('0');
        }
        s.push_str(&a_str);
        s
    } else {
        a_str.clone()
    };

    let n = blen;
    let m = c_str.len();
    let exist_pos = blen.saturating_sub(alen);

    // Convert to digit arrays; behavior is undefined for non-digits per spec.
    // Take only the last n digits of A if A is longer (matches C's N=blen usage).
    if a_padded.len() > n {
        a_padded = a_padded[a_padded.len() - n..].to_string();
    }
    let a_digits: Vec<u8> = a_padded
        .bytes()
        .take(n)
        .map(|ch| ch - b'0')
        .collect();
    let b_digits: Vec<u8> = b_str.bytes().take(n).map(|ch| ch - b'0').collect();
    let c_digits: Vec<u8> = c_str.bytes().map(|ch| ch - b'0').collect();

    let memo_size = (n + 1) * (m + 1) * 2 * 2;
    let non_size = (n + 1) * 2 * 2;

    let mut solver = Solver {
        n,
        m,
        exist_pos,
        a: a_digits,
        b: b_digits,
        c: c_digits,
        memo: vec![0; memo_size],
        number_of_number: vec![0; non_size],
    };

    let ans = solver.search(0, 0, 1, 1);
    print!("{ans}\n");
}
```
