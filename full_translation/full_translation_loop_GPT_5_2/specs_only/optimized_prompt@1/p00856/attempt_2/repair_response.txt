```rust
use std::io::{self, Read};

fn main() {
    // Read all input as whitespace-separated tokens.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n_i64 = n_opt.unwrap().parse::<i64>().unwrap_or(0);
        let t_i64 = it.next().and_then(|s| s.parse::<i64>().ok()).unwrap_or(0);
        let l_i64 = it.next().and_then(|s| s.parse::<i64>().ok()).unwrap_or(0);
        let b_i64 = it.next().and_then(|s| s.parse::<i64>().ok()).unwrap_or(0);

        // Termination condition.
        if n_i64 == 0 && t_i64 == 0 && l_i64 == 0 && b_i64 == 0 {
            break;
        }

        // Invalid set: skip and consume obstacle lists best-effort.
        if n_i64 < 0 || t_i64 < 0 || l_i64 < 0 || b_i64 < 0 {
            let l_cnt = l_i64.max(0) as usize;
            let b_cnt = b_i64.max(0) as usize;
            for _ in 0..l_cnt {
                let _ = it.next();
            }
            for _ in 0..b_cnt {
                let _ = it.next();
            }
            continue;
        }

        let n = n_i64 as usize;
        let t = t_i64 as usize;
        let l_cnt = l_i64 as usize;
        let b_cnt = b_i64 as usize;

        if l_cnt + b_cnt > n {
            for _ in 0..l_cnt {
                let _ = it.next();
            }
            for _ in 0..b_cnt {
                let _ = it.next();
            }
            continue;
        }

        // Board: 0..=N, where N is the goal (end of board).
        // Positions in input are interpreted as 1-based indices on 1..=N-1 (typical for such tasks).
        // Any obstacle at position <=0 or >=N is ignored.
        let mut is_l = vec![false; n + 1];
        let mut is_b = vec![false; n + 1];

        for _ in 0..l_cnt {
            if let Some(tok) = it.next() {
                if let Ok(p) = tok.parse::<i64>() {
                    if p > 0 && (p as usize) < n {
                        is_l[p as usize] = true;
                    }
                }
            }
        }
        for _ in 0..b_cnt {
            if let Some(tok) = it.next() {
                if let Ok(p) = tok.parse::<i64>() {
                    if p > 0 && (p as usize) < n {
                        is_b[p as usize] = true;
                    }
                }
            }
        }

        // Memoization array initialized to a maximum double value before each computation.
        // Use f64::MAX as sentinel for "uncomputed".
        let mut memo = vec![vec![f64::MAX; n + 1]; t + 1];

        // Model:
        // - Start at position 0, goal is position N.
        // - Each turn: roll two fair six-sided dice; move forward by sum in 2..=12.
        // - If landing on an 'L' position: you lose immediately (probability 0 from that branch).
        // - If landing on a 'B' position: you are sent back to start (position 0) and continue.
        // - If move would go beyond N: treat as reaching N (win).
        //
        // This matches the "Backgammon" mention and yields non-trivial probabilities.
        fn dp(
            turns_left: usize,
            pos: usize,
            n: usize,
            is_l: &[bool],
            is_b: &[bool],
            memo: &mut [Vec<f64>],
        ) -> f64 {
            if pos >= n {
                return 1.0;
            }
            if turns_left == 0 {
                return 0.0;
            }
            if memo[turns_left][pos] != f64::MAX {
                return memo[turns_left][pos];
            }

            // Distribution of sums of two dice (2..=12) out of 36.
            const COUNTS: [i32; 13] = [0, 0, 1, 2, 3, 4, 5, 6, 5, 4, 3, 2, 1];

            let mut acc = 0.0f64;
            for sum in 2..=12 {
                let ways = COUNTS[sum] as f64;
                if ways == 0.0 {
                    continue;
                }
                let mut next = pos + sum;
                if next >= n {
                    next = n;
                } else {
                    if is_l[next] {
                        // Immediate loss on 'L'.
                        acc += ways * 0.0;
                        continue;
                    }
                    if is_b[next] {
                        next = 0;
                    }
                }
                acc += ways * dp(turns_left - 1, next, n, is_l, is_b, memo);
            }

            let res = acc / 36.0;
            memo[turns_left][pos] = res;
            res
        }

        let prob = if n == 0 { 1.0 } else { dp(t, 0, n, &is_l, &is_b, &mut memo) };

        // Print with sufficient precision; judge samples show 6 decimals.
        out.push_str(&format!("{:.6}\n", prob));
    }

    print!("{out}");
}
```
