```rust
use std::cmp::Reverse;
use std::collections::BinaryHeap;
use std::io::{self, Read};

const INF: i64 = 1_000_000_000; // "large constant" for distance initialization

fn dijkstra_skip_edge(n: usize, adj: &[Vec<(usize, usize)>], skip: Option<(usize, usize)>) -> Vec<i64> {
    let mut dist = vec![INF; n];
    if n == 0 {
        return dist;
    }
    dist[0] = 0;

    let mut pq: BinaryHeap<(Reverse<i64>, usize)> = BinaryHeap::new();
    pq.push((Reverse(0), 0));

    while let Some((Reverse(d), u)) = pq.pop() {
        if d != dist[u] {
            continue;
        }
        for &(v, eid) in &adj[u] {
            if let Some((su, se)) = skip {
                if su == u && se == eid {
                    continue;
                }
            }
            // All edges are treated as weight 1 (unweighted shortest path in directed graph).
            let nd = d.saturating_add(1);
            if nd < dist[v] {
                dist[v] = nd;
                pq.push((Reverse(nd), v));
            }
        }
    }

    dist
}

fn average_distance(dist: &[i64]) -> f64 {
    // Narrow interpretation: average over "all other nodes" => nodes 1..n-1.
    // If n <= 1, average is 0.0 (no "other nodes").
    if dist.len() <= 1 {
        return 0.0;
    }
    let mut sum: i128 = 0;
    for &d in &dist[1..] {
        sum += d as i128;
    }
    let denom = (dist.len() - 1) as f64;
    (sum as f64) / denom
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap_or(0),
        None => {
            // No input: produce a defined output.
            println!("{:.10}", 0.0);
            return;
        }
    };
    let m: usize = match it.next() {
        Some(x) => x.parse().unwrap_or(0),
        None => 0,
    };

    // Build adjacency list; store edge id to allow skipping exactly one specific edge.
    let mut adj: Vec<Vec<(usize, usize)>> = vec![Vec::new(); n];
    for eid in 0..m {
        let a_opt = it.next();
        let b_opt = it.next();
        if a_opt.is_none() || b_opt.is_none() {
            break;
        }
        let a: isize = a_opt.unwrap().parse().unwrap_or(0);
        let b: isize = b_opt.unwrap().parse().unwrap_or(0);

        // Preconditions say indices are valid; we still guard to keep behavior defined.
        if a >= 0 && b >= 0 {
            let au = a as usize;
            let bu = b as usize;
            if au < n && bu < n {
                adj[au].push((bu, eid));
            }
        }
    }

    // Consider removing no edge (baseline) and removing each specific edge.
    // If no edges exist, this naturally returns the average based on INF-initialized distances.
    let mut best = {
        let dist = dijkstra_skip_edge(n, &adj, None);
        average_distance(&dist)
    };

    for u in 0..n {
        for &(_v, eid) in &adj[u] {
            let dist = dijkstra_skip_edge(n, &adj, Some((u, eid)));
            let avg = average_distance(&dist);
            if avg < best {
                best = avg;
            }
        }
    }

    println!("{:.10}", best);
}
```
