fn idx_from_char(c: u8) -> Option<usize> {
    match c {
        b'A'..=b'Z' => Some((c - b'A') as usize),
        b'a'..=b'z' => Some(26 + (c - b'a') as usize),
        _ => None,
    }
}

fn char_from_idx(i: usize) -> char {
    if i < 26 {
        (b'A' + i as u8) as char
    } else {
        (b'a' + (i as u8 - 26)) as char
    }
}
