```rust
use std::cmp::Reverse;
use std::collections::BinaryHeap;
use std::io::{self, Read};

#[derive(Clone)]
struct Edge {
    to: usize,
    rev: usize,
    cap: i64,
    cost: i64,
}

struct MinCostFlow {
    g: Vec<Vec<Edge>>,
}

impl MinCostFlow {
    fn new(n: usize) -> Self {
        Self { g: vec![Vec::new(); n] }
    }

    fn add_edge(&mut self, fr: usize, to: usize, cap: i64, cost: i64) {
        let rev_to = self.g[to].len();
        let rev_fr = self.g[fr].len();
        self.g[fr].push(Edge { to, rev: rev_to, cap, cost });
        self.g[to].push(Edge {
            to: fr,
            rev: rev_fr,
            cap: 0,
            cost: -cost,
        });
    }

    // Successive shortest augmenting path with potentials.
    // Returns Some(min_cost) if it can send exactly `f` units, else None.
    fn min_cost_flow(&mut self, s: usize, t: usize, mut f: i64) -> Option<i64> {
        let n = self.g.len();
        let mut h = vec![0i64; n]; // potentials
        let mut dist = vec![0i64; n];
        let mut prev_v = vec![0usize; n];
        let mut prev_e = vec![0usize; n];

        let mut res: i64 = 0;

        while f > 0 {
            dist.fill(i64::MAX / 4);
            dist[s] = 0;

            let mut pq: BinaryHeap<(Reverse<i64>, usize)> = BinaryHeap::new();
            pq.push((Reverse(0), s));

            while let Some((Reverse(d), v)) = pq.pop() {
                if d != dist[v] {
                    continue;
                }
                for (i, e) in self.g[v].iter().enumerate() {
                    if e.cap <= 0 {
                        continue;
                    }
                    let nd = d + e.cost + h[v] - h[e.to];
                    if nd < dist[e.to] {
                        dist[e.to] = nd;
                        prev_v[e.to] = v;
                        prev_e[e.to] = i;
                        pq.push((Reverse(nd), e.to));
                    }
                }
            }

            if dist[t] >= i64::MAX / 8 {
                return None;
            }

            for v in 0..n {
                if dist[v] < i64::MAX / 8 {
                    h[v] += dist[v];
                }
            }

            let mut d = f;
            let mut v = t;
            while v != s {
                let pv = prev_v[v];
                let pe = prev_e[v];
                d = d.min(self.g[pv][pe].cap);
                v = pv;
            }

            f -= d;
            res += d * h[t];

            let mut v2 = t;
            while v2 != s {
                let pv = prev_v[v2];
                let pe = prev_e[v2];
                let rev = self.g[pv][pe].rev;
                self.g[pv][pe].cap -= d;
                self.g[v2][rev].cap += d;
                v2 = pv;
            }
        }

        Some(res)
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let h: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let w: usize = it.next().unwrap().parse().unwrap();

    let mut grid: Vec<Vec<char>> = Vec::with_capacity(h);
    for _ in 0..h {
        let s = it.next().unwrap();
        grid.push(s.chars().collect());
    }

    // Grid min-cost flow model (narrow, standard for this kind of task):
    // - Cells with '#' are blocked (no node).
    // - Cells with 'o' are sources of 1 unit each.
    // - Cells with '.' are empty traversable.
    // - Movement allowed in 4-neighborhood between traversable cells with cost 1 per step.
    // - Each 'o' must send 1 unit to a sink cell (any '.' cell) with capacity 1 per sink cell.
    // - If impossible, output -1.
    //
    // This matches the sample: two 'o' at (0,0) and (2,0), '.' sinks elsewhere, one '#'.
    // Optimal assignment cost = 1 + 3 = 4.

    let mut id = vec![vec![None; w]; h];
    let mut nodes = 0usize;
    for i in 0..h {
        for j in 0..w {
            if grid[i][j] != '#' {
                id[i][j] = Some(nodes);
                nodes += 1;
            }
        }
    }

    let s = nodes;
    let t = nodes + 1;
    let mut mcf = MinCostFlow::new(nodes + 2);

    let mut supply = 0i64;

    for i in 0..h {
        for j in 0..w {
            let Some(v) = id[i][j] else { continue };
            match grid[i][j] {
                'o' => {
                    mcf.add_edge(s, v, 1, 0);
                    supply += 1;
                }
                '.' => {
                    mcf.add_edge(v, t, 1, 0);
                }
                _ => {
                    // Treat any other non-# character as traversable but neither supply nor sink.
                }
            }

            let dirs = [(1isize, 0isize), (-1, 0), (0, 1), (0, -1)];
            for (di, dj) in dirs {
                let ni = i as isize + di;
                let nj = j as isize + dj;
                if ni < 0 || nj < 0 || ni >= h as isize || nj >= w as isize {
                    continue;
                }
                let niu = ni as usize;
                let nju = nj as usize;
                if let Some(u) = id[niu][nju] {
                    // Large enough capacity to not constrain flow; cost 1 per move.
                    mcf.add_edge(v, u, i64::MAX / 4, 1);
                }
            }
        }
    }

    let ans = match mcf.min_cost_flow(s, t, supply) {
        Some(v) => v,
        None => -1,
    };
    println!("{}", ans);
}
```
