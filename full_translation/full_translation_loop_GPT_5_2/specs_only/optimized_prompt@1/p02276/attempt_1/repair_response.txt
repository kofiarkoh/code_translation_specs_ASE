```rust
use std::io::{self, Read};

fn main() {
    // Read all input as whitespace-separated integers.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return, // No input; behavior not specified, so exit quietly.
    };

    // Preconditions say n is positive; if not, do nothing (narrowest reasonable behavior).
    if n == 0 {
        return;
    }

    let mut a: Vec<i64> = Vec::with_capacity(n);
    for _ in 0..n {
        if let Some(s) = it.next() {
            a.push(s.parse().unwrap());
        } else {
            // Missing elements; unspecified. Stop early with what we have.
            break;
        }
    }

    if a.is_empty() {
        return;
    }

    // Partition around a pivot and place the pivot so that:
    // left <= pivot and right > pivot.
    //
    // Use Lomuto partition scheme with pivot = last element.
    // This guarantees the postcondition for the chosen pivot.
    let pivot_index = {
        let last = a.len() - 1;
        let pivot = a[last];
        let mut i: usize = 0;
        for j in 0..last {
            if a[j] <= pivot {
                a.swap(i, j);
                i += 1;
            }
        }
        a.swap(i, last);
        i
    };

    // Output: elements separated by spaces; pivot enclosed in brackets.
    let mut out = String::new();
    for (idx, val) in a.iter().enumerate() {
        if idx > 0 {
            out.push(' ');
        }
        if idx == pivot_index {
            out.push('[');
            out.push_str(&val.to_string());
            out.push(']');
        } else {
            out.push_str(&val.to_string());
        }
    }
    print!("{out}");
}
```
