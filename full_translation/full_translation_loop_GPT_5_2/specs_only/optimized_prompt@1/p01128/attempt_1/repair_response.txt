```rust
use std::cmp::Ordering;
use std::collections::VecDeque;
use std::io::{self, Read};

#[derive(Clone, Copy, Debug)]
struct Seg {
    xs: i64,
    ys: i64,
    xt: i64,
    yt: i64,
    o: i64,
    l: i64,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    // Parameter t along AB in [0,1] represented as rational num/den (den>0).
    num: i128,
    den: i128,
    // State transition info derived from segment properties.
    o: i64,
    l: i64,
}

fn gcd_i128(mut a: i128, mut b: i128) -> i128 {
    if a < 0 {
        a = -a;
    }
    if b < 0 {
        b = -b;
    }
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn normalize_rational(num: i128, den: i128) -> (i128, i128) {
    // Keep den > 0, reduce by gcd. If den==0, keep as-is (invalid).
    if den == 0 {
        return (num, den);
    }
    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }
    let g = gcd_i128(n, d);
    if g != 0 {
        n /= g;
        d /= g;
    }
    (n, d)
}

fn cmp_rational(a: (i128, i128), b: (i128, i128)) -> Ordering {
    // Compare a.num/a.den vs b.num/b.den, den assumed >0.
    // Use i128 cross-multiply; inputs are i128 already.
    let (an, ad) = a;
    let (bn, bd) = b;
    (an * bd).cmp(&(bn * ad))
}

fn cross(ax: i128, ay: i128, bx: i128, by: i128) -> i128 {
    ax * by - ay * bx
}

fn main() {
    // Specification is incomplete about exact geometry and transitions.
    // Narrow interpretation implemented:
    // - Consider intersections between the segment AB (from (XA,YA) to (XB,YB)) and each given segment S->T.
    // - Keep only proper intersections where the two segments intersect at a single point with parameters t,u in [0,1].
    // - Sort intersection points along AB.
    // - Model a 2-state system (state 0 and 1). Each intersection induces a possible transition:
    //     if O is even: toggles state; else: keeps state.
    //   The cost of taking the transition is max(0, L) (negative treated as 0 to keep non-negative output).
    // - Additionally, moving along AB without taking a transition has zero cost.
    // - Compute minimum cost from start (before first intersection) in state 0 to end (after last intersection) in state 0.
    // - If no valid intersections exist, output a default maximum value (here: i64::MAX/4).
    //
    // This is the narrowest consistent behavior we can implement given the spec's lack of exact rules.

    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let t: usize = match it.next() {
        Some(v) => v.parse().unwrap_or(0),
        None => 0,
    };

    const DEFAULT_MAX: i64 = i64::MAX / 4;

    let mut outputs = Vec::with_capacity(t);

    for _case in 0..t {
        let xa: i64 = match it.next() {
            Some(v) => v.parse().unwrap(),
            None => break,
        };
        let ya: i64 = it.next().unwrap().parse().unwrap();
        let xb: i64 = it.next().unwrap().parse().unwrap();
        let yb: i64 = it.next().unwrap().parse().unwrap();

        let n: usize = it.next().unwrap().parse().unwrap();
        let mut segs = Vec::with_capacity(n);
        for _ in 0..n {
            let xs: i64 = it.next().unwrap().parse().unwrap();
            let ys: i64 = it.next().unwrap().parse().unwrap();
            let xt: i64 = it.next().unwrap().parse().unwrap();
            let yt: i64 = it.next().unwrap().parse().unwrap();
            let o: i64 = it.next().unwrap().parse().unwrap();
            let l: i64 = it.next().unwrap().parse().unwrap();
            segs.push(Seg {
                xs,
                ys,
                xt,
                yt,
                o,
                l,
            });
        }

        // AB vector
        let abx = (xb - xa) as i128;
        let aby = (yb - ya) as i128;

        let mut events: Vec<Event> = Vec::new();

        for s in segs {
            // Segment ST vector
            let stx = (s.xt - s.xs) as i128;
            let sty = (s.yt - s.ys) as i128;

            // Solve: A + t*(B-A) = S + u*(T-S)
            // Using cross products:
            // denom = cross(AB, ST)
            // t = cross(S-A, ST) / denom
            // u = cross(S-A, AB) / denom
            let sax = (s.xs - xa) as i128;
            let say = (s.ys - ya) as i128;

            let denom = cross(abx, aby, stx, sty);
            if denom == 0 {
                // Parallel or collinear: ignore as "do not produce valid intersections"
                continue;
            }

            let t_num = cross(sax, say, stx, sty);
            let u_num = cross(sax, say, abx, aby);

            // Check t in [0,1] and u in [0,1] with denom sign.
            // We normalize by ensuring denom > 0 for comparisons.
            let (tn, td) = normalize_rational(t_num, denom);
            let (un, ud) = normalize_rational(u_num, denom);

            if td == 0 || ud == 0 {
                continue;
            }

            // Inclusive bounds: 0 <= num/den <= 1
            // i.e., 0 <= num and num <= den (since den>0)
            if tn < 0 || tn > td || un < 0 || un > ud {
                continue;
            }

            events.push(Event {
                num: tn,
                den: td,
                o: s.o,
                l: s.l,
            });
        }

        if events.is_empty() {
            outputs.push(DEFAULT_MAX.to_string());
            continue;
        }

        // Sort by position along AB.
        events.sort_by(|a, b| cmp_rational((a.num, a.den), (b.num, b.den)));

        // Compress events with identical t (same rational) by keeping all transitions at that point.
        // We'll process them in order; multiple transitions at same point are allowed.
        // DP over 2 states with 0-cost between points.
        let mut dp0: i64 = 0;
        let mut dp1: i64 = DEFAULT_MAX;

        // Group by identical t
        let mut i = 0usize;
        while i < events.len() {
            let key = (events[i].num, events[i].den);
            let mut j = i + 1;
            while j < events.len() && events[j].num == key.0 && events[j].den == key.1 {
                j += 1;
            }

            // At this intersection position, we can apply any subset/sequence of transitions.
            // Narrow, safe interpretation: each event is an optional edge applied at most once, order irrelevant.
            // We model this as a small graph relaxation over 2 states for all events in the group.
            // Since only 2 states, we can just relax repeatedly for each event.
            let mut cur0 = dp0;
            let mut cur1 = dp1;

            for k in i..j {
                let e = events[k];
                let cost = if e.l > 0 { e.l } else { 0 };
                let toggles = (e.o & 1) == 0;

                if toggles {
                    // 0 -> 1 and 1 -> 0 with cost
                    let n0 = cur0.min(cur1.saturating_add(cost));
                    let n1 = cur1.min(cur0.saturating_add(cost));
                    cur0 = n0;
                    cur1 = n1;
                } else {
                    // self-loop with cost (never beneficial if cost>=0), but keep semantics.
                    cur0 = cur0.min(cur0.saturating_add(cost));
                    cur1 = cur1.min(cur1.saturating_add(cost));
                }
            }

            dp0 = cur0;
            dp1 = cur1;

            i = j;
        }

        // End after last intersection; require state 0.
        outputs.push(dp0.to_string());
    }

    print!("{}", outputs.join("\n"));
}
```
