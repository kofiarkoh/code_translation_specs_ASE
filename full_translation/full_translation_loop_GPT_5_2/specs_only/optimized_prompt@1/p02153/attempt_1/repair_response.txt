```rust
use std::cmp::Reverse;
use std::collections::BinaryHeap;
use std::io::{self, Read};

fn main() {
    // Read entire stdin
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let h: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let w: usize = it.next().unwrap().parse().unwrap();

    let cost_normal: i64 = it.next().unwrap().parse().unwrap();
    let cost_bomb: i64 = it.next().unwrap().parse().unwrap();

    // Grid characters:
    // We only rely on: start, goal, bomb, obstacle, and "other" traversable.
    // Since the spec doesn't define exact characters, we use a narrow, common mapping:
    // 'S' = start, 'G' = goal, 'B' = bomb, '#' = obstacle, others = normal traversable.
    // If the input uses different characters, behavior is unspecified per spec.
    let mut grid: Vec<Vec<u8>> = Vec::with_capacity(h);
    let mut start: Option<(usize, usize)> = None;
    let mut goal: Option<(usize, usize)> = None;

    for r in 0..h {
        let token = it.next().unwrap_or("");
        // If the row is not exactly w chars, behavior is undefined by spec; we proceed with what we got.
        let bytes = token.as_bytes().to_vec();
        let mut row = vec![b' '; w];
        for c in 0..w {
            let ch = *bytes.get(c).unwrap_or(&b' ');
            row[c] = ch;
            if ch == b'S' {
                start = Some((r, c));
            } else if ch == b'G' {
                goal = Some((r, c));
            }
        }
        grid.push(row);
    }

    let (sr, sc) = start.unwrap();
    let (gr, gc) = goal.unwrap();

    // Exceptional behavior per spec:
    // "If the starting position is surrounded by bombs, the program will return that the target is unreachable."
    // Interpret "surrounded" as all 4-neighbors (within bounds) are bombs.
    // If there are no in-bounds neighbors (degenerate 1x1), start and goal are distinct per preconditions.
    let dirs = [(1isize, 0isize), (-1, 0), (0, 1), (0, -1)];
    let mut any_neighbor = false;
    let mut all_neighbors_bombs = true;
    for (dr, dc) in dirs {
        let nr = sr as isize + dr;
        let nc = sc as isize + dc;
        if nr >= 0 && nr < h as isize && nc >= 0 && nc < w as isize {
            any_neighbor = true;
            if grid[nr as usize][nc as usize] != b'B' {
                all_neighbors_bombs = false;
            }
        }
    }
    if any_neighbor && all_neighbors_bombs {
        println!("unreachable");
        return;
    }

    // Dijkstra over cells with non-negative costs.
    // Movement cost is based on the cell being traversed *into*:
    // - entering a bomb cell costs cost_bomb
    // - entering any other traversable cell costs cost_normal
    // Obstacles are impassable.
    let n = h * w;
    let mut dist = vec![i64::MAX; n];

    let idx = |r: usize, c: usize| -> usize { r * w + c };

    let mut pq: BinaryHeap<(Reverse<i64>, usize, usize)> = BinaryHeap::new();
    dist[idx(sr, sc)] = 0;
    pq.push((Reverse(0), sr, sc));

    while let Some((Reverse(d), r, c)) = pq.pop() {
        if d != dist[idx(r, c)] {
            continue;
        }
        if (r, c) == (gr, gc) {
            println!("{}", d);
            return;
        }

        for (dr, dc) in dirs {
            let nr = r as isize + dr;
            let nc = c as isize + dc;
            if nr < 0 || nr >= h as isize || nc < 0 || nc >= w as isize {
                continue;
            }
            let nr = nr as usize;
            let nc = nc as usize;

            let cell = grid[nr][nc];
            if cell == b'#' {
                continue;
            }

            let step = if cell == b'B' { cost_bomb } else { cost_normal };

            // Preserve wraparound-like behavior narrowly: use checked_add and saturate to i64::MAX
            // to avoid panic/UB; spec doesn't define overflow, so we choose a safe, minimal behavior.
            let nd = match d.checked_add(step) {
                Some(v) => v,
                None => i64::MAX,
            };

            let id = idx(nr, nc);
            if nd < dist[id] {
                dist[id] = nd;
                pq.push((Reverse(nd), nr, nc));
            }
        }
    }

    println!("unreachable");
}
```
