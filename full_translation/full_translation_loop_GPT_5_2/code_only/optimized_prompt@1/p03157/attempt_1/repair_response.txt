```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, token-scan like scanf("%d%d") and scanf("%s")
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let h: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let w: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut s: Vec<Vec<u8>> = Vec::with_capacity(h);
    for _ in 0..h {
        let row = match it.next() {
            Some(v) => v.as_bytes().to_vec(),
            None => return,
        };
        // C code assumes width W; we preserve behavior for valid inputs.
        s.push(row);
    }

    let mut group = vec![vec![0i32; w]; h];
    // count has indices up to H*W; each entry has 2 counters.
    let mut count = vec![[0u32; 2]; h * w + 1];

    // Iterative DFS to avoid recursion depth issues in Rust.
    let dirs: [(i32, i32); 4] = [(0, 1), (0, -1), (1, 0), (-1, 0)];
    let mut asumisu: i32 = 0;

    for i in 0..h {
        for j in 0..w {
            if group[i][j] == 0 {
                asumisu += 1;
                let gid = asumisu;

                let mut stack: Vec<(usize, usize)> = Vec::new();
                stack.push((i, j));
                while let Some((y, x)) = stack.pop() {
                    if group[y][x] == gid {
                        continue;
                    }
                    group[y][x] = gid;
                    let cur = s[y][x];
                    for (dy, dx) in dirs.iter() {
                        let ny = y as i32 + dy;
                        let nx = x as i32 + dx;
                        if 0 <= ny && ny < h as i32 && 0 <= nx && nx < w as i32 {
                            let nyu = ny as usize;
                            let nxu = nx as usize;
                            if cur != s[nyu][nxu] && group[nyu][nxu] != gid {
                                stack.push((nyu, nxu));
                            }
                        }
                    }
                }
            }
        }
    }

    for i in 0..h {
        for j in 0..w {
            let gid = group[i][j] as usize;
            let idx = if s[i][j] == b'.' { 1 } else { 0 }; // matches (S[i][j] == '.')
            count[gid][idx] = count[gid][idx].wrapping_add(1); // emulate C int overflow if it ever happens
        }
    }

    let mut mizuhasu: u64 = 0;
    for i in 1..=(asumisu as usize) {
        mizuhasu = mizuhasu.wrapping_add((count[i][0] as u64).wrapping_mul(count[i][1] as u64));
    }

    print!("{}\n", mizuhasu);
}
```
