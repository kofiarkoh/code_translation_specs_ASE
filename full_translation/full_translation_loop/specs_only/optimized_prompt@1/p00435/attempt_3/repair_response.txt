use std::io::{self, Read};

fn transform_string(input: &str) -> String {
    let mut output = String::with_capacity(input.len());
    
    for c in input.chars() {
        let transformed_char = if c.is_ascii_uppercase() {
            // Shift uppercase letters backward by 3 positions
            let shifted = c as u8 as i32 - 3;
            // Wrap around if it goes below 'A'
            if shifted < 'A' as i32 {
                (shifted + 26) as u8 as char
            } else {
                shifted as u8 as char
            }
        } else {
            // Shift all other characters forward by 23 positions
            let shifted = c as u8 as i32 + 23;
            // Wrap around if it goes above 255 (ASCII range)
            if shifted > 255 {
                (shifted - 256) as u8 as char
            } else {
                shifted as u8 as char
            }
        };
        output.push(transformed_char);
    }
    
    // No need to null-terminate the output string for Rust's String type
    output
}

fn main() {
    let mut input = String::new();
    // Read input from standard input
    io::stdin().read_to_string(&mut input).expect("Failed to read input");
    
    // Ensure the input does not exceed 999 characters
    if input.len() > 999 {
        // Behavior is not defined if input exceeds maximum length
        return;
    }
    
    // Remove any trailing newline characters that may be added by read_to_string
    input = input.trim_end_matches('\n').to_string();
    
    // Transform the input string
    let transformed = transform_string(&input);
    
    // Output the transformed string
    print!("{}", transformed);
}
