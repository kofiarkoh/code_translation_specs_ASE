struct UnionFind {
    parent: Vec<usize>,
    max_value: Vec<i32>,
    count: Vec<i32>,
}

impl UnionFind {
    fn new(size: usize) -> Self {
        UnionFind {
            parent: (0..size).collect(),
            max_value: vec![0; size],
            count: vec![1; size],
        }
    }

    fn find(&mut self, x: usize) -> usize {
        if self.parent[x] != x {
            self.parent[x] = self.find(self.parent[x]); // Path compression
        }
        self.parent[x]
    }

    fn union(&mut self, x: usize, y: usize, value: i32) {
        let root_x = self.find(x);
        let root_y = self.find(y);

        if root_x != root_y {
            // Union by size
            self.parent[root_y] = root_x;
            self.count[root_x] += self.count[root_y];
            self.max_value[root_x] = self.max_value[root_x].max(self.max_value[root_y]).max(value);
        } else {
            // Update max value if they are already connected
            self.max_value[root_x] = self.max_value[root_x].max(value);
        }
    }

    fn compute_result(&mut self) -> i32 {
        let mut result = 1;
        let mut seen = vec![false; self.parent.len()];

        for i in 0..self.parent.len() {
            let root = self.find(i);
            if !seen[root] {
                // Only consider groups with a maximum value greater than 0
                if self.max_value[root] > 0 {
                    result = (result * self.max_value[root]) % 10007;
                }
                seen[root] = true;
            }
        }
        result
    }
}

fn process_relationships(input: Vec<(usize, usize, i32)>) -> i32 {
    let max_entities = 1000; // Assuming a maximum limit based on the specification
    let mut uf = UnionFind::new(max_entities);

    for (x, y, value) in input {
        uf.union(x, y, value);
    }

    uf.compute_result()
}

fn main() {
    // Example input: number of entities and relationships
    let input = vec![(0, 1, 1), (1, 2, 2), (2, 3, 3), (3, 4, 4), (4, 5, 0)];
    let result = process_relationships(input);
    println!("{}", result); // Output the computed result
}
