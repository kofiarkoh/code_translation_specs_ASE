                for &(v, cost) in neighbors {
                    if !visited[v] {
                        let new_cost = costs[u].saturating_add(cost); // Prevent overflow
                        if new_cost < costs[v] {
                            costs[v] = new_cost; // Update cost if new cost is lower
                        }
                    }
                }
            }
        }

    }
