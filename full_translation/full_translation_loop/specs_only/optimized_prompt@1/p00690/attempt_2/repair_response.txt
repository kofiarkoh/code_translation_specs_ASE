```rust
use std::collections::{HashMap, HashSet};
use std::io::{self, BufRead};

#[derive(Debug)]
struct Connection {
    station1: usize,
    station2: usize,
    distance: usize,
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let line = line.unwrap();
        let mut iter = line.split_whitespace();
        let num_stations: usize = iter.next().unwrap().parse().unwrap();
        let num_connections: usize = iter.next().unwrap().parse().unwrap();

        // Terminate if the number of stations is zero
        if num_stations == 0 {
            return;
        }

        let mut graph: HashMap<usize, Vec<(usize, usize)>> = HashMap::new();

        for _ in 0..num_connections {
            let connection_line = reader.lines().next().unwrap().unwrap();
            let mut conn_iter = connection_line.split_whitespace();
            let station1: usize = conn_iter.next().unwrap().parse().unwrap();
            let station2: usize = conn_iter.next().unwrap().parse().unwrap();
            let distance: usize = conn_iter.next().unwrap().parse().unwrap();

            // Ignore invalid connections
            if station1 < 1 || station1 > num_stations || station2 < 1 || station2 > num_stations {
                continue;
            }

            // Add the connection in both directions
            graph.entry(station1).or_insert_with(Vec::new).push((station2, distance));
            graph.entry(station2).or_insert_with(Vec::new).push((station1, distance));
        }

        let (max_length, path) = find_longest_path(&graph);
        println!("{}", max_length);
        println!("{:?}", path);
    }
}

fn find_longest_path(graph: &HashMap<usize, Vec<(usize, usize)>>) -> (usize, Vec<usize>) {
    let mut max_length = 0;
    let mut longest_path = Vec::new();

    for &start_station in graph.keys() {
        let (length, path) = dfs(start_station, graph, &mut HashSet::new(), 0, vec![start_station]);
        if length > max_length {
            max_length = length;
            longest_path = path;
        }
    }

    (max_length, longest_path)
}

fn dfs(
    current: usize,
    graph: &HashMap<usize, Vec<(usize, usize)>>,
    visited: &mut HashSet<usize>,
    current_length: usize,
    current_path: Vec<usize>,
) -> (usize, Vec<usize>) {
    visited.insert(current);
    let mut max_length = current_length;
    let mut longest_path = current_path.clone();

    if let Some(connections) = graph.get(&current) {
        for &(neighbor, distance) in connections {
            if !visited.contains(&neighbor) {
                let (length, path) = dfs(neighbor, graph, visited, current_length + distance, {
                    let mut new_path = current_path.clone();
                    new_path.push(neighbor);
                    new_path
                });
                if length > max_length {
                    max_length = length;
                    longest_path = path;
                }
            }
        }
    }

    visited.remove(&current);
    (max_length, longest_path)
}
```
