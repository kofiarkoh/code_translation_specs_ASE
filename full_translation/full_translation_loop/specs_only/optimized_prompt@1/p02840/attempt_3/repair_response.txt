fn compute_count(n: i64, x: i64, d: i64) -> i64 {
    // Preconditions: N must be a non-negative integer.
    assert!(n >= 0);

    // Handle special cases based on the specification.
    if x == 0 && d == 0 {
        return 1; // Both X and D are zero.
    }
    if d == 0 {
        return n + 1; // D is zero.
    }
    if x == 0 {
        return n * (n - 1) / 2 + 1; // X is zero and D is non-zero.
    }

    // Calculate the count based on the input parameters.
    let mut count = 0;

    // Iterate over the range from 0 to N.
    for i in 0..=n {
        // Calculate the value based on the formula derived from X and D.
        let value = x + i * d;

        // Count the valid ranges based on the computed value.
        if value <= n {
            count += 1; // Increment count for valid ranges.
        }
    }

    count // Return the computed count.
}

fn main() {
    // Example usage of the compute_count function.
    let n: i64 = 10; // Example input
    let x: i64 = 5;  // Example input
    let d: i64 = 3;  // Example input
    let result = compute_count(n, x, d);
    println!("{}", result); // Output the result
}
