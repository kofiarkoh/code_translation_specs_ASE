```rust
use std::collections::VecDeque;

const INF: usize = usize::MAX; // Represents infinity for distance calculations

fn max_distance(height: usize, width: usize, grid: Vec<Vec<char>>, target: char, different: char) -> usize {
    let mut distances = vec![vec![INF; width]; height]; // Initialize distances to infinity
    let mut queue = VecDeque::new();

    // Initialize distances for cells containing the target character
    for i in 0..height {
        for j in 0..width {
            if grid[i][j] == target {
                distances[i][j] = 0; // Distance to itself is zero
                queue.push_back((i, j)); // Add to queue for BFS
            }
        }
    }

    // Directions for moving in the grid (up, down, left, right)
    let directions = [(1, 0), (-1, 0), (0, 1), (0, -1)];

    // Perform BFS to calculate distances
    while let Some((x, y)) = queue.pop_front() {
        for &(dx, dy) in &directions {
            let new_x = (x as isize + dx) as usize;
            let new_y = (y as isize + dy) as usize;

            // Check bounds and if the cell is different
            if new_x < height && new_y < width && grid[new_x][new_y] == different {
                // Update distance if a shorter path is found
                if distances[new_x][new_y] > distances[x][y] + 1 {
                    distances[new_x][new_y] = distances[x][y] + 1;
                    queue.push_back((new_x, new_y)); // Add to queue for further processing
                }
            }
        }
    }

    // Find the maximum distance from any target cell to the nearest different cell
    let mut max_dist = 0;
    for i in 0..height {
        for j in 0..width {
            if distances[i][j] != INF && grid[i][j] == different {
                max_dist = max_dist.max(distances[i][j]);
            }
        }
    }

    max_dist
}

fn main() {
    // Example usage
    let height = 4;
    let width = 5;
    let grid = vec![
        vec!['.', '.', '#', '.', '.'],
        vec!['.', '#', '.', '.', '.'],
        vec!['.', '.', '.', '.', '.'],
        vec!['.', '.', '#', '.', '.'],
    ];
    let target = '#';
    let different = '.';
    
    let result = max_distance(height, width, grid, target, different);
    println!("{}", result); // Output the maximum distance
}
```
