```rust
use std::cmp::Ordering;
use std::io::{self, BufRead};

const MAX_N: usize = 1_000_000;
const NIL: i32 = -1;

#[derive(Debug, Copy, Clone)]
struct Point {
    id: usize,
    x: i32,
    y: i32,
}

#[derive(Debug, Copy, Clone)]
struct Node {
    location: usize,
    l: i32,
    r: i32,
}

static mut P: [Point; MAX_N] = [Point { id: 0, x: 0, y: 0 }; MAX_N];
static mut T: [Node; MAX_N] = [Node { location: 0, l: NIL, r: NIL }; MAX_N];
static mut NP: usize = 0;
static mut N_PID: usize = 0;
static mut PID: [i32; MAX_N] = [0; MAX_N];

fn cmpx_func(l: &Point, r: &Point) -> Ordering {
    l.x.cmp(&r.x)
}

fn cmpy_func(l: &Point, r: &Point) -> Ordering {
    l.y.cmp(&r.y)
}

fn make_kd_tree(l: usize, r: usize, depth: usize) -> i32 {
    if l >= r {
        return NIL;
    }

    if r - l >= 2 {
        if depth % 2 == 0 {
            unsafe {
                P[l..r].sort_unstable_by(cmpx_func);
            }
        } else {
            unsafe {
                P[l..r].sort_unstable_by(cmpy_func);
            }
        }
    }

    let mid = (l + r) / 2;
    let s = unsafe { NP };
    unsafe {
        NP += 1;
        T[s].location = mid;
        T[s].l = make_kd_tree(l, mid, depth + 1);
        T[s].r = make_kd_tree(mid + 1, r, depth + 1);
    }
    s as i32
}

fn find(v: usize, sx: i32, tx: i32, sy: i32, ty: i32, depth: usize) {
    unsafe {
        let x = P[T[v].location].x;
        let y = P[T[v].location].y;

        if sx <= x && x <= tx && sy <= y && y <= ty {
            PID[N_PID] = P[T[v].location].id as i32;
            N_PID += 1;
        }

        if depth % 2 == 0 {
            if T[v].l != NIL {
                if sx <= x {
                    find(T[v].l as usize, sx, tx, sy, ty, depth + 1);
                }
            }
            if T[v].r != NIL {
                if x <= tx {
                    find(T[v].r as usize, sx, tx, sy, ty, depth + 1);
                }
            }
        } else {
            if T[v].l != NIL {
                if sy <= y {
                    find(T[v].l as usize, sx, tx, sy, ty, depth + 1);
                }
            }
            if T[v].r != NIL {
                if y <= ty {
                    find(T[v].r as usize, sx, tx, sy, ty, depth + 1);
                }
            }
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let n: usize = lines.next().unwrap().unwrap().parse().unwrap();
    for i in 0..n {
        let line = lines.next().unwrap().unwrap();
        let mut coords = line.split_whitespace();
        let x: i32 = coords.next().unwrap().parse().unwrap();
        let y: i32 = coords.next().unwrap().parse().unwrap();
        unsafe {
            P[i] = Point { id: i, x, y };
            T[i].l = NIL;
            T[i].r = NIL;
        }
    }

    unsafe {
        NP = 0;
        let root = make_kd_tree(0, n, 0);
        let q: usize = lines.next().unwrap().unwrap().parse().unwrap();

        for _ in 0..q {
            let line = lines.next().unwrap().unwrap();
            let mut bounds = line.split_whitespace();
            let sx: i32 = bounds.next().unwrap().parse().unwrap();
            let tx: i32 = bounds.next().unwrap().parse().unwrap();
            let sy: i32 = bounds.next().unwrap().parse().unwrap();
            let ty: i32 = bounds.next().unwrap().parse().unwrap();
            N_PID = 0;
            find(root as usize, sx, tx, sy, ty, 0);
            let pid_slice = &mut PID[..N_PID];
            pid_slice.sort_unstable();

            for id in pid_slice.iter() {
                println!("{}", id);
            }
            println!();
        }
    }
}
```
