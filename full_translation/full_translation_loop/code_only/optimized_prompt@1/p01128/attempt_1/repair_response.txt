```rust
use std::io::{self, BufRead};
use std::cmp::Ordering;

const EPS: f64 = 1e-10;

#[derive(Clone, Copy)]
struct P {
    x: f64,
    y: f64,
    o: i32,
    l: i32,
}

fn cmp(a: &P, b: &P) -> Ordering {
    if (a.x - b.x).abs() < 0.0 {
        if a.y < b.y {
            return Ordering::Less;
        }
        if b.y < a.y {
            return Ordering::Greater;
        }
    }
    if a.x < b.x {
        return Ordering::Less;
    }
    if b.x < a.x {
        return Ordering::Greater;
    }
    Ordering::Equal
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let c: i32 = lines.next().unwrap().unwrap().parse().unwrap();
    
    for _ in 0..c {
        let mut st: Vec<P> = Vec::new();
        let line = lines.next().unwrap().unwrap();
        let mut params = line.split_whitespace().map(|s| s.parse::<i32>().unwrap());
        let xa = params.next().unwrap();
        let ya = params.next().unwrap();
        let xb = params.next().unwrap();
        let yb = params.next().unwrap();
        let n = params.next().unwrap();

        for _ in 0..n {
            let line = lines.next().unwrap().unwrap();
            let mut params = line.split_whitespace().map(|s| s.parse::<i32>().unwrap());
            let xs = params.next().unwrap();
            let ys = params.next().unwrap();
            let xt = params.next().unwrap();
            let yt = params.next().unwrap();
            let o = params.next().unwrap();
            let l = params.next().unwrap();

            let tx = xs - xa;
            let ty = ys - ya;
            let b = (xb - xa) * (yt - ys) - (yb - ya) * (xt - xs);
            if b == 0 {
                continue;
            }
            let r = (yt - ys) as f64 * tx as f64 - (xt - xs) as f64 * ty as f64 / b as f64;
            let s = (yb - ya) as f64 * tx as f64 - (xb - xa) as f64 * ty as f64 / b as f64;

            if r < EPS || s < EPS || 1.0 - r < EPS || 1.0 - s < EPS {
                continue;
            }
            if r > EPS && 1.0 - r > EPS {
                st.push(P {
                    x: xa as f64 + r * (xb - xa) as f64,
                    y: ya as f64 + r * (yb - ya) as f64,
                    o,
                    l,
                });
            } else {
                st.push(P {
                    x: xs as f64 + s * (xt - xs) as f64,
                    y: ys as f64 + s * (yt - ys) as f64,
                    o,
                    l,
                });
            }
        }

        st.sort_by(|a, b| cmp(a, b));
        let mut r = 999;

        for l in 0..2 {
            let mut t = l;
            let mut b = 0;
            for i in 0..st.len() {
                if st[i].o == 1 && st[i].l != t {
                    b += 1;
                    t ^= 1;
                }
                if st[i].o != 1 && st[i].l == t {
                    b += 1;
                    t ^= 1;
                }
            }
            r = r.min(b);
        }
        println!("{}", r);
    }
}
```
