```rust
use std::io;

const MAX_SIZE: usize = 100010;

fn dfs(a: &mut [i32; MAX_SIZE], n: usize) {
    let mut t = a[n];
    while a[t as usize] != -1 {
        let temp = a[t as usize];
        a[t as usize] = -1;
        t = temp;
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let mut iter = input.split_whitespace();
    let n: usize = iter.next().unwrap().parse().unwrap();
    let k: usize = iter.next().unwrap().parse().unwrap();

    let mut a = [0; MAX_SIZE];
    input.clear();
    io::stdin().read_line(&mut input).unwrap();
    let values: Vec<i32> = input.split_whitespace().map(|s| s.parse().unwrap()).collect();
    for i in 1..=n {
        a[i] = values[i - 1];
    }

    if n == k {
        let x = a[1] - 1;
        let mut flag = true; // Use a boolean instead of an integer
        for i in 1..=n {
            flag &= (a[i] - i as i32 + n as i32) % n as i32 == x;
        }
        println!("{}", if flag { "Yes" } else { "No" });
    } else {
        let mut cnt = 0;
        for i in 1..=n {
            if a[i] != -1 {
                dfs(&mut a, i);
                cnt += 1;
            }
        }
        println!("{}", if (n - cnt) % 2 != 0 && k % 2 != 0 { "No" } else { "Yes" });
    }
}
```
