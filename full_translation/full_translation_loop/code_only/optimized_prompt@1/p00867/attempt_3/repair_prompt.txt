impl UnionFind {

        let mut id = [0; MAX];
        let mut size = [1; MAX];
        for i in 0..n {
            id[i] = i;
        }
        UnionFind { id, size }

    fn root(&mut self, i: usize) -> usize {
        let mut i = i;
        while i != self.id[i] {
            self.id[i] = self.id[self.id[i]];
            i = self.id[i];
        }
        i
    }

    fn connected(&mut self, p: usize, q: usize) -> bool {
        self.root(p) == self.root(q)
