#[derive(Debug)]
struct T {
    a: i64,
    d: i32,
}

fn in_number() -> i32 {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut n = 0;
    let mut c = handle.bytes().next().unwrap().unwrap() as char;

    while c.is_digit(10) {
        n = (n << 3) + (n << 1) + (c as i32 - '0' as i32);
        c = handle.bytes().next().unwrap().unwrap() as char;
    }
    n
}

fn cmp(a: &T, b: &T) -> std::cmp::Ordering {
    let left = a.a * b.d;
    let right = b.a * a.d;
    left.cmp(&right)
}
