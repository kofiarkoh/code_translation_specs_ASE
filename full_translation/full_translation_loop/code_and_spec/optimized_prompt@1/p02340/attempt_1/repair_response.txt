```rust
use std::io::{self, BufRead};

const DIVISOR: u64 = 1_000_000_007;
const NUM_MAX: usize = 1000;
const K_MAX: usize = 1000;

// pdp[# of partitions][# of balls]
static mut PDP: [[u64; NUM_MAX + 1]; K_MAX + 1] = [[0; NUM_MAX + 1]; K_MAX + 1];

fn init() {
    unsafe {
        PDP[0][0] = 1;
        for i in 1..=K_MAX {
            for j in 0..=NUM_MAX {
                PDP[i][j] = PDP[i - 1][j];
                if j >= i {
                    PDP[i][j] = (PDP[i][j] + PDP[i][j - i]) % DIVISOR;
                }
            }
        }
    }
}

fn get_int2() -> Result<(usize, usize), &'static str> {
    let stdin = io::stdin();
    let line = stdin.lock().lines().next().ok_or("Input reading failed")??;
    let mut parts = line.split_whitespace();
    let n = parts.next().ok_or("Missing first integer")?.parse::<usize>().map_err(|_| "Invalid integer")?;
    let k = parts.next().ok_or("Missing second integer")?.parse::<usize>().map_err(|_| "Invalid integer")?;
    Ok((n, k))
}

fn main() {
    let (n, k) = match get_int2() {
        Ok(values) => values,
        Err(e) => {
            eprintln!("{}", e);
            return;
        }
    };

    if n > NUM_MAX || k > K_MAX {
        eprintln!("Input values exceed predefined maximum limits.");
        return;
    }

    init();
    unsafe {
        println!("{}", PDP[k][n]);
    }
}
```
