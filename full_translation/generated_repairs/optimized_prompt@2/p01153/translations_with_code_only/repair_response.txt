// AOJ 2028: Gather on the Clock
// 2017.11.03 bal4u@uu

use std::io::{self, BufRead};

const MAX_N: usize = 102;

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let t: usize = lines.next().unwrap().unwrap().parse().expect("Failed to parse number of test cases");
    for _ in 0..t {
        let n: usize = lines.next().unwrap().unwrap().parse().expect("Failed to parse size of array");
        let mut a = vec![0; n]; // Use a vector to accommodate variable size
        for i in 0..n {
            a[i] = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse array element"); // Trim input to avoid invalid digits
        }

        let mut dp = vec![vec![0; n]; n]; // Use a vector for dynamic size

        for k in 1..n {
            for i in 0..n {
                for j in 0..k {
                    let x = (i + j + 1) % n; // Simplified modulo operation
                    let v = (a[x] as isize - a[i] as isize).abs(); // Use isize for subtraction
                    let total = v + dp[x][k - j - 1] + dp[i][j];
                    if total > dp[i][k] {
                        dp[i][k] = total;
                    }
                }
            }
        }

        let ans = dp.iter().flat_map(|row| row.iter()).max().unwrap_or(&0); // Find the maximum value
        println!("{}", ans);
    }
}
