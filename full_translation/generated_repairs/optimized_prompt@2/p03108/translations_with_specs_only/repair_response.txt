// Union-Find data structure to manage connected components in a graph
struct UnionFind {
    parent: Vec<usize>,
    size: Vec<usize>,
    count: usize,
}

impl UnionFind {
    // Initialize the Union-Find structure
    fn new(n: usize) -> Self {
        let parent = (0..n).collect();
        let size = vec![1; n];
        UnionFind { parent, size, count: n }
    }

    // Find the root of the component for a given node
    fn find(&mut self, node: usize) -> usize {
        if self.parent[node] != node {
            self.parent[node] = self.find(self.parent[node]); // Path compression
        }
        self.parent[node]
    }

    // Union two components
    fn union(&mut self, a: usize, b: usize) {
        let root_a = self.find(a);
        let root_b = self.find(b);
        
        if root_a != root_b {
            // Union by size
            if self.size[root_a] < self.size[root_b] {
                self.parent[root_a] = root_b;
                self.size[root_b] += self.size[root_a];
            } else {
                self.parent[root_b] = root_a;
                self.size[root_a] += self.size[root_b];
            }
            self.count -= 1; // Decrease the count of connected components
        }
    }

    // Calculate the number of connected pairs
    fn connected_pairs(&self) -> usize {
        // Using the formula size * (size - 1) / 2 for pairs
        let total_pairs = self.size.iter().map(|&s| s * (s - 1) / 2).sum::<usize>();
        total_pairs
    }
}

// Main function to process edges and output results
fn main() {
    use std::io::{self, BufRead};

    let stdin = io::stdin();
    let reader = stdin.lock();

    // Read number of nodes and edges
    let mut lines = reader.lines();
    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let n: usize = parts.next().unwrap().parse().unwrap();
    let m: usize = parts.next().unwrap().parse().unwrap();

    let mut uf = UnionFind::new(n);
    let mut results = Vec::new();

    // Read edges and process them
    for _ in 0..m {
        let edge_line = lines.next().unwrap().unwrap();
        let mut edge_parts = edge_line.split_whitespace();
        let u: usize = edge_parts.next().unwrap().parse().unwrap();
        let v: usize = edge_parts.next().unwrap().parse().unwrap();

        // Adjust for 0-based indexing
        uf.union(u - 1, v - 1);
        results.push(uf.connected_pairs());
    }

    // Output results in reverse order
    for result in results.iter().rev() {
        println!("{}", result);
    }
}
