// AOJ 1168: Off Balance

use std::io::{self, BufRead};

const MAX: usize = 602;
const MAX_TBL: usize = 100;
const MAX_MAP: usize = 60;

struct UnionFind {
    p: [usize; MAX],
    rank: [usize; MAX],
}

impl UnionFind {
    fn new() -> Self {
        UnionFind {
            p: (0..MAX).collect::<Vec<_>>().try_into().unwrap(),
            rank: [0; MAX],
        }
    }

    fn make_set(&mut self, x: usize) {
        self.p[x] = x;
        self.rank[x] = 0;
    }

    fn find_set(&mut self, x: usize) -> usize {
        if x != self.p[x] {
            self.p[x] = self.find_set(self.p[x]);
        }
        self.p[x]
    }

    fn union_set(&mut self, x: usize, y: usize) {
        let root_x = self.find_set(x);
        let root_y = self.find_set(y);
        self.link(root_x, root_y);
    }

    fn link(&mut self, x: usize, y: usize) {
        if self.rank[x] > self.rank[y] {
            self.p[y] = x;
        } else {
            self.p[x] = y;
            if self.rank[x] == self.rank[y] {
                self.rank[y] += 1;
            }
        }
    }
}

#[derive(Default)]
struct T {
    used: usize,
    m: usize,
    mx: usize,
    xl: isize,
    xr: isize,
    par: usize,
    ch: usize,
    sib: usize,
}

struct Tree {
    tbl: [T; MAX_TBL],
    sz: usize,
    tr: [isize; MAX],
}

impl Tree {
    fn new() -> Self {
        Tree {
            tbl: Default::default(),
            sz: 0,
            tr: [-1; MAX],
        }
    }

    fn mktree(&mut self, r: usize, c: usize, this: usize, h: usize, map: &Vec<Vec<char>>, rel: &Vec<Vec<usize>>) {
        if r == h - 1 {
            if map[r][c] == '.' {
                return;
            }
            if self.tbl[this].xl < 0 {
                self.tbl[this].xl = c as isize;
            } else if (c as isize) < self.tbl[this].xl {
                self.tbl[this].xl = c as isize;
            }
            if (c as isize) > self.tbl[this].xr {
                self.tbl[this].xr = c as isize;
            }
            return;
        }
        if map[r + 1][c] == '.' {
            return;
        }

        let k = rel[r + 1][c];
        let p = self.tr[k];

        if p == this as isize {
            return;
        }

        if self.tbl[this].xl < 0 {
            self.tbl[this].xl = c as isize;
        } else if (c as isize) < self.tbl[this].xl {
            self.tbl[this].xl = c as isize;
        }
        if (c as isize) > self.tbl[this].xr {
            self.tbl[this].xr = c as isize;
        }
        if self.tbl[this].par != 0 {
            return;
        }

        let s = self.tbl[p as usize].ch;
        if s == 0 {
            self.tbl[p as usize].ch = this;
        } else {
            let mut s = s;
            while self.tbl[s].sib != 0 {
                s = self.tbl[s].sib;
            }
            self.tbl[s].sib = this;
        }
        self.tbl[this].par = p as usize;
    }

    fn check(&mut self, this: usize) -> usize {
        if self.tbl[this].used != 0 {
            return 1;
        }
        let ch = self.tbl[this].ch;
        if ch != 0 && self.check(ch) == 0 {
            return 0;
        }

        let mut sib = self.tbl[this].sib;
        while sib != 0 {
            if self.check(sib) == 0 {
                return 0;
            }
            sib = self.tbl[sib].sib;
        }

        let x = 0.5 + (self.tbl[this].mx as f64) / (self.tbl[this].m as f64);
        if x <= self.tbl[this].xl as f64 || self.tbl[this].xr as f64 + 1.0 <= x {
            return 0;
        }

        let par = self.tbl[this].par;
        self.tbl[par].m += self.tbl[this].m;
        self.tbl[par].mx += self.tbl[this].mx;
        self.tbl[this].used = 1;
        1
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let mut union_find = UnionFind::new();
    let mut tree = Tree::new();
    let mut map = vec![vec![' '; 12]; MAX_MAP];
    let mut rel = vec![vec![0; 10]; MAX_MAP];
    let mv: [[isize; 2]; 4] = [[-1, 0], [1, 0], [0, -1], [0, 1]]; // Define mv here

    while let Some(line) = lines.next() {
        let line = line.unwrap();
        let mut iter = line.split_whitespace();
        let w: usize = iter.next().unwrap().parse().unwrap();
        let h: usize = iter.next().unwrap().parse().unwrap();

        if w == 0 {
            break;
        }

        for r in 0..h {
            let row = lines.next().unwrap().unwrap();
            for c in 0..w {
                map[r][c] = row.chars().nth(c).unwrap();
                let i = r * w + c;
                union_find.make_set(i);
                rel[r][c] = i;
            }
        }

        for r in 0..h {
            for c in 0..w {
                if map[r][c] == '.' {
                    continue;
                }
                for i in 0..4 {
                    let r2 = r as isize + mv[i][0];
                    let c2 = c as isize + mv[i][1];
                    if r2 >= 0 && r2 < h as isize && c2 >= 0 && c2 < w as isize {
                        if map[r2 as usize][c2 as usize] == map[r][c] {
                            union_find.union_set(rel[r][c], rel[r2 as usize][c2 as usize]);
                        }
                    }
                }
            }
        }

        tree.sz = 1;
        tree.tbl.iter_mut().for_each(|t| *t = T::default());
        tree.tr.iter_mut().for_each(|t| *t = -1);

        for r in (0..h).rev() {
            for c in 0..w {
                if map[r][c] == '.' {
                    continue;
                }
                let k = rel[r][c];
                let k = union_find.find_set(k);
                let i = if tree.tr[k] < 0 {
                    let idx = tree.sz;
                    tree.sz += 1;
                    tree.tr[k] = idx as isize;
                    &mut tree.tbl[idx]
                } else {
                    let idx = tree.tr[k] as usize;
                    &mut tree.tbl[idx]
                };
                i.m += 1;
                i.mx += c;
                tree.mktree(r, c, tree.tr[k] as usize, h, &map, &rel);
            }
        }
        tree.tbl[1].par = 1;

        println!("{}", if tree.check(1) != 0 { "STABLE" } else { "UNSTABLE" });
    }
}
