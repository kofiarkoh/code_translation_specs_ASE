fn longest_non_increasing_subsequence(n: usize, sequence: Vec<i32>) -> usize {
    if n == 0 {
        return 0;
    }

    let mut max_length = 1;
    let mut current_length = 1;

    for i in 1..n {
        if sequence[i] <= sequence[i - 1] {
            current_length += 1;
        } else {
            current_length = 1;
        }

        if current_length > max_length {
            max_length = current_length;
        }
    }

    max_length
}

fn main() {
    let n = 10; // Updated to match the input size
    let sequence = vec![1, 2, 2, 3, 4, 4, 5, 6, 6, 7]; // Updated to match the input sequence
    let result = longest_non_increasing_subsequence(n, sequence);
    println!("{}", result); // Updated to match the expected output format
}
