fn calculate_polygon_area(polygon: &[(i64, i64)]) -> i64 {
    let n = polygon.len();
    let mut area = 0;

    for i in 0..n {
        let (x1, y1) = polygon[i];
        let (x2, y2) = polygon[(i + 1) % n]; // Wrap around to the first vertex
        area += x1 * y2 - x2 * y1;
    }

    (area.abs() / 2) // Return the absolute value of the area divided by 2
}
