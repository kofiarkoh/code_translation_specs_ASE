```rust
use std::io::{self, Write};

fn main() {
    // Input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let mut iter = input.split_whitespace();
    
    // Check if we have enough inputs
    let x: i32 = match iter.next().and_then(|s| s.parse().ok()) {
        Some(value) => value,
        None => {
            eprintln!("Failed to parse x");
            return;
        }
    };
    
    let y: i32 = match iter.next().and_then(|s| s.parse().ok()) {
        Some(value) => value,
        None => {
            eprintln!("Failed to parse y");
            return;
        }
    };

    // Solve
    let mut ans: i64 = 0; // Using i64 to match the C long long type
    if x >= 0 {
        if y >= x {
            ans = y as i64 - x as i64;
        } else if y > 0 {
            ans = x as i64 - y as i64 + 2;
        } else if y >= -x {
            ans = x as i64 + y as i64 + 1;
        } else {
            ans = -y as i64 - x as i64 + 1;
        }
    } else {
        if y >= -x {
            ans = x as i64 + y as i64 + 1;
        } else if y > 0 {
            ans = -x as i64 - y as i64 + 1;
        } else if y >= x {
            ans = y as i64 - x as i64;
        } else {
            ans = x as i64 - y as i64 + 2;
        }
    }

    // Output
    println!("{}", ans);
    io::stdout().flush().expect("Failed to flush stdout"); // Ensure output is printed immediately
}
```
