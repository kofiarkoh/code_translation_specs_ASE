```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut reader = stdin.lock();

    loop {
        let mut input = String::new();
        reader.read_line(&mut input).unwrap();
        let mut nums = input.split_whitespace()
            .filter_map(|s| s.parse::<i32>().ok());
        
        let n = match nums.next() {
            Some(num) => num,
            None => break,
        };
        let m = match nums.next() {
            Some(num) => num,
            None => break,
        };

        if n == 0 && m == 0 {
            break;
        }

        let mut q = vec![vec![0; m as usize]; n as usize];
        for j in 0..n {
            let mut row_input = String::new();
            reader.read_line(&mut row_input).unwrap();
            let row_nums: Vec<i32> = row_input.split_whitespace()
                .filter_map(|s| s.parse::<i32>().ok())
                .collect();
            for i in 0..m {
                q[j as usize][i as usize] = row_nums[i as usize];
            }
        }

        let mut cnt = vec![0; m as usize];
        for i in 0..n {
            for j in 0..m {
                cnt[j as usize] += q[i as usize][j as usize];
            }
        }

        let mut maxed = vec![false; m as usize];
        for k in 0..m {
            let mut max = -1;
            for j in 0..m {
                if !maxed[j as usize] {
                    if max == -1 || cnt[j as usize] > cnt[max as usize] {
                        max = j;
                    }
                }
            }
            if max != -1 {
                if k > 0 {
                    print!(" ");
                }
                print!("{}", max + 1);
                maxed[max as usize] = true;
            }
        }
        println!();
    }
}
```
