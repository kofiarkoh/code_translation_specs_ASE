use std::io::{self, BufRead};

fn merge_sort(arr: &mut Vec<i64>, left: usize, right: usize) {
    if left + 1 < right {
        let mid = (left + right) / 2;
        merge_sort(arr, left, mid);
        merge_sort(arr, mid, right);
        merge(arr, left, mid, right);
    }
}

fn merge(arr: &mut Vec<i64>, left: usize, mid: usize, right: usize) {
    let n1 = mid - left;
    let n2 = right - mid;

    let mut left_part = vec![0; n1 + 1];
    let mut right_part = vec![0; n2 + 1];

    for i in 0..n1 {
        left_part[i] = arr[left + i];
    }
    for j in 0..n2 {
        right_part[j] = arr[mid + j];
    }

    left_part[n1] = i64::MAX; // Using i64::MAX as a sentinel value
    right_part[n2] = i64::MAX; // Using i64::MAX as a sentinel value

    let mut i = 0;
    let mut j = 0;

    for k in left..right {
        if left_part[i] <= right_part[j] {
            arr[k] = left_part[i];
            i += 1;
        } else {
            arr[k] = right_part[j];
            j += 1;
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let n: usize = lines.next().unwrap().unwrap().parse().expect("Failed to parse n");
    let mut a: Vec<i64> = Vec::with_capacity(n);

    for _ in 0..n {
        let value: i64 = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse value");
        a.push(value);
    }

    merge_sort(&mut a, 0, n);

    let (mut A, mut B) = (0, 0);

    if n % 2 == 0 {
        for i in (0..n).rev().step_by(2) {
            A += a[i];
            B += a[i - 1];
        }
    } else {
        for i in (1..n).rev().step_by(2) {
            A += a[i];
            B += a[i - 1];
        }
        A += a[0];
    }

    println!("{}", A - B);
}
