            if graph.cc[i] == c {
                graph.qu[graph.cnt] = i as i32;
                graph.cnt += 1;
            }
        }
        let mut cnt_ = 0;
        hh.clear();
        graph.dfs3(graph.qu[0] as usize, &mut visited, &mut hh, c as i32, &mut cnt_);
        graph.dfs4(graph.qu[0] as usize, &mut visited, &mut hh, c as i32, &mut cnt_);
        for g_ in 0..cnt_ {
            let h = hh[g_ as usize];
            visited.fill(0);
            let h_ = h;
            diff[h as usize] = if graph.dfs5(graph.ii[h as usize] as usize, graph.jj[h as usize] as usize, &mut visited, h_, c as i32) == 0 {
                1
            } else {
                0
            };
        }
    }

    for h in 0..m {
        println!("{}", if diff[h] != 0 { "diff" } else { "same" });
    }
}
