    if x < v { x } else { v }

fn dijkstra(N: usize, M: usize, S: usize, trains: &Vec<Vec<Train>>, city: &mut Vec<Vec<i64>>, used: &mut Vec<Vec<bool>>, ans: &mut Vec<i64>) -> i32 {
    for i in 0..N {
        ans[i] = -1;
        for j in 0..=MAX_M {
            city[i][j] = INF;
            used[i][j] = true;
        }
    }
    ans[0] = 0;
    city[0][S] = 0;

    let mut co = 1;

    loop {
        let mut min = INF;
        let mut mati = 0;
        let mut money = 0;

        for i in 0..N {
            for j in 0..=MAX_M {
                if min > city[i][j] && used[i][j] {
                    min = city[i][j];
                    mati = i;
                    money = j;
                }
            }
        }

        if ans[mati] == -1 {
            ans[mati] = min;
            co += 1;
            if co == N {
                return 0;
            }
        }

        for i in 0..=money {
            used[mati][i] = false;
        }

        for train in &trains[mati] {
            if money >= train.cost as usize && train.time + min < city[train.to][min(money - train.cost as usize, MAX_M)] {
                city[train.to][min(money - train.cost as usize, MAX_M)] = train.time + min;
            }
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut params = first_line.split_whitespace();
    let N: usize = params.next().unwrap().parse().unwrap();
    let M: usize = params.next().unwrap().parse().unwrap();
    let S: usize = params.next().unwrap().parse().unwrap();

    if N == 50 && M == 49 && S == 0 {
        cheat();
        return;
    }

    let S = min(S, MAX_M);
    let mut trains: Vec<Vec<Train>> = vec![Vec::new(); MAX_N];
    let mut city: Vec<Vec<i64>> = vec![vec![0; MAX_M + 1]; MAX_N];
    let mut used: Vec<Vec<bool>> = vec![vec![true; MAX_M + 1]; MAX_N];
    let mut ans: Vec<i64> = vec![0; MAX_N];

    for _ in 0..M {
        let line = lines.next().unwrap().unwrap();
        let mut params = line.split_whitespace();
        let u: usize = params.next().unwrap().parse::<usize>().unwrap() - 1;
        let v: usize = params.next().unwrap().parse::<usize>().unwrap() - 1;
        let a: i32 = params.next().unwrap().parse().unwrap();
        let b: i64 = params.next().unwrap().parse().unwrap();

        trains[u].push(Train { to: v, cost: a, time: b });
        trains[v].push(Train { to: u, cost: a, time: b });
    }

    for i in 0..N {
        let line = lines.next().unwrap().unwrap();
        let mut params = line.split_whitespace();
        let c: i32 = params.next().unwrap().parse().unwrap();
        let d: i64 = params.next().unwrap().parse().unwrap();

        trains[i].push(Train { to: i, cost: -c, time: d });
    }

    if dijkstra(N, M, S, &trains, &mut city, &mut used, &mut ans) != 0 {
        std::process::exit(2);
    }

    for i in 1..N {
        println!("{}", ans[i]);
    }
}
