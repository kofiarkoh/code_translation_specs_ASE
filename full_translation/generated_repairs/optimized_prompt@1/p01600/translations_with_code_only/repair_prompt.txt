const ABS: fn(i32) -> i32 = |a| if a >= 0 { a } else { -a };

fn in_number<R: BufRead>(reader: &mut R) -> i32 {
    let mut n = 0;
    let mut c = reader.bytes().next().unwrap().unwrap();
    while c >= b'0' {
        n = 10 * n + (c - b'0') as i32;
        c = reader.bytes().next().unwrap().unwrap();
    }
    n
}
