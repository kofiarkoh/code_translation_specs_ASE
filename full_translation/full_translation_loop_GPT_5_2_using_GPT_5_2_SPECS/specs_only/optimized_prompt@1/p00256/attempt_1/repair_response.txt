use std::io::{self, Read};

fn is_leap_year(y: u64) -> bool {
    (y % 4 == 0) && ((y % 100 != 0) || (y % 400 == 0))
}

fn month_len(y: u64, m: u64) -> u64 {
    match m {
        1 => 31,
        2 => {
            if is_leap_year(y) {
                29
            } else {
                28
            }
        }
        3 => 31,
        4 => 30,
        5 => 31,
        6 => 30,
        7 => 31,
        8 => 31,
        9 => 30,
        10 => 31,
        11 => 30,
        12 => 31,
        _ => 0, // Out-of-range month: behavior unspecified by spec; keep narrow.
    }
}

fn days_in_year(y: u64) -> u64 {
    if is_leap_year(y) { 366 } else { 365 }
}

// Convert Gregorian (year, month, day) to an absolute day count relative to epoch 2001-01-01.
// Day 0 corresponds to 2001-01-01.
// Spec implies years earlier than 2001 are not supported by the internal arithmetic.
fn gregorian_to_daycount(y: u64, m: u64, d: u64) -> u64 {
    let mut days: u64 = 0;

    // Add whole years from 2001 up to y-1.
    let mut yy = 2001u64;
    while yy < y {
        days = days.wrapping_add(days_in_year(yy));
        yy += 1;
    }

    // Add whole months in year y up to m-1.
    let mut mm = 1u64;
    while mm < m {
        days = days.wrapping_add(month_len(y, mm));
        mm += 1;
    }

    // Add days within month (1-based day).
    days = days.wrapping_add(d.wrapping_sub(1));
    days
}

// Convert absolute day count (relative to 2001-01-01) to Gregorian (year, month, day).
// Uses linear year advancement as implied by spec.
fn daycount_to_gregorian(mut days: u64) -> (u64, u64, u64) {
    let mut y: u64 = 2001;

    loop {
        let dy = days_in_year(y);
        if days < dy {
            break;
        }
        days -= dy;
        y += 1;
    }

    let mut m: u64 = 1;
    loop {
        let dm = month_len(y, m);
        if days < dm {
            break;
        }
        days -= dm;
        m += 1;
    }

    let d = days + 1;
    (y, m, d)
}

// Mixed-radix uses successive bases 13, 20, 20, 18, 20 (most significant to least).
// Interpret 5 components as a single day count in that mixed radix.
fn mixed_to_daycount(a: u64, b: u64, c: u64, d: u64, e: u64) -> u64 {
    // Use wrapping arithmetic to emulate potential overflow behavior being unspecified.
    let mut v = a;
    v = v.wrapping_mul(20).wrapping_add(b);
    v = v.wrapping_mul(20).wrapping_add(c);
    v = v.wrapping_mul(18).wrapping_add(d);
    v = v.wrapping_mul(20).wrapping_add(e);
    v
}

// Convert day count to 5-component mixed-radix (bases 13,20,20,18,20).
fn daycount_to_mixed(mut v: u64) -> (u64, u64, u64, u64, u64) {
    let e = v % 20;
    v /= 20;
    let d = v % 18;
    v /= 18;
    let c = v % 20;
    v /= 20;
    let b = v % 20;
    v /= 20;
    let a = v % 13;
    (a, b, c, d, e)
}

fn parse_components(token: &str) -> Vec<u64> {
    // Spec preconditions say only digits and dots; behavior otherwise unspecified.
    token
        .split('.')
        .map(|s| s.parse::<u64>().unwrap_or(0))
        .collect()
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    for tok in input.split_whitespace() {
        if tok.as_bytes().first() == Some(&b'#') {
            break;
        }

        let parts = parse_components(tok);

        if parts.len() == 3 {
            let y = parts[0];
            let m = parts[1];
            let d = parts[2];
            let daycount = gregorian_to_daycount(y, m, d);
            let (a, b, c, dd, e) = daycount_to_mixed(daycount);
            println!("{}.{}.{}.{}.{}", a, b, c, dd, e);
        } else {
            // If component count is other than 3, treat as 5-component form (spec).
            // If not exactly 5, behavior is unspecified; we interpret missing as 0 and ignore extras.
            let a = *parts.get(0).unwrap_or(&0);
            let b = *parts.get(1).unwrap_or(&0);
            let c = *parts.get(2).unwrap_or(&0);
            let d = *parts.get(3).unwrap_or(&0);
            let e = *parts.get(4).unwrap_or(&0);

            let daycount = mixed_to_daycount(a, b, c, d, e);
            let (y, m, dd) = daycount_to_gregorian(daycount);
            println!("{}.{}.{}", y, m, dd);
        }
    }
}
