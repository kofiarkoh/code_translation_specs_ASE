```rust
use std::io::{self, Read};

fn main() {
    // No explicit error handling per specification; malformed/missing input is undefined.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();

    // Store up to N elements (N <= 100 per precondition).
    let mut a: Vec<i64> = Vec::with_capacity(n);
    for _ in 0..n {
        let v: i64 = it.next().unwrap().parse().unwrap();
        a.push(v);
    }

    // Count non-overlapping adjacent equal pairs.
    let mut count: i64 = 0;
    let mut i: usize = 0;
    while i + 1 < n {
        if a[i] == a[i + 1] {
            count += 1;
            i += 2; // skip next to ensure disjoint pairs
        } else {
            i += 1;
        }
    }

    println!("{}", count);
}
```
