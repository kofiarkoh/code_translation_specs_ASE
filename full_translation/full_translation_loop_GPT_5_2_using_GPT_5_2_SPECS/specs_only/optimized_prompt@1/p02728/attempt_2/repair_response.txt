```rust
use std::io::{self, Read};

const MOD: u64 = 1_000_000_007;

fn mod_pow(mut a: u64, mut e: u64) -> u64 {
    let mut r = 1u64;
    while e > 0 {
        if (e & 1) == 1 {
            r = (r * a) % MOD;
        }
        a = (a * a) % MOD;
        e >>= 1;
    }
    r
}

fn mod_inv(x: u64) -> u64 {
    // MOD is prime; precondition states inverses are well-defined (x % MOD != 0).
    mod_pow(x % MOD, MOD - 2)
}

fn main() {
    // Input
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_whitespace();

    let n: usize = match it.next() {
        Some(t) => t.parse().unwrap(),
        None => return,
    };

    let mut g: Vec<Vec<usize>> = vec![Vec::new(); n];
    for _ in 0..n.saturating_sub(1) {
        let a: usize = it.next().unwrap().parse::<usize>().unwrap() - 1;
        let b: usize = it.next().unwrap().parse::<usize>().unwrap() - 1;
        g[a].push(b);
        g[b].push(a);
    }

    // Precompute inverses of 1..=n (mod MOD).
    // Precondition implies n < MOD so these are all invertible.
    let mut inv: Vec<u64> = vec![0; n + 1];
    for i in 1..=n {
        inv[i] = mod_inv(i as u64);
    }

    // Root the tree at 0, compute parent and traversal order (iterative DFS).
    let root = 0usize;
    let mut parent: Vec<usize> = vec![usize::MAX; n];
    let mut order: Vec<usize> = Vec::with_capacity(n);
    let mut stack: Vec<usize> = Vec::with_capacity(n);
    parent[root] = root;
    stack.push(root);
    while let Some(v) = stack.pop() {
        order.push(v);
        for &to in &g[v] {
            if to == parent[v] {
                continue;
            }
            parent[to] = v;
            stack.push(to);
        }
    }

    // Post-order: subtree sizes and dp_down[v] = product_{u in subtree(v)} inv[sub[u]] (under root=0).
    let mut sub: Vec<usize> = vec![1; n];
    let mut dp_down: Vec<u64> = vec![1; n];
    for &v in order.iter().rev() {
        let mut sz = 1usize;
        let mut prod = 1u64;
        for &to in &g[v] {
            if to == parent[v] {
                continue;
            }
            sz += sub[to];
            prod = (prod * dp_down[to]) % MOD;
        }
        sub[v] = sz;
        dp_down[v] = (prod * inv[sz]) % MOD;
    }

    // Rerooting:
    // Let dp_up[v] be the product of inv[subtree_size(x)] for all nodes x NOT in v's subtree,
    // when the tree is rooted at 0 (i.e., considering the "outside" part as seen from v).
    // Then for any root r, F(r) can be computed via reroot transitions.
    //
    // For a child c of v (in root=0 orientation), when rerooting to c:
    // - Nodes in c's subtree keep their subtree sizes: contribution dp_down[c].
    // - The rest of the tree becomes a subtree rooted at v of size n - sub[c].
    //   Its internal product is:
    //     dp_rest = inv[n - sub[c]] * dp_up[v] * inv[sub[v]] * product_{siblings s != c} dp_down[s]
    //   because within that rest-subtree, v's subtree size becomes (n - sub[c]) instead of sub[v],
    //   so we must include inv[n - sub[c]] and include contributions of all nodes outside c's subtree:
    //   - dp_up[v] accounts for nodes above v
    //   - siblings' dp_down account for nodes in other child subtrees
    //   - inv[sub[v]] accounts for v itself under root=0; we will replace it by inv[n - sub[c]].
    //
    // Therefore:
    //   F(c) = dp_down[c] * dp_rest
    // and we can also compute dp_up[c] for further propagation:
    //   dp_up[c] = dp_rest
    //
    // Base: dp_up[root] = 1 (nothing outside root's subtree).
    let mut dp_up: Vec<u64> = vec![1; n];
    dp_up[root] = 1;

    // We'll compute F(v) for all v.
    let mut f: Vec<u64> = vec![0; n];
    f[root] = dp_down[root];

    for &v in &order {
        // Gather children of v in the rooted tree.
        let mut children: Vec<usize> = Vec::new();
        for &to in &g[v] {
            if to == parent[v] {
                continue;
            }
            children.push(to);
        }
        let m = children.len();

        // Prefix/suffix products of dp_down over children.
        let mut pref: Vec<u64> = vec![1; m + 1];
        for i in 0..m {
            pref[i + 1] = (pref[i] * dp_down[children[i]]) % MOD;
        }
        let mut suf: Vec<u64> = vec![1; m + 1];
        for i in (0..m).rev() {
            suf[i] = (suf[i + 1] * dp_down[children[i]]) % MOD;
        }

        for (i, &c) in children.iter().enumerate() {
            let prod_siblings = (pref[i] * suf[i + 1]) % MOD;

            // dp_rest = inv[n - sub[c]] * (dp_up[v]) * (prod_siblings) * (inv[sub[v]])
            // but note: inv[sub[v]] is v's own factor under root=0; it should be included among "outside c".
            // Under root=c, v's factor becomes inv[n - sub[c]] instead, so we include inv[n - sub[c]] and
            // also include dp_up[v] and siblings, and include all nodes in v's original subtree excluding c,
            // which is exactly inv[sub[v]] * prod_siblings (since dp_down[v] = inv[sub[v]] * dp_down[c] * prod_siblings).
            let rest_size = n - sub[c];
            let dp_rest = (((inv[rest_size] * dp_up[v]) % MOD) * ((inv[sub[v]] * prod_siblings) % MOD)) % MOD;

            dp_up[c] = dp_rest;
            f[c] = (dp_down[c] * dp_rest) % MOD;
        }
    }

    // Multiply by n! modulo MOD and output per vertex in 1..=n order.
    let mut fact = 1u64;
    for i in 1..=n {
        fact = (fact * (i as u64)) % MOD;
    }

    let mut out = String::new();
    for v in 0..n {
        let val = (f[v] * fact) % MOD;
        out.push_str(&format!("{val}\n"));
    }
    print!("{out}");
}
```
