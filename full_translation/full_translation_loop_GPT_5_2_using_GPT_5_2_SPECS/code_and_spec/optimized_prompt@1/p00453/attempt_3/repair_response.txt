```rust
use std::io::{self, Read};

const INF: i64 = 2_100_000_000;

fn main() {
    // Fast byte parser
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let bytes = s.as_bytes();
    let mut idx = 0usize;

    #[inline]
    fn next_i64(bytes: &[u8], idx: &mut usize) -> Option<i64> {
        while *idx < bytes.len() && bytes[*idx].is_ascii_whitespace() {
            *idx += 1;
        }
        if *idx >= bytes.len() {
            return None;
        }
        let mut sign = 1i64;
        if bytes[*idx] == b'-' {
            sign = -1;
            *idx += 1;
        }
        let mut v = 0i64;
        while *idx < bytes.len() {
            let c = bytes[*idx];
            if !c.is_ascii_digit() {
                break;
            }
            v = v * 10 + (c - b'0') as i64;
            *idx += 1;
        }
        Some(sign * v)
    }

    let mut out = String::new();

    loop {
        let n = match next_i64(bytes, &mut idx) {
            Some(v) => v as usize,
            None => break,
        };
        let m = next_i64(bytes, &mut idx).unwrap() as usize;
        if n == 0 && m == 0 {
            break;
        }

        // Read stages
        let mut ic = vec![0usize; n];
        let mut a: Vec<Vec<i64>> = vec![Vec::new(); n];
        let mut b: Vec<Vec<i64>> = vec![Vec::new(); n];
        for i in 0..n {
            let cnt = next_i64(bytes, &mut idx).unwrap() as usize;
            ic[i] = cnt;
            a[i].resize(cnt, 0);
            b[i].resize(cnt, 0);
            for j in 0..cnt {
                a[i][j] = next_i64(bytes, &mut idx).unwrap();
                b[i][j] = next_i64(bytes, &mut idx).unwrap();
            }
        }

        // DP arrays:
        // dp1[used][k] for current stage j
        // dp2[used][l] for stage j+1
        // dp3[used][l] for stage j+2
        //
        // IMPORTANT: match C semantics: only used=0 is reachable at start.
        let mut dp1: Vec<Vec<i64>> = (0..=m).map(|_| vec![INF; ic[0]]).collect();
        for k in 0..ic[0] {
            dp1[0][k] = 0;
        }

        // We'll keep a copy of dp at stage n-2 for final "early finish" check.
        let mut dp_at_n2: Option<Vec<Vec<i64>>> = None;

        for j in 0..(n - 1) {
            let mut dp2: Vec<Vec<i64>> = (0..=m).map(|_| vec![INF; ic[j + 1]]).collect();
            let mut dp3: Option<Vec<Vec<i64>>> = if j + 2 < n {
                Some((0..=m).map(|_| vec![INF; ic[j + 2]]).collect())
            } else {
                None
            };

            for used in 0..=m {
                for k in 0..ic[j] {
                    let cur = dp1[used][k];
                    if cur >= INF {
                        continue;
                    }

                    // 1-step to j+1
                    for l in 0..ic[j + 1] {
                        let d = (b[j][k] + b[j + 1][l]) * (a[j][k] - a[j + 1][l]).abs();
                        let cand = cur + d;
                        if cand < dp2[used][l] {
                            dp2[used][l] = cand;
                        }
                    }

                    // 2-step to j+2
                    if let Some(ref mut dp3v) = dp3 {
                        if used < m {
                            for l in 0..ic[j + 2] {
                                let d = (b[j][k] + b[j + 2][l])
                                    * (a[j][k] - a[j + 2][l]).abs();
                                let cand = cur + d;
                                if cand < dp3v[used + 1][l] {
                                    dp3v[used + 1][l] = cand;
                                }
                            }
                        }
                    }
                }
            }

            // Before moving on, if this dp2 is stage n-2, store it.
            if j + 1 == n - 2 {
                dp_at_n2 = Some(dp2.clone());
            }

            // Next stage: dp1 becomes dp2, but we must also incorporate paths that reached j+1 via 2-step
            // from j-1 (stored in dp3 of previous iteration). Our construction already accounts for that
            // because dp1 is always the full dp for stage j, built in the previous iteration.
            //
            // To achieve that, we need to merge dp3 (which is stage j+2) into the dp that will be used
            // when j advances by 1. The simplest is:
            // - set dp1 = dp2 for next j+1
            // - additionally, carry dp3 forward by adding it into the dp2 of the next iteration when it
            //   becomes the "current stage". Instead of complex carrying, we can directly merge dp3 into
            //   a side buffer for stage j+2. But that becomes a full multi-stage DP.
            //
            // Therefore, we follow the original C DP more directly with a full DP over stages but with
            // compact storage: dp[used][stage][opt] is needed. Given constraints (<=80 stages, <=150 opts,
            // <=10 m), this is small enough and avoids timeouts from incorrect rolling.
            //
            // Mismatch note: the previous rolling approach was incorrect; we switch to faithful DP.

            // Break out to faithful DP implementation.
            // (We keep this loop only to satisfy borrow checker structure; it will never run.)
            unreachable!();
        }

        // The code above is intentionally unreachable; we implement the faithful DP below.
        // (Rust doesn't allow easy mid-function "goto"; we just re-run with correct DP.)

        // Faithful DP: dp[used][stage][opt]
        let mut dp: Vec<Vec<Vec<i64>>> = (0..=m)
            .map(|_| (0..n).map(|s| vec![INF; ic[s]]).collect())
            .collect();
        for k in 0..ic[0] {
            dp[0][0][k] = 0;
        }

        for used in 0..=m {
            for j in 0..(n - 1) {
                for k in 0..ic[j] {
                    let cur = dp[used][j][k];
                    if cur >= INF {
                        continue;
                    }
                    // 1-step
                    for l in 0..ic[j + 1] {
                        let d = (b[j][k] + b[j + 1][l]) * (a[j][k] - a[j + 1][l]).abs();
                        let cand = cur + d;
                        if cand < dp[used][j + 1][l] {
                            dp[used][j + 1][l] = cand;
                        }
                    }
                    // 2-step
                    if used < m && j + 2 < n {
                        for l in 0..ic[j + 2] {
                            let d = (b[j][k] + b[j + 2][l]) * (a[j][k] - a[j + 2][l]).abs();
                            let cand = cur + d;
                            if cand < dp[used + 1][j + 2][l] {
                                dp[used + 1][j + 2][l] = cand;
                            }
                        }
                    }
                }
            }
        }

        let mut best = INF + 210;
        for used in 0..=m {
            for &v in &dp[used][n - 1] {
                if v < best {
                    best = v;
                }
            }
            if used < m {
                for &v in &dp[used][n - 2] {
                    if v < best {
                        best = v;
                    }
                }
            }
        }

        out.push_str(&best.to_string());
        out.push('\n');
    }

    print!("{out}");
}
```
