1. Functional Description  
Computes the maximum value, over all contiguous groups of items after sorting, of: (sum of the second values in the group) minus (difference between the last and first first-values in the group). Outputs this maximum as an integer.

2. Inputs  
- Reads an integer count of items.  
- Reads that many pairs of integers:  
  - a first value (treated as a 64-bit signed integer),

3. Outputs  
- Writes a single signed 64-bit integer to standard output: the maximum achievable value defined in the Functional Description.

4. Preconditions  
- The item count is within the supported capacity (at most 500,001 items).

- The computed result fits within signed 64-bit integer range.

5. Postconditions  
Let the items be sorted in nondecreasing order by the first value, and for equal first values, by the second value in nondecreasing order. For any indices \(1 \le l \le r \le N\), define:  
- \(A_i\) = first value of sorted item \(i\)  
- \(B_i\) = second value of sorted item \(i\)  
- \(S(l,r) = \sum_{i=l}^{r} B_i\)  
Then the program outputs:  
\[
\max_{1 \le l \le r \le N} \left( S(l,r) - (A_r - A_l) \right)
\]

#include <stdio.h>

//#define getchar_unlocked()  getchar()
int in()	/* 整数値の入力 */
{
	int n = 0, c = getchar_unlocked();
	do n = 10*n + (c & 0xf), c = getchar_unlocked(); while (c >= '0');
	return n;
}

long long big()		/* long long 整数値の入力 */
{
	long long n = 0, c = getchar_unlocked();
	do n = 10*n + (c & 0xf), c = getchar_unlocked(); while (c >= '0');
	return n;
}

#define INF 0x1010101010101010LL
typedef struct { long long a; int b; } T;
T tbl[500002];
long long s[500002];

int cmp(T *a, T *b) {
	if (a->a == b->a) return a->b - b->b;
	if (a->a < b->a) return -1;
	return 1;
}

int main()
{
	int N, i;
	long long t, min, max;

	N = in();
	for (i = 1; i <= N; i++) tbl[i].a = big(), tbl[i].b = in();
	qsort(tbl+1, N, sizeof(T), cmp);

	min = INF, max = -INF;
	for (i = 1; i <= N; i++) {
		t = s[i-1] - tbl[i].a;
		if (t < min) min = t;
		s[i] = s[i-1] + tbl[i].b;
		t = s[i] - tbl[i].a - min;
		if (t > max) max = t;
	}
	printf("%lld\n", max);
	return 0;
}
