int ins(char *s, char *r)
{
	char *ss = s;
	int n;

	do *s = gc();
	while (*s++ > ' ');
	*--s = 0;
	n = s - ss;
	while (--s >= ss) *r++ = *s;
	*r = 0;
	return n;
}

// Binary Search
int search(char *s, int n, int idx, int part)
{
	int k, mid, low, high;

	low = 0, high = N;
	if (part) {
		while (low < high) {
			mid = (low + high) >> 1;
			if (strncmp(idx ? r[mid] : w[mid], s, n) <= 0) low = mid + 1; else high = mid;
		}
		k = low - 1;
		low = 0, high = N;

			mid = (low + high) >> 1;
			if (strncmp(idx ? r[mid] : w[mid], s, n) < 0) low = mid + 1; else high = mid;

	}
	return 0;
}
