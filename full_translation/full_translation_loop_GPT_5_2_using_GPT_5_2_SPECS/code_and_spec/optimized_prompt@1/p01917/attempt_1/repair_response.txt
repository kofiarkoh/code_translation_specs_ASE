use std::io::{self, Read};

fn compute_scores(values: &[i32], mapping: &[i32]) -> Vec<i32> {
    let n = values.len();
    let mut out = vec![0i32; n];
    for i in 0..n {
        let mut cnt = 0usize;
        for j in 0..n {
            if values[j] > values[i] {
                cnt += 1;
            }
        }
        // Spec says mapping length is sufficient; otherwise behavior is undefined (same as C).
        out[i] = mapping[cnt];
    }
    out
}

fn main() {
    // Read all input as whitespace-separated integers.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    // Score mapping x[0..n)
    let mut x = vec![0i32; n];
    for i in 0..n {
        x[i] = it.next().unwrap().parse().unwrap();
    }

    // Category values a[3][n]
    let mut a = vec![vec![0i32; n]; 3];
    for i in 0..n {
        a[0][i] = it.next().unwrap().parse().unwrap();
        a[1][i] = it.next().unwrap().parse().unwrap();
        a[2][i] = it.next().unwrap().parse().unwrap();
    }

    let mut ans: i32 = 1000;

    for k in 0..3 {
        // s = sum of scores from the two categories other than k
        let mut s = vec![0i32; n];
        for kk in 0..3 {
            if kk == k {
                continue;
            }
            let t = compute_scores(&a[kk], &x);
            for i in 0..n {
                s[i] += t[i];
            }
        }

        let ori = a[k][0];

        // Consider setting a[k][0] to (a[k][i] + d) for i>=1 and d in {0,1},
        // but only if it strictly increases over original.
        for i in 1..n {
            for d in 0..2 {
                let cand = a[k][i] + d;
                if cand > ori {
                    a[k][0] = cand;
                    let t = compute_scores(&a[k], &x);

                    let my_total = s[0] + t[0];
                    let mut cnt_ahead = 0;
                    for j in 0..n {
                        if s[j] + t[j] > my_total {
                            cnt_ahead += 1;
                        }
                    }
                    if cnt_ahead <= 7 {
                        let inc = cand - ori;
                        if inc < ans {
                            ans = inc;
                        }
                    }
                }
            }
        }

        // Restore original value (as per spec/state transitions).
        a[k][0] = ori;
    }

    if ans == 1000 {
        print!("Saiko\n");
    } else {
        print!("{}\n", ans);
    }
}
