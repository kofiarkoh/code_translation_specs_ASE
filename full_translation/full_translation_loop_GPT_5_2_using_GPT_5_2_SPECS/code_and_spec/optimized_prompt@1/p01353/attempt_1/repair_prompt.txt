	do n = (n<<3)+(n<<1) + (c & 0xf), c = getchar_unlocked();

int cmp(T *a, T *b) { return a->a * b->d > b->a * a->d; }

int main()
{
	int n, i, k, sz;
	long long turn, ans;

	n = in();
	for (i = 0; i <= n; i++) h[i] = in(), a[i] = in(), d[i] = in(), s[i] = in();
	
	sz = 0, ans = turn = 0;
