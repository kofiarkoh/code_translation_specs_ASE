```rust
use std::io::{self, Read};
use std::process;

fn main() {
    // Read entire stdin
    let mut input = String::new();
    if io::stdin().read_to_string(&mut input).is_err() {
        process::exit(1);
    }
    let mut it = input.split_whitespace();

    let h: usize = match it.next().and_then(|s| s.parse().ok()) {
        Some(v) => v,
        None => process::exit(1),
    };
    let w: usize = match it.next().and_then(|s| s.parse().ok()) {
        Some(v) => v,
        None => process::exit(1),
    };

    // Read grid rows; each row must provide at least w characters (per spec).
    let mut grid: Vec<Vec<u8>> = Vec::with_capacity(h);
    for _ in 0..h {
        let row = match it.next() {
            Some(s) => s.as_bytes(),
            None => process::exit(1),
        };
        // Preconditions guarantee row.len() >= w; behavior undefined otherwise.
        grid.push(row[..w].to_vec());
    }

    // group ids: 0 means unassigned; ids start at 1
    let mut group = vec![vec![0usize; w]; h];
    let mut num_groups: usize = 0;

    // Iterative DFS to avoid recursion depth issues.
    let dirs: [(isize, isize); 4] = [(0, 1), (0, -1), (1, 0), (-1, 0)];

    for y in 0..h {
        for x in 0..w {
            if group[y][x] != 0 {
                continue;
            }
            num_groups += 1;
            let gid = num_groups;

            let mut stack: Vec<(usize, usize)> = Vec::new();
            stack.push((y, x));
            while let Some((cy, cx)) = stack.pop() {
                if group[cy][cx] == gid {
                    continue;
                }
                group[cy][cx] = gid;

                let cur = grid[cy][cx];
                for (dy, dx) in dirs {
                    let ny = cy as isize + dy;
                    let nx = cx as isize + dx;
                    if ny < 0 || nx < 0 || ny >= h as isize || nx >= w as isize {
                        continue;
                    }
                    let nyu = ny as usize;
                    let nxu = nx as usize;
                    if cur != grid[nyu][nxu] && group[nyu][nxu] != gid {
                        stack.push((nyu, nxu));
                    }
                }
            }
        }
    }

    // Count dot vs non-dot per group.
    // Index 0 unused to match C behavior.
    let mut count = vec![[0u64; 2]; num_groups + 1];
    for y in 0..h {
        for x in 0..w {
            let gid = group[y][x];
            let is_dot = (grid[y][x] == b'.') as usize; // 1 if dot, 0 otherwise
            count[gid][is_dot] += 1;
        }
    }

    // Sum products using 64-bit unsigned arithmetic.
    let mut ans: u64 = 0;
    for gid in 1..=num_groups {
        ans = ans.wrapping_add(count[gid][0].wrapping_mul(count[gid][1]));
    }

    print!("{ans}\n");
}
```
