Reads multiple datasets describing a set of straight line segments on an integer grid and classifies the segments into digit categories (0–9) based solely on how segments connect at shared endpoints and on whether any segment endpoint lies strictly on another segment. - The sum of the 10 reported counts equals the number of input segments.
- Classification is determined by partitioning segments into connected components where segments are connected if they share at least one endpoint coordinate, then applying the following rules per component:
  - Components with fewer than 2 segments do not directly form a digit; their segments are counted in digit 1.
  - Components with exactly 2 segments and exactly 2 “shared-endpoint incidences” (i.e., both segments meet at exactly one common endpoint) contribute both segments to digit 4.
  - Components with exactly 4 segments and exactly 6 shared-endpoint incidences contribute all 4 segments to digit 9.

#include <stdio.h>

#define MAX 1002

int id[MAX], size[MAX];
void init(int n) { int i; for (i = 0; i < n; i++) id[i] = i, size[i] = 1; }
int root(int i) { while (i != id[i]) id[i] = id[id[i]], i = id[i]; return i; }
int connected(int p, int q) { return root(p) == root(q); }
void unite(int p, int q)
{
    int i = root(p), j = root(q); if (i == j) return;
    if (size[i] < size[j]) id[i] = j, size[j] += size[i]; else id[j] = i, size[i] += size[j];
}

#define PeqP(a,b)  (((a.x)==(b.x))&&((a.y)==(b.y)))

int pOnSeg(PP p, SEG u)
{
	int x1 = u.s.x, y1 = u.s.y, x2 = u.e.x, y2 = u.e.y;
	int d;

	if (PeqP(p, u.s) || PeqP(p, u.e)) return 0;
	if (x1 > x2) { d = x1, x1 = x2, x2 = d; d = y1, y1 = y2, y2 = d; }
	return x1 <= p.x && p.x <= x2 &&
		((y1 <= y2 && y1 <= p.y && p.y <= y2) ||
		(y1 > y2 && y2 <= p.y && p.y <= y1))
		&& (p.y - y1)*(x2 - x1) == (y2 - y1)*(p.x - x1);
}
