use std::io::{self, Read};

fn main() {
    // Read entire stdin, like repeated scanf calls.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // C code uses implicit-int globals: n, x; and main(y) where y is an int parameter.
    // We'll model them as i32 to match typical C 'int' behavior.
    let n0: i32 = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };

    let mut n = n0;
    let mut x: i32 = 0;

    // for (scanf("%d",&n); n--; printf("%d\n",(x-1)%y+1)) scanf("%d%d",&x,&y);
    while n != 0 {
        n -= 1;

        // scanf("%d%d",&x,&y)
        let xs = it.next().unwrap();
        let ys = it.next().unwrap();
        x = xs.parse::<i32>().unwrap();
        let y = ys.parse::<i32>().unwrap();

        // Emulate C's signed arithmetic and remainder behavior (trunc toward zero).
        // Use wrapping for (x-1) to mimic potential signed overflow in C as best as possible.
        let xm1 = x.wrapping_sub(1);
        let r = xm1 % y;
        let out = r.wrapping_add(1);

        print!("{}\n", out);
    }

    // C calls exit(0); normal return is equivalent.
}
