```rust
use std::io::{self, Read};

const ONE_SIXTH: f64 = 1.0 / 6.0;

fn myabs(x: i32) -> i32 {
    if x >= 0 { x } else { -x }
}

fn solve(
    rest: usize,
    turn: i32,
    n: usize,
    sq: &[u8; 100],
    memo: &mut [[f64; 101]; 101],
) -> f64 {
    // In C, memo is indexed only when turn is within [0..100] due to base cases.
    // In Rust, avoid casting negative turn to usize (would underflow to huge index).
    if turn >= 0 {
        let tu = turn as usize;
        if tu <= 100 && memo[rest][tu] != f64::MAX {
            return memo[rest][tu];
        }
    }

    let ret = if rest == 0 {
        1.0
    } else if turn <= 0 {
        0.0
    } else if turn == 1 && rest <= 6 {
        ONE_SIXTH
    } else {
        let mut acc = 0.0;
        for i in 1..=6 {
            let next_pos = myabs(rest as i32 - i) as usize;

            // C does: sq[N-next_pos] with int arithmetic.
            // If next_pos > N, then N-next_pos is negative -> out-of-bounds UB in C.
            // To emulate typical behavior without crashing, treat out-of-range as empty (0).
            let idx_i32 = n as i32 - next_pos as i32;
            let cell = if (0..(sq.len() as i32)).contains(&idx_i32) {
                sq[idx_i32 as usize]
            } else {
                0u8
            };

            match cell {
                0 => acc += ONE_SIXTH * solve(next_pos, turn - 1, n, sq, memo),
                b'L' => acc += ONE_SIXTH * solve(next_pos, turn - 2, n, sq, memo),
                b'B' => acc += ONE_SIXTH * solve(n, turn - 1, n, sq, memo),
                _ => {}
            }
        }
        acc
    };

    if turn >= 0 {
        let tu = turn as usize;
        if tu <= 100 {
            memo[rest][tu] = ret;
        }
    }
    ret
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: usize = n_opt.unwrap().parse().unwrap();
        let t: i32 = it.next().unwrap().parse().unwrap();
        let lcnt: usize = it.next().unwrap().parse().unwrap();
        let bcnt: usize = it.next().unwrap().parse().unwrap();

        if n == 0 && t == 0 && lcnt == 0 && bcnt == 0 {
            break;
        }

        let mut sq = [0u8; 100];
        let mut memo = [[f64::MAX; 101]; 101];

        for _ in 0..lcnt {
            let pos: usize = it.next().unwrap().parse().unwrap();
            if pos < sq.len() {
                sq[pos] = b'L';
            }
        }
        for _ in 0..bcnt {
            let pos: usize = it.next().unwrap().parse().unwrap();
            if pos < sq.len() {
                sq[pos] = b'B';
            }
        }

        let ret = solve(n, t, n, &sq, &mut memo);
        // printf("%lf\n", ret) prints 6 digits after decimal by default.
        out.push_str(&format!("{:.6}\n", ret));
    }

    print!("{out}");
}
```
