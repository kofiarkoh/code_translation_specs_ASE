```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, parse two integers like scanf("%d%d", &N, &K)
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n_opt = it.next();
    let k_opt = it.next();
    if n_opt.is_none() || k_opt.is_none() {
        // If scanf fails, C would leave variables uninitialized (UB).
        // We choose to do nothing rather than invent behavior.
        return;
    }

    let n: i32 = n_opt.unwrap().parse().unwrap();
    let mut k: i32 = k_opt.unwrap().parse().unwrap();

    let mut multi: i32 = 1;
    let mut i: i32 = n;

    // C uses a VLA: int denomi[N];
    // In Rust, use Vec sized by N. If N <= 0, C VLA is invalid/UB; we avoid panicking.
    if n <= 0 {
        // The C program would have undefined behavior; we exit quietly.
        return;
    }
    let mut denomi: Vec<i32> = vec![0; n as usize];

    let mut nume: f64 = 0.0; // long double -> use f64 (closest portable choice in Rust std)

    while i > 0 {
        let idx = (i - 1) as usize;
        denomi[idx] = 0;

        // while (K > multi*i) { multi *= 2; }
        // In C, signed overflow is UB; we emulate typical 2's-complement wraparound
        // using wrapping operations to preserve "int" behavior as closely as possible.
        while k > multi.wrapping_mul(i) {
            multi = multi.wrapping_mul(2);
        }

        denomi[idx] = denomi[idx].wrapping_add(multi.wrapping_mul(n));
        i -= 1;
    }

    for j in 0..(n as usize) {
        // C does integer division: denomi[0]/denomi[j]
        // Division by zero would be UB in C; Rust would panic. We avoid inventing behavior.
        if denomi[j] == 0 {
            return;
        }
        let term_int: i32 = denomi[0] / denomi[j];
        nume += term_int as f64;
    }

    if denomi[0] == 0 {
        return;
    }
    let result = nume / (denomi[0] as f64);

    // printf("%.12Lf", ...)
    print!("{:.12}", result);
}
```
