```rust
use std::io::{self, Read};

const QMAX: usize = 3005;
#[derive(Clone, Copy, Default)]
struct Que {
    t: i32,
    s: i32,
}

const INF: i32 = 0x10101010;
const MAX: usize = 110;

#[derive(Clone, Copy, Default)]
struct Edge {
    to: i32,
    rev: i32,
    cap: i32,
    cost: i32,
}

struct Scanner {
    buf: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }
    // Emulates the C in(): assumes there is a next integer token; no EOF check.
    fn next_i32(&mut self) -> i32 {
        // Skip non-digit and non-'-' (C code doesn't, but input is well-formed for AOJ).
        while self.idx < self.buf.len()
            && self.buf[self.idx] != b'-'
            && (self.buf[self.idx] < b'0' || self.buf[self.idx] > b'9')
        {
            self.idx += 1;
        }
        let mut neg = false;
        if self.idx < self.buf.len() && self.buf[self.idx] == b'-' {
            neg = true;
            self.idx += 1;
        }
        let mut n: i32 = 0;
        while self.idx < self.buf.len() {
            let c = self.buf[self.idx];
            if c < b'0' || c > b'9' {
                break;
            }
            // (c & 0xf) in C
            n = n.wrapping_mul(10).wrapping_add((c & 0x0f) as i32);
            self.idx += 1;
        }
        if neg { -n } else { n }
    }
}

fn abs_i32(a: i32) -> i32 {
    if a >= 0 { a } else { -a }
}

struct MinCostFlow {
    edge: [[Edge; MAX]; MAX],
    hi: [usize; MAX],
    v: usize,
    potential: [i32; MAX],
    dist: [i32; MAX],
    prevv: [i32; MAX],
    preve: [i32; MAX],

    // custom min-heap priority queue (matches C behavior)
    que: [Que; QMAX],
    qsize: usize,
}

impl MinCostFlow {
    fn new() -> Self {
        Self {
            edge: [[Edge::default(); MAX]; MAX],
            hi: [0; MAX],
            v: 0,
            potential: [0; MAX],
            dist: [0; MAX],
            prevv: [0; MAX],
            preve: [0; MAX],
            que: [Que::default(); QMAX],
            qsize: 0,
        }
    }

    #[inline]
    fn parent(i: usize) -> usize { i >> 1 }
    #[inline]
    fn left(i: usize) -> usize { i << 1 }
    #[inline]
    fn right(i: usize) -> usize { (i << 1) + 1 }

    fn min_heapify(&mut self, i: usize) {
        let l = Self::left(i);
        let r = Self::right(i);
        let mut min = if l < self.qsize && self.que[l].t < self.que[i].t { l } else { i };
        if r < self.qsize && self.que[r].t < self.que[min].t {
            min = r;
        }
        if min != i {
            let qt = self.que[i];
            self.que[i] = self.que[min];
            self.que[min] = qt;
            self.min_heapify(min);
        }
    }

    fn deq(&mut self) {
        self.qsize -= 1;
        self.que[0] = self.que[self.qsize];
        self.min_heapify(0);
    }

    fn enq(&mut self, s: i32, t: i32) {
        let mut i = self.qsize;
        self.qsize += 1;
        self.que[i].s = s;
        self.que[i].t = t;
        while i > 0 {
            let p = Self::parent(i);
            if self.que[p].t <= self.que[i].t {
                break;
            }
            let qt = self.que[i];
            self.que[i] = self.que[p];
            self.que[p] = qt;
            i = p;
        }
    }

    fn add_edge(&mut self, from: usize, to: usize, cap: i32, cost: i32) {
        let f = self.hi[from];
        self.hi[from] += 1;
        let t = self.hi[to];
        self.hi[to] += 1;

        self.edge[from][f] = Edge { to: to as i32, cap, cost, rev: t as i32 };
        self.edge[to][t] = Edge { to: from as i32, cap: 0, cost: -cost, rev: f as i32 };
    }

    fn min_cost_flow(&mut self, s: usize, t: usize, mut f: i32) -> i32 {
        let mut ret: i32 = 0;
        for i in 0..self.v {
            self.potential[i] = 0;
        }

        while f > 0 {
            for i in 0..self.v {
                self.dist[i] = INF;
            }
            self.qsize = 0;
            self.enq(s as i32, 0);
            self.dist[s] = 0;

            while self.qsize != 0 {
                let v = self.que[0].s as usize;
                let cur_t = self.que[0].t;
                self.deq();
                if self.dist[v] < cur_t {
                    continue;
                }
                let hv = self.hi[v];
                for i in 0..hv {
                    let ep = self.edge[v][i];
                    if ep.cap <= 0 {
                        continue;
                    }
                    let nv = ep.to as usize;
                    // nt = t + cost - potential[nv] + potential[v]
                    let nt = cur_t
                        .wrapping_add(ep.cost)
                        .wrapping_sub(self.potential[nv])
                        .wrapping_add(self.potential[v]);
                    if self.dist[nv] > nt {
                        self.dist[nv] = nt;
                        self.prevv[nv] = v as i32;
                        self.preve[nv] = i as i32;
                        self.enq(nv as i32, nt);
                    }
                }
            }

            if self.dist[t] == INF {
                return -1;
            }
            for v in 0..self.v {
                self.potential[v] = self.potential[v].wrapping_add(self.dist[v]);
            }

            let mut addf = f;
            let mut v = t;
            while v != s {
                let pv = self.prevv[v] as usize;
                let pe = self.preve[v] as usize;
                let ep = self.edge[pv][pe];
                if addf > ep.cap {
                    addf = ep.cap;
                }
                v = pv;
            }
            f -= addf;

            ret = ret.wrapping_add(addf.wrapping_mul(self.potential[t]));
            let mut v2 = t;
            while v2 != s {
                let pv = self.prevv[v2] as usize;
                let pe = self.preve[v2] as usize;

                // Update forward edge
                let mut ep = self.edge[pv][pe];
                ep.cap -= addf;
                self.edge[pv][pe] = ep;

                // Update reverse edge
                let rev = ep.rev as usize;
                let mut back = self.edge[v2][rev];
                back.cap += addf;
                self.edge[v2][rev] = back;

                v2 = pv;
            }
        }
        ret
    }
}

fn main() {
    let mut sc = Scanner::new();
    let mut out = String::new();

    let mut mcf = MinCostFlow::new();

    // Problem data arrays (match C sizes)
    let mut ax = [0i32; 101];
    let mut ay = [0i32; 101];
    let mut bx = [0i32; 6];
    let mut by = [0i32; 6];
    let mut c = [0i32; 6];
    let mut f = [0i32; 6];

    loop {
        // The original C uses: while (N = in()) { ... }
        // Here we must stop on N == 0. Input is assumed well-formed.
        let n = sc.next_i32();
        if n == 0 {
            break;
        }
        let n_usize = n as usize;
        let m = sc.next_i32() as usize;
        let b = sc.next_i32();

        for i in 0..n_usize {
            ax[i] = sc.next_i32();
            ay[i] = sc.next_i32();
        }
        for i in 0..m {
            bx[i] = sc.next_i32();
            by[i] = sc.next_i32();
            c[i] = sc.next_i32();
            f[i] = sc.next_i32();
        }

        let source = n_usize + m;
        let sink = source + 1;
        mcf.v = sink + 1;

        let mut ans = INF;
        let lim = 1usize << m;

        for s in 0..lim {
            let mut res: i32 = 0;
            let mut cnt: i32 = 0;
            for i in 0..m {
                if ((s >> i) & 1) != 0 {
                    res = res.wrapping_add(f[i]);
                    cnt += 1;
                }
            }

            let mut lb: i32 = 0;
            let mut ub: i32 = 4002;

            // helper closure to build graph like C make_edge(s, mi)
            let mut make_edge = |mcf: &mut MinCostFlow, mi: i32| {
                for i in 0..mcf.v {
                    mcf.hi[i] = 0;
                }
                for i in 0..n_usize {
                    mcf.add_edge(source, i, 1, 0);
                }
                for i in 0..m {
                    if ((1usize << i) & s) != 0 {
                        for j in 0..n_usize {
                            let mut x = abs_i32(ax[j] - bx[i]) + abs_i32(ay[j] - by[i]) - mi;
                            if x < 0 {
                                x = 0;
                            }
                            mcf.add_edge(j, n_usize + i, 1, x);
                        }
                        mcf.add_edge(n_usize + i, sink, c[i], 0);
                    }
                }
            };

            // binary search loop with "goto next" behavior
            let mut skip = false;
            while lb + 1 < ub {
                let mi = (lb + ub) >> 1;

                make_edge(&mut mcf, mi);
                let a1 = mcf
                    .min_cost_flow(source, sink, n as i32)
                    .wrapping_add(mi.wrapping_mul(b).wrapping_mul(cnt));

                make_edge(&mut mcf, mi - 1);
                let a2 = mcf
                    .min_cost_flow(source, sink, n as i32)
                    .wrapping_add((mi - 1).wrapping_mul(b).wrapping_mul(cnt));

                if a1 < 0 || a2 < 0 {
                    skip = true;
                    break;
                }
                if a1 < a2 {
                    lb = mi;
                } else {
                    ub = mi;
                }
            }
            if skip {
                continue;
            }

            make_edge(&mut mcf, lb);
            res = res
                .wrapping_add(mcf.min_cost_flow(source, sink, n as i32))
                .wrapping_add(lb.wrapping_mul(b).wrapping_mul(cnt));
            if ans > res {
                ans = res;
            }
        }

        out.push_str(&format!("{ans}\n"));
    }

    print!("{out}");
}
```
