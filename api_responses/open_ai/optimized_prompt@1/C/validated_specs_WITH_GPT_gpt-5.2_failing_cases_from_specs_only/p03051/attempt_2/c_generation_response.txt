#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

#define M (1u << 20)
#define P 1000000007LL

static long long mod_pow2(long long e) {
    /* Computes 2^e mod P, with e assumed >= 0 */
    long long base = 2LL % P;
    long long res = 1LL;
    while (e > 0) {
        if (e & 1LL) res = (res * base) % P;
        base = (base * base) % P;
        e >>= 1LL;
    }
    return res;
}

int main(void) {
    int N;
    if (scanf("%d", &N) != 1) return 0;

    /* Arrays sized per specification; allocation failure => undefined behavior per spec */
    int *s = (int *)malloc((size_t)(N + 1) * sizeof(int));
    int *Onum = (int *)malloc((size_t)(N + 1) * sizeof(int));

    int *prev = (int *)malloc((size_t)M * sizeof(int));
    long long *dp = (long long *)malloc((size_t)M * sizeof(long long));
    long long *sum = (long long *)malloc((size_t)M * sizeof(long long));
    long long *Osum = (long long *)malloc((size_t)M * sizeof(long long));

    s[0] = 0;
    Onum[0] = 0;

    for (unsigned i = 0; i < M; i++) {
        prev[i] = 0;
        dp[i] = 0;
        sum[i] = 0;
        Osum[i] = 1; /* per spec */
    }

    for (int i = 1; i <= N; i++) {
        int a;
        if (scanf("%d", &a) != 1) return 0;

        s[i] = s[i - 1] ^ a;

        /* Prefix count of zero prefix-xors */
        Onum[i] = Onum[i - 1] + (s[i] == 0);

        /* Core recurrence; assumes s[i] in [0, M-1], else UB per spec */
        int x = s[i];
        int px = prev[(unsigned)x];

        int Delta = Onum[i] - Onum[px];

        long long old_sum = sum[(unsigned)x];
        long long old_Osum = Osum[(unsigned)x];

        long long term = (old_sum * (long long)Delta) % P;

        dp[(unsigned)x] = (term + old_Osum) % P;
        Osum[(unsigned)x] = (old_Osum + term) % P;
        sum[(unsigned)x] = (old_sum + dp[(unsigned)x]) % P;
        prev[(unsigned)x] = i;
    }

    int t = s[N];
    long long ans;
    if (t != 0) {
        ans = dp[(unsigned)t] % P;
    } else {
        /* Onum[N] >= 1 when s_N == 0, so exponent is nonnegative */
        long long pow_term = mod_pow2((long long)Onum[N] - 1LL);
        long long acc = 0;
        for (unsigned x = 1; x < M; x++) {
            acc += sum[x];
            if (acc >= (1LL << 62)) acc %= P; /* avoid overflow in long accumulation */
        }
        acc %= P;
        ans = (pow_term + acc) % P;
    }

    printf("%lld\n", ans);

    free(s);
    free(Onum);
    free(prev);
    free(dp);
    free(sum);
    free(Osum);

    return 0;
}
