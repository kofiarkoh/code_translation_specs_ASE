#include <stdio.h>

int main(void) {
    enum { MAX_BYTES = 114514 };
    unsigned char buf[MAX_BYTES];

    size_t n = fread(buf, 1, MAX_BYTES, stdin);

    /* Contract says behavior is undefined if fewer than 2 bytes are read.
       We intentionally do not add explicit error handling. */
    if (buf[0] == buf[n - 1]) {
        puts("First");
    } else {
        puts("Second");
    }

    return 0;
}
