#include <stdio.h>
#include <string.h>

int a[102];
int dp[102][102];

int main()
{
	int t, n, i, j, k, v, ans;
	
	scanf("%d", &t); while (t--) {
		scanf("%d", &n);
		for (i = 0; i < n; i++) scanf("%d", a+i);
