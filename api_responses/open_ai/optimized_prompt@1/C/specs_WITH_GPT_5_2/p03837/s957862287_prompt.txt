#include <stdbool.h>

int min(int a, int b){

        return a;
    }else{
        return b;
    }
}

int main(void){

    int N, M;

    scanf("%d%d", &N, &M);

    int a[M], b[M], c[M];
    int dist[N][N];

    for(int i = 0; i < M; i++){

        b[i]--;

                dist[i][j] = 1000000000;
            }
        }
    }

        dist[a[i]][b[i]] = min(dist[a[i]][b[i]], c[i]);

        }
