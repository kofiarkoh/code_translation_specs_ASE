// AOJ 2488 Tree Construction

#include <stdio.h>
#include <string.h>

#define ABS(a)  ((a)>=0?(a):-(a))
int x[1001], y[1001];
int dp[1001][1001];

//#define getchar_unlocked()  getchar()
int in()
{

}
