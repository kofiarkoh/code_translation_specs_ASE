#include <stdio.h>

#define MAX_N 20

// Function to count the number of set bits in an integer
int count_set_bits(int num) {
    int count = 0;
    while (num) {
        count += num & 1; // Increment count if the least significant bit is set
        num >>= 1;        // Right shift to check the next bit
    }
    return count;
}

// Function to compute the maximum number of set bits based on constraints
int max_set_bits(int n, int a[], int x[][MAX_N], int y[][MAX_N]) {
    int max_bits = 0;

    // Iterate through all possible combinations of integers from 1 to n
    for (int i = 1; i < (1 << n); i++) { // 1 << n generates 2^n combinations
        int valid = 1; // Assume the combination is valid
        for (int j = 0; j < n; j++) {
            if (a[j] > 0) { // Check only groups with constraints
                for (int k = 0; k < a[j]; k++) {
                    // Check if the current combination is valid based on constraints
                    if (((i & (1 << (x[j][k] - 1))) == 0) || 
                        ((i & (1 << (y[j][k] - 1))) == 0)) {
                        valid = 0; // Mark as invalid if any constraint fails
                        break;
                    }
                }
            }
            if (!valid) break; // Exit if already invalid
        }
        if (valid) {
            // Count set bits in the valid combination
            int set_bits = count_set_bits(i);
            if (set_bits > max_bits) {
                max_bits = set_bits; // Update max if current set bits are greater
            }
        }
    }
    return max_bits; // Return the maximum count of set bits found
}
