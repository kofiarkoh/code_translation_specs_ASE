```c
#include <stdio.h>
#include <limits.h>

int main() {
    int n; // Number of integers
    scanf("%d", &n); // Read the number of integers

    int sum = 0; // To hold the sum of integers
    int min_non_multiple_of_10 = INT_MAX; // To track the smallest non-multiple of 10
    int has_non_multiple_of_10 = 0; // Flag to check if there's at least one non-multiple of 10

    for (int i = 0; i < n; i++) {
        int value;
        scanf("%d", &value); // Read each integer

        sum += value; // Update the sum

        // Check if the value is a non-multiple of 10
        if (value % 10 != 0) {
            has_non_multiple_of_10 = 1; // Set the flag
            if (value < min_non_multiple_of_10) {
                min_non_multiple_of_10 = value; // Update the minimum non-multiple of 10
            }
        }
    }

    // Determine the output based on the computed sum and conditions
    if (sum % 10 == 0) { // If the sum is divisible by 10
        if (!has_non_multiple_of_10) { // All integers are multiples of 10
            printf("0\n"); // Output is 0
        } else { // At least one integer is not a multiple of 10
            printf("%d\n", sum - min_non_multiple_of_10); // Output adjusted sum
        }
    } else { // If the sum is not divisible by 10
        printf("%d\n", sum); // Output the sum as is
    }

    return 0; // Successful execution
}
```
