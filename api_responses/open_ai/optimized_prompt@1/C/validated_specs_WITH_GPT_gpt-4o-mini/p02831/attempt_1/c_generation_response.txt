```c
#include <stdio.h>

int main() {
    int first_input, second_input;
    
    // Read inputs
    scanf("%d %d", &first_input, &second_input);
    
    // Preconditions check
    // Undefined behavior if first_input is zero or second_input is not positive
    if (first_input == 0 || second_input <= 0) {
        return 1; // Exit without output as per specification
    }

    // Initialize the smallest multiple
    int multiple = first_input; // Start with the first multiple

    // Iterate over multiples of first_input
    while (multiple % second_input != 0) {
        multiple += first_input; // Move to the next multiple
    }

    // Output the result
    printf("%d\n", multiple); // Output the smallest valid multiple
    return 0;
}
```
