#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX_NODES 100

// Function to compute the shortest paths using Floyd-Warshall algorithm
void floydWarshall(int dist[MAX_NODES][MAX_NODES], int N) {
    for (int k = 1; k <= N; k++) {
        for (int i = 1; i <= N; i++) {
            for (int j = 1; j <= N; j++) {
                if (dist[i][k] < INT_MAX && dist[k][j] < INT_MAX) {
                    if (dist[i][j] > dist[i][k] + dist[k][j]) {
                        dist[i][j] = dist[i][k] + dist[k][j];
                    }
                }
            }
        }
    }
}

// Depth-first search to find the minimum distance to visit all locations
void dfs(int current, int visited, int depth, int R, int locations[], int dist[MAX_NODES][MAX_NODES], int *minDistance, int currentDistance) {
    if (depth == R) {
        // Return to starting point
        currentDistance += dist[current][locations[0]];
        if (currentDistance < *minDistance) {
            *minDistance = currentDistance;
        }
        return;
    }

    for (int i = 0; i < R; i++) {
        if (!(visited & (1 << i))) { // Check if location is not visited
            dfs(locations[i], visited | (1 << i), depth + 1, R, locations, dist, minDistance, currentDistance + dist[current][locations[i]]);
        }
    }
}

int main() {
    int N, M, R;
    scanf("%d %d %d", &N, &M, &R);

    // Initialize distance matrix
    int dist[MAX_NODES][MAX_NODES];
    for (int i = 1; i <= N; i++) {
        for (int j = 1; j <= N; j++) {
            if (i == j) {
                dist[i][j] = 0;
            } else {
                dist[i][j] = INT_MAX; // Use INT_MAX to represent no direct path
            }
        }
    }

    // Read edges
    for (int i = 0; i < M; i++) {
        int A, B, C;
        scanf("%d %d %d", &A, &B, &C);
        if (C >= 0 && A >= 1 && A <= N && B >= 1 && B <= N) {
            dist[A][B] = (dist[A][B] < C) ? dist[A][B] : C; // Keep the minimum distance
            dist[B][A] = (dist[B][A] < C) ? dist[B][A] : C; // Undirected graph
        }
    }

    // Read locations to visit
    int locations[MAX_NODES];
    for (int i = 0; i < R; i++) {
        scanf("%d", &locations[i]);
    }

    // Compute shortest paths
    floydWarshall(dist, N);

    // Initialize minimum distance to a large value
    int minDistance = INT_MAX;

    // Start DFS from the first location
    dfs(locations[0], 1, 1, R, locations, dist, &minDistance, 0);

    // Output the minimum distance
    printf("%d\n", minDistance);

    return 0;
}
