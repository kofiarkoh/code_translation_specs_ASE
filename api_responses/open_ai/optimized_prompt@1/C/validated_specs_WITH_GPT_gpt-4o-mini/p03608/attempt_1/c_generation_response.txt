#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX_NODES 100

// Function to compute the minimum distance using Floyd-Warshall algorithm
void floydWarshall(int graph[MAX_NODES][MAX_NODES], int numNodes) {
    for (int k = 0; k < numNodes; k++) {
        for (int i = 0; i < numNodes; i++) {
            for (int j = 0; j < numNodes; j++) {
                if (graph[i][k] != INT_MAX && graph[k][j] != INT_MAX) {
                    if (graph[i][j] > graph[i][k] + graph[k][j]) {
                        graph[i][j] = graph[i][k] + graph[k][j];
                    }
                }
            }
        }
    }
}

// Depth-first search to find the minimum distance to visit all locations
void dfs(int graph[MAX_NODES][MAX_NODES], int locations[], int numLocations, int currentIndex, int visited[], int currentDistance, int *minDistance) {
    if (currentIndex == numLocations) {
        // Return to starting point
        currentDistance += graph[locations[numLocations - 1]][locations[0]];
        if (currentDistance < *minDistance) {
            *minDistance = currentDistance;
        }
        return;
    }

    for (int i = 0; i < numLocations; i++) {
        if (!visited[i]) {
            visited[i] = 1; // Mark as visited
            dfs(graph, locations, numLocations, currentIndex + 1, visited, currentDistance + graph[locations[currentIndex]][locations[i]], minDistance);
            visited[i] = 0; // Backtrack
        }
    }
}

// Main function to compute the minimum distance
int computeMinimumDistance(int numNodes, int numEdges, int numLocations, int locations[], int edges[][3]) {
    int graph[MAX_NODES][MAX_NODES];

    // Initialize the graph with maximum distances
    for (int i = 0; i < numNodes; i++) {
        for (int j = 0; j < numNodes; j++) {
            if (i == j) {
                graph[i][j] = 0; // Distance to self is zero
            } else {
                graph[i][j] = INT_MAX; // Initialize to max distance
            }
        }
    }

    // Fill the graph with edges
    for (int i = 0; i < numEdges; i++) {
        int u = edges[i][0];
        int v = edges[i][1];
        int weight = edges[i][2];
        if (weight >= 0 && u < numNodes && v < numNodes) {
            graph[u][v] = weight; // Set the distance for the edge
            graph[v][u] = weight; // Assuming undirected graph
        }
    }

    // Compute shortest paths
    floydWarshall(graph, numNodes);

    // Prepare for DFS
    int visited[MAX_NODES] = {0};
    int minDistance = INT_MAX;

    // Start DFS from the first location
    visited[0] = 1; // Mark the first location as visited
    dfs(graph, locations, numLocations, 1, visited, 0, &minDistance);

    return minDistance;
}

int main() {
    // Example usage
    int numNodes = 4;
    int numEdges = 5;
    int numLocations = 3;
    int locations[] = {0, 1, 2}; // Locations to visit
    int edges[5][3] = {
        {0, 1, 10},
        {0, 2, 15},
        {1, 2, 35},
        {1, 3, 20},
        {2, 3, 30}
    };

    int result = computeMinimumDistance(numNodes, numEdges, numLocations, locations, edges);
    printf("Minimum distance: %d\n", result);

    return 0;
}
