#include <stdio.h>
#include <stdlib.h>

int max_value(int num_items, int max_resource1, int max_resource2, 
              int items[][4]) {
    // Create a DP table to store maximum values for given resource constraints
    int **dp = (int **)malloc((max_resource1 + 1) * sizeof(int *));
    for (int i = 0; i <= max_resource1; i++) {
        dp[i] = (int *)malloc((max_resource2 + 1) * sizeof(int));
        for (int j = 0; j <= max_resource2; j++) {
            dp[i][j] = 0; // Initialize DP table with 0
        }
    }

    // Iterate through each item
    for (int i = 0; i < num_items; i++) {
        int cost1 = items[i][0]; // Resource cost in the first category
        int cost2 = items[i][1]; // Resource cost in the second category
        int value = items[i][2];  // Value gained from the item
        int gain = items[i][3];   // Additional resource gained in the second category

        // Update the DP table in reverse order to avoid overwriting results
        for (int r1 = max_resource1; r1 >= cost1; r1--) {
            for (int r2 = max_resource2; r2 >= cost2; r2--) {
                // Calculate new value if this item is included
                int new_value = dp[r1 - cost1][r2 - cost2] + value;
                // Update the DP table if the new value is greater
                if (new_value > dp[r1][r2]) {
                    dp[r1][r2] = new_value;
                }
            }
        }
    }

    // The maximum value obtainable is found at dp[max_resource1][max_resource2]
    int result = dp[max_resource1][max_resource2];

    // Free allocated memory for the DP table
    for (int i = 0; i <= max_resource1; i++) {
        free(dp[i]);
    }
    free(dp);

    return result; // Return the maximum value
}

int main() {
    // Example usage
    int num_items = 3;
    int max_resource1 = 5;
    int max_resource2 = 5;
    int items[3][4] = {
        {2, 3, 4, 1}, // Item 1
        {3, 2, 5, 2}, // Item 2
        {1, 1, 3, 0}  // Item 3
    };

    int result = max_value(num_items, max_resource1, max_resource2, items);
    printf("Maximum value obtainable: %d\n", result);

    return 0;
}
