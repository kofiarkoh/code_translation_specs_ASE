#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

void print_pairs(long long *arr, int n, long long computed_sum) {
    // Print pairs based on the computed sum and the order of elements
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            // Print pairs where the sum of the pair equals the computed sum
            if (arr[i] + arr[j] == computed_sum) {
                printf("Pair: (%lld, %lld)\n", arr[i], arr[j]);
            }
        }
    }
}

long long compute_sum(long long *arr, int n) {
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        sum += llabs(arr[i]); // Compute the sum of absolute values
    }
    return sum;
}

int compare(const void *a, const void *b) {
    // Comparison function for qsort to sort in non-decreasing order
    return (*(long long *)a > *(long long *)b) - (*(long long *)a < *(long long *)b);
}

void process_array(long long *arr, int n) {
    // Sort the array in non-decreasing order
    qsort(arr, n, sizeof(long long), compare);
    
    // Compute the sum of absolute values
    long long computed_sum = compute_sum(arr, n);
    
    // Print the computed sum
    printf("Computed Sum: %lld\n", computed_sum);
    
    // Print pairs based on the computed sum
    print_pairs(arr, n, computed_sum);
}

int main() {
    int n;
    // Read the number of elements
    scanf("%d", &n);
    
    // Allocate memory for the array
    long long *arr = (long long *)malloc(n * sizeof(long long));
    
    // Read the array elements
    for (int i = 0; i < n; i++) {
        scanf("%lld", &arr[i]);
    }
    
    // Process the array
    process_array(arr, n);
    
    // Free allocated memory
    free(arr);
    
    return 0;
}
