#include <stdio.h>

int compute_count(int n, int k) {
    // Step 1: Transform n by multiplying it by 7
    long long transformed_n = (long long)n * 7; // Use long long to prevent overflow

    // Step 2: Initialize count of integers satisfying the modular condition
    int count = 0;

    // Step 3: Check integers from 0 to transformed_n (inclusive)
    for (long long i = 0; i <= transformed_n; i++) {
        // Check if i satisfies the modular condition with k
        if (i % k == 0) {
            count++; // Increment count if condition is met
        }
    }

    // Step 4: Return the final count
    return count; // The count is guaranteed to be non-negative
}

int main() {
    int n, k;

    // Input reading (assumed valid as per specification)
    scanf("%d %d", &n, &k);

    // Compute and output the count
    int result = compute_count(n, k);
    printf("%d\n", result);

    return 0;
}
