```c
#include <stdio.h>

int main() {
    int num_cases; // Number of cases to process
    scanf("%d", &num_cases); // Read the number of cases

    // Ensure the number of cases is positive as per the specification
    if (num_cases <= 0) {
        return 1; // Exit if the input is invalid (though spec states inputs are valid)
    }

    for (int case_num = 1; case_num <= num_cases; case_num++) {
        int input_value; // Input integer for the current case
        scanf("%d", &input_value); // Read the input integer

        // Output the case header
        printf("Case %d:\n", case_num);

        // Initialize the derived value with the input value
        int derived_value = input_value;

        // Compute and output the sequence of ten derived values
        for (int i = 0; i < 10; i++) {
            derived_value = (derived_value * derived_value) / 100; // Square and divide by 100
            derived_value = derived_value % 10000; // Take modulo 10000
            printf("%d\n", derived_value); // Output the derived value
        }
    }

    return 0; // Successful execution
}
```
