```c
#include <stdio.h>

int main() {
    int h, w, k;
    
    // Read input values for h, w, and k
    scanf("%d %d %d", &h, &w, &k);
    
    // Iterate through all combinations of non-negative integers i and j
    for (int i = 0; i <= h; i++) {
        for (int j = 0; j <= w; j++) {
            // Calculate the expression (w * i) + (h * j) - (2 * i * j)
            int expression = (w * i) + (h * j) - (2 * i * j);
            
            // Check if the expression equals k
            if (expression == k) {
                printf("Yes\n");
                return 0; // Exit after finding the first valid pair
            }
        }
    }
    
    // If no valid (i, j) pair was found, output "No"
    printf("No\n");
    return 0;
}
```
