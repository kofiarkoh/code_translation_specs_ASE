```c
#include <stdio.h>

#define MAX_RANGES 6

int main() {
    int num_measurements;
    float height;
    int counts[MAX_RANGES] = {0}; // Initialize counts for each range to zero

    // Read the number of height measurements
    scanf("%d", &num_measurements);

    // Process each height measurement
    for (int i = 0; i < num_measurements; i++) {
        scanf("%f", &height);
        
        // Categorize the height into the appropriate range
        if (height < 165.0) {
            counts[0]++;
        } else if (height < 170.0) {
            counts[1]++;
        } else if (height < 175.0) {
            counts[2]++;
        } else if (height < 180.0) {
            counts[3]++;
        } else if (height < 185.0) {
            counts[4]++;
        } else {
            counts[5]++;
        }
    }

    // Output the counts as asterisks
    printf("Range 1: ");
    for (int i = 0; i < counts[0]; i++) {
        printf("*");
    }
    printf("\n");

    printf("Range 2: ");
    for (int i = 0; i < counts[1]; i++) {
        printf("*");
    }
    printf("\n");

    printf("Range 3: ");
    for (int i = 0; i < counts[2]; i++) {
        printf("*");
    }
    printf("\n");

    printf("Range 4: ");
    for (int i = 0; i < counts[3]; i++) {
        printf("*");
    }
    printf("\n");

    printf("Range 5: ");
    for (int i = 0; i < counts[4]; i++) {
        printf("*");
    }
    printf("\n");

    printf("Range 6: ");
    for (int i = 0; i < counts[5]; i++) {
        printf("*");
    }
    printf("\n");

    return 0;
}
```
