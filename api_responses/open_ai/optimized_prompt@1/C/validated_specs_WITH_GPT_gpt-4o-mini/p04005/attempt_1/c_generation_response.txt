```c
#include <stdio.h>

int main() {
    long int a, b, c;
    
    // Read three long integer values from user input
    scanf("%ld %ld %ld", &a, &b, &c);
    
    // Check if any of the values are even
    if (a % 2 == 0 || b % 2 == 0 || c % 2 == 0) {
        // If at least one value is even, output "0"
        printf("0\n");
    } else {
        // If all values are odd, find the maximum value
        long int max = a;
        if (b > max) max = b;
        if (c > max) max = c;

        // Set the maximum value to 1
        long int product = (a == max ? 1 : a) * (b == max ? 1 : b) * (c == max ? 1 : c);
        
        // Output the product of the three values with the maximum set to 1
        printf("%ld\n", product);
    }

    return 0;
}
```
