#include <stdio.h>

void process_arrays(int n, int arr1[], int m, int arr2[], long arr3[]) {
    long cumulative_sum = 0; // To hold the cumulative sum of arr1
    int upper_bound_index = 0; // To track the upper bound index

    // Iterate through each element in the second array
    for (int i = 0; i < m; i++) {
        // Update the cumulative sum up to the current upper bound index
        while (upper_bound_index < n && upper_bound_index < arr2[i]) {
            cumulative_sum += arr1[upper_bound_index];
            upper_bound_index++;
        }

        // Compare the cumulative sum with the corresponding long integer
        if (cumulative_sum >= arr3[i]) {
            printf("Yes\n");
        } else {
            printf("No\n");
        }
    }
}

int main() {
    // Example usage (not part of the specification, just for demonstration)
    int n = 5;
    int arr1[] = {1, 2, 3, 4, 5}; // Sorted array
    int m = 3;
    int arr2[] = {2, 4, 5}; // Upper bounds
    long arr3[] = {3, 10, 15}; // Long integers to compare against

    process_arrays(n, arr1, m, arr2, arr3);
    return 0;
}
