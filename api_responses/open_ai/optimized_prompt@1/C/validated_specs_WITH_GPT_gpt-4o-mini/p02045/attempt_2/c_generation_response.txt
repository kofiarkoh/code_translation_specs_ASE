#include <stdio.h>
#include <stdlib.h>

#define MAX_POINTS 128 // N can be at most 127, so we use 128 for size

// Function to calculate the movement cost based on boundary constraints
int nigeru(int x, int y, int A, int B, int C, int D, int isSrc) {
    int cost = 0;

    // Check boundary constraints for x-coordinate
    if (x < A) {
        cost += A - x; // Increase cost if x is less than A
    } else if (x > C) {
        cost += x - C; // Increase cost if x exceeds C
    }

    // Check boundary constraints for y-coordinate
    if (y < B) {
        cost += B - y; // Increase cost if y is less than B
    } else if (y > D) {
        cost += y - D; // Increase cost if y exceeds D
    }

    // If this point is the source and cost is greater than zero, decrement cost by 1
    if (isSrc && cost > 0) {
        cost -= 1;
    }

    return cost;
}

// Main function to calculate the minimum movement cost
int main() {
    int N;
    int A, B, C, D;
    int X[MAX_POINTS], Y[MAX_POINTS];

    // Read input for N, A, B, C, D
    if (scanf("%d", &N) != 1 || N <= 0 || N >= MAX_POINTS) {
        return -1; // Error if N is not positive or exceeds limits
    }
    if (scanf("%d %d %d %d", &A, &B, &C, &D) != 4) {
        return -1; // Error if not exactly four integers for A, B, C, D
    }

    // Read coordinates into arrays X and Y
    for (int i = 0; i <= N; i++) {
        if (scanf("%d %d", &X[i], &Y[i]) != 2) {
            return -1; // Error if not two integers for each coordinate pair
        }
    }

    int totalCost = 0;

    // Calculate total movement cost
    for (int i = 1; i <= N; i++) {
        int cost = nigeru(X[i], Y[i], A, B, C, D, (i == 1)); // Check if it's the source point
        int directDistance = abs(X[i] - X[i - 1]) + abs(Y[i] - Y[i - 1]);
        totalCost += (cost < directDistance) ? cost : directDistance; // Use the minimum of cost or direct distance
    }

    printf("%d\n", totalCost); // Output the total minimum cost
    return 0;
}
