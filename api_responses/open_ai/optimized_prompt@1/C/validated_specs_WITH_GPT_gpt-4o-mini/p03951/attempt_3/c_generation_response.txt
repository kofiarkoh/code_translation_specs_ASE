#include <stdio.h>

int compute_match_value(int N, const char *S, const char *T) {
    int i, j;
    int ans = 0; // Initialize answer variable

    // Iterate from the end of both strings towards the beginning
    for (i = N - 1, j = N - 1; i >= 0 && j >= 0; i--) {
        // Check for matches
        if (S[i] == T[j]) {
            ans = 2 * N - (N - i - 1); // Calculate result based on last match
            break; // Exit loop on first match found
        }
        j--; // Move j to the left for each iteration
    }

    // If no matches were found, ans remains 0
    return ans; // Return the computed result
}

int main() {
    int N;
    // Example input handling (not specified in the task, but necessary for testing)
    scanf("%d", &N);
    char S[N + 1], T[N + 1]; // +1 for null terminator
    scanf("%s %s", S, T); // Read strings

    int result = compute_match_value(N, S, T);
    printf("%d\n", result); // Output the result
    return 0;
}
