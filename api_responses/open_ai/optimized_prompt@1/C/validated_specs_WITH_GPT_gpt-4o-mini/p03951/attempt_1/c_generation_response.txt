```c
#include <stdio.h>

int compute_matching_value(int N, const char *S, const char *T) {
    // Initialize the result variable
    int ans = 0;

    // Iterate over the strings in reverse order
    for (int i = N - 1; i >= 0; i--) {
        // Check if characters match
        if (S[i] == T[N - 1 - i]) {
            ans++; // Increment the match count
        }
    }

    // Return the computed result
    return ans; // The result is guaranteed to be non-negative
}

int main() {
    // Example usage
    int N = 5; // Length of the strings
    const char *S = "abcde"; // First string
    const char *T = "edcba"; // Second string

    // Compute and print the result
    int result = compute_matching_value(N, S, T);
    printf("Matching value: %d\n", result);

    return 0;
}
```
