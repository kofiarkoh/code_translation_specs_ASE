#include <stdio.h>

#define MOD 1000000007
#define MAX 1000000

// Function to compute factorial % MOD
long long factorial[MAX + 1];

// Function to compute modular inverse using Fermat's Little Theorem
long long mod_inverse(long long a, long long p) {
    long long result = 1;
    long long exp = p - 2; // p is prime
    while (exp > 0) {
        if (exp % 2 == 1) {
            result = (result * a) % p;
        }
        a = (a * a) % p;
        exp /= 2;
    }
    return result;
}

// Function to initialize factorial and inverse factorial arrays
void initialize_factorials() {
    factorial[0] = 1;
    for (int i = 1; i <= MAX; i++) {
        factorial[i] = (factorial[i - 1] * i) % MOD;
    }
}

// Function to calculate the number of distinct paths
int calculate_paths(int x, int y) {
    // Check preconditions
    if (x < 0 || y < 0) {
        return 0; // Invalid input
    }

    int sum = x + y;
    if (sum % 3 != 0) {
        return 0; // Sum not divisible by 3
    }

    int n = sum / 3; // Calculate n
    int x_ = x - n; // Adjusted x
    int y_ = y - n; // Adjusted y

    // Check postconditions
    if (x_ < 0 || y_ < 0) {
        return 0; // Adjusted coordinates are negative
    }

    // Calculate the number of distinct paths using combinatorial formula
    long long result = (factorial[x_ + y_] * mod_inverse(factorial[x_], MOD) % MOD * mod_inverse(factorial[y_], MOD) % MOD) % MOD;
    return (int)result; // Return result as integer
}

int main() {
    initialize_factorials(); // Initialize factorials before any calculations

    int x, y;
    // Example input; in practice, you would read these values
    x = 3; // Example value for x
    y = 3; // Example value for y

    int paths = calculate_paths(x, y);
    printf("%d\n", paths); // Output the number of distinct paths

    return 0;
}
