#include <stdio.h>

int main() {
    long double a, b, c;
    
    // Read three long double values from input
    if (scanf("%Lf %Lf %Lf", &a, &b, &c) != 3) {
        // Undefined behavior if input is not valid long double values
        return 1; // Exit with error code
    }

    long double d = c - a - b; // Calculate d

    // Check the conditions for real and positive roots
    if (d * d > 4 * a * b && d > 0) {
        printf("Yes\n"); // Output "Yes" if conditions are satisfied
    } else {
        printf("No\n"); // Output "No" otherwise
    }

    return 0; // Successful execution
}
