#include <stdio.h>

int count_peaks_and_valleys(int n, int arr[]) {
    // Precondition: n must be >= 3
    // Undefined behavior if n < 3, as per specification
    int count = 0;

    // Iterate through the array from the second element to the second-to-last
    for (int i = 1; i < n - 1; i++) {
        // Check for a peak
        if (arr[i] > arr[i - 1] && arr[i] > arr[i + 1]) {
            count++;
        }
        // Check for a valley
        else if (arr[i] < arr[i - 1] && arr[i] < arr[i + 1]) {
            count++;
        }
    }

    return count; // Return the total count of peaks and valleys
}

int main() {
    int n;
    // Example input handling (not specified in the task, but necessary for testing)
    printf("Enter the number of elements (n >= 3): ");
    scanf("%d", &n);
    
    if (n < 3) {
        // Undefined behavior as per specification; handle gracefully
        printf("Error: n must be at least 3.\n");
        return 1; // Exit with error code
    }

    int arr[n]; // Declare an array of size n
    printf("Enter %d integers:\n", n);
    for (int i = 0; i < n; i++) {
        scanf("%d", &arr[i]); // Read the sequence of integers
    }

    int result = count_peaks_and_valleys(n, arr); // Count peaks and valleys
    printf("Count of peaks and valleys: %d\n", result); // Output the result

    return 0; // Successful execution
}
