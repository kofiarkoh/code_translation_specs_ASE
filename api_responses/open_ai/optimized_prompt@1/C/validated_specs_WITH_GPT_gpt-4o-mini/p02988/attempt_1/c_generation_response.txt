#include <stdio.h>

int count_peaks_and_valleys(int n, int sequence[]) {
    // Precondition check: n must be >= 3
    // Undefined behavior if n < 3, as per specification
    if (n < 3) {
        return 0; // Returning 0 to avoid out-of-bounds access
    }

    int count = 0;

    // Iterate through the sequence from the second element to the second-to-last
    for (int i = 1; i < n - 1; i++) {
        // Check for peak
        if (sequence[i] > sequence[i - 1] && sequence[i] > sequence[i + 1]) {
            count++;
        }
        // Check for valley
        else if (sequence[i] < sequence[i - 1] && sequence[i] < sequence[i + 1]) {
            count++;
        }
    }

    return count; // Return the total count of peaks and valleys
}

int main() {
    int n;
    // Example input handling (not specified in the task, but necessary for testing)
    printf("Enter the number of elements in the sequence (n >= 3): ");
    scanf("%d", &n);

    int sequence[n]; // Declare an array of size n

    printf("Enter the sequence of %d integers:\n", n);
    for (int i = 0; i < n; i++) {
        scanf("%d", &sequence[i]); // Read the sequence
    }

    int result = count_peaks_and_valleys(n, sequence);
    printf("Count of peaks and valleys: %d\n", result); // Output the result

    return 0;
}
