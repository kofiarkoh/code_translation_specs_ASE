#include <stdio.h>

long long compute_result(int N, int pairs[][2]) {
    long long cumulative_value = 0; // Initialize cumulative value

    // Iterate through the pairs to adjust the cumulative value
    for (int i = 0; i < N - 1; i++) {
        int u = pairs[i][0];
        int w = pairs[i][1];

        // Ensure the first integer of the pair is less than or equal to the second
        if (u > w) {
            // If u > w, we do not adjust the cumulative value as per the specification
            continue; // This case should not happen as per the precondition
        }

        // Adjust the cumulative value based on the pair
        cumulative_value += (long long)(w - u); // Ensure non-negative adjustment
    }

    return cumulative_value; // Return the final computed result
}

int main() {
    int N;
    scanf("%d", &N); // Read the number of iterations

    // Declare pairs array to hold N-1 pairs of integers
    int pairs[N - 1][2];

    // Read the pairs of integers
    for (int i = 0; i < N - 1; i++) {
        scanf("%d %d", &pairs[i][0], &pairs[i][1]);
    }

    // Compute the result
    long long result = compute_result(N, pairs);

    // Output the final computed result
    printf("%lld\n", result);

    return 0;
}
