```c
#include <stdio.h>
#include <math.h>

int main() {
    int a, b, c; // Declare three integer inputs
    double result; // Variable to hold the computed result

    // Read three integers from standard input
    // Note: The specification does not handle invalid input formats
    scanf("%d %d %d", &a, &b, &c); // c is unused

    // Calculate half of the product of a and b
    result = (a * b) / 2.0; // Use 2.0 to ensure floating-point division

    // Output the result rounded to the nearest integer
    // The round function is used to round to the nearest integer
    printf("%d\n", (int)round(result)); // Cast to int for output formatting

    return 0; // Indicate successful completion
}
```
