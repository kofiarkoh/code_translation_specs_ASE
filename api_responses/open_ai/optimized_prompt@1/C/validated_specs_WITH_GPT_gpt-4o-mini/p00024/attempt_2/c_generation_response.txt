#include <stdio.h>
#include <math.h>

int main() {
    float velocity;
    // Read floating-point values until EOF
    while (scanf("%f", &velocity) == 1) {
        // Compute the derived integer value based on the given formula
        // n = floor((4.9 * v^2) / (9.8 * 9.8)) + 5) / 5 + 1
        int n = (int)floor((4.9f * velocity * velocity) / (9.8f * 9.8f)) + 5; // Compute numerator
        n = n / 5 + 1; // Final computation to get the output integer
        
        // Output the computed integer
        printf("%d\n", n);
    }
    // The program terminates when EOF is reached
    return 0;
}
