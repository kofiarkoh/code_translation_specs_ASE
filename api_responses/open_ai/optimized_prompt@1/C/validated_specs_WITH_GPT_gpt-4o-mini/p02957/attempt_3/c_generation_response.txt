```c
#include <stdio.h>

int main() {
    long int a, b; // Declare two long integers for user input
    long int sum;  // Variable to hold the sum of a and b

    // Read two long integers from user input
    scanf("%ld %ld", &a, &b);

    sum = a + b; // Calculate the sum of the two integers

    // Check if the sum is even
    if (sum % 2 == 0) {
        // If even, compute the average using integer division
        printf("%ld\n", sum / 2);
    } else {
        // If odd, output "IMPOSSIBLE"
        printf("IMPOSSIBLE\n");
    }

    return 0; // Indicate successful completion
}
```
