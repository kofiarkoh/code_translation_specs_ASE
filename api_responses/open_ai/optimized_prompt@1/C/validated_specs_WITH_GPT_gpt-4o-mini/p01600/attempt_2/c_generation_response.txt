```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int minCostToConnectPoints(int n, int* x, int* y) {
    // Dynamic programming table to store minimum costs
    int** dp = (int**)malloc(n * sizeof(int*));
    for (int i = 0; i < n; i++) {
        dp[i] = (int*)malloc(n * sizeof(int));
        for (int j = 0; j < n; j++) {
            dp[i][j] = INT_MAX; // Initialize to a high value
        }
    }

    // Cost to connect a point to itself is zero
    for (int i = 0; i < n; i++) {
        dp[i][i] = 0;
    }

    // Compute costs for connecting points
    for (int len = 2; len <= n; len++) { // len is the segment length
        for (int i = 0; i <= n - len; i++) {
            int j = i + len - 1; // j is the endpoint of the segment
            for (int k = i; k < j; k++) { // k is the point to connect
                int cost = abs(x[j] - x[k]) + abs(y[j] - y[k]);
                dp[i][j] = (dp[i][j] < dp[i][k] + cost) ? dp[i][j] : (dp[i][k] + cost);
            }
        }
    }

    // The minimum cost to connect all points is stored in dp[0][n-1]
    int result = dp[0][n - 1];

    // Free allocated memory
    for (int i = 0; i < n; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main() {
    // Example usage
    int n = 4;
    int x[] = {0, 0, 1, 1};
    int y[] = {0, 1, 0, 1};
    int minCost = minCostToConnectPoints(n, x, y);
    printf("Minimum cost to connect points: %d\n", minCost);
    return 0;
}
```
