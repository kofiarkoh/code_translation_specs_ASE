#include <stdio.h>

// Function to calculate the digit sum of a positive integer
long digit_sum(long x) {
    long sum = 0;
    while (x > 0) {
        sum += x % 10; // Add the last digit to the sum
        x /= 10;       // Remove the last digit
    }
    return sum;
}

int main() {
    long n; // Number of values to compute
    // Read input
    if (scanf("%ld", &n) != 1 || n < 0) {
        return 1; // Undefined behavior for invalid input or negative n
    }

    long current_value = 1; // Initialize the first value to 1

    for (long i = 0; i < n; i++) {
        printf("%ld\n", current_value); // Output the current value

        // Calculate the next value based on the current value
        long candidate1 = current_value + digit_sum(current_value);
        long candidate2 = current_value + 2 * digit_sum(current_value);
        long candidate3 = current_value + 3 * digit_sum(current_value);

        // Find the minimum candidate that is greater than the current value
        current_value = candidate1; // Start with candidate1 as the minimum
        if (candidate2 > current_value) {
            current_value = candidate2; // Update if candidate2 is greater
        }
        if (candidate3 > current_value) {
            current_value = candidate3; // Update if candidate3 is greater
        }
    }

    return 0; // Successful execution
}
