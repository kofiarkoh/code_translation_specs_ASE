#include <stdio.h>

long long compute_pairs(int N, int *sequence) {
    long long total_pairs = 0; // Initialize total pairs to 0

    // Iterate through the sequence to compute pairs
    for (int i = 0; i < N; i++) {
        // Each element can form pairs with all subsequent elements
        // This is a simple combinatorial calculation
        total_pairs += (N - i - 1); // Pairs formed with elements after the current one
    }

    return total_pairs; // Return the total number of pairs
}

int main() {
    int N;
    // Read the number of elements
    scanf("%d", &N);
    
    // Declare the sequence array
    int sequence[100000]; // Maximum size based on the specification

    // Read the sequence of integers
    for (int i = 0; i < N; i++) {
        scanf("%d", &sequence[i]);
    }

    // Compute and print the total number of pairs
    long long result = compute_pairs(N, sequence);
    printf("%lld\n", result);

    return 0;
}
