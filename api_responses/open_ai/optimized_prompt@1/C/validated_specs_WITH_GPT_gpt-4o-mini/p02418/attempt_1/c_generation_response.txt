```c
#include <stdio.h>
#include <string.h>

#define MAX_LENGTH 100

int main() {
    char str1[MAX_LENGTH + 1]; // +1 for null terminator
    char str2[MAX_LENGTH + 1]; // +1 for null terminator
    char concatenated[2 * MAX_LENGTH + 1]; // To hold str1 concatenated with itself

    // Read the first string
    fgets(str1, sizeof(str1), stdin);
    str1[strcspn(str1, "\n")] = 0; // Remove newline character if present

    // Read the second string
    fgets(str2, sizeof(str2), stdin);
    str2[strcspn(str2, "\n")] = 0; // Remove newline character if present

    // Concatenate str1 with itself
    strcpy(concatenated, str1);
    strcat(concatenated, str1);

    // Check if str2 is a substring of the concatenated string
    if (strstr(concatenated, str2) != NULL) {
        printf("Yes\n"); // str2 is found in the concatenated string
    } else {
        printf("No\n"); // str2 is not found in the concatenated string
    }

    return 0;
}
```
