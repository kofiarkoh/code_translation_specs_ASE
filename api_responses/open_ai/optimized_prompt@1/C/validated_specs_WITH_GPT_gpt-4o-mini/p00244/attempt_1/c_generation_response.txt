#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

typedef struct Edge {
    int target;
    int cost;
    struct Edge* next;
} Edge;

typedef struct Node {
    int id;
    Edge* edges;
} Node;

typedef struct PriorityQueue {
    int* nodes;
    int* costs;
    int size;
} PriorityQueue;

void init_priority_queue(PriorityQueue* pq, int max_size) {
    pq->nodes = (int*)malloc(max_size * sizeof(int));
    pq->costs = (int*)malloc(max_size * sizeof(int));
    pq->size = 0;
}

void enqueue(PriorityQueue* pq, int node, int cost) {
    pq->nodes[pq->size] = node;
    pq->costs[pq->size] = cost;
    pq->size++;
    // Simple bubble up to maintain order (inefficient but straightforward)
    for (int i = pq->size - 1; i > 0; i--) {
        if (pq->costs[i] < pq->costs[i - 1]) {
            int temp_node = pq->nodes[i];
            int temp_cost = pq->costs[i];
            pq->nodes[i] = pq->nodes[i - 1];
            pq->costs[i] = pq->costs[i - 1];
            pq->nodes[i - 1] = temp_node;
            pq->costs[i - 1] = temp_cost;
        } else {
            break;
        }
    }
}

int dequeue(PriorityQueue* pq) {
    if (pq->size == 0) return -1; // Indicate empty queue
    int min_node = pq->nodes[0];
    pq->nodes[0] = pq->nodes[pq->size - 1];
    pq->costs[0] = pq->costs[pq->size - 1];
    pq->size--;
    return min_node;
}

int min_cost(Node* graph, int num_nodes, int start, int goal) {
    int* costs = (int*)malloc(num_nodes * sizeof(int));
    for (int i = 0; i < num_nodes; i++) {
        costs[i] = INT_MAX; // Initialize costs to infinity
    }
    costs[start] = 0;

    PriorityQueue pq;
    init_priority_queue(&pq, num_nodes);
    enqueue(&pq, start, 0);

    while (pq.size > 0) {
        int current = dequeue(&pq);
        if (current == goal) {
            free(costs);
            free(pq.nodes);
            free(pq.costs);
            return costs[goal]; // Return the cost to reach the goal
        }

        Edge* edge = graph[current].edges;
        while (edge != NULL) {
            int new_cost = costs[current] + edge->cost;
            if (new_cost < costs[edge->target]) {
                costs[edge->target] = new_cost;
                enqueue(&pq, edge->target, new_cost);
            }
            edge = edge->next;
        }
    }

    free(costs);
    free(pq.nodes);
    free(pq.costs);
    return -1; // Indicate that the goal is unreachable
}

int main() {
    int num_nodes, num_edges;
    while (scanf("%d", &num_nodes) == 1 && num_nodes > 0) {
        scanf("%d", &num_edges);
        Node* graph = (Node*)malloc(num_nodes * sizeof(Node));
        for (int i = 0; i < num_nodes; i++) {
            graph[i].id = i;
            graph[i].edges = NULL;
        }

        for (int i = 0; i < num_edges; i++) {
            int from, to, cost;
            scanf("%d %d %d", &from, &to, &cost);
            Edge* new_edge = (Edge*)malloc(sizeof(Edge));
            new_edge->target = to;
            new_edge->cost = cost;
            new_edge->next = graph[from].edges;
            graph[from].edges = new_edge;
        }

        int start = 0; // Assuming start node is 0
        int goal = num_nodes - 1; // Assuming goal node is the last node
        int result = min_cost(graph, num_nodes, start, goal);
        printf("%d\n", result);

        // Free graph memory
        for (int i = 0; i < num_nodes; i++) {
            Edge* edge = graph[i].edges;
            while (edge != NULL) {
                Edge* temp = edge;
                edge = edge->next;
                free(temp);
            }
        }
        free(graph);
    }
    return 0;
}
