#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

typedef struct Edge {
    int to;
    int cost;
    struct Edge* next;
} Edge;

typedef struct Node {
    Edge* edges;
} Node;

typedef struct PriorityQueue {
    int* nodes;
    int* costs;
    int size;
} PriorityQueue;

void init_priority_queue(PriorityQueue* pq, int max_size) {
    pq->nodes = (int*)malloc(max_size * sizeof(int));
    pq->costs = (int*)malloc(max_size * sizeof(int));
    pq->size = 0;
}

void enqueue(PriorityQueue* pq, int node, int cost) {
    pq->nodes[pq->size] = node;
    pq->costs[pq->size] = cost;
    pq->size++;
    // Simple insertion sort to maintain order (inefficient but straightforward)
    for (int i = pq->size - 1; i > 0 && pq->costs[i] < pq->costs[i - 1]; i--) {
        int temp_node = pq->nodes[i];
        int temp_cost = pq->costs[i];
        pq->nodes[i] = pq->nodes[i - 1];
        pq->costs[i] = pq->costs[i - 1];
        pq->nodes[i - 1] = temp_node;
        pq->costs[i - 1] = temp_cost;
    }
}

int dequeue(PriorityQueue* pq) {
    if (pq->size == 0) return -1; // Indicate empty queue
    int node = pq->nodes[0];
    for (int i = 1; i < pq->size; i++) {
        pq->nodes[i - 1] = pq->nodes[i];
        pq->costs[i - 1] = pq->costs[i];
    }
    pq->size--;
    return node;
}

int min_cost(int num_nodes, int num_edges, int edges[][3], int start, int goal) {
    Node* graph = (Node*)malloc(num_nodes * sizeof(Node));
    for (int i = 0; i < num_nodes; i++) {
        graph[i].edges = NULL;
    }

    // Build the graph
    for (int i = 0; i < num_edges; i++) {
        int from = edges[i][0];
        int to = edges[i][1];
        int cost = edges[i][2];
        Edge* new_edge = (Edge*)malloc(sizeof(Edge));
        new_edge->to = to;
        new_edge->cost = cost;
        new_edge->next = graph[from].edges;
        graph[from].edges = new_edge;
    }

    // Dijkstra's algorithm initialization
    int* min_costs = (int*)malloc(num_nodes * sizeof(int));
    for (int i = 0; i < num_nodes; i++) {
        min_costs[i] = INT_MAX; // Initialize costs to infinity
    }
    min_costs[start] = 0;

    PriorityQueue pq;
    init_priority_queue(&pq, num_nodes);
    enqueue(&pq, start, 0);

    while (pq.size > 0) {
        int current = dequeue(&pq);
        if (current == goal) {
            free(graph);
            free(min_costs);
            free(pq.nodes);
            free(pq.costs);
            return min_costs[goal]; // Return the minimum cost to the goal
        }

        Edge* edge = graph[current].edges;
        while (edge != NULL) {
            int new_cost = min_costs[current] + edge->cost;
            if (new_cost < min_costs[edge->to]) {
                min_costs[edge->to] = new_cost;
                enqueue(&pq, edge->to, new_cost);
            }
            edge = edge->next;
        }
    }

    // If we reach here, the goal is unreachable
    free(graph);
    free(min_costs);
    free(pq.nodes);
    free(pq.costs);
    return -1; // Indicate unreachable goal
}

int main() {
    int num_nodes, num_edges;
    while (scanf("%d %d", &num_nodes, &num_edges) == 2) {
        if (num_nodes <= 0) {
            return 0; // Terminate if zero nodes
        }
        int edges[num_edges][3];
        for (int i = 0; i < num_edges; i++) {
            scanf("%d %d %d", &edges[i][0], &edges[i][1], &edges[i][2]);
        }
        int start = 0; // Assuming start node is 0
        int goal = num_nodes - 1; // Assuming goal node is the last node
        int result = min_cost(num_nodes, num_edges, edges, start, goal);
        printf("%d\n", result);
    }
    return 0;
}
