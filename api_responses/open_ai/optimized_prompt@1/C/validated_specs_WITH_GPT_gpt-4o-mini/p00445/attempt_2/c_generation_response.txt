```c
#include <stdio.h>
#include <string.h>

#define MAX_LENGTH 10000

// Function to count overlapping occurrences of a substring in a string
int count_overlapping(const char *str, const char *substr) {
    int count = 0;
    const char *temp = str;

    // Loop through the string to find all occurrences of the substring
    while ((temp = strstr(temp, substr)) != NULL) {
        count++;
        temp++; // Move one character forward to allow for overlapping
    }

    return count;
}

int main() {
    char input[MAX_LENGTH + 1]; // Buffer to hold input strings

    // Read input until EOF
    while (fgets(input, sizeof(input), stdin) != NULL) {
        // Remove newline character if present
        input[strcspn(input, "\n")] = 0;

        // Count occurrences of "JOI" and "IOI"
        int count_JOI = count_overlapping(input, "JOI");
        int count_IOI = count_overlapping(input, "IOI");

        // Output the counts
        printf("%d %d\n", count_JOI, count_IOI);
    }

    return 0;
}
```
