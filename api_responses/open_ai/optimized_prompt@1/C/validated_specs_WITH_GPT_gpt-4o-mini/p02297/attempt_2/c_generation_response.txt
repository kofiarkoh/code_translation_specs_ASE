```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

#define MAX_VERTICES 100 // Assuming a maximum number of vertices

// Structure to hold the coordinates of a vertex
typedef struct {
    int x;
    int y;
} Vertex;

int main() {
    int n; // Number of vertices
    Vertex vertices[MAX_VERTICES]; // Array to store vertices

    // Read the number of vertices
    scanf("%d", &n);

    // Read the coordinates of each vertex
    for (int i = 0; i < n; i++) {
        scanf("%d %d", &vertices[i].x, &vertices[i].y);
    }

    // Calculate the area using the shoelace formula
    double area = 0.0;
    for (int i = 0; i < n; i++) {
        int j = (i + 1) % n; // Next vertex, wrapping around
        area += (vertices[i].x * vertices[j].y) - (vertices[i].y * vertices[j].x);
    }
    area = fabs(area) / 2.0; // Take the absolute value and divide by 2

    // Output the area formatted to one decimal place
    printf("%.1f\n", area);

    return 0;
}
```
