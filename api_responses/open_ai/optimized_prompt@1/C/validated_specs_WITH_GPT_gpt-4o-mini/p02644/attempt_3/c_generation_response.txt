#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX_SIZE 100 // Assuming a maximum grid size for simplicity

// Directions for movement: up, down, left, right
int directions[4][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

// Function to find the minimum number of moves
int min_moves(int h, int w, int k, int s[2], int g[2], char grid[MAX_SIZE][MAX_SIZE]) {
    // Check for valid input dimensions
    if (h <= 0 || w <= 0 || s[0] < 0 || s[0] >= h || s[1] < 0 || s[1] >= w ||
        g[0] < 0 || g[0] >= h || g[1] < 0 || g[1] >= w) {
        return -1; // Undefined behavior for invalid input
    }

    // Distance array initialized to infinity
    int distance[MAX_SIZE][MAX_SIZE];
    for (int i = 0; i < h; i++) {
        for (int j = 0; j < w; j++) {
            distance[i][j] = INT_MAX; // Represents infinity
        }
    }
    distance[s[0]][s[1]] = 0; // Starting position distance is 0

    // Queue for BFS
    int queue[MAX_SIZE * MAX_SIZE][2];
    int front = 0, rear = 0;

    // Enqueue starting position
    queue[rear][0] = s[0];
    queue[rear][1] = s[1];
    rear++;

    // BFS to explore the grid
    while (front < rear) {
        int x = queue[front][0];
        int y = queue[front][1];
        front++;

        // Check if we reached the goal
        if (x == g[0] && y == g[1]) {
            return distance[x][y]; // Return the distance to the goal
        }

        // Explore all four directions
        for (int d = 0; d < 4; d++) {
            int dx = directions[d][0];
            int dy = directions[d][1];

            // Move in the current direction up to k cells
            for (int step = 1; step <= k; step++) {
                int new_x = x + step * dx;
                int new_y = y + step * dy;

                // Check bounds and obstacles
                if (new_x < 0 || new_x >= h || new_y < 0 || new_y >= w || grid[new_x][new_y] == '#') {
                    break; // Stop if out of bounds or hit an obstacle
                }

                // If we found a shorter path to this cell
                if (distance[new_x][new_y] > distance[x][y] + 1) {
                    distance[new_x][new_y] = distance[x][y] + 1; // Update distance
                    queue[rear][0] = new_x; // Enqueue new position
                    queue[rear][1] = new_y;
                    rear++;
                }
            }
        }
    }

    return -1; // Target is unreachable
}

int main() {
    int h, w, k;
    int s[2], g[2];
    char grid[MAX_SIZE][MAX_SIZE];

    // Example input (to be replaced with actual input handling)
    h = 5; w = 5; k = 2;
    s[0] = 0; s[1] = 0; // Starting position
    g[0] = 4; g[1] = 4; // Target position

    // Example grid initialization (to be replaced with actual grid input)
    char example_grid[5][5] = {
        {'.', '.', '.', '.', '.'},
        {'.', '#', '#', '#', '.'},
        {'.', '.', '.', '#', '.'},
        {'.', '#', '.', '.', '.'},
        {'.', '.', '.', '.', '.'}
    };
    for (int i = 0; i < h; i++) {
        for (int j = 0; j < w; j++) {
            grid[i][j] = example_grid[i][j];
        }
    }

    int result = min_moves(h, w, k, s, g, grid);
    printf("%d\n", result); // Output the result

    return 0;
}
