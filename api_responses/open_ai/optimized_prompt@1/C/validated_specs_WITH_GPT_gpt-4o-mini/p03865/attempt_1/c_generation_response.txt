```c
#include <stdio.h>

#define MAX_LENGTH 114514

int main() {
    char buffer[MAX_LENGTH]; // Buffer to hold the input characters
    int i = 0; // Index for reading characters

    // Read characters from input until EOF or buffer limit is reached
    while (i < MAX_LENGTH - 1 && (buffer[i] = getchar()) != EOF) {
        if (buffer[i] == '\n') { // Stop reading at newline
            break;
        }
        i++;
    }
    buffer[i] = '\0'; // Null-terminate the string

    // Check if there is at least one character read
    if (i > 0) {
        // Compare the first and last characters (excluding the null terminator)
        if (buffer[0] == buffer[i - 1]) {
            printf("First\n"); // Output "First" if they are equal
        } else {
            printf("Second\n"); // Output "Second" if they are not equal
        }
    }

    return 0; // Return success
}
```
