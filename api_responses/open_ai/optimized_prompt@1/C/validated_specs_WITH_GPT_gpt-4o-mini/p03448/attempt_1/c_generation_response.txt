#include <stdio.h>

int count_combinations(int A, int B, int C, int X) {
    int count = 0; // Initialize count of valid combinations to zero

    // Iterate through all possible values of i, j, and k
    for (int i = 0; i <= A; i++) {
        for (int j = 0; j <= B; j++) {
            for (int k = 0; k <= C; k++) {
                // Check if the weighted sum equals X
                if (500 * i + 100 * j + 50 * k == X) {
                    count++; // Increment count for valid combination
                }
            }
        }
    }

    return count; // Return the total count of valid combinations
}

int main() {
    int A, B, C, X;

    // Example input; in practice, these would be read from user input or another source
    A = 10; // Maximum multiplier for the first weight
    B = 10; // Maximum multiplier for the second weight
    C = 10; // Maximum multiplier for the third weight
    X = 1000; // Target sum to be achieved

    int result = count_combinations(A, B, C, X);
    printf("%d\n", result); // Output the result

    return 0;
}
