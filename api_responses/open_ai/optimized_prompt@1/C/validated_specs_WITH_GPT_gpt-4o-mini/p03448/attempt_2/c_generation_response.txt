#include <stdio.h>

int count_combinations(int A, int B, int C, int X) {
    int count = 0; // Initialize count of valid combinations

    // Iterate through all possible values of i, j, and k
    for (int i = 0; i <= A; i++) { // i ranges from 0 to A
        for (int j = 0; j <= B; j++) { // j ranges from 0 to B
            for (int k = 0; k <= C; k++) { // k ranges from 0 to C
                // Calculate the weighted sum
                int weighted_sum = 500 * i + 100 * j + 50 * k;
                
                // Check if the weighted sum equals the target X
                if (weighted_sum == X) {
                    count++; // Increment count for valid combination
                }
            }
        }
    }

    return count; // Return the total count of valid combinations
}

int main() {
    int A, B, C, X;
    
    // Example input; in practice, these would be read from user input or another source
    A = 1; // Example maximum multiplier for the first weight
    B = 2; // Example maximum multiplier for the second weight
    C = 3; // Example maximum multiplier for the third weight
    X = 650; // Example target sum

    // Call the function and print the result
    int result = count_combinations(A, B, C, X);
    printf("%d\n", result); // Output the count of valid combinations

    return 0; // Indicate successful completion
}
