#include <stdio.h>
#include <stdlib.h>

#define MOD 1000000007 // Predefined constant for modulo operations
#define MAX_N 1000000  // Maximum value for n

// Precomputed arrays for factorials and their modular inverses
long long factorial[MAX_N + 1];
long long inverse_factorial[MAX_N + 1];

// Function to compute modular exponentiation
long long mod_exp(long long base, long long exp, long long mod) {
    long long result = 1;
    while (exp > 0) {
        if (exp % 2 == 1) {
            result = (result * base) % mod;
        }
        base = (base * base) % mod;
        exp /= 2;
    }
    return result;
}

// Function to precompute factorials and their inverses
void precompute_factorials() {
    factorial[0] = 1;
    for (int i = 1; i <= MAX_N; i++) {
        factorial[i] = (factorial[i - 1] * i) % MOD;
    }
    // Compute inverses using Fermat's Little Theorem
    inverse_factorial[MAX_N] = mod_exp(factorial[MAX_N], MOD - 2, MOD);
    for (int i = MAX_N - 1; i >= 0; i--) {
        inverse_factorial[i] = (inverse_factorial[i + 1] * (i + 1)) % MOD;
    }
}

// Function to compute n choose r (nCr)
long long nCr(int n, int r) {
    if (n < 0 || r < 0 || r > n) {
        return 0; // Return zero for invalid inputs
    }
    return (factorial[n] * inverse_factorial[r] % MOD * inverse_factorial[n - r] % MOD) % MOD;
}

// Main function to compute the combinatorial value based on input n
int main() {
    int n;
    scanf("%d", &n); // Read input

    // Check preconditions
    if (n < 0 || n > MAX_N) {
        printf("0\n"); // Output zero for invalid n
        return 0;
    }

    // Precompute factorials and inverses
    precompute_factorials();

    // Example: Compute n choose 2 (as a placeholder for the specific combinatorial value)
    long long result = nCr(n, 2); // Change 2 to the desired r value as needed

    // Output the result
    printf("%lld\n", result);
    return 0;
}
