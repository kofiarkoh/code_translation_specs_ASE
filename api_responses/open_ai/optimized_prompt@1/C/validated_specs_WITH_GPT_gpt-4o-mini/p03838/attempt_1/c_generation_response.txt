#include <stdio.h>

int main() {
    int x, y; // Declare input integers
    long long result; // Declare output as long long

    // Read input values for x and y
    scanf("%d %d", &x, &y);

    // Compute result based on the specified conditions
    if (x >= 0) { // x is non-negative
        if (y >= x) {
            result = y - x; // Case 1
        } else if (y > 0) {
            result = x - y + 2; // Case 2
        } else if (y >= -x) {
            result = x + y + 1; // Case 3
        } else {
            result = -y - x + 1; // Case 4
        }
    } else { // x is negative
        if (y >= -x) {
            result = x + y + 1; // Case 5
        } else if (y <= 0) {
            result = -x - y + 1; // Case 6
        } else if (y >= x) {
            result = y - x; // Case 7
        } else {
            result = x - y + 2; // Case 8
        }
    }

    // Output the computed result
    printf("%lld\n", result); // Print the result as long long

    return 0; // Indicate successful completion
}
