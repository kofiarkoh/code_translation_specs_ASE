#include <stdio.h>

void process(int N, int A, int B) {
    // Ensure N is within the range [0, 11] as per the specification
    N = N % 12; // Modulo operation to constrain N

    for (int i = 0; i < N; i++) {
        // Update A and B according to the specified operations
        int new_A = A - B; // A is updated by subtracting B
        B = B + new_A;     // B is updated by adding the new value of A
        A = new_A;         // Assign the new value of A
    }

    // Output the final values of A and B
    printf("%d %d\n", A, B);
}

int main() {
    int N, A, B;

    // Example input; in a real scenario, these would be read from user input or another source
    N = 5; // Example value for N
    A = 10; // Example initial value for A
    B = 3;  // Example initial value for B

    process(N, A, B); // Call the processing function

    return 0;
}
