#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int weight;
    int value;
    double ratio; // value-to-weight ratio
} Item;

// Comparator function for sorting items based on value-to-weight ratio
int compare(const void *a, const void *b) {
    Item *itemA = (Item *)a;
    Item *itemB = (Item *)b;
    // Sort in descending order
    return (itemB->ratio > itemA->ratio) - (itemB->ratio < itemA->ratio);
}

double fractional_knapsack(int num_items, double max_weight, int weights[], int values[]) {
    if (num_items <= 0 || max_weight <= 0.0) {
        return 0.0; // If no items or capacity is zero, return 0
    }

    Item *items = malloc(num_items * sizeof(Item));
    if (!items) {
        return 0.0; // Memory allocation failed
    }

    int valid_items = 0;

    // Populate the items array and calculate value-to-weight ratios
    for (int i = 0; i < num_items; i++) {
        if (values[i] > 0 && weights[i] >= 0) { // Only consider valid items
            items[valid_items].weight = weights[i];
            items[valid_items].value = values[i];
            items[valid_items].ratio = (double)values[i] / weights[i]; // Calculate ratio
            valid_items++;
        }
    }

    // Sort items based on value-to-weight ratio
    qsort(items, valid_items, sizeof(Item), compare);

    double total_value = 0.0;
    double remaining_weight = max_weight;

    // Process the sorted items
    for (int i = 0; i < valid_items && remaining_weight > 0; i++) {
        if (items[i].weight <= remaining_weight) {
            // Take the whole item
            total_value += items[i].value;
            remaining_weight -= items[i].weight;
        } else {
            // Take the fractional part of the item
            total_value += items[i].value * (remaining_weight / items[i].weight);
            remaining_weight = 0; // Knapsack is full
        }
    }

    free(items); // Free allocated memory
    return total_value; // Return the maximum value obtained
}

int main() {
    int num_items = 3;
    double max_weight = 50.0;
    int weights[] = {10, 20, 30};
    int values[] = {60, 100, 120};

    double max_value = fractional_knapsack(num_items, max_weight, weights, values);
    printf("Maximum value in the knapsack = %.2f\n", max_value);
    return 0;
}
