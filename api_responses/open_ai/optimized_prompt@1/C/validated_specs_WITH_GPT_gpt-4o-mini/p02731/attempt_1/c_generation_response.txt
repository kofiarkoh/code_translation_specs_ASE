```c
#include <stdio.h>

int main() {
    int input; // Variable to store the input integer
    double result; // Variable to store the computed result

    // Read an integer from standard input
    scanf("%d", &input);

    // Compute one-third of the input integer
    double one_third = input / 3.0; // Use 3.0 to ensure floating-point division

    // Compute the cube of one-third
    result = one_third * one_third * one_third; // Cube the value

    // Output the result formatted to twelve decimal places
    printf("%.12f\n", result);

    return 0; // Indicate successful completion
}
```
