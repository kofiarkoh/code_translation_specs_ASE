#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

unsigned int minCost(int n, int* weights) {
    // Dynamic programming table to store minimum costs
    unsigned int** dp = (unsigned int**)malloc(n * sizeof(unsigned int*));
    for (int i = 0; i < n; i++) {
        dp[i] = (unsigned int*)malloc(n * sizeof(unsigned int));
    }

    // Initialize the dp table
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            dp[i][j] = (i == j) ? 0 : UINT_MAX; // Zero cost for single weights, max for others
        }
    }

    // Precompute the prefix sums for weights
    unsigned int* prefixSum = (unsigned int*)malloc((n + 1) * sizeof(unsigned int));
    prefixSum[0] = 0;
    for (int i = 1; i <= n; i++) {
        prefixSum[i] = prefixSum[i - 1] + weights[i - 1];
    }

    // Fill the dp table
    for (int len = 2; len <= n; len++) { // length of the range
        for (int i = 0; i <= n - len; i++) { // starting index
            int j = i + len - 1; // ending index
            for (int k = i; k < j; k++) { // possible split points
                unsigned int cost = prefixSum[j + 1] - prefixSum[i]; // sum of weights from i to j
                if (dp[i][k] != UINT_MAX && dp[k + 1][j] != UINT_MAX) {
                    dp[i][j] = (dp[i][j] < dp[i][k] + dp[k + 1][j] + cost) ? dp[i][j] : (dp[i][k] + dp[k + 1][j] + cost);
                }
            }
        }
    }

    unsigned int result = dp[0][n - 1];

    // Free allocated memory
    for (int i = 0; i < n; i++) {
        free(dp[i]);
    }
    free(dp);
    free(prefixSum);

    return result;
}

int main() {
    int n;
    scanf("%d", &n); // Read number of weights
    int* weights = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &weights[i]); // Read weights
    }

    unsigned int result = minCost(n, weights);
    printf("%u\n", result); // Output the minimum cost

    free(weights);
    return 0;
}
