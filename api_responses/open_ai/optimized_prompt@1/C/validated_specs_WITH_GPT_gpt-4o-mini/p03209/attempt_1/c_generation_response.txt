// Recursive function to compute a specific value based on inputs n and k
int compute(int n, int k) {
    // Base case: if n is 0, return k
    if (n == 0) {
        return k;
    }
    // Recursive case: call compute with n-1 and k
    return compute(n - 1, k) + k; // Assuming the recursive logic adds k for each decrement of n
}

int main() {
    int n, k;
    // Example input handling (not specified in the task, but necessary for testing)
    printf("Enter two non-negative integers (n and k): ");
    scanf("%d %d", &n, &k);
    
    // Ensure preconditions are met
    if (n < 0 || k < 0) {
        // Undefined behavior for negative inputs, as per specification
        return -1; // Indicate error
    }

    // Compute the result
    int result = compute(n, k);

    printf("Result: %d\n", result);

}
