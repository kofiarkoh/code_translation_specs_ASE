#include <stdio.h>

long long compute_value(long long n, long long m) {
    // Preconditions: m must be >= 1
    if (m == 1) {
        return 0; // If m equals 1, output is 0
    } else if (m / 2 < n) {
        return m / 2; // If m/2 is less than n, output is m/2
    } else if (m / 2 >= n && m / 2 < n + 2) {
        return n; // If m/2 is between n and n+2, output is n
    } else {
        // Otherwise, output is n + (m/2 - n) / 2
        return n + (m / 2 - n) / 2;
    }
}

int main() {
    long long n, m;
    // Input reading (assumed to be valid as per specification)
    scanf("%lld %lld", &n, &m);
    
    // Compute and print the result
    long long result = compute_value(n, m);
    printf("%lld\n", result);
    
    return 0;
}
