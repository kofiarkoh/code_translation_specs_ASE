```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node* left;
    struct Node* right;
} Node;

// Function to create a new tree node
Node* createNode(int value) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->value = value;
    newNode->left = NULL;
    newNode->right = NULL;
    return newNode;
}

// Function to insert edges into the binary tree
void insertEdge(Node* root, int parent, int child) {
    if (root == NULL) return;
    if (root->value == parent) {
        if (root->left == NULL) {
            root->left = createNode(child);
        } else if (root->right == NULL) {
            root->right = createNode(child);
        }
    } else {
        insertEdge(root->left, parent, child);
        insertEdge(root->right, parent, child);
    }
}

// Function to calculate the maximum depth of the tree
int maxDepth(Node* node) {
    if (node == NULL) return 0;
    int leftDepth = maxDepth(node->left);
    int rightDepth = maxDepth(node->right);
    return (leftDepth > rightDepth ? leftDepth : rightDepth) + 1;
}

// Function to determine the winner based on the maximum depth
const char* determineWinner(int n, int edges[][2]) {
    if (n < 1) return "Invalid"; // Precondition check

    Node* root = createNode(1); // Assuming the root node is always 1
    for (int i = 0; i < n - 1; i++) {
        insertEdge(root, edges[i][0], edges[i][1]);
    }

    int depth = maxDepth(root);
    // Determine winner based on depth
    return (depth % 2 == 0) ? "Second" : "First";
}

int main() {
    int n = 5; // Example number of nodes
    int edges[4][2] = { {1, 2}, {1, 3}, {2, 4}, {2, 5} }; // Example edges

    const char* winner = determineWinner(n, edges);
    printf("%s\n", winner); // Output the winner

    return 0;
}
```
