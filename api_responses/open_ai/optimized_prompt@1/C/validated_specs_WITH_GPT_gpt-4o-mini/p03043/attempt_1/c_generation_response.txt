```c
#include <stdio.h>
#include <stdlib.h>

int main() {
    int N, K;

    // Read input values for N and K
    scanf("%d %d", &N, &K);

    // Preconditions: N and K must be positive integers
    if (N <= 0 || K <= 0) {
        return 1; // Exit if preconditions are not met
    }

    // Array of denominators
    double *denominators = (double *)malloc(N * sizeof(double));
    if (denominators == NULL) {
        return 1; // Exit if memory allocation fails
    }

    // Initialize the array of denominators to zero
    for (int i = 0; i < N; i++) {
        denominators[i] = 0.0;
    }

    // Populate the denominators based on K
    for (int i = 0; i < N; i++) {
        denominators[i] = K + i; // Example population logic
    }

    // First denominator remains constant
    double first_denominator = denominators[0];
    double cumulative_sum = 0.0;

    // Calculate the cumulative sum of ratios
    for (int i = 1; i < N; i++) {
        cumulative_sum += first_denominator / denominators[i];
    }

    // Calculate the final result
    double result = cumulative_sum / first_denominator;

    // Output the result formatted to twelve decimal places
    printf("%.12f\n", result);

    // Free allocated memory
    free(denominators);

    return 0;
}
```
