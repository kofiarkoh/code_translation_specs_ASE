```c
#include <stdio.h>

int main() {
    char input[4]; // Array to hold exactly three characters plus null terminator
    int count_o = 0; // Counter for 'o' characters
    int score; // Variable to hold the calculated score

    // Read a string of exactly three characters
    // The input is assumed to be valid as per the specification
    scanf("%3s", input); // Read up to 3 characters

    // Count occurrences of 'o'
    for (int i = 0; i < 3; i++) {
        if (input[i] == 'o') {
            count_o++;
        }
    }

    // Calculate the score based on the count of 'o'
    score = 700 + (100 * count_o);

    // Output the calculated score
    printf("%d\n", score);

    return 0; // Indicate successful completion
}
```
