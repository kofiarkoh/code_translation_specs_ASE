#include <stdio.h>
#include <stdlib.h>

int modified_absolute_value(int initial_value, int max_decrements, int decrement_value) {
    // Preconditions check: decrement_value must be greater than zero
    if (decrement_value <= 0) {
        // Undefined behavior as per specification
        return 0; // Placeholder for undefined behavior
    }

    // Calculate the total possible decrement
    int total_decrement = max_decrements * decrement_value;

    // Adjust the initial value based on the total decrement
    if (initial_value > total_decrement) {
        initial_value -= total_decrement; // Apply total decrement
    } else {
        initial_value = 0; // If total decrement is greater than initial value, it becomes zero
    }

    // The output must be the absolute value, which is already non-negative
    return initial_value; // Return the modified absolute value
}

int main() {
    // Example usage
    int initial_value = 10;
    int max_decrements = 3;
    int decrement_value = 4;

    int result = modified_absolute_value(initial_value, max_decrements, decrement_value);
    printf("Modified absolute value: %d\n", result); // Expected output: 0

}
