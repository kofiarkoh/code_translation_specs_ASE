#include <stdio.h>

int main() {
    int num_pairs;
    
    // Read the number of pairs; must be non-negative
    if (scanf("%d", &num_pairs) != 1 || num_pairs < 0) {
        // Undefined behavior if input is negative or not an integer
        return 1; // Exit with error
    }

    for (int i = 0; i < num_pairs; i++) {
        int x, y;
        
        // Read each pair of integers
        if (scanf("%d %d", &x, &y) != 2) {
            // Undefined behavior if input does not conform to expected format
            return 1; // Exit with error
        }

        // Compute the derived value using the specified formula
        // (x - 1) % y + 1
        // Ensure y is positive to avoid division/modulo by zero
        if (y <= 0) {
            return 1; // Exit with error for invalid y
        }

        int result = (x - 1) % y + 1;

        // Output the result for the current pair
        printf("%d\n", result);
    }

    return 0; // Successful execution
}
