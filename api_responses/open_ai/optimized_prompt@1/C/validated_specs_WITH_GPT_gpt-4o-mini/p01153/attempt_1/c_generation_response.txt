#include <stdio.h>

int compute_max_value(int *array, int n) {
    int max_value = 0; // Initialize max_value to 0 as per postconditions

    // Iterate through all combinations of three elements in the array
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            for (int k = 0; k < n; k++) {
                // Calculate the value based on the relationship defined
                // Here we assume a simple relationship: sum of three elements
                int current_value = array[i] + array[j] + array[k];
                
                // Update max_value if current_value is greater
                if (current_value > max_value) {
                    max_value = current_value;
                }
            }
        }
    }
    
    return max_value; // Return the maximum computed value
}

int main() {
    int test_cases;
    scanf("%d", &test_cases); // Read the number of test cases

    for (int t = 0; t < test_cases; t++) {
        int n;
        scanf("%d", &n); // Read the number of elements in the array

        int array[n]; // Declare the array with size n
        for (int i = 0; i < n; i++) {
            scanf("%d", &array[i]); // Read each element of the array
        }

        // Compute and print the maximum value for the current test case
        int result = compute_max_value(array, n);
        printf("%d\n", result); // Output the result
    }

    return 0; // Indicate successful completion
}
