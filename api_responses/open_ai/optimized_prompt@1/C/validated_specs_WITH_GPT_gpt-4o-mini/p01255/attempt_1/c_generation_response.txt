// Function to find the minimum number of colors needed
int minColors(Segment* segments, int n) {
    int* colors = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        colors[i] = -1; // Initialize all segments with no color
    }

    // Assign colors
    for (int i = 0; i < n; i++) {
        // Check colors of segments that intersect with segment i
        int usedColors[n]; // Track used colors for segment i
        for (int j = 0; j < n; j++) {
            usedColors[j] = 0; // Initialize used colors
        }

        for (int j = 0; j < n; j++) {
            if (i != j && doIntersect(segments[i], segments[j]) && colors[j] != -1) {
                usedColors[colors[j]] = 1; // Mark color as used
            }
        }

        // Find the first unused color
        int color;
        for (color = 0; color < n; color++) {
            if (!usedColors[color]) {
                break; // Found an unused color
            }
        }
        colors[i] = color; // Assign the color to segment i
    }

    // Determine the maximum color used
    int maxColor = 0;
    for (int i = 0; i < n; i++) {
        if (colors[i] > maxColor) {
            maxColor = colors[i];
        }
    }

    free(colors); // Free allocated memory
    return maxColor + 1; // Return the number of colors used (1-indexed)
}
