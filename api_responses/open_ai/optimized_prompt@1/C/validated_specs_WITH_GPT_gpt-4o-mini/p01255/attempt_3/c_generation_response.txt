#include <stdio.h>
#include <stdlib.h>

#define MAX_SEGMENTS 100 // Define a maximum number of segments

typedef struct {
    int x1, y1; // Start point
    int x2, y2; // End point
} Segment;

// Function to check if two segments intersect
int doIntersect(Segment s1, Segment s2) {
    // Check if the segments intersect or touch at endpoints
    // Using orientation method to determine intersection
    int o1 = (s1.x1 - s2.x1) * (s2.y2 - s2.y1) - (s1.y1 - s2.y1) * (s2.x2 - s2.x1);
    int o2 = (s1.x2 - s2.x1) * (s2.y2 - s2.y1) - (s1.y2 - s2.y1) * (s2.x2 - s2.x1);
    int o3 = (s2.x1 - s1.x1) * (s1.y2 - s1.y1) - (s2.y1 - s1.y1) * (s1.x2 - s1.x1);
    int o4 = (s2.x2 - s1.x1) * (s1.y2 - s1.y1) - (s2.y2 - s1.y1) * (s1.x2 - s1.x1);

    // General case
    if (o1 != 0 && o2 != 0 && o3 != 0 && o4 != 0) {
        return (o1 * o2 < 0) && (o3 * o4 < 0);
    }

    // Special case: Check for collinear points (endpoints touching)
    if (o1 == 0 && (s1.x1 >= s2.x1 && s1.x1 <= s2.x2 || s1.x2 >= s2.x1 && s1.x2 <= s2.x2)) return 1;
    if (o2 == 0 && (s1.x1 >= s2.x1 && s1.x1 <= s2.x2 || s1.x2 >= s2.x1 && s1.x2 <= s2.x2)) return 1;
    if (o3 == 0 && (s2.x1 >= s1.x1 && s2.x1 <= s1.x2 || s2.x2 >= s1.x1 && s2.x2 <= s1.x2)) return 1;
    if (o4 == 0 && (s2.x1 >= s1.x1 && s2.x1 <= s1.x2 || s2.x2 >= s1.x1 && s2.x2 <= s1.x2)) return 1;

    return 0; // No intersection
}

// Function to find the minimum number of colors needed
int minColors(Segment segments[], int n) {
    int *colors = (int *)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) colors[i] = -1; // Initialize colors to -1 (uncolored)

    for (int i = 0; i < n; i++) {
        // Check colors of adjacent segments
        int usedColors[MAX_SEGMENTS] = {0}; // Track used colors
        for (int j = 0; j < n; j++) {
            if (i != j && doIntersect(segments[i], segments[j]) && colors[j] != -1) {
                usedColors[colors[j]] = 1; // Mark color as used
            }
        }

        // Assign the first available color
        for (int c = 0; c < n; c++) {
            if (!usedColors[c]) {
                colors[i] = c; // Assign color
                break;
            }
        }
    }

    // Find the maximum color used
    int maxColor = 0;
    for (int i = 0; i < n; i++) {
        if (colors[i] > maxColor) {
            maxColor = colors[i];
        }
    }

    free(colors);
    return maxColor + 1; // Return number of colors (0-indexed)
}

int main() {
    int n;
    scanf("%d", &n); // Read number of segments
    if (n < 1 || n > MAX_SEGMENTS) return 1; // Check for valid number of segments

    Segment segments[MAX_SEGMENTS];
    for (int i = 0; i < n; i++) {
        scanf("%d %d %d %d", &segments[i].x1, &segments[i].y1, &segments[i].x2, &segments[i].y2);
    }

    int result = minColors(segments, n);
    printf("%d\n", result); // Output the minimum number of colors
    return 0;
}
