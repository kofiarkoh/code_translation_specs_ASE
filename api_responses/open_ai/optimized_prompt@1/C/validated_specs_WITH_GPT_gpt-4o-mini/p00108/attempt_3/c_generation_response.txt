#include <stdio.h>

#define MAX_SIZE 101 // Maximum size for the integer values (0 to 100)

void transform_sequence(int *sequence, int n, int *transform_count) {
    int counts[MAX_SIZE] = {0}; // Array to hold counts of each integer
    int stable = 0; // Flag to check if the sequence is stable
    int i;

    // Initialize the transformation count
    *transform_count = 0;

    while (!stable) {
        stable = 1; // Assume stable until proven otherwise

        // Count occurrences of each integer in the sequence
        for (i = 0; i < n; i++) {
            if (sequence[i] < MAX_SIZE) {
                counts[sequence[i]]++;
            }
        }

        // Create a new transformed sequence based on counts
        int new_sequence[MAX_SIZE] = {0};
        for (i = 0; i < n; i++) {
            new_sequence[i] = counts[sequence[i]];
            // If the new value differs from the old, it's not stable
            if (new_sequence[i] != sequence[i]) {
                stable = 0;
            }
        }

        // Update the original sequence with the new transformed values
        for (i = 0; i < n; i++) {
            sequence[i] = new_sequence[i];
        }

        // Reset counts for the next transformation
        for (i = 0; i < MAX_SIZE; i++) {
            counts[i] = 0;
        }

        // Increment the transformation count
        (*transform_count)++;
    }
}

int main() {
    int n;
    while (1) {
        // Read the size of the sequence
        scanf("%d", &n);
        if (n == 0) {
            break; // Termination condition
        }

        int sequence[MAX_SIZE] = {0}; // Array to hold the sequence
        for (int i = 0; i < n; i++) {
            scanf("%d", &sequence[i]); // Read each integer in the sequence
        }

        int transform_count;
        transform_sequence(sequence, n, &transform_count); // Transform the sequence

        // Output the number of transformations and the final sequence
        printf("%d\n", transform_count);
        for (int i = 0; i < n; i++) {
            printf("%d ", sequence[i]);
        }
        printf("\n");
    }
    return 0;
}
