void quicksort(card *A,int p,int r){

  int q;
  if(p<r){
    q=partition(A,p,r);
    quicksort(A,p,q-1);
    quicksort(A,q+1,r);
  }
}
int partition(card *A,int p,int r){

  int i,j;
  int x;
  card swap;

  i=p-1;
  x=A[r].val;

  for(j=p;j<r;j++){
    if(A[j].val<=x){
      i++;
      swap=A[i];
      A[i]=A[j];
      A[j]=swap;
    }
  }

  A[i+1]=A[r];
