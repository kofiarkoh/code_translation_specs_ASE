#include <stdio.h>

int count_distinct_pairs(int N, int sequence[]) {
    int count = 0; // Initialize the count of distinct pairs
    int i = 0; // Index for traversing the sequence

    while (i < N - 1) { // Ensure we don't go out of bounds
        if (sequence[i] == sequence[i + 1]) { // Check for a pair
            count++; // Increment the count for the found pair
            i += 2; // Skip the next integer
        } else {
            i++; // Move to the next integer
        }
    }

    return count; // Return the total count of distinct pairs
}

int main() {
    int N; // Number of integers in the sequence
    scanf("%d", &N); // Read the number of integers

    int sequence[N]; // Declare the sequence array

    // Read the sequence of integers
    for (int i = 0; i < N; i++) {
        scanf("%d", &sequence[i]);
    }

    int result = count_distinct_pairs(N, sequence); // Count the pairs
    printf("%d\n", result); // Output the result

    return 0; // Indicate successful completion
}
