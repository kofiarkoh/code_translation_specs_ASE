#include <stdio.h>

int count_modified_ones(int M, int N, int matrix[M][N]) {
    int count = 0; // Initialize count of modified '1's

    // Process each row of the matrix
    for (int i = 0; i < M; i++) {
        for (int j = 0; j < N; j++) {
            // Check if the current element is '1'
            if (matrix[i][j] == 1) {
                count++; // Increment count for each '1' encountered
                matrix[i][j] = 0; // Modify the '1' to '0' as per specification
            }
        }
    }

    return count; // Return the total count of modified '1's
}

int main() {
    int M, N;
    // Read dimensions of the matrix
    scanf("%d %d", &M, &N);
    
    int matrix[M][N]; // Declare the matrix

    // Read the matrix elements
    for (int i = 0; i < M; i++) {
        for (int j = 0; j < N; j++) {
            scanf("%d", &matrix[i][j]);
        }
    }

    // Call the function to count modified '1's and print the result
    int result = count_modified_ones(M, N, matrix);
    printf("%d\n", result); // Output the count of modified '1's

    return 0; // Indicate successful completion
}
