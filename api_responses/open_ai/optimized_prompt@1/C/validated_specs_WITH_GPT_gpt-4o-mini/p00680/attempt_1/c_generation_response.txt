int gauss_jordan(double A[MAX_NODES][MAX_NODES], double B[MAX_NODES], int n) {
    for (int i = 0; i < n; i++) {
        // Find pivot
        double max_val = 0;
        int pivot_row = -1;
        for (int j = i; j < n; j++) {
            if (fabs(A[j][i]) > max_val) {
                max_val = fabs(A[j][i]);
                pivot_row = j;
            }
        }
        if (pivot_row == -1 || fabs(max_val) < 1e-9) {
            return 0; // Singular matrix
        }
        // Swap rows
        if (pivot_row != i) {
            for (int j = 0; j < n; j++) {
                double temp = A[i][j];
                A[i][j] = A[pivot_row][j];
                A[pivot_row][j] = temp;
            }
            double temp = B[i];
            B[i] = B[pivot_row];
            B[pivot_row] = temp;
        }
        // Eliminate
        for (int j = 0; j < n; j++) {
            if (j != i) {
                double factor = A[j][i] / A[i][i];
                for (int k = 0; k < n; k++) {
                    A[j][k] -= factor * A[i][k];
                }
                B[j] -= factor * B[i];
            }
        }
    }
    return 1; // Success
}
