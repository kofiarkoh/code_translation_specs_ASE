#include <stdio.h>

void right_rotate(int arr[], int b, int m, int e) {
    int len = e - b; // Length of the segment to rotate
    if (len <= 0) return; // No rotation needed if segment is empty

    m = m % len; // Effective rotation
    if (m == 0) return; // No rotation needed if m is 0

    // Create a temporary array to hold the rotated values
    int temp[len];
    for (int i = 0; i < len; i++) {
        temp[(i + m) % len] = arr[b + i]; // Rotate and wrap around
    }

    // Copy the rotated values back to the original array
    for (int i = 0; i < len; i++) {
        arr[b + i] = temp[i];
    }
}

int main() {
    int n; // Number of elements in the array
    scanf("%d", &n); // Read the size of the array

    int arr[1000]; // Array to hold the integers (max size as per spec)
    for (int i = 0; i < n; i++) {
        scanf("%d", &arr[i]); // Read the array elements
    }

    int q; // Number of rotation queries
    scanf("%d", &q); // Read the number of queries

    for (int i = 0; i < q; i++) {
        int b, m, e; // Starting index, number of positions to rotate, ending index
        scanf("%d %d %d", &b, &m, &e); // Read each query
        right_rotate(arr, b, m, e); // Perform the rotation
    }

    // Output the modified array
    for (int i = 0; i < n; i++) {
        printf("%d", arr[i]);
        if (i < n - 1) {
            printf(" "); // Print space between numbers
        }
    }
    printf("\n"); // New line at the end of output

    return 0;
}
