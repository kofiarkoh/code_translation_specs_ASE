#include <ctype.h>

#define LEFT  -1
#define RIGHT -2
#define PLUS   0 
#define MINUS  1
#define MUL    2

typedef struct { long long v; char id; } T;
T stack[MAX+3], buf[MAX+3]; int top, end;

int tbl[13][3] = {

	if (c == '-') return MINUS;
	return MUL;
}

int rank(int op)
