#include <stdio.h>
#include <stdlib.h>

#define N 1000000007 // Predefined constant for modulus

// Function to compute factorial modulo N
void compute_factorials(int *factorials, int max_n) {
    factorials[0] = 1; // 0! = 1
    for (int i = 1; i <= max_n; i++) {
        factorials[i] = (factorials[i - 1] * i) % N; // Compute factorials modulo N
    }
}

// Function to compute modular inverse using Fermat's Little Theorem
int mod_inverse(int a) {
    int result = 1;
    int exponent = N - 2; // Since N is prime
    while (exponent > 0) {
        if (exponent % 2 == 1) {
            result = (result * a) % N;
        }
        a = (a * a) % N;
        exponent /= 2;
    }
    return result;
}

// Function to compute combinations C(n, r) modulo N
int combination(int n, int r, int *factorials) {
    if (n < r || r < 0) {
        return 0; // Invalid combination
    }
    return (factorials[n] * mod_inverse(factorials[r]) % N * mod_inverse(factorials[n - r]) % N) % N;
}

// Main function to compute the combinatorial value based on x, y, and k
int compute_combinatorial(int x, int y, int k) {
    if (x < 0 || y < 0 || k < 0) {
        return 0; // Invalid input
    }
    
    int n = x + y + 2 * k; // Maximum value for factorials
    int *factorials = (int *)malloc((n + 1) * sizeof(int));
    if (!factorials) {
        return 0; // Memory allocation failure
    }

    compute_factorials(factorials, n); // Populate factorials

    int result = 0; // Initialize result
    for (int i = 0; i <= k; i++) {
        int xgr = combination(x + 2 * (k - i), k - i, factorials); // Compute xgr
        int ygr = combination(y + 2 * i, i, factorials); // Compute ygr
        result = (result + (xgr * ygr) % N) % N; // Accumulate result
    }

    free(factorials); // Free allocated memory
    return result; // Return final result
}

// Example usage
int main() {
    int x = 5, y = 3, k = 2; // Example inputs
    int result = compute_combinatorial(x, y, k);
    printf("%d\n", result); // Output the result
    return 0;
}
