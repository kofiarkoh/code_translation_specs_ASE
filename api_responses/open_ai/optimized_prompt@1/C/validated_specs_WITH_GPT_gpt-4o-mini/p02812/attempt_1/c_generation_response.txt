#include <stdio.h>

int count_ABC_occurrences(int n, const char *str) {
    // Preconditions check: n must be at least 3
    if (n < 3) {
        // Undefined behavior as per specification
        return -1; // Indicating an error condition
    }

    // Initialize count of occurrences
    int count = 0;

    // Loop through the string while ensuring we do not exceed bounds defined by n
    for (int i = 0; i <= n - 3; i++) {
        // Check for the substring "ABC"
        if (str[i] == 'A' && str[i + 1] == 'B' && str[i + 2] == 'C') {
            count++; // Increment count when "ABC" is found
        }
    }

    // Postcondition: count is non-negative and reflects the total occurrences
    return count;
}

int main() {
    // Example usage
    const char *test_str = "ABCABCABC";
    int n = 9; // Length of the string

    // Call the function and print the result
    int result = count_ABC_occurrences(n, test_str);
    printf("Occurrences of 'ABC': %d\n", result);

    return 0;
}
