#include <stdio.h>

typedef struct {
    int x;
    int y;
} Point;

// Function to calculate the area of a triangle given three points
int calculate_area(Point a, Point b, Point c) {
    // Area = 0.5 * | ax(by - cy) + bx(cy - ay) + cx(ay - by) |
    return (a.x * (b.y - c.y) + b.x * (c.y - a.y) + c.x * (a.y - b.y));
}

// Function to check if two line segments intersect
int do_intersect(Point p1, Point q1, Point p2, Point q2) {
    // Using orientation to check for intersection
    // This is a simplified version and assumes no collinear points
    int o1 = (q1.y - p1.y) * (p2.x - p1.x) - (q1.x - p1.x) * (p2.y - p1.y);
    int o2 = (q1.y - p1.y) * (q2.x - p1.x) - (q1.x - p1.x) * (q2.y - p1.y);
    int o3 = (q2.y - p2.y) * (p1.x - p2.x) - (q2.x - p2.x) * (p1.y - p2.y);
    int o4 = (q2.y - p2.y) * (q1.x - p2.x) - (q2.x - p2.x) * (q1.y - p2.y);
    
    return (o1 * o2 < 0) && (o3 * o4 < 0);
}

// Function to classify the area
const char* classify_area(int area) {
    if (area < 0) area = -area; // Ensure area is positive
    if (area == 0) return "kyo"; // No area
    if (area < 10) return "syo-kichi";
    if (area < 50) return "kichi";
    if (area < 100) return "chu-kichi";
    return "dai-kichi";
}

int main() {
    Point a, b, c, d;

        return 1; // Invalid input
    }
    if (scanf("%d %d", &d.x, &d.y) != 2) {
        return 1; // Invalid input
    }

    // Check for intersection
    if (do_intersect(a, b, c, d)) {
        int area = calculate_area(a, b, c);
        const char* classification = classify_area(area);
        printf("%s\n", classification);
    } else {
        printf("kyo\n"); // No intersection
    }

    return 0;
}
