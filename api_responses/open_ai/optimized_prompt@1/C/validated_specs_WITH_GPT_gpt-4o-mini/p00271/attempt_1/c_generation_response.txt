```c
#include <stdio.h>

int main() {
    int a, b; // Variables to hold the input integers
    int differences[7]; // Array to store the differences

    // Read exactly seven pairs of integers
    for (int i = 0; i < 7; i++) {
        // Read a pair of integers from input
        scanf("%d %d", &a, &b);
        // Calculate the difference and store it
        differences[i] = a - b; // Output the difference
    }

    // Output the differences
    for (int i = 0; i < 7; i++) {
        printf("%d\n", differences[i]); // Print each difference
    }

    return 0; // Indicate successful completion
}
```
