```c
#include <stdio.h>

int main() {
    int N;
    // Read the number of boolean values
    if (scanf("%d", &N) != 1 || N <= 0) {
        return 1; // Error if N is not read successfully or is not positive
    }

    char input;
    char result = 'T'; // Start with true as the cumulative result

    // Process each boolean character
    for (int i = 0; i < N; i++) {
        if (scanf(" %c", &input) != 1) {
            return 1; // Error if character input is not read successfully
        }
        // Only process 'T' and 'F'
        if (input == 'F') {
            result = 'F'; // If any 'F' is found, the result is false
        }
        // If 'T' is found, we keep the result as 'T' unless an 'F' is encountered
    }

    // Output the final result
    printf("%c\n", result);
    return 0; // Successful termination
}
```
