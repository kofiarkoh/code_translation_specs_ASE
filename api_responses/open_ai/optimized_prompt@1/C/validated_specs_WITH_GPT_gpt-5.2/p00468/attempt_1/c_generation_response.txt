#include <stdio.h>
#include <string.h>

int main(void) {
    int n, m;

    /* Adjacency matrix sized for up to 500 individuals (1-based indexing). */
    static unsigned char adj[501][501];

    while (scanf("%d %d", &n, &m) == 2) {
        if (n == 0 && m == 0) {
            break;
        }

        /* Clear only the portion we will use for this dataset. */
        for (int i = 1; i <= n; i++) {
            memset(adj[i] + 1, 0, (size_t)n * sizeof(adj[i][1]));
        }

        for (int i = 0; i < m; i++) {
            int a, b;
            scanf("%d %d", &a, &b);
            /* Per spec, behavior is unspecified if out of bounds; we do no checks. */
            adj[a][b] = 1;
            adj[b][a] = 1;
        }

        /* Mark individuals within two steps of 1 (excluding 1). */
        unsigned char within2[501];
        memset(within2, 0, sizeof(within2));

        /* Direct neighbors of 1. */
        for (int x = 2; x <= n; x++) {
            if (adj[1][x]) within2[x] = 1;
        }

        /* Two-step via an intermediate y that is directly connected to 1. */
        for (int y = 2; y <= n; y++) {
            if (!adj[1][y]) continue;
            for (int x = 2; x <= n; x++) {
                if (adj[y][x]) within2[x] = 1;
            }
        }

        int count = 0;
        for (int x = 2; x <= n; x++) {
            if (within2[x]) count++;
        }

        printf("%d\n", count);
    }

    return 0;
}
