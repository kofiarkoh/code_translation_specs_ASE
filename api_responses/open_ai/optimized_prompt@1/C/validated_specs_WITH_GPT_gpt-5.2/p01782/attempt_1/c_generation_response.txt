```c
#include <stdio.h>
#include <string.h>

#define MAXN 52
#define MAXV (2*MAXN + 5)
#define INF  1000000000

static int char_rank(int c) {
    /* Order: 'A'..'Z' then 'a'..'z' */
    if (c >= 'A' && c <= 'Z') return c - 'A';
    if (c >= 'a' && c <= 'z') return 26 + (c - 'a');
    /* Preconditions say alphabetic; unspecified otherwise. */
    return 1000;
}

/* Hopcroft-Karp for bipartite matching: Left size = n, Right size = n.
   Adjacency is given by allowed[u][v] (0/1). */
static int bfs_hk(int n, const unsigned char allowed[MAXN][MAXN],
                  int pairU[MAXN], int pairV[MAXN], int dist[MAXN]) {
    int q[MAXN];
    int qh = 0, qt = 0;
    for (int u = 0; u < n; u++) {
        if (pairU[u] == -1) {
            dist[u] = 0;
            q[qt++] = u;
        } else {
            dist[u] = INF;
        }
    }

    int foundFree = 0;
    while (qh < qt) {
        int u = q[qh++];
        for (int v = 0; v < n; v++) {
            if (!allowed[u][v]) continue;
            int u2 = pairV[v];
            if (u2 == -1) {
                foundFree = 1;
            } else if (dist[u2] == INF) {
                dist[u2] = dist[u] + 1;
                q[qt++] = u2;
            }
        }
    }
    return foundFree;
}

static int dfs_hk(int u, int n, const unsigned char allowed[MAXN][MAXN],
                  int pairU[MAXN], int pairV[MAXN], int dist[MAXN]) {
    for (int v = 0; v < n; v++) {
        if (!allowed[u][v]) continue;
        int u2 = pairV[v];
        if (u2 == -1 || (dist[u2] == dist[u] + 1 && dfs_hk(u2, n, allowed, pairU, pairV, dist))) {
            pairU[u] = v;
            pairV[v] = u;
            return 1;
        }
    }
    dist[u] = INF;
    return 0;
}

static int max_matching(int n, const unsigned char allowed[MAXN][MAXN],
                        int pairU[MAXN], int pairV[MAXN]) {
    int dist[MAXN];
    for (int u = 0; u < n; u++) pairU[u] = -1;
    for (int v = 0; v < n; v++) pairV[v] = -1;

    int matching = 0;
    while (bfs_hk(n, allowed, pairU, pairV, dist)) {
        for (int u = 0; u < n; u++) {
            if (pairU[u] == -1) {
                if (dfs_hk(u, n, allowed, pairU, pairV, dist)) matching++;
            }
        }
    }
    return matching;
}

int main(void) {
    int N;
    if (scanf("%d", &N) != 1) return 0;

    static char grid[MAXN][MAXN];
    /* Read N lines, each with at least N alphabetic characters.
       Use " %c" to skip whitespace/newlines. */
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            char c;
            if (scanf(" %c", &c) != 1) return 0;
            grid[i][j] = c;
        }
    }

    /* We construct the lexicographically minimal multiset (sorted) by
       greedily fixing counts per character rank from smallest to largest.
       For each rank r, we try to force as many edges of rank r as possible
       while still allowing a perfect matching, given previously fixed edges. */

    static unsigned char fixed[MAXN][MAXN];   /* edges forced to be used */
    static unsigned char allowed[MAXN][MAXN]; /* edges currently allowed */
    memset(fixed, 0, sizeof(fixed));

    int counts[52];
    for (int i = 0; i < 52; i++) counts[i] = 0;

    /* Helper arrays for matching */
    int pairU[MAXN], pairV[MAXN];

    for (int r = 0; r < 52; r++) {
        /* Determine which rows/cols are already fixed by earlier ranks. */
        int rowFixed[MAXN], colFixed[MAXN];
        for (int i = 0; i < N; i++) rowFixed[i] = 0;
        for (int j = 0; j < N; j++) colFixed[j] = 0;

        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                if (fixed[i][j]) {
                    rowFixed[i] = 1;
                    colFixed[j] = 1;
                }
            }
        }

        /* Build base allowed graph: must include all fixed edges; for unfixed rows/cols,
           allow edges with rank >= r (since smaller ranks already decided). */
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                if (fixed[i][j]) {
                    allowed[i][j] = 1;
                } else if (rowFixed[i] || colFixed[j]) {
                    allowed[i][j] = 0;
                } else {
                    int rr = char_rank((unsigned char)grid[i][j]);
                    allowed[i][j] = (rr >= r) ? 1 : 0;
                }
            }
        }

        /* Collect candidate edges of exact rank r among currently free rows/cols. */
        int candU[MAXN * MAXN];
        int candV[MAXN * MAXN];
        int candCount = 0;
        for (int i = 0; i < N; i++) {
            if (rowFixed[i]) continue;
            for (int j = 0; j < N; j++) {
                if (colFixed[j]) continue;
                if (char_rank((unsigned char)grid[i][j]) == r) {
                    candU[candCount] = i;
                    candV[candCount] = j;
                    candCount++;
                }
            }
        }

        /* Greedily force edges of rank r when possible, one by one.
           This maximizes the number of rank r characters in the multiset,
           which minimizes the sorted multiset lexicographically. */
        for (int idx = 0; idx < candCount; idx++) {
            int u = candU[idx];
            int v = candV[idx];

            /* If row/col already fixed due to earlier forced edges in this rank, skip. */
            if (rowFixed[u] || colFixed[v]) continue;

            /* Try forcing (u,v): create a test allowed graph where u and v are fixed to each other. */
            static unsigned char testAllowed[MAXN][MAXN];
            for (int i = 0; i < N; i++) {
                for (int j = 0; j < N; j++) testAllowed[i][j] = allowed[i][j];
            }

            /* Enforce u->v and v<-u by removing other edges incident to u or v,
               but keep any already fixed edges (there should be none incident to u/v if free). */
            for (int j = 0; j < N; j++) {
                if (j != v) testAllowed[u][j] = 0;
            }
            for (int i = 0; i < N; i++) {
                if (i != u) testAllowed[i][v] = 0;
            }
            testAllowed[u][v] = 1;

            int m = max_matching(N, testAllowed, pairU, pairV);
            if (m == N) {
                /* Commit the force. */
                fixed[u][v] = 1;
                counts[r]++;

                rowFixed[u] = 1;
                colFixed[v] = 1;

                /* Update allowed to reflect the new fixation. */
                for (int j = 0; j < N; j++) {
                    if (j != v) allowed[u][j] = 0;
                }
                for (int i = 0; i < N; i++) {
                    if (i != u) allowed[i][v] = 0;
                }
                allowed[u][v] = 1;
            }
        }
    }

    /* At this point, fixed contains some forced edges; complete to a perfect matching
       using any remaining edges (all ranks allowed now). */
    int rowFixed[MAXN], colFixed[MAXN];
    for (int i = 0; i < N; i++) rowFixed[i] = 0;
    for (int j = 0; j < N; j++) colFixed[j] = 0;
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            if (fixed[i][j]) {
                rowFixed[i] = 1;
                colFixed[j] = 1;
            }
        }
    }

    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            if (fixed[i][j]) {
                allowed[i][j] = 1;
            } else if (rowFixed[i] || colFixed[j]) {
                allowed[i][j] = 0;
            } else {
                allowed[i][j] = 1; /* any remaining edge allowed */
            }
        }
    }

    (void)max_matching(N, allowed, pairU, pairV);

    /* Count characters from the final matching (pairU gives chosen column for each row). */
    int total = 0;
    for (int u = 0; u < N; u++) {
        int v = pairU[u];
        if (v >= 0) {
            int r = char_rank((unsigned char)grid[u][v]);
            if (r >= 0 && r < 52) counts[r]++;
            total++;
        }
    }

    /* Output sorted multiset as specified. */
    char out[MAXN + 1];
    int pos = 0;
    for (int r = 0; r < 52; r++) {
        int k = counts[r];
        while (k-- > 0 && pos < N) {
            if (r < 26) out[pos++] = (char)('A' + r);
            else out[pos++] = (char)('a' + (r - 26));
        }
    }
    /* Preconditions guarantee perfect matching; ensure N chars if possible. */
    while (pos < N) out[pos++] = 'A';
    out[N] = '\0';

    puts(out);
    return 0;
}
```
