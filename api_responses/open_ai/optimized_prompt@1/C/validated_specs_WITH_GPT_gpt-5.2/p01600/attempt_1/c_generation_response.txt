#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

static int iabs_int(int v) { return v < 0 ? -v : v; }

int main(void) {
    int n;
    if (scanf("%d", &n) != 1) return 0;

    int *x = (int *)malloc((size_t)n * sizeof(int));
    int *y = (int *)malloc((size_t)n * sizeof(int));
    if (!x || !y) return 0;

    for (int i = 0; i < n; i++) {
        if (scanf("%d %d", &x[i], &y[i]) != 2) {
            free(x);
            free(y);
            return 0;
        }
    }

    if (n <= 1) {
        printf("0");
        free(x);
        free(y);
        return 0;
    }

    /* dp[i][k] = minimum cost to build tree over contiguous interval [i..k] */
    int *dp = (int *)malloc((size_t)n * (size_t)n * sizeof(int));
    if (!dp) {
        free(x);
        free(y);
        return 0;
    }

    /* Initialize base cases and others to a large value */
    for (int i = 0; i < n; i++) {
        for (int k = 0; k < n; k++) {
            dp[i * n + k] = (i == k) ? 0 : INT_MAX / 4;
        }
    }

    /* Increasing interval length */
    for (int len = 2; len <= n; len++) {
        for (int i = 0; i + len - 1 < n; i++) {
            int k = i + len - 1;
            int best = INT_MAX / 4;

            for (int j = i; j < k; j++) {
                /* Merge cost for [i..j] and [j+1..k]:
                   |x_i - x_{j+1}| + |y_j - y_k|  (as specified) */
                int merge_cost = iabs_int(x[i] - x[j + 1]) + iabs_int(y[j] - y[k]);

                int left = dp[i * n + j];
                int right = dp[(j + 1) * n + k];

                int total = left + right + merge_cost;
                if (total < best) best = total;
            }

            dp[i * n + k] = best;
        }
    }

    printf("%d", dp[0 * n + (n - 1)]);

    free(dp);
    free(x);
    free(y);
    return 0;
}
