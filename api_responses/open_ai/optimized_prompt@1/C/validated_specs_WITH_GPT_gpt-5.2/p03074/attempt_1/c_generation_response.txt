#include <stdio.h>
#include <string.h>

int main(void) {
    int N, K;
    if (scanf("%d %d", &N, &K) != 2) return 0;

    static char s[100000 + 5];
    if (scanf("%100004s", s) != 1) return 0;

    /* Build run-length representation of the first N bits.
       We store lengths of runs in order. */
    static int run[2 * 100000 + 5];
    int m = 0;

    if (N <= 0) {
        printf("0\n");
        return 0;
    }

    int cur = 1;
    for (int i = 1; i < N; i++) {
        if (s[i] == s[i - 1]) {
            cur++;
        } else {
            run[m++] = cur;
            cur = 1;
        }
    }
    run[m++] = cur;

    /* Ensure runs start and end with a '1' run by padding with zero-length runs if needed.
       This allows considering segments that start/end at the string boundaries. */
    static int a[2 * 100000 + 7];
    int L = 0;

    if (s[0] == '0') a[L++] = 0;
    for (int i = 0; i < m; i++) a[L++] = run[i];
    if (s[N - 1] == '0') a[L++] = 0;

    /* Sliding window over runs: consider windows of at most (2K+1) runs,
       starting at even indices (1-runs) in this padded representation. */
    int maxRuns = 2 * K + 1;
    if (maxRuns > L) maxRuns = L;

    int ans = 0;
    int sum = 0;

    /* Initialize window [0, maxRuns) */
    for (int i = 0; i < maxRuns; i++) sum += a[i];
    ans = sum;

    for (int left = 0; left + maxRuns < L; left += 2) {
        /* Move window by 2 runs to keep starting on a 1-run index */
        sum -= a[left];
        sum -= a[left + 1];
        sum += a[left + maxRuns];
        if (left + maxRuns + 1 < L) sum += a[left + maxRuns + 1];
        if (sum > ans) ans = sum;
    }

    if (ans < 0) ans = 0;
    if (ans > N) ans = N;
    printf("%d\n", ans);
    return 0;
}
