```c
#include <stdio.h>
#include <string.h>

#define GRID_W 101
#define GRID_H 101
#define START_X 50
#define START_Y 50

static int in_bounds(int x, int y) {
    return (x >= 0 && x < GRID_W && y >= 0 && y < GRID_H);
}

int main(void) {
    int n;

    while (scanf("%d", &n) == 1) {
        if (n == 0) break;

        /* Fixed, small bounded grid representation: 1 means target present, 0 absent. */
        unsigned char grid[GRID_H][GRID_W];
        memset(grid, 0, sizeof(grid));

        int remaining = 0;

        /* Place targets; duplicates are not counted multiple times. */
        for (int i = 0; i < n; i++) {
            int x, y;
            if (scanf("%d %d", &x, &y) != 2) return 0;
            /* Preconditions say coordinates are within bounds; otherwise behavior undefined. */
            if (in_bounds(x, y) && grid[y][x] == 0) {
                grid[y][x] = 1;
                remaining++;
            }
        }

        int x = START_X, y = START_Y;

        /* Collect at initial position before any movement. */
        if (in_bounds(x, y) && grid[y][x]) {
            grid[y][x] = 0;
            remaining--;
        }

        int m;
        if (scanf("%d", &m) != 1) return 0;

        for (int i = 0; i < m; i++) {
            char dir[32];
            int steps;
            if (scanf("%31s %d", dir, &steps) != 2) return 0;

            int dx = 0, dy = 0;
            /* Any other token is treated as west. */
            if (strcmp(dir, "north") == 0) { dx = 0; dy = -1; }
            else if (strcmp(dir, "east") == 0) { dx = 1; dy = 0; }
            else if (strcmp(dir, "south") == 0) { dx = 0; dy = 1; }
            else { dx = -1; dy = 0; } /* west or any other token */

            /* Step counts are nonnegative per preconditions. */
            for (int s = 0; s < steps; s++) {
                x += dx;
                y += dy;

                /* Preconditions say all reached positions are within bounds. */
                if (in_bounds(x, y) && grid[y][x]) {
                    grid[y][x] = 0;
                    remaining--;
                }
            }
        }

        puts(remaining == 0 ? "Yes" : "No");
    }

    return 0;
}
```
