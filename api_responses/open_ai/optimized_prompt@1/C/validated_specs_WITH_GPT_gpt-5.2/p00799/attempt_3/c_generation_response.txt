```c
#include <stdio.h>

#define MAXN 102
#define MAXC 51
#define QCAP 102

typedef struct {
    int u[MAXC];
    int c[MAXC];
    int len;        /* number of (u,c) pairs */
    int idx;        /* current cycle index */
    int state;      /* 0 = Active, 1 = Charging */
    int rem;        /* remaining time in current state */
} Device;

int main(void) {
    int n;
    int lim;

    while (scanf("%d %d", &n, &lim) == 2) {
        if (n <= 0) {
            /* lim is read but ignored for termination purposes */
            break;
        }

        if (n > MAXN) {
            /* Behavior unspecified by spec; still attempt to read input for n devices.
               We will only store up to MAXN devices; extra devices are read/discarded. */
        }

        Device dev[MAXN];
        int i;

        for (i = 0; i < MAXN; i++) {
            dev[i].len = 0;
            dev[i].idx = 0;
            dev[i].state = 0;
            dev[i].rem = 0;
        }

        for (i = 0; i < n; i++) {
            int u;
            int c;
            int stored = (i < MAXN);

            int len = 0;
            while (scanf("%d", &u) == 1) {
                if (u <= 0) break;
                if (scanf("%d", &c) != 1) return 0;

                if (stored) {
                    if (len < MAXC) {
                        dev[i].u[len] = u;
                        dev[i].c[len] = c;
                        len++;
                    } else {
                        /* Exceeds fixed storage; unspecified. Discard extra pairs. */
                    }
                }
            }

            if (stored) {
                dev[i].len = len;
                dev[i].idx = 0;
                dev[i].state = 0; /* Active */
                /* Spec assumes at least one pair is provided; if not, rem stays 0. */
                if (len > 0) dev[i].rem = dev[i].u[0];
            }
        }

        /* If lim <= 0, perform zero time steps and output zero. */
        if (lim <= 0) {
            printf("0\n");
            continue;
        }

        /* Fixed-size circular buffer queue with overwrite-on-wrap semantics.
           No full-queue detection: enqueue always writes at tail and advances tail mod QCAP. */
        int q[QCAP];
        int qh = 0; /* head index */
        int qt = 0; /* tail index */

        long long misses = 0;

        /* Per-time-unit simulation */
        for (int t = 1; t <= lim; t++) {
            /* Track which devices just entered charging in step (A) */
            unsigned char justEntered[MAXN];
            for (i = 0; i < MAXN; i++) justEntered[i] = 0;

            /* (A) Advance active devices and enqueue new chargers */
            int upto = (n < MAXN) ? n : MAXN;
            for (i = 0; i < upto; i++) {
                if (dev[i].state == 0) { /* Active */
                    dev[i].rem -= 1;
                    if (dev[i].rem == 0) {
                        dev[i].state = 1; /* Charging */
                        if (dev[i].len > 0) {
                            dev[i].rem = dev[i].c[dev[i].idx];
                        } else {
                            dev[i].rem = 0;
                        }
                        justEntered[i] = 1;

                        /* enqueue with overwrite semantics */
                        q[qt] = i;
                        qt = (qt + 1) % QCAP;
                    }
                }
            }

            /* Determine if queue is empty after (A) */
            if (qh == qt) {
                /* (B) no misses, (C) no service */
                continue;
            }

            /* (B) Count misses among devices already charging before step (A)
               i.e., charging now but not justEntered this time unit. */
            int front = q[qh];
            for (i = 0; i < upto; i++) {
                if (dev[i].state == 1 && !justEntered[i]) {
                    if (i != front) misses += 1;
                }
            }

            /* (C) Apply one unit of charging service to the queue front */
            dev[front].rem -= 1;
            if (dev[front].rem == 0) {
                /* dequeue */
                qh = (qh + 1) % QCAP;

                /* switch to Active, advance cycle index, set remaining active time */
                dev[front].state = 0;
                if (dev[front].len > 0) {
                    dev[front].idx = (dev[front].idx + 1) % dev[front].len;
                    dev[front].rem = dev[front].u[dev[front].idx];
                } else {
                    dev[front].rem = 0;
                }
            } else {
                /* remains at front; queue indices unchanged */
            }
        }

        printf("%lld\n", misses);
    }

    return 0;
}
```
