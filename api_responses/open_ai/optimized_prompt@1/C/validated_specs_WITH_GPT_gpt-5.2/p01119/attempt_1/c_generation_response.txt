#include <stdio.h>
#include <stdint.h>

#define MAX_N 100
#define MAX_M 10
#define MAX_S 60000 /* upper bound for 3^10 = 59049 */

static int S[MAX_S];
static int S_count;

static int cmp_int(const void *a, const void *b) {
    int ia = *(const int *)a;
    int ib = *(const int *)b;
    return (ia > ib) - (ia < ib);
}

static int binary_search_int(const int *arr, int n, int x) {
    int lo = 0, hi = n - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        int v = arr[mid];
        if (v == x) return 1;
        if (v < x) lo = mid + 1;
        else hi = mid - 1;
    }
    return 0;
}

/* Enumerate all placements of m weights: each weight contributes -w, 0, or +w.
   Collect strictly positive net differences into S (duplicates allowed initially). */
static void build_S(const int *w, int m) {
    int i;
    int total = 1;
    for (i = 0; i < m; i++) total *= 3;

    S_count = 0;
    for (int mask = 0; mask < total; mask++) {
        int t = mask;
        int sum = 0;
        for (i = 0; i < m; i++) {
            int trit = t % 3;
            t /= 3;
            if (trit == 1) sum += w[i];
            else if (trit == 2) sum -= w[i];
        }
        if (sum > 0) {
            if (S_count < MAX_S) {
                S[S_count++] = sum;
            } else {
                /* Capacity assumptions violated: behavior unspecified by spec.
                   We keep code safe by truncating. */
                break;
            }
        }
    }

    /* Sort and unique S for membership tests and intersection computations. */
    if (S_count > 1) {
        qsort(S, (size_t)S_count, sizeof(S[0]), cmp_int);
        int out = 1;
        for (i = 1; i < S_count; i++) {
            if (S[i] != S[out - 1]) S[out++] = S[i];
        }
        S_count = out;
    }
}

int main(void) {
    int n, m;
    int targets[MAX_N];
    int weights[MAX_M];

    while (scanf("%d", &n) == 1) {
        if (n == 0) break;
        if (scanf("%d", &m) != 1) break;

        for (int i = 0; i < n; i++) scanf("%d", &targets[i]);
        for (int j = 0; j < m; j++) scanf("%d", &weights[j]);

        build_S(weights, m);

        /* Determine unbalanceable targets U. */
        int U[MAX_N];
        int U_count = 0;
        for (int i = 0; i < n; i++) {
            int t = targets[i];
            if (!binary_search_int(S, S_count, t)) {
                if (U_count < MAX_N) U[U_count++] = t;
            }
        }

        if (U_count == 0) {
            printf("0\n");
            continue;
        }

        /* Find minimal d >= 0 such that for all u in U, u + d in S.
           This is equivalent to finding minimal d in intersection over u of (S - u),
           with constraint d >= 0. We compute by scanning candidate d from first u. */
        int u0 = U[0];
        int best = -1;

        for (int i = 0; i < S_count; i++) {
            int s0 = S[i];
            int d = s0 - u0;
            if (d < 0) continue;

            int ok = 1;
            for (int k = 1; k < U_count; k++) {
                int need = U[k] + d;
                if (!binary_search_int(S, S_count, need)) {
                    ok = 0;
                    break;
                }
            }
            if (ok) {
                best = d;
                break; /* minimal due to S sorted ascending => d nondecreasing */
            }
        }

        printf("%d\n", best);
    }

    return 0;
}
