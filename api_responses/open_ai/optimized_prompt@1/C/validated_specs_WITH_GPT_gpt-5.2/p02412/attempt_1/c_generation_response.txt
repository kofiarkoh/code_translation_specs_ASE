```c
#include <stdio.h>

int main(void) {
    int n, x;

    while (scanf("%d %d", &n, &x) == 2) {
        if (n == 0 && x == 0) {
            break;
        }

        int count_ordered = 0;

        /* Count ordered permutations of valid triples (a,b,c) with all distinct. */
        for (int a = 1; a <= n; a++) {
            for (int b = 1; b <= n; b++) {
                if (b == a) continue;
                for (int c = 1; c <= n; c++) {
                    if (c == a || c == b) continue;
                    if (a + b + c == x) {
                        count_ordered++;
                    }
                }
            }
        }

        /* Each unordered triple of distinct integers has exactly 6 ordered permutations. */
        printf("%d\n", count_ordered / 6);
    }

    return 0;
}
```
