#include <stdio.h>
#include <limits.h>

#define MAXN 10000

static int num_digits_nonneg_int(int x) {
    /* Specification states inputs are non-negative decimal integers.
       For x == 0, decimal representation has 1 digit. */
    int d = 1;
    while (x >= 10) {
        x /= 10;
        d++;
    }
    return d;
}

static int pow10_int(int d) {
    /* Precondition: concatenation fits in program's integer arithmetic (no overflow). */
    int p = 1;
    while (d-- > 0) p *= 10;
    return p;
}

int main(void) {
    int n;
    if (scanf("%d", &n) != 1) return 0;

    int a[MAXN];
    int digits[MAXN];
    int p10[MAXN];

    /* No explicit error handling per specification; if n > MAXN behavior is unspecified. */
    for (int i = 0; i < n; i++) {
        if (scanf("%d", &a[i]) != 1) return 0;
        digits[i] = num_digits_nonneg_int(a[i]);
        p10[i] = pow10_int(digits[i]);
    }

    /* Track three smallest concatenation values seen so far (with duplicates). */
    int m1 = INT_MAX, m2 = INT_MAX, m3 = INT_MAX;

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j) continue;

            /* Concatenate decimal representation: a[i] followed by a[j]. */
            int v = a[i] * p10[j] + a[j];

            /* Insert v into (m1, m2, m3) maintaining non-decreasing order. */
            if (v <= m1) {
                m3 = m2;
                m2 = m1;
                m1 = v;
            } else if (v <= m2) {
                m3 = m2;
                m2 = v;
            } else if (v <= m3) {
                m3 = v;
            }
        }
    }

    /* Precondition guarantees at least 3 ordered distinct pairs exist (n >= 3). */
    printf("%d", m3);
    return 0;
}
