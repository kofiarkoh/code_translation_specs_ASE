```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

typedef long long ll;

typedef struct {
    int x, y;
    int id; /* special cell index */
} Cell;

typedef struct {
    int to;
    int w;
} Edge;

typedef struct {
    Edge *a;
    int sz;
    int cap;
} Adj;

typedef struct {
    ll dist;
    int v;
} HeapItem;

typedef struct {
    HeapItem *a;
    int sz;
    int cap;
} MinHeap;

static void adj_init(Adj *g, int cap_hint) {
    g->sz = 0;
    g->cap = (cap_hint > 0 ? cap_hint : 4);
    g->a = (Edge*)malloc((size_t)g->cap * sizeof(Edge));
    /* Specification allows undefined behavior on allocation failure; no checks. */
}

static void adj_push(Adj *g, int to, int w) {
    if (g->sz >= g->cap) {
        g->cap = g->cap * 2;
        g->a = (Edge*)realloc(g->a, (size_t)g->cap * sizeof(Edge));
    }
    g->a[g->sz].to = to;
    g->a[g->sz].w = w;
    g->sz++;
}

static void add_undirected(Adj *G, int u, int v, int w) {
    adj_push(&G[u], v, w);
    adj_push(&G[v], u, w);
}

static void heap_init(MinHeap *h, int cap_hint) {
    h->sz = 0;
    h->cap = (cap_hint > 0 ? cap_hint : 16);
    h->a = (HeapItem*)malloc((size_t)h->cap * sizeof(HeapItem));
}

static void heap_push(MinHeap *h, ll dist, int v) {
    if (h->sz >= h->cap) {
        h->cap = h->cap * 2;
        h->a = (HeapItem*)realloc(h->a, (size_t)h->cap * sizeof(HeapItem));
    }
    int i = h->sz++;
    h->a[i].dist = dist;
    h->a[i].v = v;
    while (i > 0) {
        int p = (i - 1) / 2;
        if (h->a[p].dist <= h->a[i].dist) break;
        HeapItem tmp = h->a[p];
        h->a[p] = h->a[i];
        h->a[i] = tmp;
        i = p;
    }
}

static int heap_empty(MinHeap *h) {
    return h->sz == 0;
}

static HeapItem heap_pop(MinHeap *h) {
    HeapItem ret = h->a[0];
    h->sz--;
    if (h->sz > 0) {
        h->a[0] = h->a[h->sz];
        int i = 0;
        for (;;) {
            int l = 2 * i + 1;
            int r = 2 * i + 2;
            int s = i;
            if (l < h->sz && h->a[l].dist < h->a[s].dist) s = l;
            if (r < h->sz && h->a[r].dist < h->a[s].dist) s = r;
            if (s == i) break;
            HeapItem tmp = h->a[s];
            h->a[s] = h->a[i];
            h->a[i] = tmp;
            i = s;
        }
    }
    return ret;
}

static int cmp_row(const void *A, const void *B) {
    const Cell *a = (const Cell*)A;
    const Cell *b = (const Cell*)B;
    if (a->x != b->x) return (a->x < b->x) ? -1 : 1;
    if (a->y != b->y) return (a->y < b->y) ? -1 : 1;
    /* Keep duplicates in some order; stable ordering not required by spec. */
    return (a->id < b->id) ? -1 : (a->id > b->id);
}

static int cmp_col(const void *A, const void *B) {
    const Cell *a = (const Cell*)A;
    const Cell *b = (const Cell*)B;
    if (a->y != b->y) return (a->y < b->y) ? -1 : 1;
    if (a->x != b->x) return (a->x < b->x) ? -1 : 1;
    return (a->id < b->id) ? -1 : (a->id > b->id);
}

int main(void) {
    int m, n, k;
    if (scanf("%d %d %d", &m, &n, &k) != 3) return 0;

    /* Read k special cells; duplicates are kept as distinct entities. */
    int cap = k + 2;
    int *xs = (int*)malloc((size_t)cap * sizeof(int));
    int *ys = (int*)malloc((size_t)cap * sizeof(int));

    int i;
    for (i = 0; i < k; i++) {
        (void)scanf("%d %d", &xs[i], &ys[i]);
    }

    /* Add (1,1) if not already present (by coordinate existence, not by multiplicity). */
    int has11 = 0, hasmn = 0;
    for (i = 0; i < k; i++) {
        if (xs[i] == 1 && ys[i] == 1) has11 = 1;
        if (xs[i] == m && ys[i] == n) hasmn = 1;
    }
    int K = k;
    if (!has11) {
        xs[K] = 1; ys[K] = 1; K++;
    }
    if (!hasmn) {
        xs[K] = m; ys[K] = n; K++;
    }

    /* Identify one chosen start cell entity and one chosen goal cell entity.
       If multiple duplicates exist, choose the first occurrence in input order;
       if added, it will be the appended one. */
    int start_id = -1, goal_id = -1;
    for (i = 0; i < K; i++) {
        if (start_id < 0 && xs[i] == 1 && ys[i] == 1) start_id = i;
        if (goal_id < 0 && xs[i] == m && ys[i] == n) goal_id = i;
    }

    int V = 2 * K;
    Adj *G = (Adj*)malloc((size_t)V * sizeof(Adj));
    for (i = 0; i < V; i++) adj_init(&G[i], 4);

    /* Intra-cell edges: p^(0) <-> p^(1) cost 1 */
    for (i = 0; i < K; i++) {
        int u0 = 2 * i;
        int u1 = 2 * i + 1;
        add_undirected(G, u0, u1, 1);
    }

    /* Build arrays for sorting */
    Cell *cells = (Cell*)malloc((size_t)K * sizeof(Cell));
    for (i = 0; i < K; i++) {
        cells[i].x = xs[i];
        cells[i].y = ys[i];
        cells[i].id = i;
    }

    /* Same-row connections between ^(0) nodes only, consecutive in row-sorted order */
    Cell *tmp = (Cell*)malloc((size_t)K * sizeof(Cell));
    memcpy(tmp, cells, (size_t)K * sizeof(Cell));
    qsort(tmp, (size_t)K, sizeof(Cell), cmp_row);
    for (i = 0; i + 1 < K; i++) {
        if (tmp[i].x == tmp[i + 1].x) {
            int id1 = tmp[i].id;
            int id2 = tmp[i + 1].id;
            int w = tmp[i].y - tmp[i + 1].y;
            if (w < 0) w = -w;
            add_undirected(G, 2 * id1, 2 * id2, w);
        }
    }

    /* Same-column connections between ^(1) nodes only, consecutive in col-sorted order */
    memcpy(tmp, cells, (size_t)K * sizeof(Cell));
    qsort(tmp, (size_t)K, sizeof(Cell), cmp_col);
    for (i = 0; i + 1 < K; i++) {
        if (tmp[i].y == tmp[i + 1].y) {
            int id1 = tmp[i].id;
            int id2 = tmp[i + 1].id;
            int w = tmp[i].x - tmp[i + 1].x;
            if (w < 0) w = -w;
            add_undirected(G, 2 * id1 + 1, 2 * id2 + 1, w);
        }
    }

    /* Dijkstra from (1,1)^(0) */
    const ll INF = (ll)LLONG_MAX / 4;
    ll *dist = (ll*)malloc((size_t)V * sizeof(ll));
    unsigned char *vis = (unsigned char*)malloc((size_t)V);
    for (i = 0; i < V; i++) {
        dist[i] = INF;
        vis[i] = 0;
    }

    int s = 2 * start_id; /* (1,1)^(0) */
    int g0 = 2 * goal_id;
    int g1 = 2 * goal_id + 1;

    MinHeap pq;
    heap_init(&pq, V > 0 ? V : 16);
    dist[s] = 0;
    heap_push(&pq, 0, s);

    ll answer = -1;
    while (!heap_empty(&pq)) {
        HeapItem it = heap_pop(&pq);
        int v = it.v;
        if (it.dist != dist[v]) continue;
        if (vis[v]) continue;
        vis[v] = 1;

        if (v == g0 || v == g1) {
            answer = dist[v];
            break; /* early termination allowed by spec */
        }

        Adj *adj = &G[v];
        for (int ei = 0; ei < adj->sz; ei++) {
            int to = adj->a[ei].to;
            int w = adj->a[ei].w;
            if (dist[v] != INF) {
                ll nd = dist[v] + (ll)w;
                if (nd < dist[to]) {
                    dist[to] = nd;
                    heap_push(&pq, nd, to);
                }
            }
        }
    }

    if (answer < 0) {
        /* If not early-terminated, compute min(dist[g0], dist[g1]) if reachable */
        ll best = dist[g0];
        if (dist[g1] < best) best = dist[g1];
        if (best != INF) answer = best;
    }

    if (answer < 0) {
        printf("-1");
    } else {
        /* Output is a single signed integer in decimal. */
        printf("%lld", answer);
    }

    return 0;
}
```
