```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAXV 20005          /* fixed internal storage limit for vertices */
#define MAXDEG 16           /* fixed per-node adjacency capacity; exceeding => undefined */
#define INF_LL ((long long)0x3f3f3f3f3f3f3f3fLL)

typedef struct {
    int x, y;
    int id;
} Node;

typedef struct {
    int to[MAXDEG];
    int w[MAXDEG];
    int deg;
} Adj;

typedef struct {
    long long d;
    int v;
} HeapItem;

static Node nodes[MAXV];
static Adj adjl[MAXV];
static long long distv[MAXV];
static unsigned char used[MAXV];

static HeapItem heap[MAXV * 8]; /* fixed-size PQ; may overflow if too many pushes => undefined */
static int hsz = 0;

static void heap_push(long long d, int v) {
    int i = ++hsz;
    heap[i].d = d;
    heap[i].v = v;
    while (i > 1) {
        int p = i >> 1;
        if (heap[p].d <= heap[i].d) break;
        HeapItem tmp = heap[p];
        heap[p] = heap[i];
        heap[i] = tmp;
        i = p;
    }
}

static HeapItem heap_pop(void) {
    HeapItem ret = heap[1];
    heap[1] = heap[hsz--];
    int i = 1;
    for (;;) {
        int l = i << 1, r = l + 1, s = i;
        if (l <= hsz && heap[l].d < heap[s].d) s = l;
        if (r <= hsz && heap[r].d < heap[s].d) s = r;
        if (s == i) break;
        HeapItem tmp = heap[i];
        heap[i] = heap[s];
        heap[s] = tmp;
        i = s;
    }
    return ret;
}

static int cmp_x_then_y(const void *a, const void *b) {
    const Node *A = (const Node *)a;
    const Node *B = (const Node *)b;
    if (A->x != B->x) return (A->x < B->x) ? -1 : 1;
    if (A->y != B->y) return (A->y < B->y) ? -1 : 1;
    return 0;
}

static int cmp_y_then_x(const void *a, const void *b) {
    const Node *A = (const Node *)a;
    const Node *B = (const Node *)b;
    if (A->y != B->y) return (A->y < B->y) ? -1 : 1;
    if (A->x != B->x) return (A->x < B->x) ? -1 : 1;
    return 0;
}

static void add_edge(int u, int v, int w) {
    /* Undirected; adjacency capacity overflow => undefined per spec */
    int du = adjl[u].deg++;
    adjl[u].to[du] = v;
    adjl[u].w[du] = w;
    int dv = adjl[v].deg++;
    adjl[v].to[dv] = u;
    adjl[v].w[dv] = w;
}

static int abs_int(int a) { return a < 0 ? -a : a; }

int main(void) {
    int m, n, k;
    if (scanf("%d %d %d", &m, &n, &k) != 3) return 0;

    int cnt = 0;
    int start_id = -1, goal_id = -1;

    /* Read provided special cells */
    for (int i = 0; i < k; i++) {
        int x, y;
        if (scanf("%d %d", &x, &y) != 2) return 0;
        if (cnt >= MAXV) {
            /* Exceeding internal storage => undefined; choose to stop reading further */
            /* Still must be compilable; behavior unspecified */
            continue;
        }
        nodes[cnt].x = x;
        nodes[cnt].y = y;
        nodes[cnt].id = cnt;
        if (x == 1 && y == 1) start_id = cnt;
        if (x == m && y == n) goal_id = cnt;
        cnt++;
    }

    /* Add corners if missing */
    if (start_id < 0) {
        if (cnt < MAXV) {
            start_id = cnt;
            nodes[cnt].x = 1;
            nodes[cnt].y = 1;
            nodes[cnt].id = cnt;
            cnt++;
        }
    }
    if (goal_id < 0) {
        if (cnt < MAXV) {
            goal_id = cnt;
            nodes[cnt].x = m;
            nodes[cnt].y = n;
            nodes[cnt].id = cnt;
            cnt++;
        }
    }

    /* Initialize adjacency */
    for (int i = 0; i < cnt; i++) {
        adjl[i].deg = 0;
    }

    /* Graph construction via sorting-based neighbor connections.
       Narrowest reasonable interpretation: connect each node to its immediate
       predecessor and successor in sorted-by-x and sorted-by-y orders, with
       non-negative edge cost = Manhattan distance. */
    Node *tmp = (Node *)malloc((size_t)cnt * sizeof(Node));
    if (!tmp) return 0;

    for (int i = 0; i < cnt; i++) tmp[i] = nodes[i];
    qsort(tmp, (size_t)cnt, sizeof(Node), cmp_x_then_y);
    for (int i = 0; i + 1 < cnt; i++) {
        int u = tmp[i].id, v = tmp[i + 1].id;
        int w = abs_int(nodes[u].x - nodes[v].x) + abs_int(nodes[u].y - nodes[v].y);
        add_edge(u, v, w);
    }

    for (int i = 0; i < cnt; i++) tmp[i] = nodes[i];
    qsort(tmp, (size_t)cnt, sizeof(Node), cmp_y_then_x);
    for (int i = 0; i + 1 < cnt; i++) {
        int u = tmp[i].id, v = tmp[i + 1].id;
        int w = abs_int(nodes[u].x - nodes[v].x) + abs_int(nodes[u].y - nodes[v].y);
        add_edge(u, v, w);
    }

    free(tmp);

    /* Dijkstra shortest path with early termination when goal finalized */
    for (int i = 0; i < cnt; i++) {
        distv[i] = INF_LL;
        used[i] = 0;
    }
    hsz = 0;

    if (start_id < 0 || goal_id < 0) {
        /* Should not happen unless storage exceeded; treat as unreachable */
        printf("-1\n");
        return 0;
    }

    distv[start_id] = 0;
    heap_push(0, start_id);

    while (hsz > 0) {
        HeapItem it = heap_pop();
        int v = it.v;
        if (used[v]) continue;
        used[v] = 1;

        if (v == goal_id) {
            printf("%lld\n", distv[v]);
            return 0;
        }

        for (int ei = 0; ei < adjl[v].deg; ei++) {
            int to = adjl[v].to[ei];
            int w = adjl[v].w[ei];
            if (used[to]) continue;
            long long nd = distv[v] + (long long)w;
            if (nd < distv[to]) {
                distv[to] = nd;
                heap_push(nd, to);
            }
        }
    }

    printf("-1\n");
    return 0;
}
```
