#include <stdio.h>
#include <string.h>

#define MAXLINE 1024

/* Grid is stored so that the first character of each input line is placed at column index 1.
   We keep a leading dummy column at index 0. */
static char grid[10][MAXLINE + 2]; /* 0..9 rows, each row has room for dummy + line + '\0' */

/* Recursive explosion per specification. Undefined behavior is allowed if px/py out of 1..8. */
static void explode(int px, int py) {
    /* Immediately clear the cell, even if it was not '1'. */
    grid[py][px] = '0';

    /* Direction order: left, up, right, down. Distances: 1..3. */
    /* left */
    for (int d = 1; d <= 3; d++) {
        int nx = px - d, ny = py;
        if (nx < 1) break;
        if (grid[ny][nx] == '1') explode(nx, ny);
    }
    /* up */
    for (int d = 1; d <= 3; d++) {
        int nx = px, ny = py - d;
        if (ny < 1) break;
        if (grid[ny][nx] == '1') explode(nx, ny);
    }
    /* right */
    for (int d = 1; d <= 3; d++) {
        int nx = px + d, ny = py;
        if (nx > 8) break;
        if (grid[ny][nx] == '1') explode(nx, ny);
    }
    /* down */
    for (int d = 1; d <= 3; d++) {
        int nx = px, ny = py + d;
        if (ny > 8) break;
        if (grid[ny][nx] == '1') explode(nx, ny);
    }
}

int main(void) {
    int N;
    if (scanf("%d", &N) != 1) return 0;

    /* Consume the remainder of the line after N, if any. */
    int ch;
    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);

    for (int dataset = 1; dataset <= N; dataset++) {
        /* Read 9 lines verbatim as C strings; store starting at column index 1. */
        for (int r = 0; r < 9; r++) {
            /* No bounds checking beyond fixed buffer; relies on input fitting. */
            if (!fgets(grid[r], (int)sizeof(grid[r]), stdin)) {
                /* Specification allows undefined behavior on malformed input; just stop. */
                return 0;
            }

            /* Place first character at internal column index 1 by shifting right by 1. */
            size_t len = strlen(grid[r]);
            /* memmove handles overlap; include terminating '\0'. */
            memmove(&grid[r][1], &grid[r][0], len + 1);
            grid[r][0] = '\0'; /* dummy */
        }

        int sx, sy;
        if (scanf("%d", &sx) != 1) return 0;
        if (scanf("%d", &sy) != 1) return 0;

        /* Consume remainder of line after sy to align with next dataset's line reads. */
        do {
            ch = getchar();
        } while (ch != '\n' && ch != EOF);

        explode(sx, sy);

        printf("Data %d:\n", dataset);
        /* Print rows 1 through 8, starting at column index 1, as stored C strings. */
        for (int r = 1; r <= 8; r++) {
            fputs(&grid[r][1], stdout);
            /* If the stored string did not include a newline, do not add one. */
        }
    }

    return 0;
}
