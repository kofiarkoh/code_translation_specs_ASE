#include <stdio.h>
#include <string.h>

static char grid[8][8];
static char marked_ch = 0;
static char cleared_ch = 0;

static void infer_state_chars(void) {
    /* Narrowest reasonable interpretation:
       - There exist two characters: one for "marked" and one for "cleared".
       - We infer them from the grid contents:
         * cleared_ch: first character that is NOT the marked character (chosen below)
         * marked_ch: the most frequent character in the grid (common in such inputs)
       If the grid is uniform, marked_ch == cleared_ch and propagation becomes a no-op
       beyond setting the start cell to cleared_ch (which equals marked_ch). */
    int freq[256] = {0};
    for (int y = 0; y < 8; y++) {
        for (int x = 0; x < 8; x++) {
            unsigned char c = (unsigned char)grid[y][x];
            freq[c]++;
        }
    }

    int best = 0;
    int bestc = 0;
    for (int c = 0; c < 256; c++) {
        if (freq[c] > best) {
            best = freq[c];
            bestc = c;
        }
    }
    marked_ch = (char)bestc;

    cleared_ch = marked_ch;
    for (int c = 0; c < 256; c++) {
        if (freq[c] > 0 && c != (unsigned char)marked_ch) {
            cleared_ch = (char)c;
            break;
        }
    }
}

static void react(int x, int y) {
    /* x,y are 0-based indices. No bounds checks beyond those implied by loops below. */
    grid[y][x] = cleared_ch;

    /* For each of 4 directions, check distances 1..3; if a cell is marked, clear it recursively. */
    static const int dx[4] = { 1, -1, 0, 0 };
    static const int dy[4] = { 0, 0, 1, -1 };

    for (int dir = 0; dir < 4; dir++) {
        for (int dist = 1; dist <= 3; dist++) {
            int nx = x + dx[dir] * dist;
            int ny = y + dy[dir] * dist;
            if (nx < 0 || nx >= 8 || ny < 0 || ny >= 8) continue;
            if (grid[ny][nx] == marked_ch) {
                react(nx, ny);
            }
        }
    }
}

int main(void) {
    int datasets;
    if (scanf("%d", &datasets) != 1) return 0;

    for (int t = 1; t <= datasets; t++) {
        char line[1024];

        for (int y = 0; y < 8; y++) {
            /* Read a token containing at least 8 characters (spec: at least 8 usable chars). */
            if (scanf("%1023s", line) != 1) return 0;
            /* Copy first 8 characters as the row. */
            for (int x = 0; x < 8; x++) grid[y][x] = line[x];
        }

        int x1, y1;
        if (scanf("%d %d", &x1, &y1) != 2) return 0;

        infer_state_chars();

        /* Coordinates are 1..8 inclusive per spec; convert to 0-based.
           No explicit error handling; out-of-range leads to undefined behavior as allowed. */
        react(x1 - 1, y1 - 1);

        printf("Data %d:\n", t);
        for (int y = 0; y < 8; y++) {
            for (int x = 0; x < 8; x++) putchar(grid[y][x]);
            putchar('\n');
        }
    }

    return 0;
}
