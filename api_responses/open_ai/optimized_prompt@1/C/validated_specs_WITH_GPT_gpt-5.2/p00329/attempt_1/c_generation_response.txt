#define MAXN 512

static unsigned char allowed[MAXN + 1][MAXN + 1]; /* allowed[row][col] */
static int match_row_of_col[MAXN + 1];            /* col -> row, 0 if none */
static int match_col_of_row[MAXN + 1];            /* row -> col, 0 if none */
static unsigned char vis_row[MAXN + 1];

static int n;

static int dfs_augment(int col) {
    int row;
    for (row = 1; row <= n; row++) {
        if (!allowed[row][col]) continue;
        if (vis_row[row]) continue;
        vis_row[row] = 1;

        if (match_col_of_row[row] == 0 || dfs_augment(match_col_of_row[row])) {
            match_row_of_col[col] = row;
            match_col_of_row[row] = col;
            return 1;
        }
    }
    return 0;
}

                return 0;

        }
