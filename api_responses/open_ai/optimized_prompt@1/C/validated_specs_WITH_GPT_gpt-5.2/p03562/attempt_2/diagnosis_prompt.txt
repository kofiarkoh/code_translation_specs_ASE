#define MOD_PRIME 1000000007u

typedef struct {
    /* Little-endian bitset: bit i is coefficient of x^i (and also integer bit i). */
    uint64_t w[(MAX_BITS + 63) / 64];
} Poly;

static void poly_zero(Poly *p) {
    memset(p->w, 0, sizeof(p->w));
}

    for (size_t i = 0; i < sizeof(p->w) / sizeof(p->w[0]); i++) {
        if (p->w[i] != 0) return 0;
    }
    return 1;
}

static void poly_copy(Poly *dst, const Poly *src) {
    memcpy(dst->w, src->w, sizeof(dst->w));
}

static void poly_xor_shifted(Poly *a, const Poly *b, int shift) {
    /* a ^= (b << shift), shift >= 0, within MAX_BITS (otherwise undefined by spec). */
    int word_shift = shift / 64;
    int bit_shift = shift % 64;
    int nwords = (int)(sizeof(a->w) / sizeof(a->w[0]));

    if (bit_shift == 0) {
        for (int i = nwords - 1; i >= word_shift; i--) {
            a->w[i] ^= b->w[i - word_shift];
        }
    } else {
        for (int i = nwords - 1; i >= word_shift; i--) {
            uint64_t lo = b->w[i - word_shift] << bit_shift;
            uint64_t hi = 0;
            if (i - word_shift - 1 >= 0) {
                hi = b->w[i - word_shift - 1] >> (64 - bit_shift);
            }
            a->w[i] ^= (lo | hi);
        }
    }
}

static int poly_deg(const Poly *p) {
    /* Returns degree (highest set bit index), or -1 for zero. */
    int nwords = (int)(sizeof(p->w) / sizeof(p->w[0]));
    for (int i = nwords - 1; i >= 0; i--) {

            /* find msb index in x */

            while (((x >> msb) & 1u) == 0u) msb--;
            return i * 64 + msb;
        }
    }
    return -1;
}

    if (dmod < 0) return; /* unspecified by contract; caller avoids */

        if (da < dmod) break;
        poly_xor_shifted(a, mod, da - dmod);
    }
}

static void poly_gcd(Poly *out, const Poly *a_in, const Poly *b_in) {
    /* Euclidean algorithm over GF(2)[x]. */
    Poly a, b, r;
    poly_copy(&a, a_in);
    poly_copy(&b, b_in);

    while (!poly_is_zero(&b)) {
        poly_copy(&r, &a);
        poly_mod(&r, &b);
        poly_copy(&a, &b);
        poly_copy(&b, &r);
    }
    poly_copy(out, &a);
}
