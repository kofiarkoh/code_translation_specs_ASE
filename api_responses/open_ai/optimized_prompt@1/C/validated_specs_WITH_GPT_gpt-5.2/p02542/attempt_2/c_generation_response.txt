#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

typedef long long ll;

enum {
    MAXN = 64,          /* fixed bounds per spec: undefined if exceeded */
    MAXM = 64,
    MAXV = MAXN * MAXM,
    MAXCAND = 100
};

/* ---------------- Min-Cost Flow (successive shortest augmenting path) ---------------- */

typedef struct Edge {
    int to;
    int rev;
    int cap;
    ll cost;
} Edge;

typedef struct Vec {
    Edge *a;
    int n, cap;
} Vec;

static void vec_push(Vec *v, Edge e) {
    if (v->n == v->cap) {
        int nc = v->cap ? v->cap * 2 : 4;
        Edge *na = (Edge*)realloc(v->a, (size_t)nc * sizeof(Edge));
        if (!na) exit(1);
        v->a = na;
        v->cap = nc;
    }
    v->a[v->n++] = e;
}

static void add_edge(Vec *g, int fr, int to, int cap, ll cost) {
    Edge f = { to, g[to].n, cap, cost };
    Edge r = { fr, g[fr].n, 0, -cost };
    vec_push(&g[fr], f);
    vec_push(&g[to], r);
}

/* Sentinel behavior per spec */
static const ll SENTINEL_NEG_CYCLE = (ll)0x3f3f3f3f3f3f3f3fLL; /* arbitrary large positive */
static const ll SENTINEL_NO_AUG    = (ll)0x5a5a5a5a5a5a5a5aLL; /* different arbitrary large positive */

/*
  Returns min cost to send exactly 'need' units from s to t.
  If detects a negative-cost cycle during initialization (Bellman-Ford), returns SENTINEL_NEG_CYCLE.
  If cannot find augmenting path to satisfy remaining deficits, returns SENTINEL_NO_AUG.
*/
static ll min_cost_flow(Vec *g, int V, int s, int t, int need) {
    ll *pot = (ll*)malloc((size_t)V * sizeof(ll));
    ll *dist = (ll*)malloc((size_t)V * sizeof(ll));
    int *pv_v = (int*)malloc((size_t)V * sizeof(int));
    int *pv_e = (int*)malloc((size_t)V * sizeof(int));
    char *inQ = (char*)malloc((size_t)V);
    int *q = (int*)malloc((size_t)V * sizeof(int));
    if (!pot || !dist || !pv_v || !pv_e || !inQ || !q) exit(1);

    /* Initialization: Bellman-Ford/SPFA from s to get potentials; detect negative cycle. */
    for (int i = 0; i < V; i++) pot[i] = (ll)0;
    for (int i = 0; i < V; i++) dist[i] = (ll)LLONG_MAX / 4;
    memset(inQ, 0, (size_t)V);
    dist[s] = 0;
    int qh = 0, qt = 0;
    q[qt++] = s;
    inQ[s] = 1;
    int *relaxCnt = (int*)calloc((size_t)V, sizeof(int));
    if (!relaxCnt) exit(1);

    while (qh != qt) {
        int v = q[qh++];
        if (qh == V) qh = 0;
        inQ[v] = 0;
        ll dv = dist[v];
        if (dv == (ll)LLONG_MAX / 4) continue;
        for (int ei = 0; ei < g[v].n; ei++) {
            Edge *e = &g[v].a[ei];
            if (e->cap <= 0) continue;
            int u = e->to;
            ll nd = dv + e->cost;
            if (nd < dist[u]) {
                dist[u] = nd;
                if (!inQ[u]) {
                    q[qt++] = u;
                    if (qt == V) qt = 0;
                    inQ[u] = 1;
                    if (++relaxCnt[u] > V) {
                        free(relaxCnt);
                        free(pot); free(dist); free(pv_v); free(pv_e); free(inQ); free(q);
                        return SENTINEL_NEG_CYCLE;
                    }
                }
            }
        }
    }
    free(relaxCnt);

    for (int i = 0; i < V; i++) {
        if (dist[i] < (ll)LLONG_MAX / 4) pot[i] = dist[i];
        else pot[i] = 0;
    }

    ll totalCost = 0;
    int flow = 0;

    /* Dijkstra with reduced costs (non-negative if potentials valid) */
    while (flow < need) {
        for (int i = 0; i < V; i++) dist[i] = (ll)LLONG_MAX / 4;
        dist[s] = 0;
        for (int i = 0; i < V; i++) { pv_v[i] = -1; pv_e[i] = -1; }

        /* Simple O(V^2) Dijkstra to avoid extra assumptions */
        char *used = (char*)calloc((size_t)V, 1);
        if (!used) exit(1);

        for (;;) {
            int v = -1;
            ll best = (ll)LLONG_MAX / 4;
            for (int i = 0; i < V; i++) {
                if (!used[i] && dist[i] < best) { best = dist[i]; v = i; }
            }
            if (v < 0) break;
            used[v] = 1;
            if (v == t) break;
            for (int ei = 0; ei < g[v].n; ei++) {
                Edge *e = &g[v].a[ei];
                if (e->cap <= 0) continue;
                int u = e->to;
                ll rcost = e->cost + pot[v] - pot[u];
                if (rcost < 0) rcost = rcost; /* keep as-is; spec doesn't forbid negative reduced costs */
                ll nd = dist[v] + rcost;
                if (nd < dist[u]) {
                    dist[u] = nd;
                    pv_v[u] = v;
                    pv_e[u] = ei;
                }
            }
        }

        free(used);

        if (pv_v[t] < 0) {
            free(pot); free(dist); free(pv_v); free(pv_e); free(inQ); free(q);
            return SENTINEL_NO_AUG;
        }

        for (int i = 0; i < V; i++) {
            if (dist[i] < (ll)LLONG_MAX / 4) pot[i] += dist[i];
        }

        int add = need - flow;
        int v = t;
        while (v != s) {
            int p = pv_v[v];
            int ei = pv_e[v];
            if (p < 0 || ei < 0) { add = 0; break; }
            if (g[p].a[ei].cap < add) add = g[p].a[ei].cap;
            v = p;
        }
        if (add <= 0) {
            free(pot); free(dist); free(pv_v); free(pv_e); free(inQ); free(q);
            return SENTINEL_NO_AUG;
        }

        v = t;
        while (v != s) {
            int p = pv_v[v];
            int ei = pv_e[v];
            Edge *e = &g[p].a[ei];
            Edge *r = &g[v].a[e->rev];
            e->cap -= add;
            r->cap += add;
            totalCost += (ll)add * e->cost;
            v = p;
        }
        flow += add;
    }

    free(pot); free(dist); free(pv_v); free(pv_e); free(inQ); free(q);
    return totalCost;
}

/* ---------------- Grid BFS candidate extraction ---------------- */

typedef struct Cand {
    int cell;   /* flattened cell index */
    int dist;   /* BFS distance */
} Cand;

int main(void) {
    int N, M;
    if (scanf("%d %d", &N, &M) != 2) return 0;

    static char grid[MAXN][MAXM + 5];
    for (int i = 0; i < N; i++) {
        if (scanf("%s", grid[i]) != 1) return 0;
    }

    /* Collect sources in fixed scan order: bottom->top, right->left */
    int *src_r = (int*)malloc((size_t)MAXV * sizeof(int));
    int *src_c = (int*)malloc((size_t)MAXV * sizeof(int));
    int S = 0;
    if (!src_r || !src_c) exit(1);

    for (int r = N - 1; r >= 0; r--) {
        for (int c = M - 1; c >= 0; c--) {
            if (grid[r][c] == 'o') {
                src_r[S] = r;
                src_c[S] = c;
                S++;
            }
        }
    }

    /* Candidate lists per source */
    Cand **cands = (Cand**)calloc((size_t)S, sizeof(Cand*));
    int *candN = (int*)calloc((size_t)S, sizeof(int));
    if (!cands || !candN) exit(1);

    /* BFS working arrays */
    int Vcells = N * M;
    int *q = (int*)malloc((size_t)Vcells * sizeof(int));
    int *dist = (int*)malloc((size_t)Vcells * sizeof(int));
    unsigned int *visTag = (unsigned int*)calloc((size_t)Vcells, sizeof(unsigned int));
    if (!q || !dist || !visTag) exit(1);
    unsigned int curTag = 1;

    for (int si = 0; si < S; si++) {
        int sr = src_r[si], sc = src_c[si];

        /* Immediately convert 'o' to '.' before BFS */
        grid[sr][sc] = '.';

        /* BFS with moves: right (c+1) and increasing row index (r+1) */
        int head = 0, tail = 0;

        /* Start cell is included as reachable (distance 0) */
        int sidx = sr * M + sc;
        q[tail++] = sidx;
        dist[sidx] = 0;
        visTag[sidx] = curTag;

        /* Record discovery order list (including start) */
        int *order = (int*)malloc((size_t)Vcells * sizeof(int));
        int ordN = 0;
        if (!order) exit(1);
        order[ordN++] = sidx;

        while (head < tail) {
            int v = q[head++];
            int r = v / M;
            int c = v % M;
            int d = dist[v];

            /* Fixed neighbor expansion order: right then down (increasing row index) */
            /* Move right */
            if (c + 1 < M) {
                int nr = r, nc = c + 1;
                int ni = nr * M + nc;
                if (visTag[ni] != curTag && grid[nr][nc] == '.') {
                    visTag[ni] = curTag;
                    dist[ni] = d + 1;
                    q[tail++] = ni;
                    order[ordN++] = ni;
                }
            }
            /* Move toward increasing row index */
            if (r + 1 < N) {
                int nr = r + 1, nc = c;
                int ni = nr * M + nc;
                if (visTag[ni] != curTag && grid[nr][nc] == '.') {
                    visTag[ni] = curTag;
                    dist[ni] = d + 1;
                    q[tail++] = ni;
                    order[ordN++] = ni;
                }
            }
        }

        /* Take up to last 100 discovered reachable cells as candidates */
        int take = ordN < MAXCAND ? ordN : MAXCAND;
        cands[si] = (Cand*)malloc((size_t)take * sizeof(Cand));
        if (!cands[si]) exit(1);
        candN[si] = take;

        for (int k = 0; k < take; k++) {
            int idx = order[ordN - take + k];
            cands[si][k].cell = idx;
            cands[si][k].dist = dist[idx];
        }

        free(order);

        curTag++;
        if (curTag == 0) { /* wrap: reset tags (narrowest reasonable handling) */
            memset(visTag, 0, (size_t)Vcells * sizeof(unsigned int));
            curTag = 1;
        }
    }

    /* ---------------- Build flow network ----------------
       Nodes:
         0: super source
         1..S: source nodes for each original 'o'
         S+1 .. S+Vcells: grid cell nodes
         S+Vcells+1: super sink
    */
    int SRC = 0;
    int baseO = 1;
    int baseCell = baseO + S;
    int SNK = baseCell + Vcells;
    int V = SNK + 1;

    Vec *g = (Vec*)calloc((size_t)V, sizeof(Vec));
    if (!g) exit(1);

    /* super source -> each o (cap 1, cost 0) */
    for (int i = 0; i < S; i++) add_edge(g, SRC, baseO + i, 1, 0);

    /* each cell -> sink (cap 1, cost 0) */
    for (int cell = 0; cell < Vcells; cell++) add_edge(g, baseCell + cell, SNK, 1, 0);

    /* o -> candidate cell edges (cap 1, cost = -distance) */
    for (int i = 0; i < S; i++) {
        for (int k = 0; k < candN[i]; k++) {
            int cell = cands[i][k].cell;
            int d = cands[i][k].dist;
            add_edge(g, baseO + i, baseCell + cell, 1, -(ll)d);
        }
    }

    ll minCost = min_cost_flow(g, V, SRC, SNK, S);
    ll ans = -minCost; /* negate per spec (maximize sum of distances) */
    printf("%lld\n", ans);

    /* cleanup */
    for (int i = 0; i < V; i++) free(g[i].a);
    free(g);
    for (int i = 0; i < S; i++) free(cands[i]);
    free(cands);
    free(candN);
    free(src_r);
    free(src_c);
    free(q);
    free(dist);
    free(visTag);

    return 0;
}
