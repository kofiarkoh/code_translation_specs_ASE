            return 0;

            /* Terminating dataset: stop with no output for this dataset. */
            return 0;

        /* Per spec, n is positive and <= 100 for datasets processed. */
        int a[100];
        for (int i = 0; i < n; i++) {
            /* If input is malformed here, behavior is unspecified by spec. */
            (void)scanf("%d", &a[i]);
        }

        /* Count inversions (adjacent swaps needed for nondecreasing order). */
        int swaps = 0; /* Spec states result fits in signed int. */
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[i] > a[j]) swaps++;
            }
        }

    }
}
