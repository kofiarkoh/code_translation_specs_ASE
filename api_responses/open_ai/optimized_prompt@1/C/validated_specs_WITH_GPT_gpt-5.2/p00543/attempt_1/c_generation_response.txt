#include <stdio.h>

#define MAX_N 1000

int main(void) {
    int n, pass_limit;
    int a[MAX_N];

    if (scanf("%d %d", &n, &pass_limit) != 2) {
        return 0; /* unspecified behavior; exit quietly */
    }

    for (int i = 0; i < n; i++) {
        if (scanf("%d", &a[i]) != 1) {
            return 0; /* unspecified behavior; exit quietly */
        }
    }

    /* For each modulus k = 1..pass_limit, enforce:
       for every adjacent pair, (left % k) <= (right % k).
       The narrowest reasonable interpretation is to repeatedly swap adjacent
       out-of-order pairs until the property holds for the whole sequence,
       i.e., a bubble-sort pass repeated to completion using key (x % k). */
    for (int k = 1; k <= pass_limit; k++) {
        int changed;
        do {
            changed = 0;
            for (int i = 0; i + 1 < n; i++) {
                int r1 = a[i] % k;
                int r2 = a[i + 1] % k;
                if (r1 > r2) {
                    int tmp = a[i];
                    a[i] = a[i + 1];
                    a[i + 1] = tmp;
                    changed = 1;
                }
            }
        } while (changed);
    }

    for (int i = 0; i < n; i++) {
        if (i) putchar(' ');
        printf("%d", a[i]);
    }
    putchar('\n');

    return 0;
}
