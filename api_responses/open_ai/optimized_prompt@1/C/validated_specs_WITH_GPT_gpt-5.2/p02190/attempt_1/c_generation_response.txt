```c
#include <stdio.h>
#include <stdlib.h>

#define MAXN 100001

static int cmp_int(const void *a, const void *b) {
    /* Avoid overflow from subtraction; define strict weak ordering. */
    int x = *(const int *)a;
    int y = *(const int *)b;
    if (x < y) return -1;
    if (x > y) return 1;
    return 0;
}

int main(void) {
    int n;
    int a[MAXN];

    /* No explicit error handling per specification. */
    if (scanf("%d", &n) != 1) return 0;

    for (int i = 0; i < n; i++) {
        if (scanf("%d", &a[i]) != 1) return 0;
    }

    qsort(a, (size_t)n, sizeof(a[0]), cmp_int);

    int distinct = 1; /* Preconditions guarantee n >= 1. */
    for (int i = 1; i < n; i++) {
        if (a[i] != a[i - 1]) distinct++;
    }

    printf("%d", distinct);
    return 0;
}
```
