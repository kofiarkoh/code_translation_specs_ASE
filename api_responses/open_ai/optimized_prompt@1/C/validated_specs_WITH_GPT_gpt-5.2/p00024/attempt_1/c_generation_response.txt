```c
#include <stdio.h>

int main(void)
{
    double v;

    while (scanf("%lf", &v) == 1) {
        /* x = (4.9 * v^2) / (9.8^2) */
        double x = (4.9 * v * v) / (9.8 * 9.8);

        /* k = truncate(x + 5) toward zero */
        int k = (int)(x + 5.0);

        /* m = truncate(k / 5) toward zero (C integer division) */
        int m = k / 5;

        /* output m + 1 */
        printf("%d\n", m + 1);
    }

    return 0;
}
```
