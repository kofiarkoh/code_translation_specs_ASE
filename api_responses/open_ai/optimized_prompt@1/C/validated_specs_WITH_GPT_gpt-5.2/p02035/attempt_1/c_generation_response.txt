```c
#include <stdio.h>
#include <stdint.h>

#define MAXN 2010

static int64_t mod;

static int64_t norm_mod(int64_t x) {
    x %= mod;
    if (x < 0) x += mod;
    return x;
}

static int64_t mul_mod(int64_t a, int64_t b) {
    /* Specification assumes no signed 64-bit overflow before reduction. */
    return (a * b) % mod;
}

static int64_t add_mod(int64_t a, int64_t b) {
    int64_t r = a + b;
    r %= mod;
    if (r < 0) r += mod;
    return r;
}

static int64_t sub_mod(int64_t a, int64_t b) {
    int64_t r = a - b;
    r %= mod;
    if (r < 0) r += mod;
    return r;
}

static int64_t pow_mod(int64_t a, int64_t e) {
    a = norm_mod(a);
    int64_t r = 1 % mod;
    while (e > 0) {
        if (e & 1) r = mul_mod(r, a);
        a = mul_mod(a, a);
        e >>= 1;
    }
    return r;
}

static int64_t fact[MAXN + 1];
static int64_t ifact[MAXN + 1];

static int64_t C(int n, int k) {
    /* Combination is 0 when out of range, per specification. */
    if (k < 0 || k > n) return 0;
    return mul_mod(fact[n], mul_mod(ifact[k], ifact[n - k]));
}

int main(void) {
    int n;
    if (scanf("%d %lld", &n, (long long *)&mod) != 2) return 0;

    /* Precompute factorials and inverse factorials up to fixed bound. */
    fact[0] = 1 % mod;
    for (int i = 1; i <= MAXN; i++) fact[i] = mul_mod(fact[i - 1], i);

    /* Modular inverse via Fermat's little theorem; requires modulus properties as specified. */
    ifact[MAXN] = pow_mod(fact[MAXN], mod - 2);
    for (int i = MAXN; i >= 1; i--) ifact[i - 1] = mul_mod(ifact[i], i);

    int64_t ans = 0;

    /* Sum over i = 1..n-1 of weighted counts. */
    for (int i = 1; i <= n - 1; i++) {
        int a = i;
        int b = n - i;

        /* number of internal configurations within each group:
           narrowest reasonable interpretation: factorial(a) * factorial(b) */
        int64_t internal = mul_mod(fact[a], fact[b]);

        /* number of ways to choose which elements belong to the first group: C(n, a) */
        int64_t choose = C(n, a);

        /* cross-group configurations satisfying an inclusion–exclusion condition over elements
           of the second group:
           narrowest reasonable interpretation consistent with "double sum over i and j" and
           "inclusion–exclusion over elements of the second group":
             sum_{j=0..b} (-1)^j * C(b, j) * (2^(a*(b-j)) - 1)
           This counts, by inclusion–exclusion over the b elements, the number of nonempty
           cross configurations where every element of the second group participates at least once,
           with 2^(a*(b-j)) representing choices of cross relations after excluding j elements. */
        int64_t cross = 0;
        for (int j = 0; j <= b; j++) {
            int64_t term = C(b, j);
            int64_t ways = sub_mod(pow_mod(2, (int64_t)a * (int64_t)(b - j)), 1);
            term = mul_mod(term, ways);
            if (j & 1) cross = sub_mod(cross, term);
            else       cross = add_mod(cross, term);
        }

        int64_t contrib = mul_mod(choose, mul_mod(internal, cross));
        ans = add_mod(ans, contrib);
    }

    ans = norm_mod(ans);
    printf("%lld\n", (long long)ans);
    return 0;
}
```
