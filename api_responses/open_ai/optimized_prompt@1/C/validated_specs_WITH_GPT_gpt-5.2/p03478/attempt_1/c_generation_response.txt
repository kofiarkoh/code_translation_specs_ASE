#include <stdio.h>

static int digitSum5(int x) {
    /* Sum of up to five least significant decimal digits.
       Higher-order digits are ignored by construction. */
    int sum = 0;
    int d;

    d = x % 10; if (d < 0) d = -d; sum += d; x /= 10;
    d = x % 10; if (d < 0) d = -d; sum += d; x /= 10;
    d = x % 10; if (d < 0) d = -d; sum += d; x /= 10;
    d = x % 10; if (d < 0) d = -d; sum += d; x /= 10;
    d = x % 10; if (d < 0) d = -d; sum += d;

    return sum;
}

int main(void) {
    int N, A, B;
    int total = 0;

    /* If input is missing/invalid, behavior is unspecified per spec. */
    (void)scanf("%d %d %d", &N, &A, &B);

    for (int i = 1; i <= N; i++) {
        int s = digitSum5(i);
        if (A <= s && s <= B) {
            total += i;
        }
    }

    printf("%d\n", total);
    return 0;
}
