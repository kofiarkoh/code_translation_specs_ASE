    long long d;
    int v;
} HeapNode;

typedef struct {
    HeapNode a[MAXH * MAXW + 5];
    int n;
} MinHeap;

static void heap_push(MinHeap *h, long long d, int v) {
    int i = ++h->n;
    h->a[i].d = d;
    h->a[i].v = v;
    while (i > 1) {
        int p = i >> 1;
        if (h->a[p].d <= h->a[i].d) break;
        HeapNode tmp = h->a[p];
        h->a[p] = h->a[i];
        h->a[i] = tmp;
        i = p;
    }
}

static HeapNode heap_pop(MinHeap *h) {
    HeapNode ret = h->a[1];
    HeapNode x = h->a[h->n--];
    int i = 1;
    while (1) {
        int l = i << 1;
        int r = l + 1;
        if (l > h->n) break;
        int m = l;
        if (r <= h->n && h->a[r].d < h->a[l].d) m = r;
        if (h->a[m].d >= x.d) break;
        h->a[i] = h->a[m];
        i = m;
    }
    h->a[i] = x;
    return ret;
}

static int heap_empty(const MinHeap *h) { return h->n == 0; }
