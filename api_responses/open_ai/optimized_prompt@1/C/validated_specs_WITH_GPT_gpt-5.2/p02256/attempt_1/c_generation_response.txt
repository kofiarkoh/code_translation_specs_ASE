```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

static uint64_t gcd_u64(uint64_t a, uint64_t b) {
    /* Euclidean algorithm; terminates when b becomes 0, avoiding division by zero. */
    while (b != 0) {
        uint64_t r = a % b;
        a = b;
        b = r;
    }
    return a;
}

int main(void) {
    long long x, y;

    /* Behavior is unspecified if input cannot be read as two integers;
       we do not add explicit handling beyond attempting to read. */
    (void)scanf("%lld %lld", &x, &y);

    /* Range check: inclusive [1, 1,000,000,000] for each input. */
    if (x < 1LL || x > 1000000000LL || y < 1LL || y > 1000000000LL) {
        /* No output required on invalid inputs. */
        return EXIT_FAILURE;
    }

    uint64_t a = (uint64_t)x;
    uint64_t b = (uint64_t)y;

    uint64_t g = gcd_u64(a, b);

    printf("%" PRIu64 "\n", g);
    return EXIT_SUCCESS;
}
```
