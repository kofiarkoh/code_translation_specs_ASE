        long long I = i - 1;

            int b = (int)((I >> p) & 1LL);
            low[p] = b;
            if (low[p] == 0) {
                I = ~I; /* full-width bitwise complement on signed 64-bit */
            }
        }

        for (int k = 1; k <= n; k++) {

        }

    }
