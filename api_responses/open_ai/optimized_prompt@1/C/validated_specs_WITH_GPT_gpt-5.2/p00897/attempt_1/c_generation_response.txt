```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define MAX_CITIES 256
#define MAX_NAME   64
#define MAX_ROADS  4096
#define MAX_CAP    1000  /* fixed upper bound on scaled capacity state space */

typedef struct {
    int to;
    int w;
    int next;
} Edge;

static char city_names[MAX_CITIES][MAX_NAME];
static int city_count;

static int head[MAX_CITIES];
static Edge edges[2 * MAX_ROADS];
static int edge_count;

static int is_refuel[MAX_CITIES];

static int dist_tbl[MAX_CITIES][MAX_CAP + 1];
static unsigned char inq[MAX_CITIES][MAX_CAP + 1];

static int q_city[MAX_CITIES * (MAX_CAP + 1)];
static int q_fuel[MAX_CITIES * (MAX_CAP + 1)];

static int get_city_id(const char *name) {
    for (int i = 0; i < city_count; i++) {
        if (strcmp(city_names[i], name) == 0) return i;
    }
    if (city_count >= MAX_CITIES) {
        /* unspecified behavior per spec if limits exceeded */
        return 0;
    }
    strncpy(city_names[city_count], name, MAX_NAME - 1);
    city_names[city_count][MAX_NAME - 1] = '\0';
    return city_count++;
}

static void add_undirected_edge(int a, int b, int w) {
    if (edge_count + 2 > (int)(sizeof(edges) / sizeof(edges[0]))) {
        /* unspecified behavior per spec if limits exceeded */
        return;
    }
    edges[edge_count].to = b;
    edges[edge_count].w = w;
    edges[edge_count].next = head[a];
    head[a] = edge_count++;

    edges[edge_count].to = a;
    edges[edge_count].w = w;
    edges[edge_count].next = head[b];
    head[b] = edge_count++;
}

int main(void) {
    int roads, refuels, basecap;

    while (scanf("%d", &roads) == 1) {
        if (roads == 0) break;

        if (scanf("%d %d", &refuels, &basecap) != 3) return 0;

        char origin_name[MAX_NAME], dest_name[MAX_NAME];
        if (scanf("%63s %63s", origin_name, dest_name) != 2) return 0;

        /* reset per dataset */
        city_count = 0;
        for (int i = 0; i < MAX_CITIES; i++) {
            head[i] = -1;
            is_refuel[i] = 0;
        }
        edge_count = 0;

        int origin = get_city_id(origin_name);
        int dest = get_city_id(dest_name);

        for (int i = 0; i < roads; i++) {
            char a_name[MAX_NAME], b_name[MAX_NAME];
            int w;
            if (scanf("%63s %63s %d", a_name, b_name, &w) != 3) return 0;
            int a = get_city_id(a_name);
            int b = get_city_id(b_name);
            add_undirected_edge(a, b, w);
        }

        for (int i = 0; i < refuels; i++) {
            char r_name[MAX_NAME];
            if (scanf("%63s", r_name) != 1) return 0;
            /* If refuel city not present in network/endpoints, it has no effect. */
            for (int j = 0; j < city_count; j++) {
                if (strcmp(city_names[j], r_name) == 0) {
                    is_refuel[j] = 1;
                    break;
                }
            }
        }

        int cap = basecap * 10;
        if (cap < 0) cap = 0; /* narrow handling; spec doesn't define negatives */
        if (cap > MAX_CAP) {
            /* unspecified behavior per spec if limits exceeded; clamp to keep state representable */
            cap = MAX_CAP;
        }

        /* initialize distances */
        for (int c = 0; c < city_count; c++) {
            for (int f = 0; f <= cap; f++) {
                dist_tbl[c][f] = INT_MAX;
                inq[c][f] = 0;
            }
        }

        /* start at origin with full capacity */
        dist_tbl[origin][cap] = 0;

        /* SPFA-like relaxation over (city, fuel) states; weights are nonnegative */
        int qh = 0, qt = 0;
        q_city[qt] = origin;
        q_fuel[qt] = cap;
        qt++;
        inq[origin][cap] = 1;

        while (qh != qt) {
            int u = q_city[qh];
            int fu_in = q_fuel[qh];
            qh++;
            if (qh >= (int)(sizeof(q_city) / sizeof(q_city[0]))) qh = 0;

            inq[u][fu_in] = 0;

            int fu = is_refuel[u] ? cap : fu_in;

            int du = dist_tbl[u][fu_in];
            if (du == INT_MAX) continue;

            for (int ei = head[u]; ei != -1; ei = edges[ei].next) {
                int v = edges[ei].to;
                int w = edges[ei].w;
                if (w < 0) continue; /* spec says nonnegative; ignore otherwise */
                if (w > fu) continue;

                int fv = fu - w;
                int nd = du + w;
                /* preserve monotone best-known distances */
                if (nd < dist_tbl[v][fv]) {
                    dist_tbl[v][fv] = nd;
                    if (!inq[v][fv]) {
                        inq[v][fv] = 1;
                        q_city[qt] = v;
                        q_fuel[qt] = fv;
                        qt++;
                        if (qt >= (int)(sizeof(q_city) / sizeof(q_city[0]))) qt = 0;
                        /* If queue overflows, behavior is unspecified by spec; we do not guard. */
                    }
                }
            }
        }

        int best = INT_MAX;
        for (int f = 0; f <= cap; f++) {
            if (dist_tbl[dest][f] < best) best = dist_tbl[dest][f];
        }

        if (best == INT_MAX) {
            printf("-1\n");
        } else {
            printf("%d\n", best);
        }
    }

    return 0;
}
```
