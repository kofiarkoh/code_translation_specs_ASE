```c
#include <stdio.h>
#include <stdlib.h>

#define MAXW 20000

typedef struct {
    int w;
    int s;
    int v;
} Item;

static int cmp_item(const void *a, const void *b) {
    const Item *ia = (const Item *)a;
    const Item *ib = (const Item *)b;

    /* Sort by (w + s) ascending; tie-break by w ascending.
       This ordering supports the standard DP for stacking feasibility. */
    long long ka = (long long)ia->w + (long long)ia->s;
    long long kb = (long long)ib->w + (long long)ib->s;
    if (ka < kb) return -1;
    if (ka > kb) return 1;
    if (ia->w < ib->w) return -1;
    if (ia->w > ib->w) return 1;
    return 0;
}

int main(void) {
    int n;
    if (scanf("%d", &n) != 1) {
        /* Undefined by spec on malformed input; just exit. */
        return 0;
    }

    /* Spec: negative item count => undefined behavior. We do not guard. */
    Item *items = (Item *)malloc((size_t)n * sizeof(Item));
    if (!items) {
        /* Undefined by spec on allocation failure; just exit. */
        return 0;
    }

    for (int i = 0; i < n; i++) {
        if (scanf("%d %d %d", &items[i].w, &items[i].s, &items[i].v) != 3) {
            /* Undefined by spec on malformed input; just exit. */
            free(items);
            return 0;
        }
    }

    qsort(items, (size_t)n, sizeof(Item), cmp_item);

    /* dp[W] = maximum total value for a feasible stack of total weight W,
       considering only weights 0..MAXW. Unreachable states are -1. */
    int *dp = (int *)malloc((MAXW + 1) * sizeof(int));
    if (!dp) {
        free(items);
        return 0;
    }

    for (int w = 0; w <= MAXW; w++) dp[w] = -1;
    dp[0] = 0;

    for (int i = 0; i < n; i++) {
        int wi = items[i].w;
        int si = items[i].s;
        int vi = items[i].v;

        /* Spec notes: if si > 20000 or intermediate weights exceed 20000,
           program may access out of bounds => undefined behavior.
           We intentionally do not clamp si to preserve that contract. */

        /* Place this item below an existing stack of weight 'above' (weight above it).
           Feasibility requires: above <= si.
           New total weight: above + wi (must be within 0..MAXW to be represented). */
        for (int above = si; above >= 0; above--) {
            if (above > MAXW) {
                /* If si > MAXW, this loop would otherwise index dp out of bounds.
                   Spec allows undefined behavior; we do not attempt to correct it.
                   This branch is only to keep compilation sane; it does not prevent UB
                   if si > MAXW because the loop bounds already imply it. */
            }
            if (dp[above] < 0) continue;
            int neww = above + wi;
            if (neww < 0 || neww > MAXW) continue;
            int cand = dp[above] + vi;
            if (cand > dp[neww]) dp[neww] = cand;
        }
    }

    int ans = 0;
    for (int w = 0; w <= MAXW; w++) {
        if (dp[w] > ans) ans = dp[w];
    }

    printf("%d\n", ans);

    free(dp);
    free(items);
    return 0;
}
```
