#include <stdio.h>
#include <math.h>

int main(void) {
    int a, b, c;

    /* Behavior is unspecified if fewer than three integers are provided. */
    if (scanf("%d %d %d", &a, &b, &c) != 3) {
        return 0;
    }

    /* Preconditions state a*b is representable in int; otherwise undefined. */
    int prod = a * b;

    /* Compute round((a*b)/2) and print with no fractional part. */
    double area = (double)prod / 2.0;
    long long out = llround(area);

    printf("%lld", out);
    return 0;
}
