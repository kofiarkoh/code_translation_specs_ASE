#include <stdio.h>
#include <stdlib.h>

typedef long long ll;

static int dsu_find(int *parent, int x) {
    while (parent[x] != x) {
        parent[x] = parent[parent[x]];
        x = parent[x];
    }
    return x;
}

static void dsu_union(int *parent, int *sz, int a, int b, ll *disconnected_pairs) {
    int ra = dsu_find(parent, a);
    int rb = dsu_find(parent, b);
    if (ra == rb) return;

    /* Union by size */
    if (sz[ra] < sz[rb]) {
        int tmp = ra; ra = rb; rb = tmp;
    }

    /* Merging two components of sizes sa and sb connects sa*sb previously disconnected pairs */
    *disconnected_pairs -= (ll)sz[ra] * (ll)sz[rb];

    parent[rb] = ra;
    sz[ra] += sz[rb];
}

int main(void) {
    int n, m;
    if (scanf("%d %d", &n, &m) != 2) return 0; /* Undefined behavior per spec if malformed */

    int *a = (int *)malloc((size_t)m * sizeof(int));
    int *b = (int *)malloc((size_t)m * sizeof(int));
    ll  *ans = (ll  *)malloc((size_t)m * sizeof(ll));
    int *parent = (int *)malloc((size_t)n * sizeof(int));
    int *sz     = (int *)malloc((size_t)n * sizeof(int));

    if (!a || !b || !ans || !parent || !sz) {
        /* Spec says no explicit error handling; exiting is a narrow, safe choice */
        return 0;
    }

    for (int i = 0; i < m; i++) {
        int x, y;
        if (scanf("%d %d", &x, &y) != 2) return 0; /* Undefined per spec if malformed */
        /* Convert 1-based to 0-based as implied by vertex set 0..n-1 */
        a[i] = x - 1;
        b[i] = y - 1;
    }

    for (int i = 0; i < n; i++) {
        parent[i] = i;
        sz[i] = 1;
    }

    ll disconnected_pairs = (ll)n * (ll)(n - 1) / 2;

    /* Process edges in reverse; ans[i] corresponds to graph with edges j > i */
    for (int i = m - 1; i >= 0; i--) {
        ans[i] = disconnected_pairs;
        dsu_union(parent, sz, a[i], b[i], &disconnected_pairs);
    }

    for (int i = 0; i < m; i++) {
        printf("%lld\n", ans[i]);
    }

    free(a);
    free(b);
    free(ans);
    free(parent);
    free(sz);
    return 0;
}
