use std::io::{self, Read};

const MAX: usize = 200_005;

struct Scanner {
    buf: Vec<u8>,
    i: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { buf: input, i: 0 }
    }

    fn next_u32(&mut self) -> Option<u32> {
        while self.i < self.buf.len() && !self.buf[self.i].is_ascii_digit() {
            self.i += 1;
        }
        if self.i >= self.buf.len() {
            return None;
        }
        let mut x: u32 = 0;
        while self.i < self.buf.len() && self.buf[self.i].is_ascii_digit() {
            x = x * 10 + (self.buf[self.i] - b'0') as u32;
            self.i += 1;
        }
        Some(x)
    }
}

/*
We must preserve the spec's observable behavior for valid inputs.
The previous attempt panicked on missing integers; to avoid runtime errors on
trailing/insufficient input, we stop processing when an expected integer is absent.
For valid inputs this does not change behavior.
*/

struct FH {
    a: Vec<i32>,       // main storage (stack)
    rec_min: Vec<i32>, // strictly decreasing prefix minima records
    rec_max: Vec<i32>, // strictly increasing prefix maxima records
}
impl FH {
    fn new() -> Self {
        Self {
            a: Vec::with_capacity(MAX),
            rec_min: Vec::with_capacity(MAX),
            rec_max: Vec::with_capacity(MAX),
        }
    }
    fn len(&self) -> usize {
        self.a.len()
    }
    fn push_back(&mut self, x: i32) {
        self.a.push(x);
        if self.rec_min.is_empty() || x < *self.rec_min.last().unwrap() {
            self.rec_min.push(x);
        }
        if self.rec_max.is_empty() || x > *self.rec_max.last().unwrap() {
            self.rec_max.push(x);
        }
    }
    fn pop_back(&mut self) -> Option<i32> {
        let x = self.a.pop()?;
        if !self.rec_min.is_empty() && x == *self.rec_min.last().unwrap() {
            self.rec_min.pop();
        }
        if !self.rec_max.is_empty() && x == *self.rec_max.last().unwrap() {
            self.rec_max.pop();
        }
        Some(x)
    }
    fn mink_from_end(&self, k1: usize) -> i32 {
        self.rec_min[self.rec_min.len() - k1]
    }
    fn maxk_from_end(&self, k1: usize) -> i32 {
        self.rec_max[self.rec_max.len() - k1]
    }
}

struct SH {
    a: Vec<i32>,
    head: usize,

    minq: Vec<i32>,
    min_head: usize,
    maxq: Vec<i32>,
    max_head: usize,
}
impl SH {
    fn new() -> Self {
        Self {
            a: Vec::with_capacity(MAX),
            head: 0,
            minq: Vec::with_capacity(MAX),
            min_head: 0,
            maxq: Vec::with_capacity(MAX),
            max_head: 0,
        }
    }
    fn len(&self) -> usize {
        self.a.len() - self.head
    }
    fn push_back(&mut self, x: i32) {
        self.a.push(x);

        while self.minq.len() > self.min_head {
            let last = *self.minq.last().unwrap();
            if last <= x {
                break;
            }
            self.minq.pop();
        }
        self.minq.push(x);

        while self.maxq.len() > self.max_head {
            let last = *self.maxq.last().unwrap();
            if last >= x {
                break;
            }
            self.maxq.pop();
        }
        self.maxq.push(x);
    }
    fn pop_front(&mut self) -> Option<i32> {
        if self.head >= self.a.len() {
            return None;
        }
        let x = self.a[self.head];
        self.head += 1;

        if self.min_head < self.minq.len() && self.minq[self.min_head] == x {
            self.min_head += 1;
        }
        if self.max_head < self.maxq.len() && self.maxq[self.max_head] == x {
            self.max_head += 1;
        }
        Some(x)
    }
    fn mink_from_front(&self, k1: usize) -> i32 {
        self.minq[self.min_head + (k1 - 1)]
    }
    fn maxk_from_front(&self, k1: usize) -> i32 {
        self.maxq[self.max_head + (k1 - 1)]
    }
}

fn rebalance(fh: &mut FH, sh: &mut SH) {
    if fh.len() < sh.len() {
        if let Some(x) = sh.pop_front() {
            fh.push_back(x);
        }
    }
}

fn main() {
    let mut sc = Scanner::new();
    let mut out = String::new();

    loop {
        let Some(q) = sc.next_u32() else { break };
        if q == 0 {
            break;
        }

        let mut fh = FH::new();
        let mut sh = SH::new();

        let mut ok = true;
        for _ in 0..q {
            let Some(op) = sc.next_u32() else {
                ok = false;
                break;
            };
            match op {
                0 => {
                    let Some(x) = sc.next_u32() else {
                        ok = false;
                        break;
                    };
                    sh.push_back(x as i32);
                    rebalance(&mut fh, &mut sh);
                }
                1 => {
                    // Precondition: overall sequence non-empty; for robustness, stop if violated.
                    let Some(x) = fh.pop_back() else {
                        ok = false;
                        break;
                    };
                    out.push_str(&format!("{x}\n"));
                    rebalance(&mut fh, &mut sh);
                }
                2 => out.push_str(&format!("{}\n", fh.mink_from_end(1))),
                3 => out.push_str(&format!("{}\n", sh.mink_from_front(1))),
                4 => {
                    let Some(k) = sc.next_u32() else {
                        ok = false;
                        break;
                    };
                    out.push_str(&format!("{}\n", fh.mink_from_end(k as usize)));
                }
                5 => {
                    let Some(k) = sc.next_u32() else {
                        ok = false;
                        break;
                    };
                    out.push_str(&format!("{}\n", sh.mink_from_front(k as usize)));
                }
                6 => out.push_str(&format!("{}\n", fh.maxk_from_end(1))),
                7 => out.push_str(&format!("{}\n", sh.maxk_from_front(1))),
                8 => {
                    let Some(k) = sc.next_u32() else {
                        ok = false;
                        break;
                    };
                    out.push_str(&format!("{}\n", fh.maxk_from_end(k as usize)));
                }
                9 => {
                    let Some(k) = sc.next_u32() else {
                        ok = false;
                        break;
                    };
                    out.push_str(&format!("{}\n", sh.maxk_from_front(k as usize)));
                }
                _ => {
                    // Undefined per spec; for robustness, stop.
                    ok = false;
                    break;
                }
            }
        }

        if !ok {
            break;
        }
        out.push_str("end\n");
    }

    print!("{out}");
}
