```rust
use std::cmp::Ordering;
use std::io::{self, Read};

const MAX_WORDS: usize = 100;
const MAX_WORD_LEN: usize = 30;
const MAX_PAGE: usize = 1000;

#[derive(Clone)]
struct Entry {
    word: String,
    pgnum: [u8; MAX_PAGE + 1],
    tm: i32,
}

fn strcmp_like(a: &str, b: &str) -> Ordering {
    let ab = a.as_bytes();
    let bb = b.as_bytes();
    let n = ab.len().min(bb.len());
    for i in 0..n {
        match ab[i].cmp(&bb[i]) {
            Ordering::Equal => {}
            non_eq => return non_eq,
        }
    }
    ab.len().cmp(&bb.len())
}

fn main() {
    // Read all input; treat it as whitespace-delimited tokens like scanf("%s %d").
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut entries: Vec<Entry> = Vec::new();

    // Consume records until scanf("%s %d") would fail.
    loop {
        let w = match it.next() {
            Some(s) => s,
            None => break,
        };
        let p_tok = match it.next() {
            Some(s) => s,
            None => break,
        };

        // Under the spec's assumptions this always succeeds and is in-range.
        // If not, stop consuming further input (closest to scanf failure).
        let page_i32 = match p_tok.parse::<i32>() {
            Ok(v) => v,
            Err(_) => break,
        };
        if page_i32 < 0 || page_i32 > MAX_PAGE as i32 {
            break;
        }
        let page = page_i32 as usize;

        // Find existing word.
        let mut idx_opt = None;
        for (i, e) in entries.iter().enumerate() {
            if e.word == w {
                idx_opt = Some(i);
                break;
            }
        }

        if let Some(i) = idx_opt {
            let e = &mut entries[i];
            e.pgnum[page] = 1;
            e.tm = e.tm.wrapping_add(1);
        } else {
            if entries.len() >= MAX_WORDS {
                // Undefined by spec; ignore additional distinct words.
                continue;
            }
            let mut word = w.to_string();
            if word.len() > MAX_WORD_LEN {
                // Undefined by spec; truncate to fit the stated storage size.
                word.truncate(MAX_WORD_LEN);
            }
            let mut e = Entry {
                word,
                pgnum: [0u8; MAX_PAGE + 1],
                tm: 0,
            };
            e.pgnum[page] = 1;
            e.tm = e.tm.wrapping_add(1);
            entries.push(e);
        }
    }

    entries.sort_by(|a, b| strcmp_like(&a.word, &b.word));

    // Stream output directly to stdout to avoid any memory-related differences.
    // (The spec only constrains observable output, but this is closer to C stdio usage.)
    let mut out = String::new();

    for e in &entries {
        out.push_str(&e.word);
        out.push('\n');

        let mut k: i32 = 0;
        for j in 0..=MAX_PAGE {
            if e.pgnum[j] == 1 {
                if k == e.tm.wrapping_sub(1) {
                    out.push_str(&j.to_string());
                    out.push('\n');
                    break;
                } else {
                    out.push_str(&j.to_string());
                    out.push(' ');
                    k = k.wrapping_add(1);
                }
            }
        }
    }

    print!("{out}");
}
```
