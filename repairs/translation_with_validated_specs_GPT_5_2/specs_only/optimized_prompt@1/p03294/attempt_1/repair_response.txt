```rust
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut it = input.split_whitespace();

    let count: i64 = it.next().unwrap().parse().unwrap();

    // Memory usage grows linearly with the count (stores the sequence).
    let mut values: Vec<i64> = Vec::with_capacity(count as usize);
    for _ in 0..count {
        let v: i64 = it.next().unwrap().parse().unwrap();
        values.push(v);
    }

    // Running total equals sum of first k integers minus k.
    let mut total: i64 = 0;
    for v in &values {
        total += *v - 1;
    }

    print!("{}", total);
}
```
