```rust
use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;

fn mod_norm(x: i64) -> i64 {
    // Preserve signed remainder behavior at the end, but keep internal values in [0, MOD).
    let r = x % MOD;
    if r < 0 { r + MOD } else { r }
}

fn pairwise_diff_sum(seq: &[i64]) -> i64 {
    // Computes sum_{0 <= i < j < n} (seq[j] - seq[i]) in input order.
    // Assumes intermediate arithmetic fits in i64 as per specification.
    let mut prefix_sum: i64 = 0;
    let mut acc: i64 = 0;
    for (i, &x) in seq.iter().enumerate() {
        let i_i64 = i as i64;
        // Contribution of x as the "later element" in pairs with all earlier elements:
        // sum_{k < i} (x - seq[k]) = i*x - prefix_sum
        acc += i_i64 * x - prefix_sum;
        prefix_sum += x;
    }
    acc
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let m: usize = it.next().unwrap().parse().unwrap();

    let mut a = Vec::with_capacity(n);
    for _ in 0..n {
        a.push(it.next().unwrap().parse::<i64>().unwrap());
    }

    let mut b = Vec::with_capacity(m);
    for _ in 0..m {
        b.push(it.next().unwrap().parse::<i64>().unwrap());
    }

    let sa = pairwise_diff_sum(&a);
    let sb = pairwise_diff_sum(&b);

    // Reduce modulo 1,000,000,007; final printed representative may be negative due to `%`.
    let pa = mod_norm(sa);
    let pb = mod_norm(sb);

    // Use i128 for the modular product to avoid overflow in Rust; spec only constrains i64
    // for intermediate arithmetic of the aggregates, not for this final multiplication.
    let prod = ((pa as i128) * (pb as i128)) % (MOD as i128);
    let out = (prod as i64) % MOD; // keep signed remainder behavior (non-negative here)
    println!("{}", out);
}
```
