```rust
use std::io::{self, Read};

const MAXN: usize = 2010;

fn mod_pow(mut a: i64, mut e: i64, m: i64) -> i64 {
    if m == 1 {
        return 0;
    }
    a %= m;
    if a < 0 {
        a += m;
    }
    let mut r: i64 = 1 % m;
    while e > 0 {
        if (e & 1) != 0 {
            r = (r * a) % m;
        }
        a = (a * a) % m;
        e >>= 1;
    }
    r
}

fn comb(a: i32, b: i32, m: i64, fact: &[i64], inv_fact: &[i64]) -> i64 {
    if a < 0 || b < 0 || b > a {
        return 0;
    }
    let au = a as usize;
    let bu = b as usize;
    if au >= fact.len() || bu >= fact.len() {
        // Specification implies no validation; return 0 for out-of-range to avoid panic.
        return 0;
    }
    (((fact[au] * inv_fact[bu]) % m) * inv_fact[au - bu]) % m
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: i32 = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };
    let m: i64 = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };

    // Precompute factorials and inverse factorials modulo m up to 2010 inclusive.
    let mut fact = vec![0i64; MAXN + 1];
    let mut inv_fact = vec![0i64; MAXN + 1];

    if m == 1 {
        // All computations mod 1 are 0; normalized output is 0.
        println!("0");
        return;
    }

    fact[0] = 1 % m;
    for i in 1..=MAXN {
        fact[i] = (fact[i - 1] * (i as i64)) % m;
    }
    inv_fact[MAXN] = mod_pow(fact[MAXN], m - 2, m);
    for i in (0..MAXN).rev() {
        inv_fact[i] = (inv_fact[i + 1] * ((i + 1) as i64)) % m;
    }

    let mut ans: i64 = 0;

    // Sum over i = 1..n-1
    for i in 1..=n - 1 {
        let ni = n - i;

        // Inclusion–exclusion subtotal over j = 0..(n-i-1)
        let mut subtotal: i64 = 0;
        for j in 0..=ni - 1 {
            let c = comb(ni, j, m, &fact, &inv_fact);
            let pow2 = mod_pow(2, (ni - j) as i64, m);
            let base = (pow2 - 1).rem_euclid(m);
            let term_pow = mod_pow(base, i as i64, m);
            let term = (c * term_pow) % m;

            if (j & 1) == 0 {
                subtotal += term;
            } else {
                subtotal -= term;
            }
            subtotal %= m;
        }
        subtotal = subtotal.rem_euclid(m);

        let cn_i = comb(n, i, m, &fact, &inv_fact);
        let e1 = (i as i64) * ((i - 1) as i64) / 2;
        let e2 = (ni as i64) * ((ni - 1) as i64) / 2;
        let p1 = mod_pow(2, e1, m);
        let p2 = mod_pow(2, e2, m);

        let mut add = cn_i;
        add = (add * p1) % m;
        add = (add * p2) % m;
        add = (add * subtotal) % m;

        ans += add;
        ans %= m;
    }

    // Print normalized to [0, m-1]
    println!("{}", ans.rem_euclid(m));
}
```
