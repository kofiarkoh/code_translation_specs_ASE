```rust
use std::collections::VecDeque;
use std::io::{self, Read};

#[derive(Default)]
struct FH {
    stack: Vec<i32>,
    min_hist: Vec<i32>,
    max_hist: Vec<i32>,
}

impl FH {
    fn new() -> Self {
        Self {
            stack: Vec::new(),
            min_hist: Vec::new(),
            max_hist: Vec::new(),
        }
    }

    fn len(&self) -> usize {
        self.stack.len()
    }

    fn push(&mut self, x: i32) {
        self.stack.push(x);

        if self.min_hist.last().map_or(true, |&m| x < m) {
            self.min_hist.push(x);
        }
        if self.max_hist.last().map_or(true, |&m| x > m) {
            self.max_hist.push(x);
        }
    }

    fn pop(&mut self) -> i32 {
        let x = self.stack.pop().unwrap();

        if self.min_hist.last().copied() == Some(x) {
            self.min_hist.pop();
        }
        if self.max_hist.last().copied() == Some(x) {
            self.max_hist.pop();
        }
        x
    }

    fn query_min_k_recent(&self, k: usize) -> i32 {
        // k is 1-based from the end of history
        let idx = self.min_hist.len() - k;
        self.min_hist[idx]
    }

    fn query_max_k_recent(&self, k: usize) -> i32 {
        let idx = self.max_hist.len() - k;
        self.max_hist[idx]
    }
}

#[derive(Default)]
struct SH {
    q: VecDeque<i32>,
    // Monotone sequences stored from front to back.
    // For min_seq: non-decreasing (front is smallest among prefix minima candidates).
    // For max_seq: non-increasing.
    min_seq: VecDeque<i32>,
    max_seq: VecDeque<i32>,
}

impl SH {
    fn new() -> Self {
        Self {
            q: VecDeque::new(),
            min_seq: VecDeque::new(),
            max_seq: VecDeque::new(),
        }
    }

    fn len(&self) -> usize {
        self.q.len()
    }

    fn push_back(&mut self, x: i32) {
        self.q.push_back(x);

        while let Some(&back) = self.min_seq.back() {
            if back > x {
                self.min_seq.pop_back();
            } else {
                break;
            }
        }
        self.min_seq.push_back(x);

        while let Some(&back) = self.max_seq.back() {
            if back < x {
                self.max_seq.pop_back();
            } else {
                break;
            }
        }
        self.max_seq.push_back(x);
    }

    fn pop_front(&mut self) -> i32 {
        let x = self.q.pop_front().unwrap();

        if self.min_seq.front().copied() == Some(x) {
            self.min_seq.pop_front();
        }
        if self.max_seq.front().copied() == Some(x) {
            self.max_seq.pop_front();
        }
        x
    }

    fn query_min_k_pos(&self, k: usize) -> i32 {
        // k is 1-based from the front of the maintained monotone sequence
        self.min_seq[k - 1]
    }

    fn query_max_k_pos(&self, k: usize) -> i32 {
        self.max_seq[k - 1]
    }
}

fn rebalance_if_needed(fh: &mut FH, sh: &mut SH) {
    // Only move from SH -> FH, and only when FH becomes smaller than SH.
    if fh.len() < sh.len() {
        let x = sh.pop_front();
        fh.push(x);
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let q_opt = it.next();
        if q_opt.is_none() {
            break;
        }
        let q: usize = q_opt.unwrap().parse().unwrap();
        if q == 0 {
            break;
        }

        let mut fh = FH::new();
        let mut sh = SH::new();

        for _ in 0..q {
            let op: i32 = it.next().unwrap().parse().unwrap();
            match op {
                0 => {
                    let x: i32 = it.next().unwrap().parse().unwrap();
                    sh.push_back(x);
                    rebalance_if_needed(&mut fh, &mut sh);
                }
                1 => {
                    let v = fh.pop();
                    rebalance_if_needed(&mut fh, &mut sh);
                    out.push_str(&format!("{v}\n"));
                }
                2 => {
                    let v = fh.query_min_k_recent(1);
                    out.push_str(&format!("{v}\n"));
                }
                3 => {
                    let v = sh.query_min_k_pos(1);
                    out.push_str(&format!("{v}\n"));
                }
                4 => {
                    let k: usize = it.next().unwrap().parse().unwrap();
                    let v = fh.query_min_k_recent(k);
                    out.push_str(&format!("{v}\n"));
                }
                5 => {
                    let k: usize = it.next().unwrap().parse().unwrap();
                    let v = sh.query_min_k_pos(k);
                    out.push_str(&format!("{v}\n"));
                }
                6 => {
                    let v = fh.query_max_k_recent(1);
                    out.push_str(&format!("{v}\n"));
                }
                7 => {
                    let v = sh.query_max_k_pos(1);
                    out.push_str(&format!("{v}\n"));
                }
                8 => {
                    let k: usize = it.next().unwrap().parse().unwrap();
                    let v = fh.query_max_k_recent(k);
                    out.push_str(&format!("{v}\n"));
                }
                9 => {
                    let k: usize = it.next().unwrap().parse().unwrap();
                    let v = sh.query_max_k_pos(k);
                    out.push_str(&format!("{v}\n"));
                }
                _ => {
                    // Undefined behavior per spec for malformed input/opcodes.
                    // We keep it unreachable for valid inputs.
                    unreachable!();
                }
            }
        }

        out.push_str("end\n");
    }

    print!("{out}");
}
```
