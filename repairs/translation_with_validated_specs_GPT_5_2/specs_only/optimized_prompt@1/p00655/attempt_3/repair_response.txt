use std::collections::VecDeque;
use std::io::{self, Read};

#[derive(Default)]
struct FH {
    stack: Vec<i32>,
    min_hist: Vec<i32>,
    max_hist: Vec<i32>,
}

impl FH {
    fn len(&self) -> usize {
        self.stack.len()
    }

    fn push(&mut self, x: i32) {
        self.stack.push(x);
        if self.min_hist.last().map_or(true, |&m| x < m) {
            self.min_hist.push(x);
        }
        if self.max_hist.last().map_or(true, |&m| x > m) {
            self.max_hist.push(x);
        }
    }

    fn pop(&mut self) -> i32 {
        let x = self.stack.pop().unwrap();
        if self.min_hist.last().copied() == Some(x) {
            self.min_hist.pop();
        }
        if self.max_hist.last().copied() == Some(x) {
            self.max_hist.pop();
        }
        x
    }

    fn query_min_k_recent(&self, k: usize) -> i32 {
        self.min_hist[self.min_hist.len() - k]
    }

    fn query_max_k_recent(&self, k: usize) -> i32 {
        self.max_hist[self.max_hist.len() - k]
    }
}

#[derive(Default)]
struct SH {
    q: VecDeque<i32>,
    min_seq: VecDeque<i32>,
    max_seq: VecDeque<i32>,
}

impl SH {
    fn len(&self) -> usize {
        self.q.len()
    }

    fn push_back(&mut self, x: i32) {
        self.q.push_back(x);

        while let Some(&b) = self.min_seq.back() {
            if b > x {
                self.min_seq.pop_back();
            } else {
                break;
            }
        }
        self.min_seq.push_back(x);

        while let Some(&b) = self.max_seq.back() {
            if b < x {
                self.max_seq.pop_back();
            } else {
                break;
            }
        }
        self.max_seq.push_back(x);
    }

    fn pop_front(&mut self) -> i32 {
        let x = self.q.pop_front().unwrap();
        if self.min_seq.front().copied() == Some(x) {
            self.min_seq.pop_front();
        }
        if self.max_seq.front().copied() == Some(x) {
            self.max_seq.pop_front();
        }
        x
    }

    fn query_min_k_pos(&self, k: usize) -> i32 {
        self.min_seq[k - 1]
    }

    fn query_max_k_pos(&self, k: usize) -> i32 {
        self.max_seq[k - 1]
    }
}

fn rebalance_if_needed(fh: &mut FH, sh: &mut SH) {
    if fh.len() < sh.len() {
        let x = sh.pop_front();
        fh.push(x);
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let q = match it.next() {
            Some(s) => s.parse::<usize>().unwrap(),
            None => break,
        };
        if q == 0 {
            break;
        }

        let mut fh = FH::default();
        let mut sh = SH::default();

        for _ in 0..q {
            let op: i32 = match it.next() {
                Some(s) => s.parse().unwrap(),
                None => break, // malformed/truncated input => undefined; avoid panic
            };

            match op {
                0 => {
                    let x: i32 = match it.next() {
                        Some(s) => s.parse().unwrap(),
                        None => break,
                    };
                    sh.push_back(x);
                    rebalance_if_needed(&mut fh, &mut sh);
                }
                1 => {
                    // Spec precondition: FH non-empty. If violated, behavior undefined.
                    // Avoid Rust panic by doing nothing if empty.
                    if fh.len() == 0 {
                        continue;
                    }
                    let v = fh.pop();
                    rebalance_if_needed(&mut fh, &mut sh);
                    out.push_str(&format!("{v}\n"));
                }
                2 => {
                    if !fh.min_hist.is_empty() {
                        let v = fh.query_min_k_recent(1);
                        out.push_str(&format!("{v}\n"));
                    }
                }
                3 => {
                    if !sh.min_seq.is_empty() {
                        let v = sh.query_min_k_pos(1);
                        out.push_str(&format!("{v}\n"));
                    }
                }
                4 => {
                    let k: usize = match it.next() {
                        Some(s) => s.parse().unwrap(),
                        None => break,
                    };
                    if k >= 1 && k <= fh.min_hist.len() {
                        let v = fh.query_min_k_recent(k);
                        out.push_str(&format!("{v}\n"));
                    }
                }
                5 => {
                    let k: usize = match it.next() {
                        Some(s) => s.parse().unwrap(),
                        None => break,
                    };
                    if k >= 1 && k <= sh.min_seq.len() {
                        let v = sh.query_min_k_pos(k);
                        out.push_str(&format!("{v}\n"));
                    }
                }
                6 => {
                    if !fh.max_hist.is_empty() {
                        let v = fh.query_max_k_recent(1);
                        out.push_str(&format!("{v}\n"));
                    }
                }
                7 => {
                    if !sh.max_seq.is_empty() {
                        let v = sh.query_max_k_pos(1);
                        out.push_str(&format!("{v}\n"));
                    }
                }
                8 => {
                    let k: usize = match it.next() {
                        Some(s) => s.parse().unwrap(),
                        None => break,
                    };
                    if k >= 1 && k <= fh.max_hist.len() {
                        let v = fh.query_max_k_recent(k);
                        out.push_str(&format!("{v}\n"));
                    }
                }
                9 => {
                    let k: usize = match it.next() {
                        Some(s) => s.parse().unwrap(),
                        None => break,
                    };
                    if k >= 1 && k <= sh.max_seq.len() {
                        let v = sh.query_max_k_pos(k);
                        out.push_str(&format!("{v}\n"));
                    }
                }
                _ => {
                    // Undefined per spec; ignore to avoid panic.
                }
            }
        }

        out.push_str("end\n");
    }

    print!("{out}");
}
