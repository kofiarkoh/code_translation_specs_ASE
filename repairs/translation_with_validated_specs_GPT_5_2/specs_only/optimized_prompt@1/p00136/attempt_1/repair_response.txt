```rust
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i64 = it.next().unwrap().parse().unwrap();

    let mut c1: usize = 0;
    let mut c2: usize = 0;
    let mut c3: usize = 0;
    let mut c4: usize = 0;
    let mut c5: usize = 0;
    let mut c6: usize = 0;

    // Behavior for negative n is unspecified by the spec; we do not add handling.
    for _ in 0..(n as usize) {
        let h: f64 = it.next().unwrap().parse().unwrap();

        // Use ordered comparisons exactly as specified.
        if h < 165.0 {
            c1 += 1;
        } else if h < 170.0 {
            c2 += 1;
        } else if h < 175.0 {
            c3 += 1;
        } else if h < 180.0 {
            c4 += 1;
        } else if h < 185.0 {
            c5 += 1;
        } else {
            c6 += 1;
        }
    }

    // Exact output format: six lines, "k:" followed immediately by '*' repeated count times.
    println!("1:{}", "*".repeat(c1));
    println!("2:{}", "*".repeat(c2));
    println!("3:{}", "*".repeat(c3));
    println!("4:{}", "*".repeat(c4));
    println!("5:{}", "*".repeat(c5));
    println!("6:{}", "*".repeat(c6));
}
```
