```rust
use std::io::{self, Read};

const MAX: usize = 100;
const PGMAX: usize = 1001; // 0..=1000

#[derive(Clone)]
struct Entry {
    word: Vec<u8>,     // store raw bytes to emulate C strcmp-style ordering on bytes
    pg: [u8; PGMAX],   // 0/1 presence
    tm: i32,           // count of records for this word (not distinct pages)
}

fn main() {
    // Read all input and split on ASCII whitespace, approximating scanf("%s %d").
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut entries: Vec<Entry> = Vec::new();

    loop {
        let w = match it.next() {
            Some(s) => s.as_bytes().to_vec(),
            None => break,
        };
        let p_str = match it.next() {
            Some(s) => s,
            None => break,
        };
        let p: i32 = match p_str.parse() {
            Ok(v) => v,
            Err(_) => break, // scanf("%d") would fail; stop consuming further
        };

        // Find existing entry by exact byte equality (strcmp equality on bytes).
        let mut idx = None;
        for (i, e) in entries.iter().enumerate() {
            if e.word == w {
                idx = Some(i);
                break;
            }
        }

        if let Some(i) = idx {
            if (0..PGMAX as i32).contains(&p) {
                entries[i].pg[p as usize] = 1;
            }
            entries[i].tm = entries[i].tm.wrapping_add(1);
        } else {
            // Exceeding MAX is UB in the C version; here we avoid OOB by ignoring extra distinct words.
            if entries.len() >= MAX {
                continue;
            }
            let mut e = Entry {
                word: w,
                // C version leaves these uninitialized; Rust must initialize. Choose zeroed state.
                pg: [0u8; PGMAX],
                tm: 0,
            };
            if (0..PGMAX as i32).contains(&p) {
                e.pg[p as usize] = 1;
            }
            e.tm = e.tm.wrapping_add(1);
            entries.push(e);
        }
    }

    // Sort ascending lexicographic order by raw bytes (strcmp-like).
    entries.sort_by(|a, b| a.word.cmp(&b.word));

    let mut out = String::new();

    for e in &entries {
        // Words are printed as-is; if non-UTF8 bytes existed, C would still print them.
        // Here, input came from UTF-8 String, so bytes are valid UTF-8.
        out.push_str(std::str::from_utf8(&e.word).unwrap());
        out.push('\n');

        // Scan pages 0..PGMAX-1 and print indices with pg[j]==1.
        // Stop after printing exactly tm such indices; if tm is larger, no terminating newline is printed.
        let mut printed: i32 = 0;
        for j in 0..PGMAX {
            if e.pg[j] == 1 {
                printed = printed.wrapping_add(1);
                if printed == e.tm {
                    out.push_str(&j.to_string());
                    out.push('\n');
                    break;
                } else {
                    out.push_str(&j.to_string());
                    out.push(' ');
                }
            }
        }
    }

    print!("{out}");
}
```
