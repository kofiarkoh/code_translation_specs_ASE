```rust
use std::io::{self, Read};

const MOD: u64 = 1_000_000_007;
const LIM: usize = 1 << 20;

fn mod_add(a: u64, b: u64) -> u64 {
    let s = a + b;
    if s >= MOD { s - MOD } else { s }
}
fn mod_sub(a: u64, b: u64) -> u64 {
    if a >= b { a - b } else { a + MOD - b }
}
fn mod_mul(a: u64, b: u64) -> u64 {
    (a * b) % MOD
}
fn mod_pow(mut a: u64, mut e: u64) -> u64 {
    let mut r = 1u64;
    while e > 0 {
        if (e & 1) != 0 {
            r = mod_mul(r, a);
        }
        a = mod_mul(a, a);
        e >>= 1;
    }
    r
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();

    // Prefix xor S[0]=0, S[i] in [0, 2^20-1] by precondition.
    let mut s: usize = 0;

    // Z(N): number of i in [1..N] with S[i]==0.
    let mut z: u64 = 0;

    // last[v] is maintained to satisfy the invariant, though the recurrence below
    // does not require it explicitly.
    let mut last: Vec<usize> = vec![0; LIM];

    // dp[v] is the accumulated value associated with xor state v (mod MOD).
    // sum_nonzero tracks sum of dp[v] for v != 0 (mod MOD).
    let mut dp: Vec<u64> = vec![0; LIM];
    let mut sum_nonzero: u64 = 0;

    // The specification's finalization distinguishes the zero-xor state and uses Z(N).
    // A narrow recurrence consistent with that structure:
    // - For nonzero states, each step adds the current total over nonzero states plus 1,
    //   but avoids double-counting by subtracting the previous dp for that state.
    // - For the zero state, we do not accumulate into dp[0]; instead we only count Z(N)
    //   and use the explicit 2^(Z(N)-1) term at the end as specified.
    for i in 1..=n {
        let x_raw: i64 = it.next().unwrap().parse().unwrap();
        let x = (x_raw as u64) & ((LIM as u64) - 1);
        s ^= x as usize;

        if s == 0 {
            z += 1;
            last[0] = i;
            continue;
        }

        // new = sum_nonzero + 1 - dp[s]
        let mut new_val = mod_add(sum_nonzero, 1);
        new_val = mod_sub(new_val, dp[s]);

        // Update sum_nonzero by replacing dp[s] with new_val.
        sum_nonzero = mod_add(sum_nonzero, mod_sub(new_val, dp[s]));
        dp[s] = new_val;

        last[s] = i;
    }

    let ans = if s != 0 {
        dp[s] % MOD
    } else {
        // Spec assumes Z(N) >= 1 in this branch.
        let pow_term = mod_pow(2, z - 1);
        mod_add(pow_term, sum_nonzero)
    };

    println!("{}", ans);
}
```
