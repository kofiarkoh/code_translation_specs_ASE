use std::io::{self, Read};

fn main() {
    // Read entire stdin, then split into whitespace-delimited tokens.
    // This matches the spec's token-based processing and clean EOF termination.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut out = String::new();

    for token in input.split_whitespace() {
        // Spec assumes tokens are at most 99 characters; if longer, behavior is undefined.
        // In Rust we won't corrupt memory; we simply proceed.
        let bytes = token.as_bytes();
        let mut i = 0usize;

        // Transform left-to-right.
        while i < bytes.len() {
            if bytes[i] == b'@' {
                // Spec precondition: '@' is followed by a digit and then a character.
                // If not, behavior is undefined; we choose the narrowest behavior:
                // stop processing this token to avoid panics/out-of-bounds.
                if i + 2 >= bytes.len() {
                    break;
                }
                let d = bytes[i + 1];
                if !(b'0'..=b'9').contains(&d) {
                    break;
                }
                let count = (d - b'0') as usize;
                let c = bytes[i + 2] as char;
                for _ in 0..count {
                    out.push(c);
                }
                i += 3;
            } else {
                out.push(bytes[i] as char);
                i += 1;
            }
        }

        out.push('\n');
    }

    print!("{out}");
}
