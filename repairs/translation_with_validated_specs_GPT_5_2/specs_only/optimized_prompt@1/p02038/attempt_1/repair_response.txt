```rust
use std::io::{self, Read};
use std::process;

const CAPACITY: usize = 1000;

fn main() {
    let mut input = String::new();
    if io::stdin().read_to_string(&mut input).is_err() {
        process::exit(1);
    }

    let mut it = input.split_whitespace();

    let n: usize = match it.next().and_then(|s| s.parse::<usize>().ok()) {
        Some(v) => v,
        None => process::exit(1),
    };

    // Behavior is undefined by the specification if n < 2 or n > CAPACITY.
    // We intentionally do not add explicit handling beyond fixed storage.
    let mut tokens: Vec<char> = Vec::with_capacity(CAPACITY);

    for _ in 0..n {
        let tok = match it.next() {
            Some(t) => t,
            None => process::exit(1),
        };
        let ch = tok.chars().next().unwrap_or('\0'); // token provides at least one character per preconditions
        tokens.push(ch);
    }

    // Left-associative reduction with rule:
    // result is false only when left operand is true and right operand is false; otherwise true.
    // Intermediate result is always 'T' or 'F'.
    let mut acc = if tokens[0] == 'T' { 'T' } else { 'F' };
    for &ch in tokens.iter().skip(1) {
        let right_is_f = ch == 'F';
        if acc == 'T' && right_is_f {
            acc = 'F';
        } else {
            acc = 'T';
        }
    }

    print!("{}\n", acc);
}
```
