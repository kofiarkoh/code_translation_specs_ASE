```rust
use std::io::{self, Read};

fn is_prime(n: i64) -> bool {
    // Specification defines primality by trial division from i=2 while i*i <= n,
    // and prime iff no divisors are found in that range.
    // Preconditions assume n is such that i*i does not overflow; we preserve that
    // by using i <= n / i instead of i*i <= n (same logical condition for n>0).
    if n < 2 {
        return false;
    }
    let mut i: i64 = 2;
    while i <= n / i {
        if n % i == 0 {
            return false;
        }
        i += 1;
    }
    true
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).ok();

    let mut out = String::new();

    for tok in input.split_whitespace() {
        let x: i64 = match tok.parse() {
            Ok(v) => v,
            Err(_) => break, // unspecified behavior on read/parse failure; terminate processing
        };

        if x == 0 {
            break; // sentinel: no output for this value
        }

        // Per specification, behavior is only defined/terminating for x >= 2.
        // We implement the described decrement/increment searches directly.
        let mut l = x;
        while !is_prime(l) {
            l -= 1;
        }

        let mut u = x;
        while !is_prime(u) {
            u += 1;
        }

        let diff = u - l;
        out.push_str(&diff.to_string());
        out.push('\n');
    }

    print!("{out}");
}
```
