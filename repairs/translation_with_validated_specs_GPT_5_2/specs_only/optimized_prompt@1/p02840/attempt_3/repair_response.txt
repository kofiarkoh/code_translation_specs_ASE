```rust
use std::cmp::max;
use std::collections::BTreeMap;
use std::io::{self, Read};

fn main() {
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_whitespace();

    let n: i64 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let x: i64 = it.next().unwrap().parse().unwrap();
    let d: i64 = it.next().unwrap().parse().unwrap();

    // Special cases per specification.
    if x == 0 && d == 0 {
        println!("1");
        return;
    }
    if d == 0 {
        println!("{}", n + 1);
        return;
    }
    if x == 0 {
        let nn = n as i128;
        println!("{}", nn * (nn - 1) / 2 + 1);
        return;
    }

    // Normalize D to be positive; this preserves the set of achievable sums.
    let mut dd = d;
    let mut xx = x;
    if dd < 0 {
        dd = -dd;
        xx = -xx;
    }

    // Euclidean division: xx = a*dd + b, with 0 <= b < dd.
    let a = xx.div_euclid(dd);
    let b = xx.rem_euclid(dd);

    // For each subset size m, subset sum is:
    //   m*xx + dd*s, where s is sum of chosen indices.
    // Rewrite:
    //   m*xx + dd*s = dd*(s + m*a) + (m*b mod dd)
    //
    // For fixed m, s ranges over a contiguous interval [L(m), R(m)] (inclusive):
    //   L(m) = m(m-1)/2
    //   R(m) = m(2N - m - 1)/2
    //
    // Then q = s + m*a ranges over [L(m)+m*a, R(m)+m*a].
    // Group by remainder r = (m*b) mod dd, merge q-intervals, sum lengths.
    let mut mp: BTreeMap<i64, Vec<(i64, i64)>> = BTreeMap::new();

    for m in 0..=n {
        let mm = m as i128;
        let nn = n as i128;

        let l_s = mm * (mm - 1) / 2;
        let r_s = mm * (2 * nn - mm - 1) / 2;

        let shift = (m as i128) * (a as i128);
        let l_q = l_s + shift;
        let r_q = r_s + shift;

        let r = (((m as i128) * (b as i128)) % (dd as i128)) as i64;

        mp.entry(r)
            .or_default()
            .push((l_q as i64, r_q as i64));
    }

    let mut ans: i128 = 0;
    for (_r, mut intervals) in mp {
        intervals.sort_by(|(l1, r1), (l2, r2)| l1.cmp(l2).then(r1.cmp(r2)));

        let mut cur_l: i64 = 0;
        let mut cur_r: i64 = -1;
        let mut first = true;

        for (l, r) in intervals {
            if first {
                cur_l = l;
                cur_r = r;
                first = false;
                continue;
            }
            if l <= cur_r + 1 {
                cur_r = max(cur_r, r);
            } else {
                ans += (cur_r as i128) - (cur_l as i128) + 1;
                cur_l = l;
                cur_r = r;
            }
        }
        if !first {
            ans += (cur_r as i128) - (cur_l as i128) + 1;
        }
    }

    println!("{}", ans);
}
```
