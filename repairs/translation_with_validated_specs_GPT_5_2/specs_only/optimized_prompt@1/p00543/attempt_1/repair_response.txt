```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin; behavior is unspecified on malformed/missing input per spec.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let m: i64 = it.next().unwrap().parse().unwrap();

    let mut a: Vec<i64> = Vec::with_capacity(n);
    for _ in 0..n {
        let v: i64 = it.next().unwrap().parse().unwrap();
        a.push(v);
    }

    // For k = 1..m, perform exactly one left-to-right sweep of adjacent pairs.
    // Remainder semantics follow Rust's `%` for signed integers (truncating division),
    // matching typical C-like remainder behavior for negative values.
    for k in 1..=m {
        if n >= 2 {
            for i in 0..(n - 1) {
                let left_rem = a[i] % k;
                let right_rem = a[i + 1] % k;
                if left_rem > right_rem {
                    a.swap(i, i + 1);
                }
            }
        }
    }

    let mut out = String::new();
    for v in a {
        out.push_str(&format!("{v}\n"));
    }
    print!("{out}");
}
```
