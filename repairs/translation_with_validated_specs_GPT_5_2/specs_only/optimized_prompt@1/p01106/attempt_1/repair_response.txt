```rust
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).ok();

    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: i32 = match n_opt.unwrap().parse() {
            Ok(v) => v,
            Err(_) => break,
        };
        let i: i64 = match it.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => break,
        };
        let j: i64 = match it.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => break,
        };

        if n <= 0 {
            break;
        }

        let n_usize = n as usize;

        // Fixed-buffer style per spec (original program had size 62 and wrote up to n+1).
        // We keep a similar bound but allocate dynamically; behavior for n > 60 is still
        // "meaningless/undefined" per spec, but we do not intentionally crash.
        let mut low: Vec<i64> = vec![0; n_usize.max(1)];

        // Working values are signed 64-bit; complement is full-width two's-complement.
        let mut I: i64 = i.wrapping_sub(1);
        let mut J: i64 = j.wrapping_sub(1);

        // Phase A: derive low control sequence from I, stored by bit position p.
        for k in 1..=n_usize {
            let p = n_usize - k; // p goes n-1 down to 0
            // In C, right shift of signed negative is implementation-defined; intended behavior
            // here is arithmetic shift as on typical platforms. Rust's >> on i64 is arithmetic.
            let b = (I >> (p as u32)) & 1;
            low[p] = b;
            if low[p] == 0 {
                I = !I;
            }
        }

        // Phase B: consume low in forward index order while p decreases.
        for k in 1..=n_usize {
            let p = n_usize - k;
            let c = (J >> (p as u32)) & 1;
            let ctrl = low[k - 1];
            if c == ctrl {
                out.push('L');
            } else {
                out.push('R');
            }
            if ctrl == 0 {
                J = !J;
            }
        }
        out.push('\n');
    }

    print!("{out}");
}
```
