```rust
use std::io::{self, Read, Write};

fn main() {
    const CAP: usize = 114_514;
    let mut buf = vec![0u8; CAP];

    // Perform exactly one read from stdin.
    let l = io::stdin().read(&mut buf).unwrap() as usize;

    // Per spec, behavior is undefined if l < 2; we do not add explicit handling.
    // Decision rule:
    // n = L - 1
    // eq = (buf[0] == buf[n-1]) where n-1 == L-2
    // output "First" if ((n + eq) & 1) != 0 else "Second"
    let n = l - 1;
    let eq: usize = if buf[0] == buf[n - 1] { 1 } else { 0 };
    let cond = (n + eq) & 1;

    let mut stdout = io::stdout();
    if cond != 0 {
        let _ = stdout.write_all(b"First\n");
    } else {
        let _ = stdout.write_all(b"Second\n");
    }
}
```
