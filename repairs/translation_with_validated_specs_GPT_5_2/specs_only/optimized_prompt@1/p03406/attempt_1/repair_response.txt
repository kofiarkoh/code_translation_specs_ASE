use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;
const MAX: usize = 1 << 16;

fn mod_pow(mut a: i64, mut e: i64) -> i64 {
    let mut r: i64 = 1;
    a %= MOD;
    while e > 0 {
        if (e & 1) != 0 {
            r = (r * a) % MOD;
        }
        a = (a * a) % MOD;
        e >>= 1;
    }
    r
}

fn choose(n: i64, k: i64, ff: &[i64], inv: &[i64]) -> i64 {
    if k > n {
        return 0;
    }
    // Specification guarantees lookups are within precomputed range when preconditions hold.
    let ni = n as usize;
    let ki = k as usize;
    let nki = (n - k) as usize;
    (((ff[ni] * inv[ki]) % MOD) * inv[nki]) % MOD
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return, // unspecified; keep minimal behavior
    };
    let m: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let nn: usize = 1usize << n;

    // Precompute factorials and inverse factorials up to 2^16.
    let mut ff = vec![0i64; MAX + 1];
    let mut inv = vec![0i64; MAX + 1];
    ff[0] = 1;
    for i in 1..=MAX {
        ff[i] = (ff[i - 1] * (i as i64)) % MOD;
    }
    inv[MAX] = mod_pow(ff[MAX], MOD - 2);
    for i in (1..=MAX).rev() {
        inv[i - 1] = (inv[i] * (i as i64)) % MOD;
    }

    // Read v and transform to aa[h] = 2^n - v[h].
    let mut aa = vec![0i64; m];
    for h in 0..m {
        let v: i64 = it.next().unwrap().parse().unwrap();
        aa[h] = (nn as i64) - v;
    }

    // DP over masks b in [0, 2^n - 1].
    let mut dp = vec![0i64; nn];
    dp[0] = 1;

    // Process h from m-1 down to 0, update dp in-place, b descending.
    for h in (0..m).rev() {
        let aah = aa[h];
        for b in (0..nn).rev() {
            let cur = dp[b];
            if cur == 0 {
                continue;
            }
            for i in 0..n {
                if (b & (1usize << i)) == 0 {
                    let bp = b | (1usize << i);
                    // Transition factor depends on numeric b.
                    let n1 = aah - (b as i64);
                    let k1 = (1i64 << i) - 1;
                    let add = (cur * choose(n1, k1, &ff, &inv)) % MOD;
                    dp[bp] += add;
                    if dp[bp] >= MOD {
                        dp[bp] -= MOD;
                    }
                }
            }
        }
    }

    // Final alternating-sign aggregation.
    let mut ans: i64 = 0;
    for b in 0..nn {
        let mut x = dp[b];
        if x == 0 {
            continue;
        }
        let mut n_rem: i64 = ((nn - 1 - b) as i64);
        for i in 0..n {
            if (b & (1usize << i)) == 0 {
                let k = 1i64 << i;
                x = (x * choose(n_rem, k, &ff, &inv)) % MOD;
                n_rem -= k;
            }
        }

        // Sign determined by parity of (2n - popcount(b)), equivalent to parity of popcount(b).
        let pc = (b as u32).count_ones() & 1;
        if pc == 0 {
            ans += x;
        } else {
            ans -= x;
        }
        ans %= MOD;
    }

    // Final scaling by ∏ (ff[1<<i] * 2) for i=0..n-1.
    for i in 0..n {
        ans = (ans * ff[1usize << i]) % MOD;
        ans = (ans * 2) % MOD;
    }

    if ans < 0 {
        ans += MOD;
    }
    println!("{}", ans);
}
