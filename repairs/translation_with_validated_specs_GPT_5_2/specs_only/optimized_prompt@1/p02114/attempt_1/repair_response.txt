```rust
use std::collections::VecDeque;
use std::io::{self, Read};

fn main() {
    // Read entire stdin; input is assumed well-formed per specification.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    // Read n*n cells as tokens; each token is treated as a character cell.
    // Only the first char of each token is used (narrowest reasonable interpretation).
    let mut grid: Vec<Vec<u8>> = vec![vec![b'.'; n]; n];
    for i in 0..n {
        for j in 0..n {
            let tok = it.next().unwrap();
            let ch = tok.as_bytes()[0];
            grid[i][j] = ch;
        }
    }

    let mut visited = vec![vec![false; n]; n];
    let mut components: i64 = 0;

    let dirs: [(isize, isize); 4] = [(-1, 0), (1, 0), (0, -1), (0, 1)];

    for i in 0..n {
        for j in 0..n {
            if !visited[i][j] && grid[i][j] == b'o' {
                components += 1;
                visited[i][j] = true;

                let mut q = VecDeque::new();
                q.push_back((i, j));

                while let Some((x, y)) = q.pop_front() {
                    for (dx, dy) in dirs {
                        let nx = x as isize + dx;
                        let ny = y as isize + dy;
                        if nx < 0 || ny < 0 || nx >= n as isize || ny >= n as isize {
                            continue;
                        }
                        let nxu = nx as usize;
                        let nyu = ny as usize;
                        if !visited[nxu][nyu] && grid[nxu][nyu] == b'o' {
                            visited[nxu][nyu] = true;
                            q.push_back((nxu, nyu));
                        }
                    }
                }
            }
        }
    }

    // Output floor((C-1)/3). For C=0, this yields floor(-1/3) = -1 in math,
    // but integer division in Rust truncates toward zero. The specification does not
    // define behavior for C=0 explicitly; we keep the narrowest interpretation by
    // using integer arithmetic on signed integers as written.
    let ans = (components - 1) / 3;
    println!("{}", ans);
}
```
