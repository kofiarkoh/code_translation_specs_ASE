fn tarjan_scc(n: usize, adj: &Vec<Vec<usize>>) -> (usize, Vec<usize>, Vec<Vec<usize>>) {
    let mut index: i32 = 0;
    let mut stack: Vec<usize> = Vec::new();
    let mut on_stack = vec![false; n];
    let mut idx = vec![-1i32; n];
    let mut low = vec![0i32; n];

    let mut comp_id = vec![usize::MAX; n];
    let mut comps: Vec<Vec<usize>> = Vec::new();

    fn strongconnect(
        v: usize,
        index: &mut i32,
        stack: &mut Vec<usize>,
        on_stack: &mut [bool],
        idx: &mut [i32],
        low: &mut [i32],
        adj: &Vec<Vec<usize>>,
        comp_id: &mut [usize],
        comps: &mut Vec<Vec<usize>>,
    ) {
        idx[v] = *index;
        low[v] = *index;
        *index += 1;

        stack.push(v);
        on_stack[v] = true;

        for &w in &adj[v] {
            if idx[w] == -1 {
                strongconnect(w, index, stack, on_stack, idx, low, adj, comp_id, comps);
                low[v] = low[v].min(low[w]);
            } else if on_stack[w] {
                low[v] = low[v].min(idx[w]);
            }
        }

        if low[v] == idx[v] {
            let mut comp: Vec<usize> = Vec::new();
            loop {
                let w = stack.pop().unwrap();
                on_stack[w] = false;
                comp_id[w] = comps.len();
                comp.push(w);
                if w == v {
                    break;
                }
            }
            comps.push(comp);
        }
    }

    for v in 0..n {
        if idx[v] == -1 {
            strongconnect(
                v,
                &mut index,
                &mut stack,
                &mut on_stack,
                &mut idx,
                &mut low,
                adj,
                &mut comp_id,
                &mut comps,
            );
        }
    }

    (comps.len(), comp_id, comps)
}
