```rust
use std::collections::{HashMap, VecDeque};
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Edge {
    u: usize,
    v: usize,
}

fn tarjan_scc(n: usize, adj: &[Vec<usize>]) -> (usize, Vec<usize>, Vec<Vec<usize>>) {
    let mut index: i32 = 0;
    let mut stack: Vec<usize> = Vec::new();
    let mut on_stack = vec![false; n];
    let mut idx = vec![-1i32; n];
    let mut low = vec![0i32; n];

    let mut comp_id = vec![usize::MAX; n];
    let mut comps: Vec<Vec<usize>> = Vec::new();

    fn strongconnect(
        v: usize,
        index: &mut i32,
        stack: &mut Vec<usize>,
        on_stack: &mut [bool],
        idx: &mut [i32],
        low: &mut [i32],
        adj: &[Vec<usize>],
        comp_id: &mut [usize],
        comps: &mut Vec<Vec<usize>>,
    ) {
        idx[v] = *index;
        low[v] = *index;
        *index += 1;

        stack.push(v);
        on_stack[v] = true;

        for &w in &adj[v] {
            if idx[w] == -1 {
                strongconnect(w, index, stack, on_stack, idx, low, adj, comp_id, comps);
                low[v] = low[v].min(low[w]);
            } else if on_stack[w] {
                low[v] = low[v].min(idx[w]);
            }
        }

        if low[v] == idx[v] {
            let mut comp: Vec<usize> = Vec::new();
            loop {
                let w = stack.pop().unwrap();
                on_stack[w] = false;
                comp_id[w] = comps.len();
                comp.push(w);
                if w == v {
                    break;
                }
            }
            comps.push(comp);
        }
    }

    for v in 0..n {
        if idx[v] == -1 {
            strongconnect(
                v,
                &mut index,
                &mut stack,
                &mut on_stack,
                &mut idx,
                &mut low,
                adj,
                &mut comp_id,
                &mut comps,
            );
        }
    }

    (comps.len(), comp_id, comps)
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let m: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let mut edges: Vec<Edge> = Vec::with_capacity(m);
    let mut adj: Vec<Vec<usize>> = vec![Vec::new(); n];

    for _ in 0..m {
        let u1: usize = it.next().unwrap().parse().unwrap();
        let v1: usize = it.next().unwrap().parse().unwrap();
        let u = u1 - 1;
        let v = v1 - 1;
        edges.push(Edge { u, v });
        adj[u].push(v);
    }

    let (k, comp, comps) = tarjan_scc(n, &adj);

    let mut is_diff = vec![false; m];

    // Inter-SCC: group by ordered SCC pair (S,T)
    let mut inter_map: HashMap<(usize, usize), Vec<usize>> = HashMap::new();
    // Intra-SCC: list of edge indices per SCC
    let mut intra_by_scc: Vec<Vec<usize>> = vec![Vec::new(); k];

    for (i, e) in edges.iter().enumerate() {
        let s = comp[e.u];
        let t = comp[e.v];
        if s == t {
            intra_by_scc[s].push(i);
        } else {
            inter_map.entry((s, t)).or_default().push(i);
        }
    }

    // Condensation adjacency (unique arcs)
    let mut cond_adj: Vec<Vec<usize>> = vec![Vec::new(); k];
    let mut has_arc: HashMap<(usize, usize), ()> = HashMap::new();

    for (&(s, t), idxs) in inter_map.iter() {
        // Rule A1: duplicates for same (S,T) => all diff
        if idxs.len() > 1 {
            for &ei in idxs {
                is_diff[ei] = true;
            }
        }
        if !has_arc.contains_key(&(s, t)) {
            has_arc.insert((s, t), ());
            cond_adj[s].push(t);
        }
    }

    // Rule A2: longer SCC-level route.
    //
    // The intended check is: for an arc S->T, if there exists a path from S to T
    // of length >= 2 in the condensation graph, i.e., via at least one intermediate SCC.
    //
    // This is equivalent to: there exists some neighbor X of S with X != T such that T is reachable from X.
    // We implement this by BFS from each S, but we must ignore the direct edge S->T when evaluating reachability.
    // (Otherwise dist[T] would be 1 and we'd miss cases where an alternative longer path exists but BFS
    // discovers T first via the direct arc.)
    let mut dist = vec![-1i32; k];
    for s in 0..k {
        // BFS from s
        dist.fill(-1);
        let mut q: VecDeque<usize> = VecDeque::new();
        dist[s] = 0;
        q.push_back(s);

        while let Some(x) = q.pop_front() {
            let dx = dist[x];
            for &y in &cond_adj[x] {
                if dist[y] == -1 {
                    dist[y] = dx + 1;
                    q.push_back(y);
                }
            }
        }

        // For each direct arc (s,t), check if there is an alternative path of length >= 2.
        // We do this by starting from all neighbors of s except t, and seeing if t is reachable.
        // Since K<=1000, we can do a small BFS per (s,t) using the condensation graph.
        // This stays faithful to the spec's "at least one intermediate SCC" requirement.
        let out_neighbors = cond_adj[s].clone();
        for &t in &out_neighbors {
            // If already diff due to duplicates, we still keep it diff; no need to skip.
            // Determine if there exists a path s -> x -> ... -> t with x != t.
            let mut seen = vec![false; k];
            let mut qq: VecDeque<usize> = VecDeque::new();

            seen[s] = true;
            // Seed with neighbors of s excluding t (this enforces at least one intermediate SCC).
            for &x in &cond_adj[s] {
                if x == t {
                    continue;
                }
                if !seen[x] {
                    seen[x] = true;
                    qq.push_back(x);
                }
            }

            let mut found = false;
            while let Some(x) = qq.pop_front() {
                if x == t {
                    found = true;
                    break;
                }
                for &y in &cond_adj[x] {
                    if !seen[y] {
                        seen[y] = true;
                        qq.push_back(y);
                    }
                }
            }

            if found {
                if let Some(idxs) = inter_map.get(&(s, t)) {
                    for &ei in idxs {
                        is_diff[ei] = true;
                    }
                }
            } else {
                // If not found, keep as-is (could still be diff due to duplicates).
                // Note: dist-based info is not used here because it doesn't exclude the direct arc.
                let _ = &dist; // keep dist computed; mirrors "evaluated on condensation DAG" but not relied upon.
            }
        }
    }

    // Intra-SCC edges: diff if there exists an alternative path u=>v within SCC excluding the edge itself.
    // Implemented by per-edge BFS within the SCC-induced subgraph, skipping that edge index.
    let mut intra_adj_by_scc: Vec<Vec<Vec<(usize, usize)>>> = vec![Vec::new(); k];
    let mut local_maps: Vec<HashMap<usize, usize>> = vec![HashMap::new(); k];

    for scc in 0..k {
        let mut map_local: HashMap<usize, usize> = HashMap::new();
        for &v in &comps[scc] {
            map_local.insert(v, map_local.len());
        }
        let sz = map_local.len();
        let mut ladj: Vec<Vec<(usize, usize)>> = vec![Vec::new(); sz];
        for &ei in &intra_by_scc[scc] {
            let e = edges[ei];
            let lu = *map_local.get(&e.u).unwrap();
            let lv = *map_local.get(&e.v).unwrap();
            ladj[lu].push((lv, ei));
        }
        local_maps[scc] = map_local;
        intra_adj_by_scc[scc] = ladj;
    }

    for scc in 0..k {
        let map_local = &local_maps[scc];
        let ladj = &intra_adj_by_scc[scc];
        let sz = ladj.len();
        let mut seen = vec![false; sz];

        for &ei in &intra_by_scc[scc] {
            let e = edges[ei];
            let src = *map_local.get(&e.u).unwrap();
            let dst = *map_local.get(&e.v).unwrap();

            seen.fill(false);
            let mut q: VecDeque<usize> = VecDeque::new();
            seen[src] = true;
            q.push_back(src);

            let mut found = false;
            while let Some(x) = q.pop_front() {
                if x == dst {
                    found = true;
                    break;
                }
                for &(y, edge_idx) in &ladj[x] {
                    if edge_idx == ei {
                        continue;
                    }
                    if !seen[y] {
                        seen[y] = true;
                        q.push_back(y);
                    }
                }
            }

            if found {
                is_diff[ei] = true;
            }
        }
    }

    let mut out = String::new();
    for i in 0..m {
        if is_diff[i] {
            out.push_str("diff\n");
        } else {
            out.push_str("same\n");
        }
    }
    print!("{out}");
}
```
