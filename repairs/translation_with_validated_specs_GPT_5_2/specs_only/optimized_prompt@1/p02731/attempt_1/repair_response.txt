```rust
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let l: i64 = input.split_whitespace().next().unwrap().parse().unwrap();

    let x = (l as f64) / 3.0;
    let y = x * x * x;

    println!("{:.12}", y);
}
```
