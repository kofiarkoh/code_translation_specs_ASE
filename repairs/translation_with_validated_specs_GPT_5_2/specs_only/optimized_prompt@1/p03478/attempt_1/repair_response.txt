```rust
use std::io::{self, Read};

fn digit_sum5(mut x: i64) -> i64 {
    // Sum of the five least significant decimal digits; higher digits ignored.
    // Missing higher digits are effectively 0 because we only take 5 digits.
    let mut s = 0i64;
    for _ in 0..5 {
        s += x % 10;
        x /= 10;
    }
    s
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut it = input.split_whitespace();
    let n: i64 = it.next().unwrap().parse().unwrap();
    let a: i64 = it.next().unwrap().parse().unwrap();
    let b: i64 = it.next().unwrap().parse().unwrap();

    let mut total: i64 = 0;
    for i in 1..=n {
        let ds = digit_sum5(i);
        if a <= ds && ds <= b {
            total += i;
        }
    }

    print!("{}", total);
}
```
