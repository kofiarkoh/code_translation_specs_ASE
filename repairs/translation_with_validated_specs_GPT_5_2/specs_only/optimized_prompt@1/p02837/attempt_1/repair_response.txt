```rust
use std::io::{self, Read};

fn main() {
    // Read all input as whitespace-separated integers.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return, // Undefined by spec; choose to do nothing if no input.
    };

    // Store assertions: for each i, a list of (target_index_0_based, value_0_or_1).
    let mut assertions: Vec<Vec<(usize, u8)>> = vec![Vec::new(); n];

    for i in 0..n {
        let a_i: usize = it.next().unwrap().parse().unwrap();
        let mut v = Vec::with_capacity(a_i);
        for _ in 0..a_i {
            let t_1based: usize = it.next().unwrap().parse().unwrap();
            let val: u8 = it.next().unwrap().parse().unwrap();
            // Preconditions guarantee t in [1..n] and val in {0,1}; otherwise undefined.
            v.push((t_1based - 1, val));
        }
        assertions[i] = v;
    }

    // Enumerate all subsets of {0..n-1} using a bitmask.
    // n <= 20, so 1<<n fits in u32; use u32 to mirror typical C integer bit operations.
    let total: u32 = if n >= 32 { 0 } else { 1u32.wrapping_shl(n as u32) };

    let mut best: u32 = 0;

    for mask in 0..total {
        let mut ok = true;

        for j in 0..n {
            // If entity j is selected in this subset, its assertions must match.
            if ((mask >> j) & 1) != 0 {
                for &(t, val) in &assertions[j] {
                    let in_subset = ((mask >> t) & 1) as u8;
                    if in_subset != val {
                        ok = false;
                        break;
                    }
                }
                if !ok {
                    break;
                }
            }
        }

        if ok {
            let cnt = mask.count_ones();
            if cnt > best {
                best = cnt;
            }
        }
    }

    println!("{}", best);
}
```
