```rust
use std::cmp::min;
use std::collections::VecDeque;
use std::io::{self, Read};

fn main() {
    // Read entire stdin; specification provides no malformed-input handling.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let h: usize = it.next().unwrap().parse().unwrap();
    let w: usize = it.next().unwrap().parse().unwrap();
    let k: usize = it.next().unwrap().parse().unwrap();

    let sr1: usize = it.next().unwrap().parse().unwrap();
    let sc1: usize = it.next().unwrap().parse().unwrap();
    let gr1: usize = it.next().unwrap().parse().unwrap();
    let gc1: usize = it.next().unwrap().parse().unwrap();

    // Convert 1-based to 0-based. Preconditions guarantee in-bounds.
    let sr = sr1 - 1;
    let sc = sc1 - 1;
    let gr = gr1 - 1;
    let gc = gc1 - 1;

    let mut grid: Vec<Vec<u8>> = Vec::with_capacity(h);
    for _ in 0..h {
        let row = it.next().unwrap().as_bytes().to_vec();
        // Preconditions guarantee exact width and only passable/blocked symbols.
        grid.push(row);
    }

    // Postcondition: distance is 0 when start == goal (within bounds).
    if sr == gr && sc == gc {
        println!("0");
        return;
    }

    // Distances: -1 means unvisited/unreachable so far.
    let mut dist = vec![vec![-1i32; w]; h];
    dist[sr][sc] = 0;

    // Fixed-capacity queue is assumed sufficient by precondition; VecDeque is unbounded,
    // but this preserves observable behavior under the stated assumption.
    let mut q: VecDeque<(usize, usize)> = VecDeque::new();
    q.push_back((sr, sc));

    let dirs: [(isize, isize); 4] = [(-1, 0), (1, 0), (0, -1), (0, 1)];

    while let Some((r, c)) = q.pop_front() {
        let base = dist[r][c];
        // Explore up to k cells in each direction, stopping at bounds or blocked.
        for (dr, dc) in dirs {
            for step in 1..=k {
                let nr = r as isize + dr * step as isize;
                let nc = c as isize + dc * step as isize;
                if nr < 0 || nr >= h as isize || nc < 0 || nc >= w as isize {
                    break;
                }
                let ur = nr as usize;
                let uc = nc as usize;

                // Blocked cell stops further movement in this direction.
                if grid[ur][uc] == b'#' {
                    break;
                }

                // If already visited with a strictly smaller distance, further cells in this
                // direction cannot be improved via this path (uniform cost BFS).
                let nd = base + 1;
                let cur = dist[ur][uc];
                if cur != -1 {
                    if cur < nd {
                        break;
                    }
                    if cur == nd {
                        continue;
                    }
                    // cur > nd should not happen in BFS, but if it does, update narrowly.
                }

                dist[ur][uc] = nd;
                if ur == gr && uc == gc {
                    println!("{}", nd);
                    return;
                }
                q.push_back((ur, uc));
            }
        }
    }

    println!("-1");
}
```
