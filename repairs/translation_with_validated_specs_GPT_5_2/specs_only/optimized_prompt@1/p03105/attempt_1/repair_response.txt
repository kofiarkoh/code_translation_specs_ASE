```rust
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut it = input.split_whitespace();

    let a: i64 = it.next().unwrap().parse().unwrap();
    let b: i64 = it.next().unwrap().parse().unwrap();
    let c: i64 = it.next().unwrap().parse().unwrap();

    // Preconditions state `a` is non-zero; if it is zero, behavior is undefined in the spec.
    // Rust would panic on division by zero; we do not add explicit error handling.
    let q = b / a; // truncates toward zero for signed integers, matching the specification

    let out = if q < c { q } else { c };
    println!("{}", out);
}
```
