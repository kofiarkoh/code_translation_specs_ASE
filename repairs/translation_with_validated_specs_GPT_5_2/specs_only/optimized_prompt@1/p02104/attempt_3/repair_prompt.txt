    let mut occ: Vec<Vec<usize>> = vec![Vec::new(); n + 1];

        let v: usize = it.next().unwrap().parse().unwrap();
        // Spec: if v is out of 1..=n, behavior is undefined; we do not guard.
        occ[v].push(i);
    }

    let mut ans: Vec<usize> = vec![0; n + 1];

    // Collect all values that are missing (no occurrences).
    let mut missing: Vec<usize> = Vec::new();

        if occ[v].is_empty() {
            missing.push(v);
        }
    }

    // For each missing value, try to "steal" one extra occurrence from some value
    // that appears at least twice. The stolen position must not be the one kept

    // To match the expected behavior, we:
    // - process missing values in increasing order,
    // - for each missing value, scan donor values in increasing order,

        for donor in 1..=n {
            if occ[donor].len() >= 2 {

                let stolen = occ[donor].remove(0);
