```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin as whitespace-separated tokens.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break; // Unspecified behavior on malformed input; we just stop.
        }
        let n: usize = n_opt.unwrap().parse().unwrap();
        if n == 0 {
            break;
        }

        // Fixed maximum storage capacity is mentioned; we allocate exactly n here.
        // Preconditions guarantee n <= 100000.
        let mut b: Vec<i64> = Vec::with_capacity(n);
        for _ in 0..n {
            let v: i64 = it.next().unwrap().parse().unwrap();
            b.push(v);
        }

        // Transform in-place according to the specification.
        // For each odd i, if b[i-1] != b[i], overwrite a suffix of b[0..i-1]
        // consisting of consecutive elements immediately preceding i that are not
        // already equal to b[i], stopping at first earlier position equal to b[i]
        // or at start.
        for i in (1..n).step_by(2) {
            if b[i - 1] != b[i] {
                let target = b[i];
                let mut j = i;
                while j > 0 {
                    let k = j - 1;
                    if b[k] == target {
                        break;
                    }
                    b[k] = target;
                    j -= 1;
                }
            }
        }

        let zero_count = b.iter().filter(|&&x| x == 0).count();
        out.push_str(&zero_count.to_string());
        out.push('\n');
    }

    print!("{out}");
}
```
