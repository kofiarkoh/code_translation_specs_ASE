1. Functional Description  
The program reads an integer \(N\) and an \(N \times N\) grid of alphabetic characters. It selects exactly one cell from each row and exactly one cell from each column (i.e., a perfect matching between rows and columns). The \(N\) selected cells determine a multiset of \(N\) characters.

The program chooses the matching by solving an assignment/min-cost-flow problem with a **52-dimensional integer cost vector** ordered by:
\[
\text{‘A’, ‘B’, …, ‘Z’, ‘a’, ‘b’, …, ‘z’}.
\]
Selecting a cell whose character is the \(k\)-th character in this order contributes a cost vector that is \(-1\) in component \(k\) and \(0\) elsewhere. The total cost of a matching is the componentwise sum of these per-cell vectors, i.e. \((-c_0, -c_1, \dots, -c_{51})\) where \(c_k\) is the number of selected occurrences of the \(k\)-th character.

Among all perfect matchings, the program selects one whose total cost vector is **lexicographically smallest** under the fixed component order (compare component 0 first, then 1, etc.). If multiple matchings have exactly the same total cost vector, the program may output any one of them; no additional canonical tie-breaking among equal-cost matchings is specified by the program’s logic beyond what arises from its internal shortest-path/augmentation procedure.

Finally, it outputs the selected multiset as a string sorted in the same fixed order (all uppercase ‘A’..‘Z’, then all lowercase ‘a’..‘z’), with each character repeated according to its count in the chosen matching.

#include <stdio.h>

typedef struct { signed char a[52]; } T;
T mx[10000]; int sz;
#define mx_zero   0
#define mx_inf    1

int  mx_new() { return sz++; }
void mx_clear(int a) { memset(mx+a, 0, sizeof(T)); }
void mx_add1(int a, int b) { int i; for (i = 0; i < 52; i++) mx[a].a[i] += mx[b].a[i]; }
void mx_add2(int s, int a, int b) { int i; for (i = 0; i < 52; i++) mx[s].a[i] = mx[a].a[i]+mx[b].a[i]; }
void mx_sub1(int a, int b) { int i; for (i = 0; i < 52; i++) mx[a].a[i] -= mx[b].a[i]; }
void mx_sub2(int d, int a, int b) { int i; for (i = 0; i < 52; i++) mx[d].a[i] = mx[a].a[i]-mx[b].a[i]; }
void mx_mul2(int m, int a, int k) { int i; for (i = 0; i < 52; i++) mx[m].a[i] = mx[a].a[i] * k; }
void mx_cpy(int a, int b) { memcpy(mx+a, mx+b, sizeof(T)); }
int  mx_cmp(int a, int b) {
	int k, i; for (i = 0; i < 52; i++) if (k = mx[a].a[i] - mx[b].a[i]) return k; return 0; }

#define MAX 105
typedef struct { int to, rev; int cap; int cost; } EDGE;
EDGE edge[MAX][MAX]; int hi[MAX];
int  V;				// 頂点数
int  dist[MAX], prevv[MAX], preve[MAX];

void add_edge(int from, int to, int cap, int cost, int rcost)
{
	int f, t;
	EDGE *e;

	f = hi[from]++, t = hi[to]++;
	e = &edge[from][f], e->to = to, e->cap = cap, e->cost = cost, e->rev = t;
	e = &edge[to][t], e->to = from, e->cap = 0, e->cost = rcost, e->rev = f;
}

int minCostFlow(int S, int T, int F)
{
	int i, v, nv, d, update;
	int t = mx_new(), ret;
	EDGE *e;

	for (i = 0; i < V; i++) dist[i] = mx_new(); 
	ret = mx_new();
	while (F > 0) {
		for (v = 0; v < V; v++) mx_cpy(dist[v], mx_inf);
		mx_clear(dist[S]);
		do {
			update = 0;
			for (v = 0; v < V; v++) {
				if (mx_cmp(dist[v], mx_inf) == 0) continue;
				for (i = 0; i < hi[v]; i++) {
					e = &edge[v][i];
					if (e->cap <= 0) continue;
					nv = e->to;
					mx_add2(t, dist[v], e->cost);
					if (mx_cmp(dist[nv], t) > 0) {
						mx_cpy(dist[nv], t), prevv[nv] = v, preve[e->to] = i;
						update = 1;
					}
				}
			}
		} while (update);

		d = F;
		for (v = T; v != S; v = prevv[v]) {
			e = &edge[prevv[v]][preve[v]];
			if (d > e->cap) d = e->cap;
		}
		F -= d;

		mx_mul2(t, dist[T], d);
		mx_add1(ret, t);
		for (v = T; v != S; v = prevv[v]) {
			e = &edge[prevv[v]][preve[v]];
			e->cap -= d;
			edge[v][e->rev].cap += d;
		}
	}
	return ret;
}
