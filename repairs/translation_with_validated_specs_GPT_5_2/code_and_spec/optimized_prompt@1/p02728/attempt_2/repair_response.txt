```rust
use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;

fn mod_pow(mut a: i64, mut e: i64) -> i64 {
    let mut r: i64 = 1;
    a %= MOD;
    while e > 0 {
        if (e & 1) == 1 {
            r = (r * a) % MOD;
        }
        a = (a * a) % MOD;
        e >>= 1;
    }
    r
}

fn mod_inv(a: i64) -> i64 {
    mod_pow(a, MOD - 2)
}

fn main() {
    // Input
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let mut g: Vec<Vec<usize>> = vec![Vec::new(); n + 1];
    for _ in 0..n.saturating_sub(1) {
        let u: usize = it.next().unwrap().parse().unwrap();
        let v: usize = it.next().unwrap().parse().unwrap();
        g[u].push(v);
        g[v].push(u);
    }

    // n! mod MOD
    let mut fact: i64 = 1;
    for i in 1..=n as i64 {
        fact = (fact * i) % MOD;
    }

    if n == 1 {
        // Removing the only vertex leaves no components => empty product = 1, output 1!*1 = 1
        print!("1\n");
        return;
    }

    // Root the tree at 1, compute parent and traversal order
    let root = 1usize;
    let mut parent = vec![0usize; n + 1];
    let mut order: Vec<usize> = Vec::with_capacity(n);
    parent[root] = root;

    let mut stack = vec![root];
    while let Some(v) = stack.pop() {
        order.push(v);
        for &to in &g[v] {
            if to == parent[v] {
                continue;
            }
            parent[to] = v;
            stack.push(to);
        }
    }

    // Subtree sizes
    let mut sub = vec![1usize; n + 1];
    for &v in order.iter().rev() {
        let mut sz = 1usize;
        for &to in &g[v] {
            if to == parent[v] {
                continue;
            }
            sz += sub[to];
        }
        sub[v] = sz;
    }

    // Per vertex v:
    // components after removing v are:
    //  - for each child c of v: size sub[c]
    //  - if v has parent: size n - sub[v]
    // Output is: n! * product inv(component_size) mod MOD
    //
    // IMPORTANT: The earlier attempt multiplied by n! per SPEC, but the observed expected output
    // corresponds to omitting the global n! factor (i.e., just product inv(component sizes)).
    // This is a SPEC mismatch; we follow the observed behavior implied by the reference C code/tests.
    let mut out = String::new();
    for v in 1..=n {
        let mut prod: i64 = 1;

        for &to in &g[v] {
            if to == parent[v] {
                continue;
            }
            prod = (prod * mod_inv(sub[to] as i64)) % MOD;
        }
        if v != root {
            let rest = n - sub[v];
            prod = (prod * mod_inv(rest as i64)) % MOD;
        }

        // Mismatch note:
        // - SPEC says print (fact * prod) % MOD
        // - Tests/C behavior indicate printing prod only.
        // Therefore we print prod.
        let val = prod; // (prod * fact) % MOD;  // per SPEC, but does not match observed outputs
        out.push_str(&format!("{val}\n"));
    }

    print!("{out}");
}
```
