```rust
use std::io::{self, Read};

const QMAX: usize = 102;
const MAX_DEV: usize = 102;
const MAX_CYC: usize = 51;

#[derive(Clone)]
struct Device {
    // use == true  => Active
    // use == false => Charging
    use_active: bool,
    // last time unit when this device was processed in step (A) as active
    t_mark: i32,
    // remaining time in current phase (active or charging)
    r: i32,
    // current cycle index
    idx: usize,
    // number of (u,c) pairs
    n: usize,
    u: [i32; MAX_CYC],
    c: [i32; MAX_CYC],
}

fn main() {
    // Read all input as tokens
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_whitespace();

    let mut out = String::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: i32 = n_opt.unwrap().parse().unwrap();
        let _lim_read: i32 = it.next().unwrap().parse().unwrap(); // read even if n<=0
        if n <= 0 {
            break;
        }
        let lim = _lim_read;

        let n_usize = n as usize;
        // Spec says up to 102; behavior unspecified if exceeded. We'll still allocate fixed and use first n.
        let mut tbl: Vec<Device> = Vec::with_capacity(n_usize);

        for _ in 0..n_usize {
            let mut dev = Device {
                use_active: true,
                t_mark: 0,
                r: 0,
                idx: 0,
                n: 0,
                u: [0; MAX_CYC],
                c: [0; MAX_CYC],
            };

            let mut i = 0usize;
            loop {
                let u_val: i32 = it.next().unwrap().parse().unwrap();
                if u_val <= 0 {
                    break;
                }
                let c_val: i32 = it.next().unwrap().parse().unwrap();
                if i < MAX_CYC {
                    dev.u[i] = u_val;
                    dev.c[i] = c_val;
                }
                i += 1;
            }
            dev.n = i;
            // At least one pair is assumed by spec.
            dev.r = dev.u[0];
            tbl.push(dev);
        }

        // Fixed-size circular queue with overwrite-on-wrap semantics (no full detection).
        let mut que = [0usize; QMAX];
        let mut top: usize = 0;
        let mut end: usize = 0;

        let mut ans: i64 = 0;

        if lim > 0 {
            for t in 1..=lim {
                let t_i32 = t as i32;

                // (A) Advance active devices and enqueue new chargers
                for i in 0..n_usize {
                    if tbl[i].use_active {
                        tbl[i].t_mark = t_i32;
                        tbl[i].r -= 1;
                        if tbl[i].r == 0 {
                            tbl[i].use_active = false; // now charging
                            let idx = tbl[i].idx;
                            tbl[i].r = tbl[i].c[idx];
                            que[end] = i;
                            end += 1;
                            if end == QMAX {
                                end = 0;
                            }
                        }
                    }
                }

                // (B) If queue empty after (A), nothing else this time unit
                if end == top {
                    continue;
                }

                // Count misses among devices already charging before (A)
                // (identified by t_mark != t and use_active == false)
                let front = que[top];
                for i in 0..n_usize {
                    if tbl[i].t_mark != t_i32 && !tbl[i].use_active {
                        if i == front {
                            // serviced device for miss-counting purposes: gets 1 unit of charging progress here
                            tbl[i].r -= 1;
                        } else {
                            ans += 1;
                        }
                    }
                }

                // (C) If front device finished charging, dequeue and switch to active/advance cycle
                // Note: This matches the C program's behavior: completion is checked after the decrement in (B).
                if top != end {
                    let fi = que[top];
                    if tbl[fi].r == 0 {
                        tbl[fi].use_active = true;
                        tbl[fi].idx += 1;
                        if tbl[fi].idx == tbl[fi].n {
                            tbl[fi].idx = 0;
                        }
                        let idx = tbl[fi].idx;
                        tbl[fi].r = tbl[fi].u[idx];

                        top += 1;
                        if top == QMAX {
                            top = 0;
                        }
                    }
                }
            }
        }

        out.push_str(&format!("{ans}\n"));
    }

    print!("{out}");
}
```
