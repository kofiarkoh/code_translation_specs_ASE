```rust
use std::io::{self, Read};

fn max_i32(a: i32, b: i32) -> i32 {
    if a > b { a } else { b }
}

fn main() {
    // Input
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let m: usize = it.next().unwrap().parse().unwrap();

    let mut ii = vec![0usize; m];
    let mut jj = vec![0usize; m];

    // Vertex adjacency lists storing edge indices (original edge id h)
    let mut ae: Vec<Vec<usize>> = vec![Vec::new(); n]; // outgoing
    let mut af: Vec<Vec<usize>> = vec![Vec::new(); n]; // incoming (reverse)

    for h in 0..m {
        let mut u: usize = it.next().unwrap().parse().unwrap();
        let mut v: usize = it.next().unwrap().parse().unwrap();
        u -= 1;
        v -= 1;
        ii[h] = u;
        jj[h] = v;
        ae[u].push(h);
        af[v].push(h);
    }

    // --- SCC (Kosaraju), mirroring the C program's semantics ---
    // cc: 0 = unvisited, -1 = visited in first pass, >=0 = SCC id
    let mut cc: Vec<i32> = vec![0; n];
    let mut qu: Vec<usize> = vec![0; n];
    let mut cnt: i32;

    // dfs1 recursion translated to iterative, but preserves "postorder push into qu[--cnt]"
    cnt = n as i32;
    for start in 0..n {
        if cc[start] != 0 {
            continue;
        }
        // stack of (vertex, next outgoing index)
        let mut stack: Vec<(usize, usize)> = Vec::new();
        cc[start] = -1;
        stack.push((start, 0));
        while let Some((v, idx)) = stack.pop() {
            if idx < ae[v].len() {
                stack.push((v, idx + 1));
                let h = ae[v][idx];
                let to = jj[h];
                if cc[to] == 0 {
                    cc[to] = -1;
                    stack.push((to, 0));
                }
            } else {
                cnt -= 1;
                qu[cnt as usize] = v;
            }
        }
    }

    // dfs2 on reverse graph, in qu[0..n) order
    let mut n_scc: usize = 0;
    for g in 0..n {
        let v = qu[g];
        if cc[v] != -1 {
            continue;
        }
        let cid = n_scc as i32;
        n_scc += 1;

        let mut stack: Vec<usize> = vec![v];
        cc[v] = cid;
        while let Some(x) = stack.pop() {
            for &h in &af[x] {
                let from = ii[h];
                if cc[from] == -1 {
                    cc[from] = cid;
                    stack.push(from);
                }
            }
        }
    }

    // --- Build SCC-level adjacency lists ae_ (like C's ae_) storing original edge ids h ---
    let mut ae_scc: Vec<Vec<usize>> = vec![Vec::new(); n_scc];
    for h in 0..m {
        let s = cc[ii[h]] as usize;
        let t = cc[jj[h]] as usize;
        if s != t {
            ae_scc[s].push(h);
        }
    }

    let mut diff = vec![false; m];

    // --- Inter-SCC classification: follow the C program exactly ---
    // 1) mark duplicates between same ordered SCC pair (s,t)
    // 2) mark if there exists a "longer SCC-level route" per the C's dd DP (not general reachability)
    //
    // Note: This intentionally follows the C code's dd propagation order (i from s..n_scc-1),
    // which depends on SCC numbering. This matches the observed expected outputs.
    let mut dd: Vec<i32> = vec![0; n_scc];
    for s in 0..n_scc {
        // duplicates
        dd.fill(0);
        for &h in &ae_scc[s] {
            let t = cc[jj[h]] as usize;
            dd[t] += 1;
        }
        for &h in &ae_scc[s] {
            let t = cc[jj[h]] as usize;
            if dd[t] > 1 {
                diff[h] = true;
            }
        }

        // dd DP for "longer route"
        dd.fill(-1);
        dd[s] = 0;
        for i in s..n_scc {
            if dd[i] != -1 {
                let di = dd[i];
                for &h in &ae_scc[i] {
                    let t = cc[jj[h]] as usize;
                    dd[t] = max_i32(dd[t], di + 1);
                }
            }
        }
        for &h in &ae_scc[s] {
            let t = cc[jj[h]] as usize;
            if dd[t] > 1 {
                diff[h] = true;
            }
        }
    }

    // --- Intra-SCC classification: follow the C program exactly ---
    // The C code:
    // - for each SCC c, picks qu[0] as a start vertex in that SCC
    // - runs dfs3 (forward) and dfs4 (reverse) to collect a set of edge ids hh[]
    // - for each such edge h, sets diff[h] = !dfs5(src, dst) where dfs5 searches for an
    //   alternative path excluding h within the SCC.
    //
    // This means: intra-SCC edges are marked diff iff they are "bridge-like" w.r.t. reachability
    // from their source to destination inside the SCC (no alternative path).
    let mut visited: Vec<u8> = vec![0; n]; // 0/1/2 as in C
    let mut hh: Vec<usize> = Vec::new();

    for c in 0..n_scc {
        // collect vertices of SCC c into qu[0..cnt)
        let mut verts: Vec<usize> = Vec::new();
        for v in 0..n {
            if cc[v] as usize == c {
                verts.push(v);
            }
        }
        if verts.is_empty() {
            continue;
        }

        hh.clear();

        // dfs3 from verts[0]
        // visited[v]=1; traverse outgoing edges within SCC to unvisited nodes; record edge ids
        visited.fill(0);
        {
            let start = verts[0];
            let mut stack: Vec<(usize, usize)> = Vec::new();
            visited[start] = 1;
            stack.push((start, 0));
            while let Some((v, idx)) = stack.pop() {
                if idx < ae[v].len() {
                    stack.push((v, idx + 1));
                    let h = ae[v][idx];
                    let to = jj[h];
                    if visited[to] == 0 && cc[to] as usize == c {
                        hh.push(h);
                        visited[to] = 1;
                        stack.push((to, 0));
                    }
                }
            }
        }

        // dfs4 from verts[0] on reverse edges:
        // visited[v]=2; traverse incoming edges to nodes with visited==1; record edge ids
        {
            let start = verts[0];
            let mut stack: Vec<usize> = Vec::new();
            visited[start] = 2;
            stack.push(start);
            while let Some(v) = stack.pop() {
                for &h in &af[v] {
                    let from = ii[h];
                    if visited[from] == 1 && cc[from] as usize == c {
                        hh.push(h);
                        visited[from] = 2;
                        stack.push(from);
                    }
                }
            }
        }

        // For each collected edge h, run dfs5 excluding that edge.
        // Reset visited[verts[*]] = 0 before each dfs5 (as in C).
        for &h_excl in &hh {
            for &v in &verts {
                visited[v] = 0;
            }

            let src = ii[h_excl];
            let dst = jj[h_excl];

            // dfs5: returns 1 if dst reachable from src within SCC c without using h_excl
            let mut found = false;
            let mut stack: Vec<usize> = vec![src];
            while let Some(v) = stack.pop() {
                if v == dst {
                    found = true;
                    break;
                }
                if visited[v] != 0 {
                    continue;
                }
                visited[v] = 1;
                for &h in &ae[v] {
                    if h == h_excl {
                        continue;
                    }
                    let to = jj[h];
                    if cc[to] as usize == c && visited[to] == 0 {
                        stack.push(to);
                    }
                }
            }

            // C: diff[h] = !dfs5(...)
            diff[h_excl] = !found;
        }
    }

    // Output
    let mut out = String::new();
    for h in 0..m {
        if diff[h] {
            out.push_str("diff\n");
        } else {
            out.push_str("same\n");
        }
    }
    print!("{out}");
}
```
