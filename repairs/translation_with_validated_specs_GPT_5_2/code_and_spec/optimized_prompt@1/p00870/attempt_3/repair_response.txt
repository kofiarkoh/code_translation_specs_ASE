```rust
use std::io::{self, Read};

fn main() {
    // Token-based parsing (scanf-like) is sufficient for AOJ inputs.
    // Note: The spec mentions trimming a trailing control char (< ' ') from each text fragment.
    // With whitespace tokenization, such chars typically won't appear, but we still apply the rule.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let n = match it.next() {
            Some(x) => x.parse::<usize>().unwrap(),
            None => break,
        };
        let m = it.next().unwrap().parse::<usize>().unwrap();

        if n == 0 && m == 0 {
            break;
        }

        let mut patterns: Vec<Vec<u8>> = Vec::with_capacity(n);
        let mut lens: Vec<usize> = Vec::with_capacity(n);
        for _ in 0..n {
            let s = it.next().unwrap();
            let b = s.as_bytes().to_vec();
            lens.push(b.len());
            patterns.push(b);
        }

        let mut text: Vec<u8> = Vec::new();
        for _ in 0..m {
            let mut frag = it.next().unwrap().as_bytes().to_vec();
            if let Some(&last) = frag.last() {
                if last < b' ' {
                    frag.pop();
                }
            }
            text.extend_from_slice(&frag);
        }

        let l_s = text.len();
        let loop_mask = 1usize << n;
        let full = loop_mask - 1;

        // Match the C program's fixed table width (4096) regardless of n.
        // This also avoids any potential mismatch with the original storage assumptions.
        const WIDTH: usize = 4096;
        let mut flag = vec![0u8; (l_s + 1) * WIDTH];

        // C special-cases (0, empty) as reachable without storing it; we store it explicitly.
        flag[0 * WIDTH + 0] = 1;

        for i in 0..l_s {
            let base = i * WIDTH;
            for j in 0..loop_mask {
                if flag[base + j] == 0 {
                    continue;
                }
                for k in 0..n {
                    let mask = 1usize << k;
                    if (j & mask) != 0 {
                        continue;
                    }
                    let l = lens[k];
                    if i + l > l_s {
                        continue;
                    }
                    if text[i..i + l] == patterns[k][..] {
                        let pat = j | mask;
                        flag[(i + l) * WIDTH + pat] = 1;
                    }
                }
            }
        }

        let mut cnt: usize = 0;
        for p in 0..=l_s {
            cnt += flag[p * WIDTH + full] as usize;
        }

        out.push_str(&format!("{cnt}\n"));
    }

    print!("{out}");
}
```
