use std::io::{self, Read};

#[derive(Clone)]
struct Node {
    edge_to: Vec<usize>,
    edge_cost: Vec<i32>,
    cost: i32,    // -1 means unreachable/unknown
    visited: bool,
}

fn dijkstra_noheap(nodes: &mut [Node]) {
    let n = nodes.len();
    for i in 0..n {
        nodes[i].cost = -1;
        nodes[i].visited = false;
    }
    if n > 0 {
        nodes[0].cost = 0;
    }

    loop {
        // Select the unvisited vertex with smallest nonnegative cost.
        let mut next_idx: Option<usize> = None;
        for i in 0..n {
            if nodes[i].visited || nodes[i].cost < 0 {
                continue;
            }
            match next_idx {
                None => next_idx = Some(i),
                Some(j) => {
                    if nodes[i].cost < nodes[j].cost {
                        next_idx = Some(i);
                    }
                }
            }
        }

        let u_idx = match next_idx {
            None => break,
            Some(i) => i,
        };

        nodes[u_idx].visited = true;

        // Snapshot edges and current cost to avoid borrow conflicts while relaxing.
        let u_cost = nodes[u_idx].cost;
        let tos = nodes[u_idx].edge_to.clone();
        let costs = nodes[u_idx].edge_cost.clone();

        for (to, w) in tos.into_iter().zip(costs.into_iter()) {
            let new_cost = u_cost + w;
            let cur = nodes[to].cost;
            if cur < 0 || new_cost < cur {
                nodes[to].cost = new_cost;
            }
        }
    }
}

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut s = String::new();
        io::stdin().read_to_string(&mut s).unwrap();
        Self {
            input: s.into_bytes(),
            idx: 0,
        }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut n: i32 = 0;
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_digit() {
            n = n * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        n * sign
    }

    fn next_usize(&mut self) -> usize {
        self.next_i32() as usize
    }
}

fn main() {
    let mut sc = Scanner::new();
    let n_i32 = sc.next_i32();
    let n = if n_i32 < 0 { 0 } else { n_i32 as usize };

    let mut nodes = vec![
        Node {
            edge_to: Vec::new(),
            edge_cost: Vec::new(),
            cost: -1,
            visited: false,
        };
        n
    ];

    // Input provides exactly N vertex descriptions, each U appears exactly once.
    for _ in 0..n {
        let u = sc.next_usize();
        let k = sc.next_usize();
        nodes[u].edge_to = Vec::with_capacity(k);
        nodes[u].edge_cost = Vec::with_capacity(k);
        for _ in 0..k {
            let v = sc.next_usize();
            let w = sc.next_i32();
            nodes[u].edge_to.push(v);
            nodes[u].edge_cost.push(w);
        }
    }

    dijkstra_noheap(&mut nodes);

    let mut out = String::new();
    for i in 0..n {
        out.push_str(&format!("{} {}\n", i, nodes[i].cost));
    }
    print!("{out}");
}
