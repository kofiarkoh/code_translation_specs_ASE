```rust
use std::io::{self, Read};

fn main() {
    let mut buf = [0u8; 114_514];

    // One raw read from stdin, like C's read(0, buf, 114514).
    // If it errors, treat as 0 bytes (C would be -1; spec says UB anyway).
    let l = io::stdin().read(&mut buf).unwrap_or(0);

    // For l < 2, C has undefined behavior; avoid panicking in Rust.
    if l < 2 {
        print!("Second\n");
        return;
    }

    // C: n = read(...); n--;
    let n = l - 1;

    // C: (buf[0] == buf[n-1]) where n-1 == l-2
    let eq: usize = if buf[0] == buf[n - 1] { 1 } else { 0 };

    // Critical fix: match the C expression exactly:
    // puts( n + (buf[0]==buf[n-1]) & 1 ? "First" : "Second");
    // In C, '+' has higher precedence than '&', so it parses as:
    // ((n + (buf[0]==buf[n-1])) & 1) ? ...
    let cond = (n + eq) & 1;

    if cond != 0 {
        print!("First\n");
    } else {
        print!("Second\n");
    }
}
```
