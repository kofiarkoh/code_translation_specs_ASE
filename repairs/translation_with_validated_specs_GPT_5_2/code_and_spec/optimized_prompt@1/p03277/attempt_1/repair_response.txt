```rust
use std::io::{self, Read};

type Int64 = i64;

/// Fenwick tree (BIT) for prefix sums over indices 1..=n.
struct Fenwick {
    n: usize,
    bit: Vec<i32>,
}

impl Fenwick {
    fn new(n: usize) -> Self {
        Fenwick { n, bit: vec![0; n + 1] }
    }

    fn reset(&mut self) {
        // Preserve capacity; reinitialize contents to 0.
        self.bit.fill(0);
    }

    fn add(&mut self, mut idx: usize, delta: i32) {
        while idx <= self.n {
            self.bit[idx] += delta;
            idx += idx & (!idx + 1); // lowbit
        }
    }

    fn sum(&self, mut idx: usize) -> i32 {
        let mut res = 0i32;
        while idx > 0 {
            res += self.bit[idx];
            idx &= idx - 1;
        }
        res
    }
}

/// Counts the number of contiguous subarrays whose median (upper median) is >= m.
/// This matches the C code's `count` function semantics.
fn count_ge_median(m: i32, a: &[i32], fw: &mut Fenwick) -> Int64 {
    fw.reset();

    let n = a.len() as i32;
    // Coordinate range is 1..=2n+1, with initial sum at n+1.
    let mut sum_idx: i32 = n + 1;

    fw.add(sum_idx as usize, 1);

    let mut cnt: Int64 = 0;
    for &v in a {
        sum_idx += if v < m { -1 } else { 1 };
        cnt += fw.sum(sum_idx as usize) as Int64;
        fw.add(sum_idx as usize, 1);
    }
    cnt
}

fn main() {
    // Read all input.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let mut a = Vec::with_capacity(n);
    for _ in 0..n {
        let v: i32 = it.next().unwrap().parse().unwrap();
        a.push(v);
    }

    // BIT size: 2*n + 1 (indices 1..=2n+1).
    let mut fw = Fenwick::new(2 * n + 1);

    // Binary search for greatest l such that at least half of subarrays
    // have median >= l (upper median convention).
    let mut l: i32 = 1;
    let mut r: i32 = 1_000_000_000 + 1;

    while r - l > 1 {
        let m = l + (r - l) / 2;
        let cnt = count_ge_median(m, &a, &mut fw);
        let total = (n as Int64) * ((n as Int64) + 1);

        // C condition: 4*cnt >= n*(n+1)
        if 4 * cnt >= total {
            l = m;
        } else {
            r = m;
        }
    }

    print!("{l}\n");
}
```
