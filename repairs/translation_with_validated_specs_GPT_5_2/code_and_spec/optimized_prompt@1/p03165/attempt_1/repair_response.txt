```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, then parse two whitespace-delimited strings.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let s1 = it.next().unwrap_or("");
    let s2 = it.next().unwrap_or("");

    // Work at the byte level to match C's char-by-char behavior (ASCII/byte semantics).
    let a = s1.as_bytes();
    let b = s2.as_bytes();
    let n = a.len();
    let m = b.len();

    // DP table: (n+1) x (m+1), initialized to 0.
    // arr[i][j] = LCS length of a[..i] and b[..j]
    let mut arr = vec![vec![0i32; m + 1]; n + 1];

    for i in 1..=n {
        for j in 1..=m {
            if a[i - 1] == b[j - 1] {
                arr[i][j] = arr[i - 1][j - 1] + 1;
            } else {
                // max with tie-breaking matching C: if equal, prefer arr[i-1][j]
                let up = arr[i - 1][j];
                let left = arr[i][j - 1];
                arr[i][j] = if up >= left { up } else { left };
            }
        }
    }

    let len = arr[n][m] as usize;
    if len == 0 {
        // Spec: if maximum length is zero, output nothing.
        return;
    }

    // Reconstruct one LCS with the same deterministic tie-breaking as the C code:
    // - If chars match: take diagonal
    // - Else if arr[r][c] == arr[r-1][c]: move up
    // - Else: move left
    let mut ans = vec![0u8; len];
    let mut ind: isize = (len as isize) - 1;
    let mut r = n;
    let mut c = m;

    while r >= 1 && c >= 1 {
        if a[r - 1] == b[c - 1] {
            ans[ind as usize] = a[r - 1];
            r -= 1;
            c -= 1;
            ind -= 1;
        } else if arr[r][c] == arr[r - 1][c] {
            r -= 1;
        } else {
            c -= 1;
        }
    }

    // Output exactly the LCS characters, no newline (matches C).
    print!("{}", String::from_utf8_lossy(&ans));
}
```
