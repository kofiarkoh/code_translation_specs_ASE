```rust
use std::cmp::min;
use std::io::{self, Read};

const INF: i32 = 0x10101010;

#[derive(Clone)]
struct Edge {
    to: usize,
    rev: usize,
    cap: i32,
    cost: i32,
}

struct MinCostFlow {
    g: Vec<Vec<Edge>>,
    potential: Vec<i32>,
    dist: Vec<i32>,
    prevv: Vec<usize>,
    preve: Vec<usize>,
}

impl MinCostFlow {
    fn new(v: usize) -> Self {
        Self {
            g: vec![Vec::new(); v],
            potential: vec![0; v],
            dist: vec![INF; v],
            prevv: vec![0; v],
            preve: vec![0; v],
        }
    }

    fn reset(&mut self, v: usize) {
        self.g.clear();
        self.g.resize_with(v, Vec::new);
        self.potential.fill(0);
        self.dist.fill(INF);
        // prevv/preve will be overwritten as needed
        if self.prevv.len() != v {
            self.prevv.resize(v, 0);
            self.preve.resize(v, 0);
        }
    }

    fn add_edge(&mut self, from: usize, to: usize, cap: i32, cost: i32) {
        let rev_to = self.g[to].len();
        let rev_from = self.g[from].len();
        self.g[from].push(Edge {
            to,
            rev: rev_to,
            cap,
            cost,
        });
        self.g[to].push(Edge {
            to: from,
            rev: rev_from,
            cap: 0,
            cost: -cost,
        });
    }

    // Returns -1 if cannot send full flow.
    fn min_cost_flow(&mut self, s: usize, t: usize, mut f: i32) -> i32 {
        let v = self.g.len();
        self.potential.fill(0);
        let mut ret: i32 = 0;

        while f > 0 {
            self.dist.fill(INF);

            // Dijkstra with potentials; use BinaryHeap as min-heap via Reverse.
            use std::cmp::Reverse;
            use std::collections::BinaryHeap;
            let mut pq: BinaryHeap<(Reverse<i32>, usize)> = BinaryHeap::new();

            self.dist[s] = 0;
            pq.push((Reverse(0), s));

            while let Some((Reverse(d), vtx)) = pq.pop() {
                if self.dist[vtx] < d {
                    continue;
                }
                for (i, e) in self.g[vtx].iter().enumerate() {
                    if e.cap <= 0 {
                        continue;
                    }
                    let nv = e.to;
                    let nd = d + e.cost - self.potential[nv] + self.potential[vtx];
                    if self.dist[nv] > nd {
                        self.dist[nv] = nd;
                        self.prevv[nv] = vtx;
                        self.preve[nv] = i;
                        pq.push((Reverse(nd), nv));
                    }
                }
            }

            if self.dist[t] == INF {
                return -1;
            }
            for i in 0..v {
                if self.dist[i] < INF {
                    self.potential[i] += self.dist[i];
                }
            }

            let mut addf = f;
            let mut vtx = t;
            while vtx != s {
                let pv = self.prevv[vtx];
                let pe = self.preve[vtx];
                let cap = self.g[pv][pe].cap;
                if addf > cap {
                    addf = cap;
                }
                vtx = pv;
            }

            f -= addf;
            ret += addf * self.potential[t];

            // Apply augmentation (needs mutable borrows; do via indices and temporary rev info).
            let mut vtx2 = t;
            while vtx2 != s {
                let pv = self.prevv[vtx2];
                let pe = self.preve[vtx2];
                let rev = self.g[pv][pe].rev;

                // Decrease forward cap
                self.g[pv][pe].cap -= addf;
                // Increase reverse cap
                self.g[vtx2][rev].cap += addf;

                vtx2 = pv;
            }
        }

        ret
    }
}

fn abs_i32(x: i32) -> i32 {
    if x >= 0 { x } else { -x }
}

fn main() {
    // Read all input, parse as i32 tokens.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: usize = n_opt.unwrap().parse().unwrap();
        if n == 0 {
            break;
        }
        let m: usize = it.next().unwrap().parse().unwrap();
        let b: i32 = it.next().unwrap().parse().unwrap();

        let mut ax = vec![0i32; n];
        let mut ay = vec![0i32; n];
        for i in 0..n {
            ax[i] = it.next().unwrap().parse().unwrap();
            ay[i] = it.next().unwrap().parse().unwrap();
        }

        let mut bx = vec![0i32; m];
        let mut by = vec![0i32; m];
        let mut cap = vec![0i32; m];
        let mut fixed = vec![0i32; m];
        for i in 0..m {
            bx[i] = it.next().unwrap().parse().unwrap();
            by[i] = it.next().unwrap().parse().unwrap();
            cap[i] = it.next().unwrap().parse().unwrap();
            fixed[i] = it.next().unwrap().parse().unwrap();
        }

        let source = n + m;
        let sink = source + 1;
        let v = sink + 1;

        let mut mcf = MinCostFlow::new(v);

        let mut ans: i32 = INF;
        let lim = 1usize << m;

        // Build graph for subset s and offset mi.
        let mut make_edge = |mcf: &mut MinCostFlow, s_mask: usize, mi: i32| {
            mcf.reset(v);
            // demand nodes: 0..n
            // facility nodes: n..n+m
            // source: n+m, sink: n+m+1
            for i in 0..n {
                mcf.add_edge(source, i, 1, 0);
            }
            for i in 0..m {
                if (s_mask >> i) & 1 == 1 {
                    for j in 0..n {
                        let mut x = abs_i32(ax[j] - bx[i]) + abs_i32(ay[j] - by[i]) - mi;
                        if x < 0 {
                            x = 0;
                        }
                        mcf.add_edge(j, n + i, 1, x);
                    }
                    mcf.add_edge(n + i, sink, cap[i], 0);
                }
            }
        };

        for s_mask in 0..lim {
            let mut res_fixed: i32 = 0;
            let mut cnt: i32 = 0;
            for i in 0..m {
                if (s_mask >> i) & 1 == 1 {
                    res_fixed += fixed[i];
                    cnt += 1;
                }
            }

            // Binary search for best integer mi, using the same scheme as C.
            let mut lb: i32 = 0;
            let mut ub: i32 = 4002;

            let mut skip_subset = false;
            while lb + 1 < ub {
                let mi = (lb + ub) >> 1;

                make_edge(&mut mcf, s_mask, mi);
                let flow1 = mcf.min_cost_flow(source, sink, n as i32);
                let a1 = if flow1 < 0 { -1 } else { flow1 + mi * b * cnt };

                make_edge(&mut mcf, s_mask, mi - 1);
                let flow2 = mcf.min_cost_flow(source, sink, n as i32);
                let a2 = if flow2 < 0 { -1 } else { flow2 + (mi - 1) * b * cnt };

                if a1 < 0 || a2 < 0 {
                    skip_subset = true;
                    break; // matches C's goto next
                }
                if a1 < a2 {
                    lb = mi;
                } else {
                    ub = mi;
                }
            }
            if skip_subset {
                continue;
            }

            make_edge(&mut mcf, s_mask, lb);
            let flow = mcf.min_cost_flow(source, sink, n as i32);
            // Per spec, infeasible subsets are skipped.
            if flow < 0 {
                continue;
            }
            let total = res_fixed + flow + lb * b * cnt;
            if ans > total {
                ans = total;
            }
        }

        out.push_str(&format!("{ans}\n"));
    }

    print!("{out}");
}
```
