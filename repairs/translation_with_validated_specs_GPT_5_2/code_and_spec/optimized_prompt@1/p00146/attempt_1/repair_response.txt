```rust
use std::io::{self, Read};

fn main() {
    // Read all input
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    // Spec precondition: effectively n <= 16. No explicit error handling.
    let mut a = vec![0i32; n];
    let mut b = vec![0i32; n];
    let mut c = vec![0i32; n];
    for i in 0..n {
        a[i] = it.next().unwrap().parse().unwrap();
        b[i] = it.next().unwrap().parse().unwrap();
        c[i] = it.next().unwrap().parse().unwrap();
    }

    let states = 1usize << n;
    let inf = 1e18f64;

    // d[last][mask] = minimal cost to visit exactly 'mask' and end at 'last'
    let mut d = vec![vec![inf; states]; n];
    let mut fr = vec![vec![-1i32; states]; n];

    // f(i, mask, k): step cost from i to k given already-visited set mask (including i)
    let step_cost = |i: usize, mask: usize, k: usize, b: &Vec<i32>, c: &Vec<i32>, n: usize| -> f64 {
        if mask == 0 {
            return 0.0;
        }
        let mut w: i32 = 0;
        for l in 0..n {
            if (mask & (1usize << l)) != 0 {
                w += c[l] * 20;
            }
        }
        let dist = (b[k] - b[i]).abs() as f64 / 2000.0;
        dist * (w as f64 + 70.0)
    };

    for i in 0..n {
        d[i][1usize << i] = 0.0;
    }

    for mask in 0..states {
        for i in 0..n {
            // NOTE: The C code has a precedence bug: `if((1<<i)&j==0)continue;`
            // parses as `(1<<i) & (j==0)`, which is not the intended subset check.
            // The SPEC defines the intended DP invariant, so we implement the correct check:
            if (mask & (1usize << i)) == 0 {
                continue;
            }
            let base = d[i][mask];
            if base >= inf / 2.0 {
                continue;
            }
            for k in 0..n {
                if (mask & (1usize << k)) != 0 {
                    continue;
                }
                let nmask = mask | (1usize << k);
                let cand = base + step_cost(i, mask, k, &b, &c, n);
                if d[k][nmask] > cand {
                    d[k][nmask] = cand;
                    fr[k][nmask] = i as i32;
                }
            }
        }
    }

    let full = states - 1;
    // Choose best endpoint (ties unspecified; we keep first minimal as in C's strict <)
    let mut best_end = 0usize;
    for i in 0..n {
        if d[i][full] < d[best_end][full] {
            best_end = i;
        }
    }

    // Reconstruct path
    let mut r = vec![0usize; n];
    r[n - 1] = best_end;
    let mut mask = full;
    for t in 1..n {
        let cur = r[n - t];
        let prev = fr[cur][mask];
        r[n - t - 1] = prev as usize;
        mask &= !(1usize << cur);
    }

    // Output: first attributes in order, space-separated, newline-terminated
    let mut out = String::new();
    for i in 0..n {
        if i + 1 == n {
            out.push_str(&format!("{}\n", a[r[i]]));
        } else {
            out.push_str(&format!("{} ", a[r[i]]));
        }
    }
    print!("{out}");
}
```
