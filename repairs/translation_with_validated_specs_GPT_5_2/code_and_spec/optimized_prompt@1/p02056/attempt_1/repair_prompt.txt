- the shortest-path distance from a designated source vertex to that vertex in an undirected weighted graph,  
- the shortest-path distance from the source to a corresponding “paired” vertex in a second layer of the same graph,  
- a per-vertex parameter, and  
- a per-vertex offset parameter.  
The program outputs one maximum value per category index.

  - a per-vertex value used to define a cross-layer edge weight  
  - a per-vertex category index  
  - a per-vertex additive term used in the final score  
- m triples describing undirected weighted edges between vertices: two 1-based vertex indices and a nonnegative edge weight (assumption: weights are nonnegative for shortest-path correctness).

3. Outputs  
- Exactly k lines, each containing a signed 64-bit integer.

heap* new_binary_heap (const size_t val_size, int (*cmp_func) (const void *, const void *)) {
  heap *h = (heap *) calloc (1, sizeof (heap));

  h->max_size = 1;

  return h;
}

int is_empty (const heap *h) {
  return h->heap_size == 0;

void push (heap * const h, const void *val) {
  if (h->heap_size == h->max_size) {
    h->max_size = 2 * h->max_size + 1;
    h->array = realloc (h->array, h->val_size * (h->max_size + 1));
  }
  h->heap_size++;
  uint8_t * const array = (uint8_t *) h->array;
  size_t k = h->heap_size;
  const size_t val_size = h->val_size;
  int (* const cmp) (const void *, const void *) = h->cmp;
  heap_func_copy(array + k * val_size, val, val_size);
  while (k > 1) {
    size_t parent = k / 2;
    if (cmp (array + parent * val_size, array + k * val_size) <= 0) {
      return;
    }
    heap_func_swap (array + parent * val_size, array + k * val_size, val_size);
    k = parent;
  }
}

void pop (heap * const h, void *res) {
  uint8_t * const array = (uint8_t *) h->array;
  const size_t val_size = h->val_size;
  if (res != NULL) {
    heap_func_copy (res, array + val_size, val_size);
  }
  heap_func_copy (array + val_size, array + val_size * h->heap_size, val_size);
  h->heap_size--;
  int (* const cmp) (const void *, const void *) = h->cmp;
  const size_t n = h->heap_size;
  size_t k = 1;
  while (2 * k  + 1 <= n) {
    const int c = cmp (array + val_size * 2 * k, array + val_size * (2 * k + 1));
    const size_t next = 2 * k + (c <= 0 ? 0 : 1);

  }
}
