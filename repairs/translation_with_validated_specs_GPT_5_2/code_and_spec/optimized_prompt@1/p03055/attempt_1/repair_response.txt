```rust
use std::cmp::Reverse;
use std::collections::BinaryHeap;
use std::io::{self, Read};

fn bfs_farthest(n: usize, a: &[i32], b: &[i32], c: &[usize], start: usize) -> (usize, Vec<i32>) {
    // depth[v] == -1 means unvisited; otherwise positive depth starting at 1.
    let mut depth = vec![-1i32; n];
    let mut q: Vec<usize> = Vec::with_capacity(n);

    depth[start] = 1;
    q.push(start);

    while let Some(v) = q.pop() {
        // Find the first index `idx` in c such that a[c[idx]] >= v (lower_bound).
        let mut lo: isize = -1;
        let mut hi: isize = (2 * n - 2) as isize; // sentinel index exists at hi
        while hi - lo > 1 {
            let mid = (hi + lo) / 2;
            if a[c[mid as usize]] < v as i32 {
                lo = mid;
            } else {
                hi = mid;
            }
        }

        let mut idx = hi as usize;
        while a[c[idx]] == v as i32 {
            let to = b[c[idx]] as usize;
            if depth[to] < 0 {
                depth[to] = depth[v] + 1;
                q.push(to);
            }
            idx += 1;
        }
    }

    let mut far = start;
    for i in 0..n {
        if depth[i] > depth[far] {
            far = i;
        }
    }
    (far, depth)
}

fn main() {
    // Read all input.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();

    // Store directed edges in arrays a,b of length 2*(n-1), plus one sentinel at index 2n-2.
    let m = 2 * n - 2;
    let mut a: Vec<i32> = vec![0; m + 1];
    let mut b: Vec<i32> = vec![0; m + 1];

    for i in 0..(n.saturating_sub(1)) {
        let u: i32 = it.next().unwrap().parse::<i32>().unwrap() - 1;
        let v: i32 = it.next().unwrap().parse::<i32>().unwrap() - 1;
        let e = 2 * i;
        a[e] = u;
        b[e] = v;
        a[e + 1] = v;
        b[e + 1] = u;
    }

    // The C code heap-sorts edge indices by key a[idx] (ascending), then uses binary search on that.
    // We reproduce that ordering using a BinaryHeap of (a[idx], idx) with Reverse for min-heap behavior.
    let mut heap: BinaryHeap<(Reverse<i32>, usize)> = BinaryHeap::with_capacity(m);
    for idx in 0..m {
        heap.push((Reverse(a[idx]), idx));
    }

    let mut c: Vec<usize> = vec![0; m + 1];
    for i in 0..m {
        c[i] = heap.pop().unwrap().1;
    }

    // Sentinel: ensures binary search and scan can safely read a[c[m]].
    c[m] = m;
    a[m] = -1;
    b[m] = 0;

    // First BFS from arbitrary node 0 to find farthest s.
    let (s, _) = bfs_farthest(n, &a, &b, &c, 0);

    // Second BFS from s to get diameter length in vertices.
    let (_, depth2) = bfs_farthest(n, &a, &b, &c, s);
    let mut d_vertices: i32 = 0;
    for &dv in &depth2 {
        if dv > d_vertices {
            d_vertices = dv;
        }
    }

    // SPEC: output "Second" iff D mod 3 == 0, else "First".
    // Here D is the number of vertices on the diameter path, which equals max depth (starting at 1).
    if d_vertices % 3 == 0 {
        print!("Second\n");
    } else {
        print!("First\n");
    }

    // Note: The C code prints "First" iff (max+1)%3>0, which corresponds to (D+1)%3!=0.
    // That conflicts with the SPEC; we follow the SPEC as the source of truth.
}
```
