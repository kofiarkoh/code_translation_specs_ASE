```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: usize = n_opt.unwrap().parse().unwrap();
        let q: usize = it.next().unwrap().parse().unwrap();
        if n == 0 {
            break;
        }

        // Parent array: 1..=N, root is 1.
        // Spec: parent chain always reaches 1.
        let mut parent = vec![0usize; n + 1];
        for i in 1..=n {
            parent[i] = i; // default, will be overwritten for i>=2
        }
        for k in 2..=n {
            let p: usize = it.next().unwrap().parse().unwrap();
            parent[k] = p;
        }

        // Store queries to process in reverse (as in C code).
        let mut qtype = vec!['?'; q];
        let mut qnode = vec![0usize; q];

        // Mark counts (multiset). Node 1 always marked.
        let mut mark = vec![0i32; n + 1];
        mark[1] = 1;

        for i in 0..q {
            let typ_str = it.next().unwrap();
            let typ = typ_str.chars().next().unwrap();
            let x: usize = it.next().unwrap().parse().unwrap();
            qtype[i] = typ;
            qnode[i] = x;
            if typ == 'M' {
                mark[x] += 1;
            }
        }

        // Process in reverse:
        // - Reverse of mark is decrement.
        // - Ask: climb parents until a marked node is found, add its index.
        // This matches the C program's behavior; note it relies on the spec's
        // precondition that a marked ancestor exists (otherwise it may loop).
        let mut sum: i64 = 0;
        for i in (0..q).rev() {
            if qtype[i] == 'M' {
                let x = qnode[i];
                mark[x] -= 1;
            } else {
                let mut t = qnode[i];
                while mark[t] == 0 {
                    t = parent[t];
                }
                sum += t as i64;
            }
        }

        print!("{sum}\n");
    }
}
```
