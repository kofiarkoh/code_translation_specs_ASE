```rust
use std::io::{self, Read};

const MAX: usize = 1002; // supports n up to 1001 (1-based segments), plus 0
const GRID: usize = MAX; // C uses f[MAX][MAX] and uid[MAX][MAX]

#[derive(Copy, Clone, Default, Debug, PartialEq, Eq)]
struct PP {
    x: i32,
    y: i32,
}

#[derive(Copy, Clone, Default, Debug)]
struct Seg {
    s: PP,
    e: PP,
}

#[derive(Copy, Clone, Default)]
struct T {
    n: u8,          // number of segments in component (C: char)
    u: [Seg; 5],    // stores up to 5 segments (precondition: component size <= 5)
    p: u8,          // count of shared endpoint occurrences in component (C: char)
}

struct UnionFind {
    id: Vec<usize>,
    size: Vec<usize>,
}
impl UnionFind {
    fn new(n: usize) -> Self {
        let mut id = Vec::with_capacity(n);
        let mut size = Vec::with_capacity(n);
        for i in 0..n {
            id.push(i);
            size.push(1);
        }
        Self { id, size }
    }
    fn root(&mut self, mut i: usize) -> usize {
        while i != self.id[i] {
            let p = self.id[i];
            self.id[i] = self.id[p];
            i = self.id[i];
        }
        i
    }
    fn unite(&mut self, p: usize, q: usize) {
        let i = self.root(p);
        let j = self.root(q);
        if i == j {
            return;
        }
        if self.size[i] < self.size[j] {
            self.id[i] = j;
            self.size[j] += self.size[i];
        } else {
            self.id[j] = i;
            self.size[i] += self.size[j];
        }
    }
}

fn vset(x: i32, y: i32) -> PP {
    PP { x, y }
}
fn vsub(a: PP, b: PP) -> PP {
    PP {
        x: a.x - b.x,
        y: a.y - b.y,
    }
}
// Complex multiplication: (ax + i ay) * (bx + i by)
fn vmul(a: PP, b: PP) -> PP {
    PP {
        x: a.x * b.x - a.y * b.y,
        y: a.x * b.y + a.y * b.x,
    }
}
fn dot(a: PP, b: PP) -> i32 {
    a.x * b.x + a.y * b.y
}

// Point-on-segment interior test (strictly excludes endpoints), matching C's pOnSeg.
fn p_on_seg(p: PP, seg: Seg) -> bool {
    if p == seg.s || p == seg.e {
        return false;
    }
    let mut x1 = seg.s.x;
    let mut y1 = seg.s.y;
    let mut x2 = seg.e.x;
    let mut y2 = seg.e.y;

    // C swaps endpoints if x1 > x2 (and swaps y accordingly).
    if x1 > x2 {
        std::mem::swap(&mut x1, &mut x2);
        std::mem::swap(&mut y1, &mut y2);
    }

    if !(x1 <= p.x && p.x <= x2) {
        return false;
    }
    let y_in_range = if y1 <= y2 {
        y1 <= p.y && p.y <= y2
    } else {
        y2 <= p.y && p.y <= y1
    };
    if !y_in_range {
        return false;
    }

    // Collinearity check
    (p.y - y1) * (x2 - x1) == (y2 - y1) * (p.x - x1)
}

fn check_256(id: &T, f: &[[u8; GRID]; GRID], ans: &mut [i32; 10]) {
    // Implements the C orientation/sidedness logic for digits 2/5/6.
    let mut p0 = [PP::default(); 2];
    let mut p1 = [PP::default(); 2];
    let mut p2 = [PP::default(); 2];

    let mut j = 0usize;
    for i in 0..(id.n as usize) {
        let seg = id.u[i];
        let sx = seg.s.x as usize;
        let sy = seg.s.y as usize;
        let ex = seg.e.x as usize;
        let ey = seg.e.y as usize;

        if f[sx][sy] == 1 {
            p0[j] = seg.s;
            p1[j] = seg.e;
            j += 1;
        } else if f[ex][ey] == 1 {
            p0[j] = seg.e;
            p1[j] = seg.s;
            j += 1;
        }
    }

    for i in 0..(id.n as usize) {
        let seg = id.u[i];

        if seg.s != p0[0] && seg.e == p1[0] {
            p2[0] = seg.s;
        } else if seg.e != p0[0] && seg.s == p1[0] {
            p2[0] = seg.e;
        }

        if seg.s != p0[1] && seg.e == p1[1] {
            p2[1] = seg.s;
        } else if seg.e != p0[1] && seg.s == p1[1] {
            p2[1] = seg.e;
        }
    }

    let p = vmul(vsub(p0[0], p1[0]), vset(0, 1));
    let k1 = dot(vsub(p2[0], p1[0]), p);

    let p = vmul(vsub(p0[1], p1[1]), vset(0, 1));
    let k2 = dot(vsub(p2[1], p1[1]), p);

    if k1 > 0 && k2 > 0 {
        ans[2] += 1;
    } else if k1 < 0 && k2 < 0 {
        ans[5] += 1;
    } else {
        ans[6] += 1;
    }
}

fn check_08(
    id: &T,
    n: usize,
    uf: &mut UnionFind,
    u: &[Seg; MAX],
    ans: &mut [i32; 10],
) {
    // Spec: if any segment endpoint (from any segment) lies strictly in interior of any segment
    // of this component => 8 else 0.
    // C iterates only roots, but checks endpoints of u[i] for those i.
    for i in 1..=n {
        if uf.root(i) == i {
            for j in 0..(id.n as usize) {
                if p_on_seg(u[i].s, id.u[j]) {
                    ans[8] += 1;
                    return;
                }
                if p_on_seg(u[i].e, id.u[j]) {
                    ans[8] += 1;
                    return;
                }
            }
        }
    }
    ans[0] += 1;
}

fn check_37(
    id: &T,
    n: usize,
    uf: &mut UnionFind,
    u: &[Seg; MAX],
    f: &[[u8; GRID]; GRID],
    ans: &mut [i32; 10],
) {
    // Spec: if there exists an endpoint of any segment that lies strictly in the interior of a
    // segment of this component whose both endpoints are shared points => 3 else 7.
    for i in 0..(id.n as usize) {
        let seg = id.u[i];
        let sx = seg.s.x as usize;
        let sy = seg.s.y as usize;
        let ex = seg.e.x as usize;
        let ey = seg.e.y as usize;

        if f[sx][sy] > 1 && f[ex][ey] > 1 {
            for j in 1..=n {
                if uf.root(j) == j {
                    if p_on_seg(u[j].s, seg) {
                        ans[3] += 1;
                        return;
                    }
                    if p_on_seg(u[j].e, seg) {
                        ans[3] += 1;
                        return;
                    }
                }
            }
        }
    }
    ans[7] += 1;
}

fn main() {
    // Fast integer parsing (non-negative), matching C's in().
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let bytes = input.as_bytes();
    let mut idx = 0usize;

    let mut next_u32 = || -> Option<u32> {
        while idx < bytes.len() && (bytes[idx] < b'0' || bytes[idx] > b'9') {
            idx += 1;
        }
        if idx >= bytes.len() {
            return None;
        }
        let mut n: u32 = 0;
        while idx < bytes.len() && (b'0'..=b'9').contains(&bytes[idx]) {
            n = n * 10 + (bytes[idx] - b'0') as u32;
            idx += 1;
        }
        Some(n)
    };

    let mut out = String::new();

    loop {
        let n_opt = next_u32();
        if n_opt.is_none() {
            break;
        }
        let n = n_opt.unwrap() as usize;
        if n == 0 {
            break;
        }

        let mut ans = [0i32; 10];
        let mut t = vec![T::default(); n + 1]; // 0..=n
        let mut f = [[0u8; GRID]; GRID];
        let mut uid = [[0u16; GRID]; GRID];
        let mut uf = UnionFind::new(n + 1);

        let mut u = [Seg::default(); MAX];

        for i in 1..=n {
            let x1 = next_u32().unwrap() as usize;
            let y1 = next_u32().unwrap() as usize;
            let x2 = next_u32().unwrap() as usize;
            let y2 = next_u32().unwrap() as usize;

            u[i].s = PP {
                x: x1 as i32,
                y: y1 as i32,
            };
            u[i].e = PP {
                x: x2 as i32,
                y: y2 as i32,
            };

            if f[x1][y1] != 0 {
                uf.unite(uid[x1][y1] as usize, i);
            } else {
                uid[x1][y1] = i as u16;
            }
            f[x1][y1] = f[x1][y1].wrapping_add(1);

            if f[x2][y2] != 0 {
                uf.unite(uid[x2][y2] as usize, i);
            } else {
                uid[x2][y2] = i as u16;
            }
            f[x2][y2] = f[x2][y2].wrapping_add(1);
        }

        // Build per-component summaries (stores up to 5 segments; precondition says OK).
        for i in 1..=n {
            let j = uf.root(i);
            let k = t[j].n as usize;
            t[j].n = t[j].n.wrapping_add(1);
            if k < 5 {
                t[j].u[k] = u[i];
            }
            let sx = u[i].s.x as usize;
            let sy = u[i].s.y as usize;
            let ex = u[i].e.x as usize;
            let ey = u[i].e.y as usize;
            if f[sx][sy] > 1 {
                t[j].p = t[j].p.wrapping_add(1);
            }
            if f[ex][ey] > 1 {
                t[j].p = t[j].p.wrapping_add(1);
            }
        }

        let mut s: i32 = 0;
        for i in 1..=n {
            if t[i].n >= 2 {
                s += t[i].n as i32;
                if t[i].n == 2 && t[i].p == 2 {
                    ans[4] += 1;
                } else if t[i].n == 4 && t[i].p == 6 {
                    ans[9] += 1;
                } else if t[i].n == 4 && t[i].p == 8 {
                    check_08(&t[i], n, &mut uf, &u, &mut ans);
                } else if t[i].n == 5 && t[i].p == 8 {
                    check_256(&t[i], &f, &mut ans);
                } else if t[i].n == 3 && t[i].p == 4 {
                    check_37(&t[i], n, &mut uf, &u, &f, &mut ans);
                }
            }
        }

        // Matches C's accounting: subtract components counted as 4/3/8 by adding those counts.
        // (Each such component was included in s by its segment count, but contributes 1 digit.)
        s += ans[4] + ans[3] + ans[8];
        ans[1] = n as i32 - s;

        out.push_str(&format!("{}", ans[0]));
        for i in 1..10 {
            out.push_str(&format!(" {}", ans[i]));
        }
        out.push('\n');
    }

    print!("{out}");
}
```
