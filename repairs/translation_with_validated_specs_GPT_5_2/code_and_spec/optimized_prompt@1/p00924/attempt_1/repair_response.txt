use std::io::{self, Read};

fn solve(bit: &mut [u8], runlen: &mut [i32], len: usize, start: u8) -> i32 {
    // Mirrors the C recursion exactly, including:
    // - base cases for len==1 and len==2
    // - skipping zero-length runs while toggling start
    // - mismatch behavior: find first required bit in current suffix, add its offset,
    //   flip that found bit in place, then advance suffix by 1 (without shifting).
    if len == 1 {
        return 0;
    }
    if runlen[0] == 0 {
        let next_start = if start == b'1' { b'0' } else { b'1' };
        // runlenlen is not used for bounds checking in the C code; same here.
        return solve(bit, &mut runlen[1..], len, next_start);
    }
    if len == 2 {
        return if bit[0] == start { 0 } else { 1 };
    }

    if bit[0] == start {
        runlen[0] -= 1;
        return solve(&mut bit[1..], runlen, len - 1, start);
    }

    // Mismatch case: locate first occurrence of required bit in current suffix.
    let required = start;
    let mut pos: usize = 0;
    while pos < len && bit[pos] != required {
        pos += 1;
    }
    // Per spec, if not found behavior is undefined; we mimic C by indexing anyway.
    runlen[0] -= 1;
    bit[pos] = if required == b'0' { b'1' } else { b'0' };
    (pos as i32) + solve(&mut bit[1..], runlen, len - 1, start)
}

fn main() {
    // Read all input; C uses scanf with whitespace skipping for both ints and chars.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let m: usize = it.next().unwrap().parse().unwrap();

    let mut b: Vec<u8> = vec![0u8; n];
    let mut c1: i32 = 0;
    for i in 0..n {
        let tok = it.next().unwrap();
        let ch = tok.as_bytes()[0];
        b[i] = ch;
        if ch == b'1' {
            c1 += 1;
        }
    }

    let mut p: Vec<i32> = vec![0i32; m];
    let mut ce: i32 = 0;
    for i in 0..m {
        let val: i32 = it.next().unwrap().parse().unwrap();
        p[i] = val;
        if (i & 1) == 0 {
            ce += val;
        }
    }

    let ret: i32;
    if c1 * 2 == n as i32 {
        let mut b1 = b.clone();
        let mut p1 = p.clone();
        let ret1 = solve(&mut b, &mut p, n, b'1');
        let ret2 = solve(&mut b1, &mut p1, n, b'0');
        ret = ret1.min(ret2);
    } else if ce == c1 {
        ret = solve(&mut b, &mut p, n, b'1');
    } else {
        ret = solve(&mut b, &mut p, n, b'0');
    }

    print!("{ret}\n");
}
