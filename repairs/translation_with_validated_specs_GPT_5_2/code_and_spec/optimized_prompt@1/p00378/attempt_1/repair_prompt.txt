int main()

	int c, ans;

	scanf("%d%d%d", &A, &B, &X);
	if (A >= 2 * B) ans = ((X - 1) / 500 + 1) * B;
	else {
		ans = X / 1000 * A;
		c = X % 1000;
		if (c) {
