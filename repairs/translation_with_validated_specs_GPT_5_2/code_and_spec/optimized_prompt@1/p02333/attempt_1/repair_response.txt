```rust
use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;
const MAX_M: usize = 2000;

fn mod_pow(mut a: i64, mut n: i64, m: i64) -> i64 {
    a %= m;
    let mut x: i64 = 1;
    while n > 0 {
        if (n & 1) == 1 {
            x = (x * a) % m;
        }
        a = (a * a) % m;
        n >>= 1;
    }
    x
}

fn mod_inv(a: i64, p: i64) -> i64 {
    // p is prime per spec; inverse via Fermat's little theorem
    mod_pow(a, p - 2, p)
}

fn make_fact(n: usize, m: i64) -> (Vec<i64>, Vec<i64>) {
    let mut fact = vec![0i64; n + 1];
    let mut invfact = vec![0i64; n + 1];

    fact[0] = 1;
    for i in 0..n {
        fact[i + 1] = (fact[i] * (i as i64 + 1)) % m;
    }

    invfact[n] = mod_inv(fact[n], m);
    for i in (1..=n).rev() {
        invfact[i - 1] = (invfact[i] * i as i64) % m;
    }

    (fact, invfact)
}

fn choose(n: i32, r: i32, fact: &[i64], invfact: &[i64], m: i64) -> i64 {
    if n < 0 || r < 0 || n - r < 0 {
        return 0;
    }
    let n = n as usize;
    let r = r as usize;
    let nr = (n - r) as usize;
    (((fact[n] * invfact[r]) % m) * invfact[nr]) % m
}

fn main() {
    let (fact, invfact) = make_fact(MAX_M, MOD);

    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i64 = it.next().unwrap().parse::<i64>().unwrap();
    let m: i32 = it.next().unwrap().parse::<i32>().unwrap();

    let mut s: i64 = 0;
    for i in 0..=m {
        let c = choose(m, i, &fact, &invfact, MOD);
        let p = mod_pow(i as i64, n, MOD);
        let sign = if ((m - i) & 1) == 0 { 1i64 } else { -1i64 };
        s = (s + c * p % MOD * sign) % MOD;
    }

    // Canonical non-negative residue
    let ans = (s + MOD) % MOD;
    println!("{}", ans);
}
```
