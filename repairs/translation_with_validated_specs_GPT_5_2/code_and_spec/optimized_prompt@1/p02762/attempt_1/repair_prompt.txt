Computes, for each entity in a population, the number of other entities in the same connected component (under a given set of undirected connections) that are eligible to be counted as potential new connections, excluding the entity itself, its directly connected neighbors, and certain disallowed same-component pairs.

#include <stdio.h>
#define N_MAX 100000
#define L_MAX 20
void fill(int *a, int l, int r, int x) {
	for(int i = l; i <= r; ++i) {
		a[i] = x;
	}
	return;
}
int find_root(int *a, int x) {
	int s[L_MAX + 1];
	int k = 0;
	while(a[x] >= 1) {
		s[++k] = x;
		x = a[x];
	}
	while(k) {
		a[s[k--]] = x;
	}
	return x;
}
void unite_set(int *a, int x, int y) {
	x = find_root(a, x);
	y = find_root(a, y);
	if(a[x] < a[y]) {
		a[x] += a[y];
		a[y] = x;
	} else {
		a[y] += a[x];
		a[x] = y;
	}
	return;
}
int set_size(int *a, int x) {
	return -a[find_root(a, x)];
}
