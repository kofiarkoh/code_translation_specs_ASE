#include <stdio.h>
#include <string.h>

#define N	4000
#define L	62
#define K	((N + L - 1) / L)
#define MD	998244353

void shift(long long *aa, long long *bb, int s) {
	int h, q, r;

	memset(bb, 0, K * sizeof *bb);
	q = s / L, r = s % L;
	for (h = 0; h + q < K; h++) {
		bb[h + q] |= (aa[h] & (1LL << L - r) - 1) << r;
		if (h + q + 1 < K && r > 0)
			bb[h + q + 1] |= aa[h] >> L - r;
	}
}

void xor(long long *aa, long long *bb) {
	int h;

	for (h = 0; h < K; h++)
		aa[h] ^= bb[h];
}

int main() {

	scanf("%d%s", &m, s), n = strlen(s);

		if (s[i] == '1')

			if (s[i] == '1')

		}
	}

		l--;
