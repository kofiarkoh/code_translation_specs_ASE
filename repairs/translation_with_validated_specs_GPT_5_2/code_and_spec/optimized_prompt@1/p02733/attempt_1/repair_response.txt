use std::io::{self, Read};

fn main() {
    // Read all input
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_whitespace();

    let h: usize = it.next().unwrap().parse().unwrap();
    let w: usize = it.next().unwrap().parse().unwrap();
    let k: i32 = it.next().unwrap().parse().unwrap();

    let mut grid: Vec<Vec<u8>> = Vec::with_capacity(h);
    for _ in 0..h {
        let row = it.next().unwrap().as_bytes().to_vec();
        grid.push(row);
    }

    // Enumerate all horizontal cut configurations via bitmask over (h-1) gaps.
    // For each configuration, greedily place vertical cuts minimally.
    let mut ans: i32 = i32::MAX;

    // Precompute per-column per-row value (0/1) as i32 for faster sums.
    let mut a = vec![vec![0i32; w]; h];
    for i in 0..h {
        for j in 0..w {
            a[i][j] = if grid[i][j] == b'1' { 1 } else { 0 };
        }
    }

    let max_mask = 1usize << (h.saturating_sub(1));
    for mask in 0..max_mask {
        // Build groups (row ranges) induced by horizontal cuts.
        let mut starts: Vec<usize> = Vec::new();
        let mut ends: Vec<usize> = Vec::new();
        let mut cur_start = 0usize;
        for r in 0..h - 1 {
            if (mask >> r) & 1 == 1 {
                starts.push(cur_start);
                ends.push(r);
                cur_start = r + 1;
            }
        }
        starts.push(cur_start);
        ends.push(h - 1);
        let g = starts.len();

        // Horizontal cuts count = number of groups - 1
        let mut cuts: i32 = (g as i32) - 1;

        // Greedy vertical cuts: maintain counts per group for current vertical segment.
        let mut counts = vec![0i32; g];
        let mut feasible = true;

        for col in 0..w {
            // Add this column's ones to each group
            let mut exceeded = false;
            for gi in 0..g {
                let mut add = 0i32;
                for r in starts[gi]..=ends[gi] {
                    add += a[r][col];
                }
                counts[gi] += add;
                if counts[gi] > k {
                    exceeded = true;
                }
            }

            if exceeded {
                // Need a vertical cut before this column.
                // Reset counts to just this column; if still exceeds, infeasible for this mask.
                cuts += 1;
                for gi in 0..g {
                    let mut sum = 0i32;
                    for r in starts[gi]..=ends[gi] {
                        sum += a[r][col];
                    }
                    counts[gi] = sum;
                    if counts[gi] > k {
                        feasible = false;
                        break;
                    }
                }
                if !feasible {
                    break;
                }
            }
        }

        if feasible && cuts < ans {
            ans = cuts;
        }
    }

    // Per spec, input is feasible; no special error handling required.
    println!("{}", ans);
}
