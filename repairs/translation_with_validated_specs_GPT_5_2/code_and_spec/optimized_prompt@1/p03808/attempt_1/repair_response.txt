```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i64 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    // A has length N+1 because we set A[N] = A[0]
    let mut a: Vec<i64> = vec![0; (n as usize) + 1];
    for i in 0..(n as usize) {
        a[i] = it.next().unwrap().parse().unwrap();
    }
    a[n as usize] = a[0];

    // diff indexed 0..=N (diff[0] used as accumulator later)
    let mut diff: Vec<i64> = vec![0; (n as usize) + 1];

    let mut sum: i64 = 0;
    for i in 1..=n as usize {
        sum += a[i];
        diff[i] = a[i] - a[i - 1];
    }

    let t: i64 = n * (n + 1) / 2;
    if sum % t != 0 {
        print!("NO\n");
        return;
    }

    let s: i64 = sum / t;

    diff[0] = 0;
    let mut i: usize = 1;
    while i <= n as usize {
        // Integer division semantics match C for i64 in Rust (truncates toward zero).
        diff[i] = (s - diff[i]) / n;
        if diff[i] < 0 || diff[i] > s {
            break;
        }
        diff[0] += diff[i];
        i += 1;
    }

    if i <= n as usize || diff[0] != s {
        print!("NO\n");
        return;
    }

    for i in 1..=n as usize {
        let weight: i64 = n - (i as i64) + 1;
        a[n as usize] -= diff[i] * weight;
    }

    if a[n as usize] != 0 {
        // Note: case-sensitive mismatch is intentional per spec/C: "No" here.
        print!("No\n");
    } else {
        print!("YES\n");
    }
}
```
