#define MAX 50001

int makePrimeTable( int * table )
{
        int i, j;
        for ( i = 0 ; i < MAX ; i++ ) {
                table[i] = 0;
        }

        for ( i = 2 ; i < MAX ; i++ ) {
                if ( table[i] == 0 ) {
                        for ( j = 2 ; i * j < MAX ; j++ ) {

}
