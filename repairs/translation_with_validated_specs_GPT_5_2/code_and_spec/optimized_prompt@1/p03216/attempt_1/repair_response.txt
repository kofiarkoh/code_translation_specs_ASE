```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin as whitespace-delimited tokens (matches scanf("%d %s %d ...")).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let s_token = it.next().unwrap();
    // SPEC: assumes s_token length >= n; otherwise undefined. We follow that assumption.
    let bytes = s_token.as_bytes();

    let q: usize = it.next().unwrap().parse().unwrap();
    let mut ks: Vec<usize> = Vec::with_capacity(q);
    for _ in 0..q {
        ks.push(it.next().unwrap().parse().unwrap());
    }

    let mut out = String::new();

    for &k in &ks {
        // Per-query fresh counters.
        let mut ans: i64 = 0;
        let mut d: i64 = 0;
        let mut m: i64 = 0;
        let mut dm: i64 = 0;

        // Initialization over [0, k-1]
        for i in 0..k {
            match bytes[i] {
                b'D' => d += 1,
                b'M' => {
                    m += 1;
                    dm += d;
                }
                b'C' => ans += dm,
                _ => {}
            }
        }

        // Slide window: i = k .. n-1
        for i in k..n {
            // Remove leaving character s[i-k] in the specified order.
            match bytes[i - k] {
                b'D' => {
                    dm -= m;
                    d -= 1;
                }
                b'M' => {
                    m -= 1;
                }
                _ => {}
            }

            // Add entering character s[i]
            match bytes[i] {
                b'D' => d += 1,
                b'M' => {
                    m += 1;
                    dm += d;
                }
                b'C' => ans += dm,
                _ => {}
            }
        }

        out.push_str(&format!("{ans}\n"));
    }

    print!("{out}");
}
```
