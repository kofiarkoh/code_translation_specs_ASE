use std::io::{self, Read};

const MAX_N: usize = 100_003;

fn main() {
    // Read entire stdin, parse as non-negative integers (behavior unspecified if malformed/negative).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let k: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    // Spec: if N < K, output 0.
    if n < k {
        print!("0\n");
        return;
    }

    // Fixed-size internal storage bounded by compile-time maximum (per spec).
    // Preconditions say N and K fit; otherwise behavior unspecified.
    let mut a = vec![0usize; n];
    for i in 0..n {
        a[i] = it.next().unwrap().parse::<usize>().unwrap();
    }

    // Frequency array for values 1..=K (index 0 unused).
    let mut f = vec![0usize; k + 1];

    let mut s = 0usize; // number of distinct values in 1..=K currently present in window
    let mut i = 0usize;
    let mut j = 0usize;

    // Expand right until we have all 1..=K or reach end.
    while s < k && j < n {
        let t = a[j];
        j += 1;
        if t <= k {
            if f[t] == 0 {
                s += 1;
            }
            f[t] += 1;
        }
    }

    if s < k {
        // No window contains all 1..=K.
        print!("0\n");
        return;
    }

    // Shrink from left to make window minimal w.r.t. left boundary.
    while i < j {
        let t = a[i];
        if t > k || f[t] > 1 {
            if t <= k {
                f[t] -= 1;
            }
            i += 1;
        } else {
            break;
        }
    }

    let mut ans = j - i;

    // Slide window: expand right by one, then shrink left to minimal, update answer.
    while j < n {
        let t = a[j];
        j += 1;
        if t <= k {
            f[t] += 1;
        }

        while i < j {
            let t2 = a[i];
            if t2 > k || f[t2] > 1 {
                if t2 <= k {
                    f[t2] -= 1;
                }
                i += 1;
            } else {
                break;
            }
        }

        let len = j - i;
        if len < ans {
            ans = len;
        }
    }

    print!("{ans}\n");
}
