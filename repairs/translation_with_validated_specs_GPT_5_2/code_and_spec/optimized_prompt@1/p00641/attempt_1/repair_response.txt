use std::io::{self, Read};

const MOD: i32 = 10007;

struct DSU {
    parent: Vec<usize>,
    size: Vec<usize>,
}

impl DSU {
    fn new(n: usize) -> Self {
        let mut parent = Vec::with_capacity(n);
        let mut size = Vec::with_capacity(n);
        for i in 0..n {
            parent.push(i);
            size.push(1);
        }
        Self { parent, size }
    }

    fn root(&mut self, mut x: usize) -> usize {
        // Path compression by halving (matches C's id[i] = id[id[i]])
        while self.parent[x] != x {
            let p = self.parent[x];
            let gp = self.parent[p];
            self.parent[x] = gp;
            x = gp;
        }
        x
    }

    fn unite(&mut self, a: usize, b: usize) {
        let mut ra = self.root(a);
        let mut rb = self.root(b);
        if ra == rb {
            return;
        }
        if self.size[ra] < self.size[rb] {
            std::mem::swap(&mut ra, &mut rb);
        }
        self.parent[rb] = ra;
        self.size[ra] += self.size[rb];
    }
}

struct Scanner {
    buf: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }

    // Spec assumes non-negative integers and well-formed input.
    fn next_usize(&mut self) -> Option<usize> {
        while self.idx < self.buf.len() && self.buf[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.buf.len() {
            return None;
        }
        let mut n: usize = 0;
        while self.idx < self.buf.len() {
            let c = self.buf[self.idx];
            if !c.is_ascii_digit() {
                break;
            }
            n = n * 10 + (c - b'0') as usize;
            self.idx += 1;
        }
        Some(n)
    }
}

fn main() {
    let mut sc = Scanner::new();
    let mut out = String::new();

    loop {
        let n = match sc.next_usize() {
            Some(v) => v,
            None => break,
        };
        if n == 0 {
            break;
        }

        let mut dsu = DSU::new(n);
        let mut f0 = vec![0usize; n];
        let mut f1 = vec![0usize; n];

        for i in 0..n {
            let a = sc.next_usize().unwrap();
            f0[i] = sc.next_usize().unwrap();
            let b = sc.next_usize().unwrap();
            f1[i] = sc.next_usize().unwrap();
            dsu.unite(i, a);
            dsu.unite(i, b);
        }

        // Per component (indexed by root id): track maximum attribute and its count.
        let mut cnt = vec![0usize; n];
        let mut maxv = vec![0usize; n];

        for i in 0..n {
            let k = dsu.root(i);

            // check(k, f0[i])
            let v = f0[i];
            if cnt[k] == 0 || v > maxv[k] {
                cnt[k] = 1;
                maxv[k] = v;
            } else if v == maxv[k] {
                cnt[k] += 1;
            }

            // check(k, f1[i])
            let v = f1[i];
            if cnt[k] == 0 || v > maxv[k] {
                cnt[k] = 1;
                maxv[k] = v;
            } else if v == maxv[k] {
                cnt[k] += 1;
            }
        }

        let mut ans: i32 = 1;
        for i in 0..n {
            if cnt[i] != 0 {
                let factor = (cnt[i] >> 1) as i32; // floor(T/2)
                ans = (ans * factor) % MOD;
            }
        }

        out.push_str(&format!("{ans}\n"));
    }

    print!("{out}");
}
