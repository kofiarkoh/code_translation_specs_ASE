```rust
use std::io::{self, Read};

fn main() {
    // Read all input (matches scanf-style whitespace handling).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let a: i64 = it.next().unwrap().parse().unwrap();

    // C code uses 1-based indexing for x; keep that to preserve semantics.
    let mut x = vec![0i64; n + 1];
    let mut sum: i64 = 0;
    for i in 1..=n {
        let v: i64 = it.next().unwrap().parse().unwrap();
        x[i] = v;
        sum += v;
    }

    // Spec precondition: sum is non-negative, a is non-zero.
    // No explicit error handling per spec; we assume inputs satisfy preconditions.
    let sum_usize: usize = sum as usize;

    // dp[j][k][s] = number of ways to choose exactly k elements from first j elements with sum s.
    // C uses a huge stack VLA; in Rust we allocate on the heap.
    // This preserves functional behavior; memory failure behavior remains unspecified (panic/abort).
    let mut dp = vec![vec![vec![0i64; sum_usize + 1]; n + 1]; n + 1];

    for j in 0..=n {
        for k in 0..=n {
            for s in 0..=sum_usize {
                dp[j][k][s] = if j >= 1 && (s as i64) < x[j] {
                    dp[j - 1][k][s]
                } else if j >= 1 && k >= 1 && (s as i64) >= x[j] {
                    let prev = dp[j - 1][k][s];
                    let idx = (s as i64 - x[j]) as usize;
                    prev + dp[j - 1][k - 1][idx]
                } else if j == 0 && k == 0 && s == 0 {
                    1
                } else {
                    0
                };
            }
        }
    }

    let mut ans: i64 = 0;

    // C computes max = min(sum/A, N) using integer division.
    // Note: if A were negative, C's sum/A truncates toward zero; we mirror that with Rust's /.
    let max_k_i64 = {
        let q = sum / a;
        if q < n as i64 { q } else { n as i64 }
    };

    // If max_k_i64 <= 0, loop does nothing (same as C for k=1..=max).
    if max_k_i64 >= 1 {
        for k in 1..=(max_k_i64 as usize) {
            let target_sum = (k as i64) * a;
            // Spec: only count if within achievable non-negative sum range.
            if target_sum >= 0 && (target_sum as usize) <= sum_usize {
                ans += dp[n][k][target_sum as usize];
            }
        }
    }

    print!("{ans}\n");
}
```
