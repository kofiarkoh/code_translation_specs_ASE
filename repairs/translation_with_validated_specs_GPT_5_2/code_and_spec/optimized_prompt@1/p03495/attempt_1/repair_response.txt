```rust
use std::io::{self, Read};

fn main() {
    // Read all input as whitespace-separated integers.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let k: isize = it.next().unwrap().parse().unwrap();

    // Frequency domain is [1, N] inclusive per spec.
    let mut freq = vec![0i64; n + 1];
    for _ in 0..n {
        let a: usize = it.next().unwrap().parse().unwrap();
        freq[a] += 1;
    }

    // Collect only positive frequencies (distinct values present).
    let mut present: Vec<i64> = freq.into_iter().skip(1).filter(|&x| x > 0).collect();
    let distinct = present.len() as isize;

    // If K >= distinct_count, answer is 0 (spec).
    let mut ans: i64 = 0;
    if k < distinct {
        present.sort_unstable(); // ascending
        let remove_distinct = (distinct - k) as usize;
        ans = present.iter().take(remove_distinct).sum();
    }

    print!("{ans}\n");

    // Note: The C code sorts an array of length N (including zeros) and then sums
    // indices 1..=N-K with a `c[i] > 0` guard, which can remove too few when many
    // zeros exist. The spec requires removing the (distinct_count - K) least
    // frequent *present* values, which is implemented here.
}
```
