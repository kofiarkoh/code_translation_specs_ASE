```rust
use std::io::{self, Read};

#[derive(Clone, Copy, Default)]
struct Node {
    used: bool,
    m: i32,
    mx: i32,
    xl: i32,
    xr: i32,
    par: usize,
    ch: usize,
    sib: usize,
}

struct DSU {
    p: Vec<usize>,
    rank: Vec<u8>,
}
impl DSU {
    fn new(n: usize) -> Self {
        let mut p = Vec::with_capacity(n);
        for i in 0..n {
            p.push(i);
        }
        Self {
            p,
            rank: vec![0; n],
        }
    }
    fn find(&mut self, x: usize) -> usize {
        if self.p[x] != x {
            let root = self.find(self.p[x]);
            self.p[x] = root;
        }
        self.p[x]
    }
    fn union(&mut self, a: usize, b: usize) {
        let mut x = self.find(a);
        let mut y = self.find(b);
        if x == y {
            return;
        }
        let rx = self.rank[x];
        let ry = self.rank[y];
        if rx > ry {
            self.p[y] = x;
        } else {
            self.p[x] = y;
            if rx == ry {
                self.rank[y] = ry + 1;
            }
        }
    }
}

fn mktree(
    r: usize,
    c: usize,
    this: usize,
    w: usize,
    h: usize,
    map: &[Vec<u8>],
    rel: &[Vec<usize>],
    tr: &[isize],
    tbl: &mut [Node],
) {
    if r == h - 1 {
        if map[r][c] == b'.' {
            return;
        }
        if tbl[this].xl < 0 {
            tbl[this].xl = c as i32;
        } else if (c as i32) < tbl[this].xl {
            tbl[this].xl = c as i32;
        }
        if (c as i32) > tbl[this].xr {
            tbl[this].xr = c as i32;
        }
        return;
    }
    if map[r + 1][c] == b'.' {
        return;
    }

    let k = rel[r + 1][c];
    let p = tr[k] as usize;
    if p == this {
        return;
    }

    if tbl[this].xl < 0 {
        tbl[this].xl = c as i32;
    } else if (c as i32) < tbl[this].xl {
        tbl[this].xl = c as i32;
    }
    if (c as i32) > tbl[this].xr {
        tbl[this].xr = c as i32;
    }
    if tbl[this].par != 0 {
        return;
    }

    // Append `this` to parent's child list (preserving scan order).
    if tbl[p].ch == 0 {
        tbl[p].ch = this;
    } else {
        let mut s = tbl[p].ch;
        while tbl[s].sib != 0 {
            s = tbl[s].sib;
        }
        tbl[s].sib = this;
    }
    tbl[this].par = p;

    // `w` is unused here but kept in signature symmetry with C intent.
    let _ = w;
}

fn check(this: usize, tbl: &mut [Node]) -> bool {
    if tbl[this].used {
        return true;
    }

    let ch = tbl[this].ch;
    if ch != 0 && !check(ch, tbl) {
        return false;
    }

    let mut sib = tbl[this].sib;
    while sib != 0 {
        if !check(sib, tbl) {
            return false;
        }
        sib = tbl[sib].sib;
    }

    // x = 0.5 + mx/m (real division)
    let x = 0.5f64 + (tbl[this].mx as f64) / (tbl[this].m as f64);
    if x <= (tbl[this].xl as f64) || (tbl[this].xr as f64 + 1.0) <= x {
        return false;
    }

    let par = tbl[this].par;
    tbl[par].m += tbl[this].m;
    tbl[par].mx += tbl[this].mx;
    tbl[this].used = true;
    true
}

fn main() {
    // Read all input, whitespace-delimited like scanf("%d%d") and scanf("%s").
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let w_opt = it.next();
        if w_opt.is_none() {
            break;
        }
        let w: i32 = w_opt.unwrap().parse().unwrap();
        let h: i32 = it.next().unwrap().parse().unwrap();
        if w <= 0 {
            break;
        }
        let (w, h) = (w as usize, h as usize);

        // Grid: h rows, each at least w chars (per spec; no validation).
        let mut map: Vec<Vec<u8>> = Vec::with_capacity(h);
        for _ in 0..h {
            let s = it.next().unwrap();
            map.push(s.as_bytes().to_vec());
        }

        // rel[r][c] initially unique id r*w+c
        let n = w * h;
        let mut rel: Vec<Vec<usize>> = vec![vec![0; w]; h];
        for r in 0..h {
            for c in 0..w {
                rel[r][c] = r * w + c;
            }
        }

        // Union-find over same-character 4-neighborhood.
        let mut dsu = DSU::new(n);
        let mv = [(-1i32, 0i32), (0, 1), (1, 0), (0, -1)];
        for r in 0..h {
            for c in 0..w {
                if map[r][c] == b'.' {
                    continue;
                }
                for (dr, dc) in mv {
                    let r2 = r as i32 + dr;
                    let c2 = c as i32 + dc;
                    if r2 >= 0 && r2 < h as i32 && c2 >= 0 && c2 < w as i32 {
                        let (rr, cc) = (r2 as usize, c2 as usize);
                        if map[rr][cc] == map[r][c] {
                            let a = rel[r][c];
                            let b = rel[rr][cc];
                            dsu.union(a, b);
                        }
                    }
                }
            }
        }

        // Build components and the single-parent support structure by bottom-up scan.
        // tbl is 1-based; node 0 is a dummy.
        let mut tbl: Vec<Node> = vec![Node::default(); 100];
        let mut tr: Vec<isize> = vec![-1; n]; // maps DSU root -> node id
        let mut sz: usize = 1;

        for r in (0..h).rev() {
            for c in 0..w {
                if map[r][c] == b'.' {
                    continue;
                }
                let k = dsu.find(rel[r][c]);
                rel[r][c] = k;

                let i: usize;
                if tr[k] < 0 {
                    i = sz;
                    sz += 1;
                    tr[k] = i as isize;
                    tbl[i].xl = -1; // sentinel like C
                    // other fields already zeroed
                } else {
                    i = tr[k] as usize;
                }

                tbl[i].m += 1;
                tbl[i].mx += c as i32;

                mktree(r, c, i, w, h, &map, &rel, &tr, &mut tbl);
            }
        }

        // Root selection: node 1 is forced to be its own parent.
        // Per spec, input guarantees at least one occupied cell; otherwise behavior undefined.
        tbl[1].par = 1;

        if check(1, &mut tbl) {
            out.push_str("STABLE\n");
        } else {
            out.push_str("UNSTABLE\n");
        }
    }

    print!("{out}");
}
```
