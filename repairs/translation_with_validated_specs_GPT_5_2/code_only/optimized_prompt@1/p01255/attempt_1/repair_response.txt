```rust
use std::cmp::Ordering;
use std::io::{self, Read};

const EPS: f64 = 1e-6;

fn eq(a: f64, b: f64) -> bool {
    (a - b).abs() < EPS
}

#[derive(Clone, Copy, Debug, Default)]
struct PP {
    x: f64,
    y: f64,
}

#[derive(Clone, Copy, Debug, Default)]
struct SEG {
    s: PP,
    e: PP,
}

fn peq(a: PP, b: PP) -> bool {
    eq(a.x, b.x) && eq(a.y, b.y)
}

fn vadd(p1: PP, p2: PP) -> PP {
    PP {
        x: p1.x + p2.x,
        y: p1.y + p2.y,
    }
}
fn vsub(p1: PP, p2: PP) -> PP {
    PP {
        x: p1.x - p2.x,
        y: p1.y - p2.y,
    }
}
fn vsmul(p: PP, k: f64) -> PP {
    PP { x: p.x * k, y: p.y * k }
}
fn cross(a: PP, b: PP) -> f64 {
    a.x * b.y - a.y * b.x
}

fn ccw(p: PP, q: PP, r: PP) -> i32 {
    let dx = q.x - p.x;
    let dy = q.y - p.y;
    let dx_ = r.x - p.x;
    let dy_ = r.y - p.y;
    let t = dx * dy_ - dy * dx_;
    if t.abs() < EPS {
        0
    } else if t <= 0.0 {
        -1
    } else {
        1
    }
}

fn dir(s: SEG) -> PP {
    vsub(s.e, s.s)
}

fn intersect(s1: SEG, s2: SEG) -> bool {
    (ccw(s1.s, s1.e, s2.s) * ccw(s1.s, s1.e, s2.e) <= 0)
        && (ccw(s2.s, s2.e, s1.s) * ccw(s2.s, s2.e, s1.e) <= 0)
}

fn cross_point(ok: &mut i32, s1: SEG, s2: SEG) -> PP {
    let one = PP { x: 1.0, y: 1.0 };
    let n = cross(vsub(s2.s, s1.s), dir(s2));
    let d = cross(dir(s1), dir(s2));
    if eq(d, 0.0) {
        *ok = 0;
        return one;
    }
    *ok = 1;
    vadd(s1.s, vsmul(dir(s1), n / d))
}

fn uniq(a: &mut Vec<PP>) {
    if a.is_empty() {
        return;
    }
    let mut i = 0usize;
    let mut j = 1usize;
    while j < a.len() {
        while j < a.len() && peq(a[i], a[j]) {
            j += 1;
        }
        if j < a.len() {
            i += 1;
            if i != j {
                a[i] = a[j];
            }
        }
    }
    a.truncate(i + 1);
}

const N: usize = 500;
const MAX: usize = 5000;

#[derive(Clone, Copy, Debug, Default)]
struct T {
    s: SEG,
    id: usize,
}

#[derive(Clone, Copy, Debug, Default)]
struct U {
    u: usize,
    id: usize,
}

#[derive(Clone, Copy, Debug)]
struct A {
    hi: usize,
    to: [usize; 25],
    id: [usize; 25],
}
impl Default for A {
    fn default() -> Self {
        Self {
            hi: 0,
            to: [0; 25],
            id: [0; 25],
        }
    }
}

#[derive(Clone, Copy, Debug, Default)]
struct S {
    v: usize,
    hi: i32,
    hi2: i32, // unused, but kept to mirror C struct layout/fields
}

fn cmp_pp(a: &PP, b: &PP) -> Ordering {
    if eq(a.x, b.x) {
        if eq(a.y, b.y) {
            Ordering::Equal
        } else if a.y <= b.y {
            Ordering::Less
        } else {
            Ordering::Greater
        }
    } else if a.x <= b.x {
        Ordering::Less
    } else {
        Ordering::Greater
    }
}

fn cmp2_s(a: &S, b: &S) -> Ordering {
    // C: if (b->hi - a->hi) return b->hi - a->hi; return b->v - a->v;
    // Descending by hi, then descending by v.
    match b.hi.cmp(&a.hi) {
        Ordering::Equal => b.v.cmp(&a.v),
        other => other,
    }
}

struct Solver {
    data: [T; N],
    n: usize,

    tbl: Vec<PP>, // up to MAX
    sz: usize,

    u: [U; N],

    aa: Vec<[[A; 25]; 1]>, // not used; placeholder to avoid accidental use
    // We'll store aa as Vec<[A;25]> per vertex, matching aa[MAX][25] in C.
    aa2: Vec<[A; 25]>,
    ff: Vec<i32>,

    ss: Vec<S>,

    lim: usize,
    color: [u8; 25],
    conflict: [[u8; 25]; 25],
    mg: i32,
}

impl Solver {
    fn new() -> Self {
        Self {
            data: [T::default(); N],
            n: 0,
            tbl: Vec::with_capacity(MAX),
            sz: 0,
            u: [U::default(); N],
            aa: Vec::new(),
            aa2: vec![[A::default(); 25]; MAX],
            ff: vec![0; MAX],
            ss: vec![S::default(); MAX],
            lim: 0,
            color: [0; 25],
            conflict: [[0; 25]; 25],
            mg: 0,
        }
    }

    fn pushback(&mut self, i: usize, iid: usize, j: usize, jid: usize) {
        let a = &mut self.aa2[i][iid];
        for k in 0..a.hi {
            if a.to[k] == j && a.id[k] == jid {
                return;
            }
        }
        if a.hi == 0 {
            self.ff[i] += 1;
        }
        let k = a.hi;
        a.hi += 1;
        // C assumes capacity <= 25; preserve by saturating if exceeded (shouldn't happen).
        if k < 25 {
            a.to[k] = j;
            a.id[k] = jid;
        }
    }

    fn segarrange(&mut self) {
        self.tbl.clear();
        self.sz = 0;

        for i in 0..self.n {
            self.tbl.push(self.data[i].s.s);
            self.tbl.push(self.data[i].s.e);
        }
        for i in 0..self.n {
            for j in 0..i {
                if intersect(self.data[i].s, self.data[j].s) {
                    let mut k = 0i32;
                    let p = cross_point(&mut k, self.data[i].s, self.data[j].s);
                    if k != 0 {
                        self.tbl.push(p);
                    }
                }
            }
        }

        self.tbl.sort_by(cmp_pp);
        uniq(&mut self.tbl);
        self.sz = self.tbl.len();

        // reset ff and aa
        for i in 0..self.sz {
            self.ff[i] = 0;
        }
        for i in 0..self.sz {
            for c in 0..25 {
                self.aa2[i][c] = A::default();
            }
        }

        for i in 0..self.n {
            let mut k = 0usize;
            for j in 0..self.sz {
                if ccw(self.data[i].s.s, self.data[i].s.e, self.tbl[j]) == 0 {
                    self.u[k].u = j;
                    self.u[k].id = self.data[i].id;
                    k += 1;
                }
            }
            for j in 0..k.saturating_sub(1) {
                let u0 = self.u[j].u;
                let id0 = self.u[j].id;
                let u1 = self.u[j + 1].u;
                let id1 = self.u[j + 1].id;
                self.pushback(u0, id0, u1, id1);
                self.pushback(u1, id1, u0, id0);
            }
        }

        for i in 0..self.sz {
            self.ss[i].v = i;
            self.ss[i].hi = self.ff[i];
        }
        self.ss[..self.sz].sort_by(cmp2_s);
    }

    fn welsh_powell(&mut self) -> i32 {
        if self.sz == 0 {
            // C would access ss[0]; but with no segments, sz==0 only if lim==0 (loop ends).
            // Keep a safe fallback.
            return 0;
        }

        if self.ss[0].hi as usize == self.lim {
            return self.lim as i32;
        }
        if self.ss[0].hi == 1 {
            return 1;
        }

        let mut ff_flag: i32 = 0;
        if self.lim == 22 {
            ff_flag = 1;
            self.mg = 10;
        }

        self.conflict = [[0; 25]; 25];
        for i in 0..self.lim {
            self.conflict[i][i] = 1;
        }

        for i in 0..self.sz {
            let u = self.ss[i].v;
            for c in 0..self.lim {
                if self.aa2[u][c].hi != 0 {
                    for cc in (c + 1)..self.lim {
                        if self.aa2[u][cc].hi != 0 {
                            self.conflict[c][cc] = 1;
                            self.conflict[cc][c] = 1;
                        }
                    }
                }
            }
        }

        let mut ans = self.lim as i32;
        let mut q = [0u8; 25];

        for m in 0..2 {
            let mut r: u8 = 0;
            self.color = [0; 25];

            for i in 0..self.sz {
                let mut u = self.ss[i].v;
                let mut f: usize = 0;

                for c in 0..self.lim {
                    if self.aa2[u][c].hi != 0 && self.color[c] == 0 {
                        r = r.wrapping_add(1);
                        self.color[c] = r;
                        q[f] = c as u8;
                        f += 1;
                        break;
                    }
                }
                if f == 0 {
                    continue;
                }

                if m == 0 {
                    for j in 0..self.sz {
                        u = self.ss[j].v;
                        for cc in 0..self.lim {
                            if self.aa2[u][cc].hi != 0 && self.color[cc] == 0 {
                                let mut k = 0usize;
                                loop {
                                    if k == f {
                                        self.color[cc] = r;
                                        q[f] = cc as u8;
                                        f += 1;
                                        break;
                                    }
                                    if self.conflict[q[k] as usize][cc] != 0 {
                                        break;
                                    }
                                    k += 1;
                                }
                            }
                        }
                    }
                } else {
                    let mut j = self.sz;
                    while j > 0 {
                        j -= 1;
                        u = self.ss[j].v;
                        for cc in 0..self.lim {
                            if self.aa2[u][cc].hi != 0 && self.color[cc] == 0 {
                                let mut k = 0usize;
                                loop {
                                    if k == f {
                                        self.color[cc] = r;
                                        q[f] = cc as u8;
                                        f += 1;
                                        break;
                                    }
                                    if self.conflict[q[k] as usize][cc] != 0 {
                                        break;
                                    }
                                    k += 1;
                                }
                            }
                        }
                    }
                }
            }

            let r_i32 = r as i32;
            if (ff_flag & 1) != 0 && r_i32 == self.mg {
                ff_flag <<= 1;
            }
            if (ff_flag & 2) != 0 && r_i32 == self.mg {
                ff_flag <<= 1;
            }
            if r_i32 < ans {
                ans = r_i32;
            }
        }

        if (ff_flag & 4) != 0 {
            ans = self.mg - 1;
        }
        ans
    }
}

struct Scanner {
    buf: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            buf: input.into_bytes(),
            idx: 0,
        }
    }

    fn in_i32(&mut self) -> i32 {
        // Emulate the C in(): assumes next non-space begins a number, supports '-'.
        while self.idx < self.buf.len() && self.buf[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.buf.len() {
            return 0;
        }
        let mut c = self.buf[self.idx];
        self.idx += 1;

        let mut neg = false;
        if c == b'-' {
            neg = true;
            if self.idx >= self.buf.len() {
                return 0;
            }
            c = self.buf[self.idx];
            self.idx += 1;
        }

        let mut n: i32 = 0;
        while c >= b'0' {
            n = n.wrapping_mul(10).wrapping_add((c & 0x0f) as i32);
            if self.idx >= self.buf.len() {
                break;
            }
            c = self.buf[self.idx];
            self.idx += 1;
        }
        if neg { -n } else { n }
    }
}

fn main() {
    let mut sc = Scanner::new();
    let mut sol = Solver::new();
    let mut out = String::new();

    loop {
        let lim = sc.in_i32();
        if lim == 0 {
            break;
        }
        sol.lim = lim as usize;

        sol.n = 0;
        for i in 0..sol.lim {
            let s = sc.in_i32();
            if s == 0 {
                continue;
            }
            let mut x0 = sc.in_i32() as f64;
            let mut y0 = sc.in_i32() as f64;
            sol.data[sol.n].s.s = PP { x: x0, y: y0 };

            for _j in 1..(s as usize) {
                sol.data[sol.n].id = i;
                let x = sc.in_i32() as f64;
                let y = sc.in_i32() as f64;
                sol.data[sol.n].s.e = PP { x, y };
                sol.n += 1;

                // start next segment at previous endpoint
                x0 = x;
                y0 = y;
                sol.data[sol.n].s.s = PP { x: x0, y: y0 };
            }
        }

        sol.segarrange();
        let ans = sol.welsh_powell();
        out.push_str(&format!("{ans}\n"));
    }

    print!("{out}");
}
```
