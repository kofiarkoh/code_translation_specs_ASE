```rust
use std::f64;
use std::io::{self, Read};

const ONE_SIXTH: f64 = 1.0 / 6.0;

fn myabs(x: i32) -> i32 {
    if x >= 0 { x } else { -x }
}

fn solve(
    rest: usize,
    turn: i32,
    n: usize,
    sq: &[u8; 100],
    memo: &mut [[f64; 101]; 101],
) -> f64 {
    // In the original C, memo is initialized to DBL_MAX and used as "unset".
    if memo[rest][turn as usize] != f64::MAX {
        return memo[rest][turn as usize];
    }

    let ret = if rest == 0 {
        1.0
    } else if turn <= 0 {
        0.0
    } else if turn == 1 && rest <= 6 {
        ONE_SIXTH
    } else {
        let mut acc = 0.0;
        for i in 1..=6 {
            // C uses int and myabs; rest is non-negative in calls.
            let next_pos_i = myabs(rest as i32 - i);
            let next_pos = next_pos_i as usize;

            // Board index: sq[N - next_pos]
            let idx = n - next_pos;
            match sq[idx] {
                0 => {
                    acc += ONE_SIXTH * solve(next_pos, turn - 1, n, sq, memo);
                }
                b'L' => {
                    acc += ONE_SIXTH * solve(next_pos, turn - 2, n, sq, memo);
                }
                b'B' => {
                    acc += ONE_SIXTH * solve(n, turn - 1, n, sq, memo);
                }
                _ => {
                    // Unreachable given how sq is filled; keep C-like behavior (treat as normal).
                    acc += ONE_SIXTH * solve(next_pos, turn - 1, n, sq, memo);
                }
            }
        }
        acc
    };

    memo[rest][turn as usize] = ret;
    ret
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: usize = n_opt.unwrap().parse().unwrap();
        let t: i32 = it.next().unwrap().parse().unwrap();
        let lcnt: usize = it.next().unwrap().parse().unwrap();
        let bcnt: usize = it.next().unwrap().parse().unwrap();

        if n == 0 && t == 0 && lcnt == 0 && bcnt == 0 {
            break;
        }

        let mut sq = [0u8; 100];
        let mut memo = [[f64::MAX; 101]; 101];

        for _ in 0..lcnt {
            let lpos: usize = it.next().unwrap().parse().unwrap();
            sq[lpos] = b'L';
        }
        for _ in 0..bcnt {
            let bpos: usize = it.next().unwrap().parse().unwrap();
            sq[bpos] = b'B';
        }

        let ret = solve(n, t, n, &sq, &mut memo);
        // Match C's printf("%lf\n", ret);
        println!("{}", ret);
    }
}
```
