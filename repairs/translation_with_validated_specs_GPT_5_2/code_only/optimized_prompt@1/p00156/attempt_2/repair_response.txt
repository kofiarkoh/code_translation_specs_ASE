```rust
use std::io::{self, Read};

fn main() {
    // We must emulate scanf("%d %d") and scanf(" %c") precisely enough.
    // split_whitespace() is NOT sufficient for " %c" because the grid may contain
    // characters like '.' without whitespace separation; scanf reads single chars.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let bytes = input.as_bytes();
    let mut p: usize = 0;

    fn is_ws(b: u8) -> bool {
        matches!(b, b' ' | b'\n' | b'\r' | b'\t' | 0x0b | 0x0c)
    }

    fn read_i32(bytes: &[u8], p: &mut usize) -> Option<i32> {
        while *p < bytes.len() && is_ws(bytes[*p]) {
            *p += 1;
        }
        if *p >= bytes.len() {
            return None;
        }
        let mut sign = 1i32;
        if bytes[*p] == b'-' {
            sign = -1;
            *p += 1;
        }
        let mut any = false;
        let mut val: i32 = 0;
        while *p < bytes.len() {
            let b = bytes[*p];
            if b.is_ascii_digit() {
                any = true;
                // C int overflow is UB; inputs are expected to fit.
                val = val * 10 + (b - b'0') as i32;
                *p += 1;
            } else {
                break;
            }
        }
        if !any {
            None
        } else {
            Some(val * sign)
        }
    }

    fn read_char_skip_ws(bytes: &[u8], p: &mut usize) -> Option<u8> {
        while *p < bytes.len() && is_ws(bytes[*p]) {
            *p += 1;
        }
        if *p >= bytes.len() {
            None
        } else {
            let b = bytes[*p];
            *p += 1;
            Some(b)
        }
    }

    // Global arrays in C: int x[200000], y[200000];
    let mut x = vec![0i32; 200_000];
    let mut y = vec![0i32; 200_000];

    let dx = [0i32, 1, 0, -1];
    let dy = [1i32, 0, -1, 0];

    loop {
        let w = match read_i32(bytes, &mut p) {
            Some(v) => v,
            None => break,
        };
        let h = match read_i32(bytes, &mut p) {
            Some(v) => v,
            None => break,
        };
        if w == 0 && h == 0 {
            break;
        }

        // char m[101][101];
        let mut m = vec![vec![b'\0'; 101]; 101];
        // int c[101][101] = {0};
        let mut c = vec![vec![0i32; 101]; 101];

        // In the original C, if '&' is never found, t is uninitialized (UB).
        // We keep deterministic behavior by initializing t=0; typical testcases include '&'.
        let mut t: usize = 0;

        for i in 0..(h as usize) {
            for j in 0..(w as usize) {
                let ch = read_char_skip_ws(bytes, &mut p).unwrap_or(b'\0');
                m[i][j] = ch;
                if m[i][j] == b'&' {
                    m[i][j] = b'.';
                    y[0] = i as i32;
                    t = 0;
                    x[t] = j as i32; // x[t=0]=j
                    c[i][j] = 1;
                }
            }
        }

        // for(r=1; r-t; t++) { ... }
        let mut r: usize = 1;
        while r != t {
            let cx = x[t];
            let cy = y[t];
            let cxu = cx as usize;
            let cyu = cy as usize;

            for k in 0..4 {
                let nx = cx + dx[k];
                let ny = cy + dy[k];
                if nx < 0 || nx >= w || ny < 0 || ny >= h {
                    continue;
                }
                let nxu = nx as usize;
                let nyu = ny as usize;

                if m[cyu][cxu] == m[nyu][nxu] {
                    if c[cyu][cxu] < c[nyu][nxu] || c[nyu][nxu] == 0 {
                        c[nyu][nxu] = c[cyu][cxu];
                        x[r] = nx;
                        y[r] = ny;
                        r += 1;
                    }
                } else {
                    let cand = c[cyu][cxu] + 1;
                    if cand < c[nyu][nxu] || c[nyu][nxu] == 0 {
                        c[nyu][nxu] = cand;
                        x[r] = nx;
                        y[r] = ny;
                        r += 1;
                    }
                }
            }
            t += 1;
        }

        let mut minv: i32 = w;
        for i in 0..(w as usize) {
            let v1 = c[0][i] / 2;
            if v1 < minv {
                minv = v1;
            }
            let v2 = c[(h as usize) - 1][i] / 2;
            if v2 < minv {
                minv = v2;
            }
        }
        for i in 0..(h as usize) {
            let v1 = c[i][0] / 2;
            if v1 < minv {
                minv = v1;
            }
            let v2 = c[i][(w as usize) - 1] / 2;
            if v2 < minv {
                minv = v2;
            }
        }

        print!("{minv}\n");
    }
}
```
