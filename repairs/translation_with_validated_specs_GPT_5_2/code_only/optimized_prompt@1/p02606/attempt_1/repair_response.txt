```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    // Parse three i32 values (matches C's `int` on typical platforms; preserves truncation range)
    let mut it = input.split_whitespace();
    let l: i32 = it.next().unwrap().parse().unwrap();
    let r: i32 = it.next().unwrap().parse().unwrap();
    let d: i32 = it.next().unwrap().parse().unwrap();

    // Emulate C `int` arithmetic with potential overflow on (L - 1) by using wrapping_sub.
    // Division by zero will panic in Rust; in C it's undefined behavior. We keep it as-is.
    let res: i32 = r / d - l.wrapping_sub(1) / d;

    // Match exact output formatting: one integer followed by '\n'
    print!("{}\n", res);

    // `fflush(stdout)` equivalent: ensure output is flushed.
    use std::io::Write;
    io::stdout().flush().unwrap();
}
```
