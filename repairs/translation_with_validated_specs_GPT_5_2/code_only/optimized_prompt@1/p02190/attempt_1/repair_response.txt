```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // The C code assumes scanf succeeds; if input is missing, behavior is undefined.
    // Here we unwrap to mirror "must be present" behavior.
    let n: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;

    let mut a: Vec<i32> = Vec::with_capacity(n);
    for _ in 0..n {
        let v = it.next().unwrap().parse::<i32>().unwrap();
        a.push(v);
    }

    // qsort with comparator equivalent to integer ascending order.
    a.sort();

    // The C code reads a[0] unconditionally; if n == 0 it's undefined.
    // We mirror that by indexing directly (will panic in Rust if n == 0).
    let mut prev = a[0];
    let mut ret: i32 = 1;

    for i in 1..n {
        if a[i] != prev {
            ret += 1;
        }
        prev = a[i];
    }

    print!("{}\n", ret);
}
```
