```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin (scanf-style whitespace-separated parsing)
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    // C code uses fixed array p[100][2]; we mirror that capacity.
    // If n > 100, the C code would write out of bounds (UB). We clamp to 100
    // to avoid panic; this is a best-effort emulation without invoking UB.
    let m = n.min(100);

    let mut p = vec![[0i32; 2]; 100];
    for i in 0..m {
        let x: i32 = it.next().unwrap().parse().unwrap();
        let y: i32 = it.next().unwrap().parse().unwrap();
        p[i][0] = x;
        p[i][1] = y;
    }

    // If n > 100, consume remaining points to preserve input consumption behavior.
    for _ in m..n {
        let _x: i32 = it.next().unwrap().parse().unwrap();
        let _y: i32 = it.next().unwrap().parse().unwrap();
    }

    let mut menseki: f64 = 0.0;

    // The C expression does 32-bit signed int arithmetic, which can overflow (UB in C).
    // We emulate typical two's-complement wraparound using wrapping_* on i32.
    for i in 0..m {
        let j = (i + 1) % m;
        let dx = p[i][0].wrapping_sub(p[j][0]);
        let sy = p[i][1].wrapping_add(p[j][1]);
        let prod = dx.wrapping_mul(sy);
        menseki += prod as f64;
    }

    // Match printf("%.1lf\n", menseki/2.0);
    println!("{:.1}", menseki / 2.0);
}
```
