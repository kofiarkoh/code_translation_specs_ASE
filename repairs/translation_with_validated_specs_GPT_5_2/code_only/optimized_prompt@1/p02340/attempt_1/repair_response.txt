```rust
use std::io::{self, Read};

const NUM_MAX: usize = 1000;
const K_MAX: usize = 1000;
const DIVISOR: u64 = 1_000_000_007;

fn init() -> [[u64; NUM_MAX + 1]; K_MAX + 1] {
    let mut pdp = [[0u64; NUM_MAX + 1]; K_MAX + 1];
    pdp[0][0] = 1;

    for i in 1..=K_MAX {
        for j in 0..=NUM_MAX {
            pdp[i][j] = pdp[i - 1][j];
            if j >= i {
                // Matches C: (pdp[i][j] + pdp[i][j-i]) % DIVISOR
                pdp[i][j] = (pdp[i][j] + pdp[i][j - i]) % DIVISOR;
            }
        }
    }
    pdp
}

fn main() {
    // C reads a single line with fgets(BUF_SIZE=20) and then sscanf("%d %d", ...).
    // We emulate the "single line" behavior by only considering input up to the first '\n'.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).ok();
    let line = input.splitn(2, '\n').next().unwrap_or("");

    // sscanf with "%d %d" leaves values unchanged if parsing fails; in C, n/k are uninitialized.
    // In Rust we must initialize; choose 0 defaults to keep behavior defined.
    let mut it = line.split_whitespace();
    let n: usize = it.next().and_then(|s| s.parse::<i32>().ok()).unwrap_or(0).max(0) as usize;
    let k: usize = it.next().and_then(|s| s.parse::<i32>().ok()).unwrap_or(0).max(0) as usize;

    let pdp = init();

    // C would have undefined behavior if k>1000 or n>1000 (out-of-bounds).
    // We clamp to avoid panic while keeping a reasonable approximation.
    let kk = k.min(K_MAX);
    let nn = n.min(NUM_MAX);

    // C prints (int)pdp[k][n] with "%d\n". Values are < 1e9+7, so i32 is safe.
    let out = pdp[kk][nn] as i32;
    print!("{out}\n");
}
```
