```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let mut m: i32 = it.next().unwrap().parse().unwrap();

    // c[610][610] initialized to 0
    let mut c = vec![vec![0i32; 610]; 610];
    // e[610]
    let mut e = vec![-1i32; 610];
    // d[1010] initialized to 0.0 then set as in C
    let mut d = vec![0f64; 1010];

    // scanf("%d %d",&n,&m);
    // for(i=0;i<=n;i++)d[i]=1e9;
    for i in 0..=n {
        d[i as usize] = 1e9;
    }
    d[n as usize] = 0.0;

    // for(i=0;i<=n;i++)e[i]=-1;
    for i in 0..=n {
        e[i as usize] = -1;
    }

    // while(m--){ scanf("%d %d",&i,&j); c[i][j]=1; }
    while m > 0 {
        m -= 1;
        let i: i32 = it.next().unwrap().parse().unwrap();
        let j: i32 = it.next().unwrap().parse().unwrap();
        c[i as usize][j as usize] = 1;
    }

    // for(i=n-1;i;i--){
    //   s=b=a=0;
    //   for(j=i;j<=n;j++){
    //     if(c[i][j]){
    //       s+=d[j]+1;
    //       if(b==0||d[b]<d[j])b=j;
    //       a++;
    //     }
    //   }
    //   d[i]=s/a;
    //   if(a>1)e[i]=b;
    // }
    for i in (1..=n - 1).rev() {
        let mut s: f64 = 0.0;
        let mut b: i32 = 0;
        let mut a: i32 = 0;
        for j in i..=n {
            if c[i as usize][j as usize] != 0 {
                s += d[j as usize] + 1.0;
                if b == 0 || d[b as usize] < d[j as usize] {
                    b = j;
                }
                a += 1;
            }
        }
        // In the original C, if a==0 this is division by zero (undefined behavior).
        // We preserve the observable behavior as closely as possible by using IEEE-754:
        // 0.0/0.0 => NaN, x/0.0 => +/-inf.
        d[i as usize] = s / (a as f64);
        if a > 1 {
            e[i as usize] = b;
        }
    }

    let mut minv: f64 = d[1];

    // for(k=1;k<=n;k++){
    //   if(e[k]==-1)continue;
    //   for(j=0;j<n;j++)d[j]=1e9;
    //   d[n]=0;
    //   for(i=n-1;i;i--){
    //     s=a=0;
    //     for(j=i;j<=n;j++){
    //       if(c[i][j]==0||(i==k&&j==e[k]))continue;
    //       s+=d[j]+1;
    //       a++;
    //     }
    //     d[i]=s/a;
    //   }
    //   if(min>d[1])min=d[1];
    // }
    for k in 1..=n {
        if e[k as usize] == -1 {
            continue;
        }
        for j in 0..n {
            d[j as usize] = 1e9;
        }
        d[n as usize] = 0.0;

        for i in (1..=n - 1).rev() {
            let mut s: f64 = 0.0;
            let mut a: i32 = 0;
            for j in i..=n {
                if c[i as usize][j as usize] == 0
                    || (i == k && j == e[k as usize])
                {
                    continue;
                }
                s += d[j as usize] + 1.0;
                a += 1;
            }
            d[i as usize] = s / (a as f64);
        }

        if minv > d[1] {
            minv = d[1];
        }
    }

    // printf("%.10lf\n",min);
    println!("{:.10}", minv);
}
```
