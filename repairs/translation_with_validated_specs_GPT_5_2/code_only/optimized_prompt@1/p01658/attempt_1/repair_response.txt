```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, parse as i32 like scanf("%d")
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let m: i32 = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let n: i32 = it.next().unwrap().parse().unwrap();

    // C globals: int N,M,A[106],i,j,l,r,a=0;
    // Use i32 to match typical C int behavior; indices converted to usize when used.
    let mut a: i32 = 0;
    let mut arr = vec![0i32; 106];

    for i in 0..m {
        for j in 0..n {
            let v: i32 = it.next().unwrap().parse().unwrap();
            let idx = j as usize;
            arr[idx] = v;
            if i != 0 {
                arr[idx] ^= 1;
            }
        }

        let mut l: i32 = 0;
        let mut r: i32 = n - 1;

        while l <= r {
            let li = l as usize;
            let ri = r as usize;

            if arr[li] == 1 {
                a += 1;
                l += 1; // corresponds to A[++l] in C (pre-increment)
                // In the original C, this can go out of bounds when l was r==N-1.
                // The provided code assumes inputs avoid that; we mirror behavior for valid cases.
                arr[l as usize] ^= 1;
            } else if arr[ri] == 1 {
                a += 1;
                r -= 1; // corresponds to A[--r] in C (pre-decrement)
                arr[r as usize] ^= 1;
            } else {
                l += 1; // corresponds to A[++l]
                arr[l as usize] ^= 1;
            }
        }
    }

    print!("{}\n", a);
}
```
