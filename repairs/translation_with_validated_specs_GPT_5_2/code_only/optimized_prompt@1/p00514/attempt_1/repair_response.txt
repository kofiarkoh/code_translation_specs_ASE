```rust
use std::io::{self, Read};

const TABAICHO_SIZE: usize = 8667;
const TABAICHO_BLOCK: i64 = 1_000_000_000;

fn tabaicho_set(a: &mut [i32; TABAICHO_SIZE], num: i32) {
    a[0] = num;
    for i in 1..TABAICHO_SIZE {
        a[i] = 0;
    }
}

fn tabaicho_mul(a: &mut [i32; TABAICHO_SIZE], num: i32) {
    let mut carry: i64 = 0;
    let num64 = num as i64;
    for i in 0..TABAICHO_SIZE {
        let mut calctemp: i64 = (a[i] as i64) * num64;
        calctemp += carry;
        a[i] = (calctemp % TABAICHO_BLOCK) as i32;
        carry = calctemp / TABAICHO_BLOCK;
    }
}

fn tabaicho_div(a: &mut [i32; TABAICHO_SIZE], num: i32) {
    let mut remainder: i64 = 0;
    let num64 = num as i64;
    for i in (0..TABAICHO_SIZE).rev() {
        remainder = remainder * TABAICHO_BLOCK + (a[i] as i64);
        a[i] = (remainder / num64) as i32;
        remainder %= num64;
    }
}

fn tabaicho_print(a: &[i32; TABAICHO_SIZE]) {
    // Match the C behavior: it assumes there is some i with a[i] > 0.
    // In this program, this is true for the branch that calls print.
    let mut i: isize = (TABAICHO_SIZE as isize) - 1;
    while i >= 0 {
        if a[i as usize] > 0 {
            break;
        }
        i -= 1;
    }

    // If somehow all digits are zero, C would have UB (out-of-bounds access).
    // We avoid panicking but keep output reasonable; this path should not occur.
    if i < 0 {
        println!("0");
        return;
    }

    print!("{}", a[i as usize]);
    i -= 1;
    while i >= 0 {
        print!("{:09}", a[i as usize]);
        i -= 1;
    }
    println!();
}

fn main() {
    // Read all input, parse as C scanf("%d%d%d") would (whitespace-separated).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };
    let m: i32 = it.next().unwrap().parse().unwrap();
    let mut r: i32 = it.next().unwrap().parse().unwrap();

    // r -= n*m; in C int arithmetic (potential overflow is UB in C).
    // We use wrapping to avoid Rust panics and to approximate typical 2's complement behavior.
    r = r.wrapping_sub(n.wrapping_mul(m));

    if r < 0 {
        println!("0");
    } else if r == 0 {
        println!("1");
    } else {
        // Compute (n+r-1)C(n-1) using big integer in base 1e9 blocks.
        let mut tabaicho_main = [0i32; TABAICHO_SIZE];
        tabaicho_set(&mut tabaicho_main, 1);

        let mut n2: i32 = n - 1;
        // if ((n+r-1)-(n-1) < n-1) n2 = (n+r-1)-(n-1);
        // Note: (n+r-1)-(n-1) == r, but keep structure.
        let left = (n.wrapping_add(r).wrapping_sub(1)).wrapping_sub(n.wrapping_sub(1));
        if left < n.wrapping_sub(1) {
            n2 = left;
        }

        let mut j: i32 = n.wrapping_add(r).wrapping_sub(1);
        let mut i: i32 = 0;
        while i < n2 {
            tabaicho_mul(&mut tabaicho_main, j);
            i += 1;
            j = j.wrapping_sub(1);
        }

        let mut d: i32 = 1;
        while d <= n2 {
            tabaicho_div(&mut tabaicho_main, d);
            d += 1;
        }

        tabaicho_print(&tabaicho_main);
    }
}
```
