```rust
use std::cmp::Ordering;
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Node {
    w: i32,
    s: i32,
    v: i32,
}

fn main() {
    // Read entire stdin (matches scanf-style whitespace handling).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let mut p: Vec<Node> = Vec::with_capacity(n);
    for _ in 0..n {
        let w: i32 = it.next().unwrap().parse().unwrap();
        let s: i32 = it.next().unwrap().parse().unwrap();
        let v: i32 = it.next().unwrap().parse().unwrap();
        p.push(Node { w, s, v });
    }

    // C comparator: (p->s+p->w)-(q->s+q->w) with int arithmetic.
    // In Rust, we emulate potential 32-bit signed overflow by using wrapping ops.
    p.sort_by(|a, b| {
        let sa = a.s.wrapping_add(a.w);
        let sb = b.s.wrapping_add(b.w);
        let diff = sa.wrapping_sub(sb);
        if diff < 0 {
            Ordering::Less
        } else if diff > 0 {
            Ordering::Greater
        } else {
            Ordering::Equal
        }
    });

    const M: usize = 20000;
    let mut dp: Vec<i64> = vec![0; M + 1];

    for item in p.iter() {
        // The C code assumes s >= 0 and k=j+w within [0..M].
        // We preserve behavior for valid inputs; for invalid ones, we avoid panics.
        let s = item.s;
        if s < 0 {
            continue;
        }
        let mut j = s as i32;
        while j >= 0 {
            let k = j.wrapping_add(item.w);
            if k >= 0 && (k as usize) <= M {
                let ju = j as usize;
                let ku = k as usize;
                let cand = dp[ju].wrapping_add(item.v as i64);
                if cand > dp[ku] {
                    dp[ku] = cand;
                }
            }
            j -= 1;
        }
    }

    let mut ans: i64 = 0;
    for &x in dp.iter() {
        if x > ans {
            ans = x;
        }
    }

    print!("{}\n", ans);
}
```
