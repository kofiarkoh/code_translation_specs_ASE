```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, tokenized like scanf("%d"/"%s")
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let c: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let n: usize = it.next().unwrap().parse().unwrap();

    // A: grid of '0'/'1' bytes
    let mut a: Vec<Vec<u8>> = vec![vec![0u8; n]; n];
    for i in 0..n {
        let s = it.next().unwrap().as_bytes();
        for j in 0..n {
            a[i][j] = s[j];
        }
    }

    // t: check results (0/1) stored as i32 to mirror C char arithmetic in total updates
    let mut t: Vec<Vec<i32>> = vec![vec![0i32; n]; n];
    let mut total: i32 = 0;

    // check(x,y): whether a[x][y] equals its vertical and horizontal mirror positions
    let mut check = |x: usize, y: usize, a_ref: &Vec<Vec<u8>>| -> i32 {
        if a_ref[x][y] == a_ref[n - 1 - x][y] && a_ref[x][y] == a_ref[x][n - 1 - y] {
            1
        } else {
            0
        }
    };

    // initialize t and total
    for i in 0..n {
        for j in 0..n {
            let v = check(i, j, &a);
            t[i][j] = v;
            total += v;
        }
    }

    let solve = |total_val: i32| -> i32 {
        if total_val == (n * n) as i32 { 1 } else { 0 }
    };

    let mut ans: i32 = solve(total);

    // update(x,y): recompute t[x][y] and adjust total
    let mut update = |x: usize, y: usize, a_ref: &Vec<Vec<u8>>, t_ref: &mut Vec<Vec<i32>>, total_ref: &mut i32| {
        *total_ref -= t_ref[x][y];
        let v = check(x, y, a_ref);
        t_ref[x][y] = v;
        *total_ref += v;
    };

    // change(x,y): flip a[x][y] and update affected positions
    let mut change = |x: usize, y: usize, a_ref: &mut Vec<Vec<u8>>, t_ref: &mut Vec<Vec<i32>>, total_ref: &mut i32| {
        a_ref[x][y] = if a_ref[x][y] == b'0' { b'1' } else { b'0' };
        update(x, y, a_ref, t_ref, total_ref);
        update(n - 1 - x, y, a_ref, t_ref, total_ref);
        update(x, n - 1 - y, a_ref, t_ref, total_ref);
    };

    // process C-1 diffs
    for _ in 0..(c.saturating_sub(1)) {
        let k: usize = it.next().unwrap().parse().unwrap();
        for _ in 0..k {
            let mut x: usize = it.next().unwrap().parse().unwrap();
            let mut y: usize = it.next().unwrap().parse().unwrap();
            // convert 1-based to 0-based
            x -= 1;
            y -= 1;
            change(x, y, &mut a, &mut t, &mut total);
        }
        ans += solve(total);
    }

    print!("{ans}\n");
}
```
