```rust
use std::io::{self, Read};

const MAX: usize = 13;

fn main() {
    // Read entire stdin, then parse as whitespace-separated integers (like scanf("%d")).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut numline = [0i32; MAX];
    let mut tmpline = [0i32; MAX];
    let mut num = [0i32; 101];

    let mut out = String::new();

    while let Some(tok) = it.next() {
        let n: usize = match tok.parse::<i32>() {
            Ok(v) => v as usize,
            Err(_) => break,
        };
        if n == 0 {
            break;
        }

        for i in 0..n {
            if let Some(t) = it.next() {
                numline[i] = t.parse::<i32>().unwrap();
            } else {
                // If input is truncated, mimic scanf failure by stopping.
                print!("{out}");
                return;
            }
        }

        let mut flag: i32 = 1;
        let mut cnt: i32 = 0;

        while flag != 0 {
            // reset counts
            for v in num.iter_mut() {
                *v = 0;
            }

            // count occurrences: num[numline[i]]++
            // C code assumes indices are within 0..=100; we preserve behavior by indexing directly.
            for i in 0..n {
                let idx = numline[i] as usize;
                num[idx] += 1;
            }

            // build tmpline[i] = num[numline[i]]
            for i in 0..n {
                let idx = numline[i] as usize;
                tmpline[i] = num[idx];
            }

            cnt += 1;
            flag = 0;

            // count positions where tmpline[i] == numline[i]
            for i in 0..n {
                if tmpline[i] == numline[i] {
                    flag += 1;
                }
            }

            if flag == n as i32 {
                flag = 0;
            } else {
                // copy tmpline -> numline
                for i in 0..n {
                    numline[i] = tmpline[i];
                }
                flag = 1;
            }
        }

        out.push_str(&(cnt - 1).to_string());
        out.push('\n');
        for i in 0..n {
            if i + 1 == n {
                out.push_str(&tmpline[i].to_string());
                out.push('\n');
            } else {
                out.push_str(&tmpline[i].to_string());
                out.push(' ');
            }
        }
    }

    print!("{out}");
}
```
