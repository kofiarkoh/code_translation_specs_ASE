```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, like scanf would consume from stdin.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // C uses `long` for n, p, ans. On typical platforms this is 64-bit, but it can be 32-bit.
    // We use i64 and emulate C's signed overflow behavior with wrapping operations.
    let n: i64 = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return, // If scanf fails to read, C would leave globals as-is; here we just exit.
    };
    let mut p: i64 = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };

    let mut ans: i64 = 1;

    // for (long i=2; i*i<=p; i++)
    // Use wrapping_mul to mimic potential signed overflow in i*i (C UB), but keep loop structure.
    let mut i: i64 = 2;
    while i.wrapping_mul(i) <= p {
        let mut cnt: i64 = 0;
        // while (p%i<1) cnt++, p/=i;
        // In C, p%i<1 is true when p%i == 0 (since remainder has same sign as p).
        // This loop can be non-terminating for some inputs (e.g., p==0), matching C behavior.
        while p % i < 1 {
            cnt = cnt.wrapping_add(1);
            p /= i;
        }
        if cnt >= n {
            // for (int j=0; j<cnt/n; j++) ans*=i;
            // Note: if n==0, C would have UB (division by zero). We mirror by panicking.
            let times = cnt / n;
            let mut j: i64 = 0;
            while j < times {
                ans = ans.wrapping_mul(i);
                j += 1;
            }
        }
        i = i.wrapping_add(1);
    }

    // if (p>1 && n==1) ans*=p;
    if p > 1 && n == 1 {
        ans = ans.wrapping_mul(p);
    }

    // printf("%ld\n", ans);
    print!("{ans}\n");
}
```
