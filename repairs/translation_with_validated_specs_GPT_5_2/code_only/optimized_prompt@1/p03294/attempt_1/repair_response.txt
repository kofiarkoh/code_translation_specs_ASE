```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, parse as C-like whitespace-separated integers.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // If input is missing, C's scanf would leave N uninitialized (UB).
    // Here we default to 0 to keep the program defined in Rust.
    let n: i32 = it.next().and_then(|s| s.parse().ok()).unwrap_or(0);

    // C uses a VLA `int a[N];` but only to store values; we can avoid storing.
    // Preserve C `int` overflow behavior by using wrapping arithmetic on i32.
    let mut ans: i32 = 0;

    for _ in 0..(n.max(0) as usize) {
        let ai: i32 = it.next().and_then(|s| s.parse().ok()).unwrap_or(0);
        ans = ans.wrapping_add(ai.wrapping_sub(1));
    }

    // Match C's `printf("%d", ans);` (no trailing newline).
    print!("{}", ans);
}
```
