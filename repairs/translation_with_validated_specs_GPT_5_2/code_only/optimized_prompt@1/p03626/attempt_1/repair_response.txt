```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin and split by whitespace (matches scanf("%d"...) and scanf("%s%s"...)).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let s1 = it.next().unwrap_or("").as_bytes().to_vec();
    let s2 = it.next().unwrap_or("").as_bytes().to_vec();

    // C code assumes N >= 1 and strings have length at least N.
    // We preserve that assumption; if violated, behavior is unspecified in C anyway.
    let mut ans: i64 = 1;
    let mut t: i32;

    if s1[0] == s2[0] {
        ans = 3;
        t = 1;
    } else {
        ans = 6;
        t = 0;
    }

    let m: i64 = 1_000_000_007;

    // Note: The original C loop accesses S1[i+1] even when i == N-1, which is out-of-bounds.
    // That is undefined behavior in C. To emulate the *intended* behavior (common in this known
    // pattern), we only perform the S1[i] == S1[i+1] check when i + 1 < N.
    for i in 1..n {
        if s1[i] == s2[i] {
            if t == 1 {
                ans = (ans * 2) % m;
            }
            t = 1;
        }
        if i + 1 < n && s1[i] == s1[i + 1] {
            if t == 1 {
                ans = (ans * 2) % m;
            } else {
                ans = (ans * 3) % m;
            }
            t = 0;
        }
    }

    print!("{}\n", ans);
}
```
