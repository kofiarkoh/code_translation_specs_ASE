```rust
use std::io::{self, Read};

const M: i64 = 998_244_353;
const N: usize = 90310;

fn main() {
    // Read entire stdin, parse as C scanf("%d") would (whitespace-separated ints).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // C: *d = *e = scanf("%d");
    // scanf returns number of items successfully read (0 or 1 here, or EOF=-1).
    // With split_whitespace, we can only distinguish "has int" vs "no token".
    // If no first int exists, treat as EOF => -1.
    let first = match it.next() {
        Some(tok) => {
            // If token isn't a valid int, C scanf would return 0; here we treat as 0.
            tok.parse::<i32>().map(|_| 1i64).unwrap_or(0i64)
        }
        None => -1i64,
    };

    let mut d = vec![0i64; N];
    let mut e = vec![0i64; N];
    d[0] = first;
    e[0] = first;

    let mut s: i32 = 0;
    let mut r: i64 = 1;

    // C: for (...; ~scanf("%d",&a);) { ... }
    // Loop continues while scanf result != -1 (EOF). If parse fails, C would return 0,
    // and ~0 != 0, so it would still loop; but it would not consume input.
    // With tokenization, we cannot emulate "stuck" behavior; we treat non-ints as absent.
    while let Some(tok) = it.next() {
        let a: i32 = match tok.parse() {
            Ok(v) => v,
            Err(_) => break,
        };

        // for(t=s; ~t; --t) ...
        // Here t is a signed long in C; loop runs for t = s down to 0 inclusive.
        let mut t: i32 = s;
        while t >= 0 {
            let idx = (t + a) as usize;
            let ti = t as usize;

            // d[t+a] += d[t]*2%M;
            d[idx] = d[idx].wrapping_add((d[ti].wrapping_mul(2) % M));
            // e[t+a] += e[t]%M;
            e[idx] = e[idx].wrapping_add(e[ti] % M);

            t -= 1;
        }

        // s += a; r = r*3%M;
        s = s.wrapping_add(a);
        r = (r.wrapping_mul(3)) % M;
    }

    // for(t=s/2; ~t;) r=(r-d[t--]*3)%M;
    let mut t: i32 = s / 2;
    while t >= 0 {
        let ti = t as usize;
        r = (r.wrapping_sub(d[ti].wrapping_mul(3))) % M;
        t -= 1;
    }

    // printf("%d",(r-~s%2*e[s/2]*3+M)%M);
    // In C, ~s%2 is either 1 (if s even) or 0 (if s odd), because:
    //   if s even: ~s is odd => (~s)%2 == 1
    //   if s odd:  ~s is even => (~s)%2 == 0
    let even_factor: i64 = if (s & 1) == 0 { 1 } else { 0 };
    let es = e[(s / 2) as usize];

    let ans = (r
        .wrapping_sub(even_factor.wrapping_mul(es).wrapping_mul(3))
        .wrapping_add(M))
        % M;

    // C prints with "%d" (int). We'll print as i64 decimal without newline, matching.
    print!("{}", ans);
}
```
