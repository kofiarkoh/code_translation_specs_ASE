```rust
use std::io::{self, Read};

fn counter(z: i32) -> i32 {
    if z == 0 {
        0
    } else {
        // In C, right shift of signed negative is implementation-defined.
        // Here z is always non-negative in this program (subset mask), so Rust's >> matches.
        counter(z >> 1) + (z & 1)
    }
}

fn main() {
    // Read entire stdin, parse as integers like scanf("%d")
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut a = [0i32; 20];
    let mut x = [[0i32; 20]; 20];
    let mut y = [[0i32; 20]; 20];

    for i in 0..(n as usize) {
        a[i] = it.next().unwrap().parse().unwrap();
        for j in 0..(a[i] as usize) {
            x[i][j] = it.next().unwrap().parse().unwrap();
            y[i][j] = it.next().unwrap().parse().unwrap();
        }
    }

    let mut max_v: i32 = 0;
    let mut flag: i32 = 1;

    // C uses (1<<n) with int; shifting by >= bitwidth is undefined in C.
    // We emulate the intended behavior for typical constraints (n <= 20 here).
    let limit: i32 = 1i32 << n;

    for i in 0..limit {
        flag = 1;
        for j in 0..(n as usize) {
            if (i & (1i32 << (j as i32))) == 0 {
                continue;
            }
            for k in 0..(a[j] as usize) {
                // Preserve C operator precedence:
                // (i>>(x[j][k]-1)&1)^y[j][k]
                let bit = (i >> (x[j][k] - 1)) & 1;
                let cond = (bit ^ y[j][k]) != 0;
                if cond {
                    flag = 0;
                    break;
                }
            }
        }
        if flag == 1 {
            let count = counter(i);
            if max_v < count {
                max_v = count;
            }
        }
    }

    print!("{}", max_v);
}
```
