use std::io::{self, Read};

fn main() {
    // Read entire stdin, parse as i64 (matches C's long long on typical platforms).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let k: i64 = input.split_whitespace().next().unwrap().parse().unwrap();

    if k >= 50 {
        let n: usize = 50;
        let rem: i64 = k / 50;
        let x: i64 = 49 + rem;
        let t: usize = (k % 50) as usize;

        let mut a = vec![x; n];

        for i in 0..t {
            a[i] = a[i].wrapping_add(50);
            for j in 0..n {
                if j != i {
                    a[j] = a[j].wrapping_sub(1);
                }
            }
        }

        // Match C output formatting exactly: "50\n" then numbers each followed by a space, no final newline.
        print!("50\n");
        for i in 0..n {
            print!("{} ", a[i]);
        }
    } else if k >= 2 {
        let n: i64 = k;
        print!("{}\n", n);
        // C prints each number followed by a space, no final newline.
        for _ in 0..n {
            print!("{} ", n);
        }
    } else if k == 0 {
        // Exact literal output including trailing newline.
        print!("4\n3 3 3 3\n");
    } else {
        // Exact literal output including trailing newline.
        print!("3\n1 0 3\n");
    }
}
