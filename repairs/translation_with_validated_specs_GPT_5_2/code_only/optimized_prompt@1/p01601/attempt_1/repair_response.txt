use std::io::{self, Read};

fn rev(mut n: i32) -> i32 {
    let mut out: i32 = 0;
    while n > 0 {
        // In C, signed overflow is undefined; we use wrapping ops to emulate
        // typical two's-complement wraparound behavior if it occurs.
        out = out.wrapping_mul(10).wrapping_add(n % 10);
        n /= 10;
    }
    out
}

fn main() {
    // Read entire stdin, parse first integer like scanf("%d", &n)
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: i32 = match it.next() {
        Some(s) => s.parse::<i32>().unwrap_or(0),
        None => return, // scanf would fail; no output
    };

    let mut i: i32 = 0;

    loop {
        // if (n - i >= 0) { if (rev(n - i) == n - i) ... }
        // Use wrapping arithmetic to avoid Rust panics and better match C behavior.
        let ni = n.wrapping_sub(i);
        if ni >= 0 {
            if rev(ni) == ni {
                print!("{}\n", ni);
                break;
            }
        }

        let pi = n.wrapping_add(i);
        if rev(pi) == pi {
            print!("{}\n", pi);
            break;
        }

        i = i.wrapping_add(1);
    }
}
