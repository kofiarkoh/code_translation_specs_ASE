```rust
use std::io::{self, Read};

const HASH: usize = 100_003;

#[derive(Clone, Copy)]
struct Node {
    next: Option<usize>,
    v: i64,
    l: i64,
    r: i64,
}

fn gcd(mut a: i64, mut b: i64) -> i64 {
    // Mirrors the C logic (including swap via XOR, but done safely here).
    if a > b {
        std::mem::swap(&mut a, &mut b);
    }
    if a == 0 {
        b
    } else {
        gcd(b % a, a)
    }
}

fn main() {
    // Read all input, parse as i64 (C long long).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: i64 = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };
    let mut x: i64 = it.next().unwrap().parse().unwrap();
    let mut d: i64 = it.next().unwrap().parse().unwrap();

    if x == 0 || d == 0 {
        if x == 0 && d == 0 {
            print!("1\n");
        } else if d == 0 {
            print!("{}\n", n + 1);
        } else {
            // n*(n-1)/2 + 1
            print!("{}\n", n * (n - 1) / 2 + 1);
        }
        return;
    }

    // Use i64::abs; note that abs(i64::MIN) would overflow in Rust, but the C code
    // uses llabs which is also undefined/implementation-defined for LLONG_MIN.
    // We keep behavior close by using wrapping_abs via i128 conversion.
    let ax = (x as i128).abs() as i64;
    let ad = (d as i128).abs() as i64;

    let g = gcd(ax, ad);
    x /= g;
    d /= g;

    let y: i64;
    if d < 0 {
        y = x;
        // X = X + D*(N-1) with original (negative) D
        x = x.wrapping_add(d.wrapping_mul(n.wrapping_sub(1)));
        d = d.wrapping_mul(-1);
    } else {
        y = x.wrapping_add(d.wrapping_mul(n.wrapping_sub(1)));
    }

    // Hash table of linked lists by indices into `nodes`.
    let mut hash: Vec<Option<usize>> = vec![None; HASH];
    let mut nodes: Vec<Node> = vec![
        Node {
            next: None,
            v: 0,
            l: 0,
            r: d,
        };
        (n as usize) + 1
    ];

    hash[0] = Some(0);

    for i in 1..=n {
        // L = (X*2 + D*(i-1))*i/2
        // R = (Y*2 - D*(i-1))*i/2 + D
        // Use wrapping ops to emulate C signed overflow behavior as closely as possible.
        let ii = i;
        let im1 = i.wrapping_sub(1);

        let l_num = x
            .wrapping_mul(2)
            .wrapping_add(d.wrapping_mul(im1))
            .wrapping_mul(ii);
        let l = l_num / 2;

        let r_num = y
            .wrapping_mul(2)
            .wrapping_sub(d.wrapping_mul(im1))
            .wrapping_mul(ii);
        let r = r_num / 2 + d;

        // V = (L % D + D) % D
        let mut v = l % d;
        v = (v + d) % d;

        let h = (v.rem_euclid(HASH as i64)) as usize;

        // Find existing interval with same v that overlaps [L, R].
        let mut cur = hash[h];
        let mut found: Option<usize> = None;
        while let Some(idx) = cur {
            let p = nodes[idx];
            if p.v == v && p.l <= r && p.r >= l {
                found = Some(idx);
                break;
            }
            cur = p.next;
        }

        if let Some(idx) = found {
            // Merge intervals: p->l = min(p->l, L), p->r = max(p->r, R)
            let pl = nodes[idx].l;
            let pr = nodes[idx].r;
            nodes[idx].l = if pl < l { pl } else { l };
            nodes[idx].r = if pr > r { pr } else { r };
        } else {
            // Insert new node at head.
            let next = hash[h];
            nodes[i as usize] = Node {
                next,
                v,
                l,
                r,
            };
            hash[h] = Some(i as usize);
        }
    }

    let mut ans: i64 = 0;
    for h in 0..HASH {
        let mut cur = hash[h];
        while let Some(idx) = cur {
            let p = nodes[idx];
            ans = ans.wrapping_add((p.r - p.l) / d);
            cur = p.next;
        }
    }

    print!("{}\n", ans);
}
```
