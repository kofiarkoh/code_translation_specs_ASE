```rust
use std::io::{self, Read};

fn main() {
    // Read all input
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let n_usize = n as usize;

    // Arrays sized as in C (globals). Use i32 to match C int.
    let mut a = vec![0i32; 400_005];
    let mut b = vec![0i32; 400_005];
    let mut h = vec![0usize; 400_005]; // heap stores indices into a/b/c; use usize for indexing
    let mut l: usize = 0;

    // Read edges; preserve the odd i*=2; ...; i/=2 pattern exactly.
    // In C, loop: for (i=0; i<n-1; i++) { i*=2; ...; i/=2; }
    // This effectively uses indices 0,2,4,...,2n-4.
    let mut i: i32 = 0;
    while i < n - 1 {
        i *= 2;
        let idx = i as usize;

        let mut ai: i32 = it.next().unwrap().parse().unwrap();
        let mut bi: i32 = it.next().unwrap().parse().unwrap();
        ai -= 1;
        bi -= 1;

        a[idx] = ai;
        b[idx] = bi;
        a[idx + 1] = b[idx];
        b[idx + 1] = a[idx];

        i /= 2;
        i += 1;
    }

    // Heap helpers capturing a/h/l by closure.
    let mut comp_h = |x: usize, y: usize, a: &Vec<i32>, h: &Vec<usize>| -> i32 {
        if a[h[x]] > a[h[y]] {
            1
        } else {
            -1
        }
    };
    let mut swap_h = |x: usize, y: usize, h: &mut Vec<usize>| {
        let f = h[x];
        h[x] = h[y];
        h[y] = f;
    };

    let mut push = |ne: usize,
                    a: &Vec<i32>,
                    h: &mut Vec<usize>,
                    l: &mut usize,
                    comp_h: &mut dyn FnMut(usize, usize, &Vec<i32>, &Vec<usize>) -> i32,
                    swap_h: &mut dyn FnMut(usize, usize, &mut Vec<usize>)| {
        h[*l] = ne;
        let mut p = *l;
        *l += 1;
        while p > 0 {
            let parent = (p - 1) / 2;
            if comp_h(parent, p, a, h) > 0 {
                swap_h(parent, p, h);
            }
            p = parent;
        }
    };

    let mut pop = |a: &Vec<i32>,
                   h: &mut Vec<usize>,
                   l: &mut usize,
                   comp_h: &mut dyn FnMut(usize, usize, &Vec<i32>, &Vec<usize>) -> i32,
                   swap_h: &mut dyn FnMut(usize, usize, &mut Vec<usize>)| -> usize {
        *l -= 1;
        swap_h(0, *l, h);
        let mut p: usize = 0;
        loop {
            if 2 * p + 2 < *l {
                if comp_h(2 * p + 1, 2 * p + 2, a, h) > 0 {
                    if comp_h(p, 2 * p + 2, a, h) > 0 {
                        swap_h(p, 2 * p + 2, h);
                    }
                    p = 2 * p + 2;
                } else {
                    if comp_h(p, 2 * p + 1, a, h) > 0 {
                        swap_h(p, 2 * p + 1, h);
                    }
                    p = 2 * p + 1;
                }
            } else if 2 * p + 1 < *l {
                if comp_h(p, 2 * p + 1, a, h) > 0 {
                    swap_h(p, 2 * p + 1, h);
                }
                p = 2 * p + 1;
            } else {
                break;
            }
        }
        h[*l]
    };

    // Build heap with indices [0, 2n-3] (count = 2n-2)
    l = 0;
    let m = (2 * n - 2) as usize;
    for idx in 0..m {
        push(
            idx,
            &a,
            &mut h,
            &mut l,
            &mut comp_h,
            &mut swap_h,
        );
    }

    // Pop into c
    let mut c = vec![0usize; 400_005];
    for idx in 0..m {
        c[idx] = pop(&a, &mut h, &mut l, &mut comp_h, &mut swap_h);
    }

    // Sentinel
    c[m] = m;
    a[m] = -1;

    // BFS-like traversal using sorted-by-a edge list in c
    let mut depth = vec![-1i32; 200_005];
    let mut q = vec![0i32; 200_005];

    depth[0] = 1;
    q[0] = 0;
    l = 1;

    while l > 0 {
        l -= 1;
        let v = q[l] as i32;

        let mut min_i: i32 = -1;
        let mut max_i: i32 = (2 * n - 2) as i32;

        while max_i - min_i > 1 {
            let mid = (max_i + min_i) / 2;
            let cmid = c[mid as usize];
            if a[cmid] < v {
                min_i = mid;
            } else {
                max_i = mid;
            }
        }

        // In C this assumes there is at least one entry with a == v; for a tree and both directions, true.
        while a[c[max_i as usize]] == v {
            let eidx = c[max_i as usize];
            let to = b[eidx] as usize;
            if depth[to] < 0 {
                depth[to] = depth[v as usize] + 1;
                q[l] = b[eidx];
                l += 1;
            }
            max_i += 1;
        }
    }

    // Find farthest node s
    let mut s: usize = 0;
    for v in 0..n_usize {
        if depth[s] < depth[v] {
            s = v;
        }
    }

    // Second BFS from s
    for v in 0..n_usize {
        depth[v] = -1;
    }
    depth[s] = 1;
    q[0] = s as i32;
    l = 1;

    while l > 0 {
        l -= 1;
        let v = q[l] as i32;

        let mut min_i: i32 = -1;
        let mut max_i: i32 = (2 * n - 2) as i32;

        while max_i - min_i > 1 {
            let mid = (max_i + min_i) / 2;
            let cmid = c[mid as usize];
            if a[cmid] < v {
                min_i = mid;
            } else {
                max_i = mid;
            }
        }

        while a[c[max_i as usize]] == v {
            let eidx = c[max_i as usize];
            let to = b[eidx] as usize;
            if depth[to] < 0 {
                depth[to] = depth[v as usize] + 1;
                q[l] = b[eidx];
                l += 1;
            }
            max_i += 1;
        }
    }

    // Compute max depth
    let mut maxd: i32 = 0;
    for v in 0..n_usize {
        if maxd < depth[v] {
            maxd = depth[v];
        }
    }

    if (maxd + 1) % 3 > 0 {
        print!("First\n");
    } else {
        print!("Second\n");
    }
}
```
