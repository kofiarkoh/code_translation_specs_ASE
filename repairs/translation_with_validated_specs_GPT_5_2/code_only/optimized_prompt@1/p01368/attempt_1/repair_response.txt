```rust
use std::io::{self, Read};

const MAX: usize = 2002;
const INF: i32 = 0x1010;

struct Scanner {
    buf: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }
    fn next_u32(&mut self) -> u32 {
        // Emulate the C in(): it assumes the next char is a digit and reads until c < '0'.
        // We keep behavior close: skip non-digits first (robust for whitespace/newlines),
        // then parse consecutive digits.
        while self.idx < self.buf.len() && (self.buf[self.idx] < b'0' || self.buf[self.idx] > b'9')
        {
            self.idx += 1;
        }
        let mut n: u32 = 0;
        while self.idx < self.buf.len() {
            let c = self.buf[self.idx];
            if c < b'0' || c > b'9' {
                break;
            }
            n = n * 10 + (c - b'0') as u32;
            self.idx += 1;
        }
        n
    }
}

fn bpm(
    u: usize,
    hi: &[i16; MAX],
    to: &[[i16; MAX]; MAX],
    seen: &mut [u8; MAX],
    matchv: &mut [i16; MAX],
) -> bool {
    let limit = hi[u] as usize;
    for i in 0..limit {
        let v = to[u][i] as usize;
        if seen[v] != 0 {
            continue;
        }
        seen[v] = 1;
        if matchv[v] < 0 || bpm(matchv[v] as usize, hi, to, seen, matchv) {
            matchv[u] = v as i16;
            matchv[v] = u as i16;
            return true;
        }
    }
    false
}

fn main() {
    let mut sc = Scanner::new();
    let mut out = String::new();

    // Global-like arrays (as in the C code)
    let mut dist = [[INF; 102]; 102];
    let mut p = [0usize; 1002];
    let mut t = [0i32; 1002];

    let mut hi = [0i16; MAX];
    let mut to = [[0i16; MAX]; MAX];
    let mut seen = [0u8; MAX];
    let mut matchv = [-1i16; MAX];

    loop {
        let n = sc.next_u32() as usize;
        if n == 0 {
            break;
        }
        let m = sc.next_u32() as usize;
        let l = sc.next_u32() as usize;

        // memset(dist, INF, sizeof(dist)) where INF=0x1010 and dist is int:
        // In C this sets each byte to 0x10, yielding 0x10101010 for 32-bit int.
        // But the code compares against INF (0x1010) and sets dist[k][k]=0, and
        // only uses entries that are != INF. The intended "infinite" is INF itself.
        // We'll follow the logical intent: initialize to INF.
        for i in 0..102 {
            for j in 0..102 {
                dist[i][j] = INF;
            }
        }
        for k in 0..n {
            dist[k][k] = 0;
        }
        for _ in 0..m {
            let u = sc.next_u32() as usize;
            let v = sc.next_u32() as usize;
            let d = sc.next_u32() as i32;
            dist[u][v] = d;
            dist[v][u] = d;
        }

        // Floyd-Warshall with INF checks
        for k in 0..n {
            for u in 0..n {
                if dist[u][k] != INF {
                    for v in 0..n {
                        if dist[k][v] != INF {
                            let d = dist[u][k] + dist[k][v];
                            if dist[u][v] > d {
                                dist[u][v] = d;
                            }
                        }
                    }
                }
            }
        }

        // Build bipartite graph
        hi.fill(0);
        for k in 0..l {
            p[k] = sc.next_u32() as usize;
            t[k] = sc.next_u32() as i32;
        }
        for u in 0..l {
            for v in 0..l {
                if u != v {
                    // C: if (dist[p[u]][p[v]] <= t[v]-t[u]) to[u][hi[u]++] = v+L;
                    let rhs = t[v] - t[u];
                    if dist[p[u]][p[v]] <= rhs {
                        let idx = hi[u] as usize;
                        to[u][idx] = (v + l) as i16;
                        hi[u] += 1;
                    }
                }
            }
        }

        // memset(match, -1, L << 2) where match is short:
        // This sets 4*L bytes to 0xFF, i.e., sets first 2*L shorts to -1.
        // We'll emulate that exact range.
        for i in 0..(2 * l).min(MAX) {
            matchv[i] = -1;
        }

        let mut k_ans = l as i32;
        for u in 0..l {
            // memset(seen, 0, L << 1) => zero first 2*L bytes
            // seen is char[], so that's first 2*L entries.
            for i in 0..(2 * l).min(MAX) {
                seen[i] = 0;
            }
            if bpm(u, &hi, &to, &mut seen, &mut matchv) {
                k_ans -= 1;
            }
        }

        out.push_str(&format!("{}\n", k_ans));
    }

    print!("{out}");
}
```
