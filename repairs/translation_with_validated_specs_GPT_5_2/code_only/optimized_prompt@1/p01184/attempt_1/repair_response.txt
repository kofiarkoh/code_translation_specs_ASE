```rust
use std::io::{self, Read};

const HASHSIZ: usize = 61;

#[derive(Clone, Copy)]
struct HashEntry {
    s_idx: Option<usize>, // index into `names`
    id: i32,
}

fn hash_index(s: &str) -> usize {
    let bytes = s.as_bytes();
    let w = bytes.len() as i32;
    // C code assumes non-empty strings (scanf("%s") never yields empty).
    let first = bytes[0] as i32;
    let last = bytes[bytes.len() - 1] as i32;
    let i = (101 * first + 103 * last + 107 * w) % (HASHSIZ as i32);
    i as usize
}

fn lookup(hash: &[HashEntry; HASHSIZ], names: &[String], s: &str) -> i32 {
    let mut tp = hash_index(s);
    loop {
        match hash[tp].s_idx {
            None => return -1,
            Some(idx) => {
                if names[idx] == s {
                    return hash[tp].id;
                }
                tp += 1;
                if tp == HASHSIZ {
                    tp = 0;
                }
            }
        }
    }
}

fn insert(hash: &mut [HashEntry; HASHSIZ], names: &[String], s_idx: usize, id: i32) -> bool {
    let s = &names[s_idx];
    let mut tp = hash_index(s);
    loop {
        match hash[tp].s_idx {
            Some(idx) => {
                if names[idx] == *s {
                    return false;
                }
                tp += 1;
                if tp == HASHSIZ {
                    tp = 0;
                }
            }
            None => {
                hash[tp].s_idx = Some(s_idx);
                hash[tp].id = id;
                return true;
            }
        }
    }
}

fn check(pat: i32, f: &[i32], m: usize) -> bool {
    for i in 0..m {
        if (f[i] & pat) == 0 {
            return false;
        }
    }

    // g has capacity 6 in C; pat is limited to <= 5 bits set by construction.
    let mut g = [0i32; 6];
    let mut s_cnt = 0usize;
    let mut k = pat;
    let mut bit = 0i32;
    while k != 0 {
        if (k & 1) != 0 {
            g[s_cnt] = 1 << bit;
            s_cnt += 1;
        }
        bit += 1;
        k >>= 1;
    }

    for i in 0..m {
        for j in 0..s_cnt {
            if (g[j] & f[i]) != 0 {
                g[j] |= f[i] & pat;
            }
        }
    }

    let mut kk = g[0];
    for i in 1..s_cnt {
        if (g[i] & kk) != 0 {
            kk |= g[i];
        }
    }
    kk == pat
}

fn main() {
    // Read all input and tokenize like scanf with %d/%s (whitespace-separated).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut cno = 0;
    let mut out = String::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: i32 = n_opt.unwrap().parse().unwrap();
        let m: i32 = match it.next() {
            Some(v) => v.parse().unwrap(),
            None => break,
        };
        if n <= 0 {
            break;
        }

        if cno > 0 {
            out.push('\n');
        }
        cno += 1;

        let mut hash = [HashEntry { s_idx: None, id: 0 }; HASHSIZ];
        let mut f = [0i32; 22];

        let mut names: Vec<String> = Vec::new();
        let mut id: i32 = 1;
        let mut sz: usize = 0;

        let mut nn = n;
        while nn > 0 {
            nn -= 1;
            let s = it.next().unwrap().to_string();
            names.push(s);
            let idx = names.len() - 1;
            if insert(&mut hash, &names, idx, id) {
                sz += 1;
                // C uses int; shifting left is safe here (sz <= 31 by array size).
                id <<= 1;
            }
        }

        let mm = m as usize;
        for i in 0..mm {
            let mut j: i32 = it.next().unwrap().parse().unwrap();
            while j > 0 {
                j -= 1;
                let buf = it.next().unwrap();
                let lid = lookup(&hash, &names, buf);
                // In C, OR with -1 would set all bits; we preserve that behavior.
                f[i] |= lid;
            }
        }

        let mut ans: i32 = -1;
        let mut ans_size: i32 = 0;

        'done: for i in 1..=5 {
            let mut j: i32 = (1 << i) - 1;
            let limit: i32 = 1 << (sz as i32);
            while j < limit {
                if check(j, &f, mm) {
                    ans = j;
                    ans_size = i;
                    break 'done;
                }
                // Gosper's hack as in C, using i32 arithmetic.
                let x = j & !(j - 1);
                let y = j + x;
                j = (((j & !y) / x) >> 1) | y;
            }
        }

        if ans < 0 {
            out.push_str("Impossible\n");
        } else {
            out.push_str(&format!("{}\n", ans_size));
            let mut bit_id: i32 = 1;
            for i in 0..sz {
                if (ans & bit_id) != 0 {
                    out.push_str(&names[i]);
                    out.push('\n');
                }
                bit_id <<= 1;
            }
        }
    }

    print!("{out}");
}
```
