```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, like typical competitive-programming C usage with scanf.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut it = input.split_whitespace();
    let mut a: i32 = it.next().unwrap().parse().unwrap();
    let b: i32 = it.next().unwrap().parse().unwrap();
    let mut c: i32 = it.next().unwrap().parse().unwrap();

    // In C, signed overflow is UB; we use wrapping ops to emulate typical 2's-complement behavior.
    a = a.wrapping_sub(b);

    while a > 0 {
        a = a.wrapping_sub(1);
        c = c.wrapping_sub(1);
        if c == 0 {
            break;
        }
    }

    print!("{}\n", c);
}
```
