```rust
use std::io::{self, Read};

const N: usize = 4000;
const L: usize = 62;
const K: usize = (N + L - 1) / L;
const MD: i32 = 998_244_353;

fn shift(aa: &[i64; K], bb: &mut [i64; K], s: i32) {
    bb.fill(0);
    let s = s as usize;
    let q = s / L;
    let r = s % L;

    for h in 0..K {
        if h + q >= K {
            break;
        }
        // In C: (1LL << (L - r)) - 1, with L=62 and r in [0,61], so shift is always valid.
        let mask: i64 = (1i64 << (L - r)) - 1;
        bb[h + q] |= (aa[h] & mask) << r;
        if h + q + 1 < K && r > 0 {
            bb[h + q + 1] |= aa[h] >> (L - r);
        }
    }
}

fn xor_inplace(aa: &mut [i64; K], bb: &[i64; K]) {
    for h in 0..K {
        aa[h] ^= bb[h];
    }
}

fn get_bit(arr: &[i64; K], idx: i32) -> bool {
    if idx < 0 {
        return false;
    }
    let i = idx as usize;
    let w = i / L;
    let b = i % L;
    ((arr[w] >> b) & 1) != 0
}

fn main() {
    // Read all input, tokenize by whitespace (matches scanf("%d%s") / scanf("%s")).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let m: usize = it.next().unwrap().parse().unwrap();
    let s0 = it.next().unwrap();
    let n = s0.len();

    let mut aa = [0i64; K];
    let mut bb_ = [0i64; K];
    let mut cc = [0i64; K];
    let mut dd_ = [0i64; K]; // static => zero-initialized, like C

    // Parse initial polynomial/string into aa (bit 0 is least significant, from right end).
    for (i, ch) in s0.bytes().enumerate() {
        if ch == b'1' {
            let pos = (n - 1 - i) as usize;
            aa[pos / L] |= 1i64 << (pos % L);
        }
    }

    // In C, bb and dd are pointers that can be swapped.
    // Here we keep two buffers and a flag indicating which is current.
    let mut dd_is_dd_ = true; // true => dd = dd_, bb = bb_; false => dd = bb_, bb = dd_

    for _ in 0..m {
        let s = it.next().unwrap();
        let l = s.len();

        // bb = 0
        if dd_is_dd_ {
            bb_.fill(0);
        } else {
            dd_.fill(0);
        }

        // Fill current bb from string
        for (i, ch) in s.bytes().enumerate() {
            if ch == b'1' {
                let pos = (l - 1 - i) as usize;
                if dd_is_dd_ {
                    bb_[pos / L] |= 1i64 << (pos % L);
                } else {
                    dd_[pos / L] |= 1i64 << (pos % L);
                }
            }
        }

        let mut i: i32 = (N as i32) - 1;
        let mut j: i32 = (N as i32) - 1;

        loop {
            // while (i >= 0 && (dd[i/L] & (1LL << (i%L))) == 0) i--;
            while i >= 0 {
                let dd_ref: &[i64; K] = if dd_is_dd_ { &dd_ } else { &bb_ };
                if get_bit(dd_ref, i) {
                    break;
                }
                i -= 1;
            }
            // while (j >= 0 && (bb[j/L] & (1LL << (j%L))) == 0) j--;
            while j >= 0 {
                let bb_ref: &[i64; K] = if dd_is_dd_ { &bb_ } else { &dd_ };
                if get_bit(bb_ref, j) {
                    break;
                }
                j -= 1;
            }

            if i < j {
                // swap dd and bb pointers, and swap i and j
                dd_is_dd_ = !dd_is_dd_;
                std::mem::swap(&mut i, &mut j);
            }
            if j < 0 {
                break;
            }

            let (bb_ref, dd_mut): (&[i64; K], &mut [i64; K]) = if dd_is_dd_ {
                (&bb_, &mut dd_)
            } else {
                (&dd_, &mut bb_)
            };

            shift(bb_ref, &mut cc, i - j);
            xor_inplace(dd_mut, &cc);
        }
    }

    // Determine l = degree(dd)+1 within N bits
    let dd_ref: &[i64; K] = if dd_is_dd_ { &dd_ } else { &bb_ };
    let mut l: i32 = N as i32;
    while l > 0 && !get_bit(dd_ref, l - 1) {
        l -= 1;
    }

    // Precompute powers of 2 mod MD for i=0..n-1
    let mut p2 = vec![0i32; N];
    let mut p: i32 = 1;
    for i in 0..n {
        p2[i] = p;
        p = ((p as i64 * 2) % (MD as i64)) as i32;
    }

    // bb = 0 (this is a different bb than earlier; in C it's the current bb pointer)
    if dd_is_dd_ {
        bb_.fill(0);
    } else {
        dd_.fill(0);
    }

    let mut ans: i32 = 0;
    let mut i: i32 = (n as i32) - 1;

    while i >= 0 {
        let iu = i as usize;
        let x = ((aa[iu / L] >> (iu % L)) & 1) as i32;

        let bb_ref: &[i64; K] = if dd_is_dd_ { &bb_ } else { &dd_ };
        let y = ((bb_ref[iu / L] >> (iu % L)) & 1) as i32;

        if i >= l - 1 && x == 1 {
            let idx = (i - l + 1) as usize;
            ans = (ans + p2[idx]) % MD;
        }

        if x != y {
            if i >= l - 1 {
                let dd_ref2: &[i64; K] = if dd_is_dd_ { &dd_ } else { &bb_ };
                shift(dd_ref2, &mut cc, i - l + 1);
                let bb_mut: &mut [i64; K] = if dd_is_dd_ { &mut bb_ } else { &mut dd_ };
                xor_inplace(bb_mut, &cc);
            } else {
                if x > y {
                    ans = (ans + 1) % MD;
                }
                break;
            }
        }

        i -= 1;
    }

    if i < 0 {
        ans = (ans + 1) % MD;
    }

    print!("{}\n", ans);
}
```
