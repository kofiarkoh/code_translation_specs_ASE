```rust
use std::cmp::max;
use std::io::{self, Read};

const N: usize = 1000;
const M: usize = 200000;
const OMAX: usize = 1 + M * 3;

static mut OO: [i32; OMAX] = [0; OMAX];
static mut OH: [i32; OMAX] = [0; OMAX];
static mut LINK_NEXT: i32 = 1;

fn link(o: i32, h: i32) -> i32 {
    // Uses global arrays like the C code; safe because indices are bounded by construction.
    unsafe {
        let idx = LINK_NEXT as usize;
        OO[idx] = o;
        OH[idx] = h;
        LINK_NEXT += 1;
        idx as i32
    }
}

static mut II: [i32; M] = [0; M];
static mut JJ: [i32; M] = [0; M];

static mut AE: [i32; N] = [0; N];
static mut AF: [i32; N] = [0; N];
static mut AE_: [i32; N] = [0; N];

static mut CC: [i32; N] = [0; N];
static mut QU: [i32; N] = [0; N];
static mut CNT: i32 = 0;

fn dfs1(i: i32) {
    unsafe {
        let iu = i as usize;
        if CC[iu] != 0 {
            return;
        }
        CC[iu] = -1;
        let mut o = AE[iu];
        while o != 0 {
            let h = OH[o as usize];
            let j = JJ[h as usize];
            dfs1(j);
            o = OO[o as usize];
        }
        CNT -= 1;
        QU[CNT as usize] = i;
    }
}

fn dfs2(j: i32, c: i32) {
    unsafe {
        let ju = j as usize;
        if CC[ju] != -1 {
            return;
        }
        CC[ju] = c;
        let mut o = AF[ju];
        while o != 0 {
            let h = OH[o as usize];
            let i = II[h as usize];
            dfs2(i, c);
            o = OO[o as usize];
        }
    }
}

static mut VISITED: [u8; N] = [0; N];
static mut C_CUR: i32 = 0;
static mut HH: [i32; N * 2] = [0; N * 2];
static mut CNT_: i32 = 0;

fn dfs3(i: i32) {
    unsafe {
        let iu = i as usize;
        VISITED[iu] = 1;
        let mut o = AE[iu];
        while o != 0 {
            let h = OH[o as usize];
            let j = JJ[h as usize];
            let ju = j as usize;
            if VISITED[ju] == 0 && CC[ju] == C_CUR {
                HH[CNT_ as usize] = h;
                CNT_ += 1;
                dfs3(j);
            }
            o = OO[o as usize];
        }
    }
}

fn dfs4(j: i32) {
    unsafe {
        let ju = j as usize;
        VISITED[ju] = 2;
        let mut o = AF[ju];
        while o != 0 {
            let h = OH[o as usize];
            let i = II[h as usize];
            let iu = i as usize;
            if VISITED[iu] == 1 && CC[iu] == C_CUR {
                HH[CNT_ as usize] = h;
                CNT_ += 1;
                dfs4(i);
            }
            o = OO[o as usize];
        }
    }
}

static mut H_EXCL: i32 = 0;

fn dfs5(i: i32, t: i32) -> i32 {
    unsafe {
        if i == t {
            return 1;
        }
        let iu = i as usize;
        if VISITED[iu] != 0 {
            return 0;
        }
        VISITED[iu] = 1;
        let mut o = AE[iu];
        while o != 0 {
            let h = OH[o as usize];
            let j = JJ[h as usize];
            if h != H_EXCL && CC[j as usize] == C_CUR && dfs5(j, t) != 0 {
                return 1;
            }
            o = OO[o as usize];
        }
        0
    }
}

fn main() {
    // Read all input (scanf-like whitespace handling).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let m: usize = it.next().unwrap().parse().unwrap();

    // These are "static" in C; here we allocate on stack but sizes are fixed and small enough.
    let mut dd: [i32; N] = [0; N];
    let mut diff: Vec<u8> = vec![0; m];

    unsafe {
        // Build graph adjacency lists using the shared OO/OH pool.
        for h in 0..m {
            let mut i: i32 = it.next().unwrap().parse().unwrap();
            let mut j: i32 = it.next().unwrap().parse().unwrap();
            i -= 1;
            j -= 1;
            II[h] = i;
            JJ[h] = j;
            let iu = i as usize;
            let ju = j as usize;
            AE[iu] = link(AE[iu], h as i32);
            AF[ju] = link(AF[ju], h as i32);
        }

        // Kosaraju: first pass order.
        CNT = n as i32;
        for i in 0..n {
            if CC[i] == 0 {
                dfs1(i as i32);
            }
        }

        // Second pass: assign components.
        let mut n_: i32 = 0;
        for g in 0..n {
            let i = QU[g];
            if CC[i as usize] == -1 {
                dfs2(i, n_);
                n_ += 1;
            }
        }
        let n_comp = n_ as usize;

        // Build SCC DAG adjacency (edges between components), stored in AE_ using original edge ids.
        for h in 0..m {
            let i = CC[II[h] as usize];
            let j = CC[JJ[h] as usize];
            if i != j {
                AE_[i as usize] = link(AE_[i as usize], h as i32);
            }
        }

        // For each component s, mark edges that are "diff" due to multiple edges to same target
        // or due to existence of a path of length > 1 from s to that target in the SCC DAG.
        for s in 0..n_comp {
            // memset(dd, 0, n_ * sizeof *dd);
            for x in dd[..n_comp].iter_mut() {
                *x = 0;
            }
            let mut o = AE_[s];
            while o != 0 {
                let h = OH[o as usize] as usize;
                let j = CC[JJ[h] as usize] as usize;
                dd[j] += 1;
                o = OO[o as usize];
            }
            let mut o2 = AE_[s];
            while o2 != 0 {
                let h = OH[o2 as usize] as usize;
                let j = CC[JJ[h] as usize] as usize;
                if dd[j] > 1 {
                    diff[h] = 1;
                }
                o2 = OO[o2 as usize];
            }

            // memset(dd, -1, n_ * sizeof *dd), dd[s] = 0;
            for x in dd[..n_comp].iter_mut() {
                *x = -1;
            }
            dd[s] = 0;

            // DP over component indices (as in C), not a topological order.
            for i in s..n_comp {
                if dd[i] != -1 {
                    let mut o3 = AE_[i];
                    while o3 != 0 {
                        let h = OH[o3 as usize] as usize;
                        let j = CC[JJ[h] as usize] as usize;
                        dd[j] = max(dd[j], dd[i] + 1);
                        o3 = OO[o3 as usize];
                    }
                }
            }

            let mut o4 = AE_[s];
            while o4 != 0 {
                let h = OH[o4 as usize] as usize;
                let j = CC[JJ[h] as usize] as usize;
                if dd[j] > 1 {
                    diff[h] = 1;
                }
                o4 = OO[o4 as usize];
            }
        }

        // For each SCC, run the intra-SCC logic.
        for c in 0..n_comp {
            C_CUR = c as i32;

            // Collect vertices in this SCC into QU[0..cnt).
            CNT = 0;
            for i in 0..n {
                if CC[i] == C_CUR {
                    QU[CNT as usize] = i as i32;
                    CNT += 1;
                }
            }
            if CNT == 0 {
                continue;
            }

            CNT_ = 0;
            dfs3(QU[0]);
            dfs4(QU[0]);

            // For each candidate edge h in HH, test if removing it disconnects ii[h] -> jj[h] within SCC.
            for g_ in 0..(CNT_ as usize) {
                let h = HH[g_] as usize;

                // Reset visited for vertices in this SCC.
                for g in 0..(CNT as usize) {
                    VISITED[QU[g] as usize] = 0;
                }

                H_EXCL = h as i32;
                let res = dfs5(II[h], JJ[h]);
                diff[h] = if res == 0 { 1 } else { 0 };
            }
        }

        // Output.
        let mut out = String::new();
        for h in 0..m {
            if diff[h] != 0 {
                out.push_str("diff\n");
            } else {
                out.push_str("same\n");
            }
        }
        print!("{out}");
    }
}
```
