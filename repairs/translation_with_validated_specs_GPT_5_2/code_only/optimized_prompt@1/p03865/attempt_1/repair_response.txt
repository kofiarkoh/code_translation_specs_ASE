```rust
use std::io::{self, Read};

fn main() {
    // Read up to 114514 bytes from stdin (like read(0, buf, 114514)).
    let mut buf = vec![0u8; 114_514];
    let n_read = match io::stdin().read(&mut buf) {
        Ok(n) => n,
        Err(_) => 0, // C code ignores read errors; approximate by treating as 0 bytes read.
    };

    // C: n = read(...); n--;
    // Use wrapping to emulate C's potential underflow/overflow behavior.
    let n = (n_read as isize).wrapping_sub(1);

    // C expression:
    // puts(n + (buf[0] == buf[n-1]) & 1 ? "First" : "Second");
    //
    // Operator precedence in C:
    //   + has higher precedence than &
    // so it's: ((n + (buf[0] == buf[n-1])) & 1) ? ...
    //
    // Also, buf[n-1] uses the decremented n, i.e. index (n-1).
    // If n_read is 0 or 1, this is out-of-bounds in C (UB). We must pick a behavior.
    // Here we avoid panics and approximate by treating the comparison as false when indices
    // are invalid, which keeps I/O deterministic.
    let eq_as_int: isize = if n_read >= 2 {
        let idx = (n - 1) as usize; // with n_read>=2, n = n_read-1 >=1, so idx is valid.
        if buf[0] == buf[idx] { 1 } else { 0 }
    } else {
        0
    };

    let cond = (n.wrapping_add(eq_as_int) & 1) != 0;
    if cond {
        println!("First");
    } else {
        println!("Second");
    }
}
```
