use std::io::{self, Read};

const N_MAX: usize = 100000;
const L_MAX: usize = 20;

fn fill(a: &mut [i32], l: usize, r: usize, x: i32) {
    for i in l..=r {
        a[i] = x;
    }
}

fn find_root(a: &mut [i32], mut x: usize) -> usize {
    // Mirrors the C code's fixed-size stack for path compression.
    // Assumes the path length does not exceed L_MAX (as in the original).
    let mut s = [0usize; L_MAX + 1];
    let mut k: usize = 0;

    while a[x] >= 1 {
        k += 1;
        // If k exceeds L_MAX, the C code would write out of bounds (UB).
        // We clamp to preserve memory safety; behavior in such cases is unspecified in C anyway.
        if k <= L_MAX {
            s[k] = x;
        }
        x = a[x] as usize;
    }

    while k > 0 {
        if k <= L_MAX {
            a[s[k]] = x as i32;
        }
        k -= 1;
    }
    x
}

fn unite_set(a: &mut [i32], x: usize, y: usize) {
    let mut x = find_root(a, x);
    let mut y = find_root(a, y);
    if a[x] < a[y] {
        a[x] += a[y];
        a[y] = x as i32;
    } else {
        a[y] += a[x];
        a[x] = y as i32;
    }
}

fn set_size(a: &mut [i32], x: usize) -> i32 {
    let r = find_root(a, x);
    -a[r]
}

fn main() {
    // Read all input, parse as i32 like scanf("%i").
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;
    let mut m: i32 = it.next().unwrap().parse::<i32>().unwrap();
    let mut k: i32 = it.next().unwrap().parse::<i32>().unwrap();

    // Arrays sized exactly as in C (N_MAX + 1), but we only use indices 1..=n.
    let mut s = vec![0i32; N_MAX + 1];
    fill(&mut s, 1, n, -1);

    let mut rmv = vec![0i32; N_MAX + 1];
    fill(&mut rmv, 1, n, 1);

    while m > 0 {
        m -= 1;
        let a: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;
        let b: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;

        let ra = find_root(&mut s, a);
        let rb = find_root(&mut s, b);
        if ra != rb {
            unite_set(&mut s, a, b);
        }
        rmv[a] += 1;
        rmv[b] += 1;
    }

    while k > 0 {
        k -= 1;
        let c: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;
        let d: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;

        let rc = find_root(&mut s, c);
        let rd = find_root(&mut s, d);
        if rc == rd {
            rmv[c] += 1;
            rmv[d] += 1;
        }
    }

    let mut out = String::new();
    for i in 1..=n {
        let val = set_size(&mut s, i) - rmv[i];
        if i == n {
            out.push_str(&format!("{}\n", val));
        } else {
            out.push_str(&format!("{} ", val));
        }
    }
    print!("{out}");
}
