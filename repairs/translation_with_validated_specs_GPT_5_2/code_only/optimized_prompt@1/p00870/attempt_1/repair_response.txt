```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, then tokenize like scanf("%d %d") / scanf("%s")
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: usize = n_opt.unwrap().parse().unwrap();
        let m: usize = match it.next() {
            Some(x) => x.parse().unwrap(),
            None => break,
        };
        if n == 0 && m == 0 {
            break;
        }

        // e: up to 12 strings, each up to 20 chars in C (plus NUL).
        // We'll store as bytes for strncmp-like behavior.
        let mut e: Vec<Vec<u8>> = Vec::with_capacity(n);
        let mut len: Vec<usize> = Vec::with_capacity(n);
        for _ in 0..n {
            let s = it.next().unwrap_or("");
            let b = s.as_bytes().to_vec();
            len.push(b.len());
            e.push(b);
        }

        // Build concatenated string t from m tokens.
        // The C code has a check for trailing control char, but with scanf("%s")
        // that never happens; we keep the behavior as close as possible by not
        // introducing extra trimming beyond whitespace tokenization.
        let mut t: Vec<u8> = Vec::new();
        for _ in 0..m {
            let s = it.next().unwrap_or("");
            t.extend_from_slice(s.as_bytes());
        }
        let l_s = t.len();

        // loop = 1<<n (n <= 12 => <= 4096)
        let loop_mask: usize = 1usize << n;

        // flag[5001][4096] in C, but we only need indices up to l_s and masks up to (1<<n)-1.
        // Use u8 to mimic C char storage (0/1).
        let mut flag = vec![vec![0u8; loop_mask]; l_s + 1];

        for i in 0..l_s {
            for j in 0..loop_mask {
                if j != 0 && flag[i][j] == 0 {
                    continue;
                }
                let mut mask = 1usize;
                for k in 0..n {
                    if (j & mask) != 0 {
                        mask <<= 1;
                        continue;
                    }
                    let l = len[k];
                    if i + l <= l_s {
                        // strncmp(e[k], &t[i], l) == 0
                        if t[i..i + l] == e[k][..] {
                            let pat = j | mask;
                            flag[i + l][pat] = 1;
                        }
                    }
                    mask <<= 1;
                }
            }
        }

        let mut cnt: i32 = 0;
        let full = loop_mask - 1;
        for i in 0..=l_s {
            cnt += flag[i][full] as i32;
        }

        print!("{cnt}\n");
    }
}
```
