```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, parse as whitespace-separated integers (like scanf("%d")).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // Fixed-size arrays to match the C code's 0..=500 indexing.
    let mut student = vec![[0i32; 501]; 501];
    let mut list = [0i32; 501];

    loop {
        let n_opt = it.next();
        let m_opt = it.next();
        if n_opt.is_none() || m_opt.is_none() {
            break; // If input ends unexpectedly, stop (scanf would fail).
        }
        let n: i32 = n_opt.unwrap().parse().unwrap();
        let m: i32 = m_opt.unwrap().parse().unwrap();

        if (n | m) == 0 {
            break;
        }

        // Reset arrays exactly as in C.
        for i in 0..=500usize {
            list[i] = 0;
            for j in 0..=500usize {
                student[i][j] = 0;
            }
        }

        for _ in 0..m {
            let p1: usize = it.next().unwrap().parse().unwrap();
            let p2: usize = it.next().unwrap().parse().unwrap();
            student[p1][p2] = 1;
            student[p2][p1] = 1;
        }

        let n_usize = n as usize;

        for i in 2..=n_usize {
            if student[1][i] == 1 {
                list[i] = 1;
            }
        }

        for i in 2..=n_usize {
            if student[1][i] == 1 {
                for j in 2..=n_usize {
                    if student[i][j] == 1 {
                        list[j] = 1;
                    }
                }
            }
        }

        let mut count: i32 = 0;
        for i in 2..=n_usize {
            count += list[i];
        }
        print!("{count}\n");
    }
}
```
