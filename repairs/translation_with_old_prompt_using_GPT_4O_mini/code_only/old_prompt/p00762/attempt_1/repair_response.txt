```rust
use std::io::{self, Write};

const N_MAX: usize = 4;

static RIGHT: [[u8; 7]; 7] = [
    [0, 0, 0, 0, 0, 0, 0], // top
    [0, 0, 3, 5, 2, 4, 0], // 1
    [0, 4, 0, 1, 6, 0, 3], // 2
    [0, 2, 6, 0, 0, 1, 5], // 3
    [0, 5, 1, 0, 0, 6, 2], // 4
    [0, 3, 0, 6, 1, 0, 4], // 5
    [0, 0, 4, 2, 5, 3, 0], // 6
];

fn right(top: usize, front: usize) -> u8 {
    RIGHT[top][front]
}

fn bottom(top: usize) -> usize {
    7 - top
}

fn back(front: usize) -> usize {
    7 - front
}

fn left(top: usize, front: usize) -> usize {
    7 - right(top, front)
}

struct Maps {
    height_map: [[u8; N_MAX * 2 + 1]; N_MAX * 2 + 1],
    top_map: [[u8; N_MAX * 2 + 2]; N_MAX * 2 + 1],
}

impl Maps {
    fn new() -> Self {
        Maps {
            height_map: [[0; N_MAX * 2 + 1]; N_MAX * 2 + 1],
            top_map: [[0; N_MAX * 2 + 2]; N_MAX * 2 + 1],
        }
    }

    fn put(&mut self, top: u8, front: u8) {
        let (mut x, mut y) = (N_MAX, N_MAX);
        let mut newfront;
        let mut newtop;
        let mut newx;
        let mut newy;
        let mut max;

        loop {
            max = 0;

            // front
            if self.height_map[y - 1][x] < self.height_map[y][x] && front > 3 && front > max {
                newfront = top;
                newtop = back(front as usize);
                newx = x;
                newy = y - 1;
                max = front;
            }

            // back
            if self.height_map[y + 1][x] < self.height_map[y][x] && back(front as usize) > 3 && back(front as usize) > max {
                newfront = bottom(top as usize);
                newtop = front;
                newx = x;
                newy = y + 1;
                max = back(front as usize);
            }

            // left
            if self.height_map[y][x - 1] < self.height_map[y][x] && left(top as usize, front as usize) > 3 && left(top as usize, front as usize) > max {
                newfront = front;
                newtop = right(top as usize, front as usize);
                newx = x - 1;
                newy = y;
                max = left(top as usize, front as usize);
            }

            // right
            if self.height_map[y][x + 1] < self.height_map[y][x] && right(top as usize, front as usize) > 3 && right(top as usize, front as usize) > max {
                newfront = front;
                newtop = left(top as usize, front as usize);
                newx = x + 1;
                newy = y;
                max = right(top as usize, front as usize);
            }

            if max == 0 {
                break;
            } else {
                front = newfront;
                top = newtop;
                x = newx;
                y = newy;
            }
        }

        self.height_map[y][x] += 1;
        self.top_map[y][x] = top;
    }

    fn count_and_print(&self) {
        let mut count = [0; 7];

        for y in 0..N_MAX * 2 + 1 {
            for x in 0..N_MAX * 2 + 1 {
                count[self.top_map[y][x] as usize] += 1;
            }
        }

        for x in 1..6 {
            print!("{} ", count[x]);
        }
        println!("{}", count[6]);
    }
}

fn main() {
    let stdin = io::stdin();
    let mut input = String::new();

    loop {
        input.clear();
        stdin.read_line(&mut input).unwrap();
        let n: i32 = input.trim().parse().unwrap();
        if n == 0 {
            break;
        }

        let mut maps = Maps::new();

        for _ in 0..n {
            input.clear();
            stdin.read_line(&mut input).unwrap();
            let mut iter = input.split_whitespace();
            let t: u8 = iter.next().unwrap().parse().unwrap();
            let f: u8 = iter.next().unwrap().parse().unwrap();
            maps.put(t, f);
        }

        maps.count_and_print();
    }
}
```
