```rust
use std::io;

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let n: usize = input.trim().parse().unwrap();

    let mut num = vec![0; 1000]; // Create a vector with 1000 elements initialized to 0
    for i in 1..=n {
        input.clear();
        io::stdin().read_line(&mut input).unwrap();
        num[i] = input.trim().parse().unwrap();
    }

    let mut count = 0;
    for i in 1..(n - 1) {
        if num[i] < num[i + 1] {
            if num[i + 1] < num[i + 2] {
                count += 1;
            }
        } else if num[i] > num[i + 1] {
            if num[i + 1] > num[i + 2] {
                count += 1;
            }
        }
    }

    println!("{}", count);
}
```
