use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();
    
    // Read the first line to get `n`
    let first_line = lines.next().unwrap().unwrap();
    let mut n: i32 = first_line.trim().parse().unwrap();
    
    // Create a vector to simulate the array `m`
    let mut m = vec![0; 1 << 18];
    let mut p: *mut i32 = m.as_mut_ptr(); // Pointer to the start of the array

    // Read pairs of `a` and `b` until EOF
    while let Some(Ok(line)) = lines.next() {
        let nums: Vec<i32> = line.split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();
        
        if nums.len() < 2 {
            break; // If we don't have two numbers, exit the loop
        }
        
        let a = nums[0];
        let b = nums[1];

        // Emulate the behavior of the original C code
        if !p.is_null() && n != 0 {
            n = 0;
        } else {
            unsafe {
                *p += 1; // Increment the value pointed to by `p`
            }
        }

        p = m.as_mut_ptr().add(if a - 1 != 0 {
            if b - n != 0 { 0 } else { a }
        } else {
            b
        });
    }

    // Output the result based on the value of `n`
    if n != 0 {
        println!("IMPOSSIBLE");
    } else {
        println!("POSSIBLE");
    }
}
