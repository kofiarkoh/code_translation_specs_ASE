```rust
use std::io::{self, BufRead};

const D4: [i64; 5] = [0, 1, 0, -1, 0];

fn main() {
    let stdin = io::stdin();
    let mut reader = stdin.lock();

    loop {
        let mut n = String::new();
        reader.read_line(&mut n).unwrap();
        let mut n: i64 = n.trim().parse().unwrap();
        if n == 0 {
            break;
        }

        let mut a = [[0; 25]; 25];
        for _ in 0..n {
            let mut coords = String::new();
            reader.read_line(&mut coords).unwrap();
            let mut iter = coords.split_whitespace();
            let x: usize = iter.next().unwrap().parse().unwrap();
            let y: usize = iter.next().unwrap().parse().unwrap();
            a[x][y] = 1;
        }

        let mut m = String::new();
        reader.read_line(&mut m).unwrap();
        let m: i64 = m.trim().parse().unwrap();

        let mut px: i64 = 10; // Change to i64 to match D4 type
        let mut py: i64 = 10; // Change to i64 to match D4 type
        if a[px as usize][py as usize] == 1 {
            a[px as usize][py as usize] = 0;
            n -= 1;
        }

        for _ in 0..m {
            let mut command = String::new();
            reader.read_line(&mut command).unwrap();
            let mut iter = command.split_whitespace();
            let c: char = iter.next().unwrap().chars().next().unwrap();
            let q: i64 = iter.next().unwrap().parse().unwrap();

            let d = match c {
                'N' => 0,
                'E' => 1,
                'S' => 2,
                'W' => 3,
                _ => unreachable!(), // This should not happen based on the input specification
            };

            for _ in 0..q {
                px += D4[d];
                py += D4[d + 1];
                if a[px as usize][py as usize] == 1 {
                    a[px as usize][py as usize] = 0;
                    n -= 1;
                }
            }
        }

        println!("{}", if n > 0 { "No" } else { "Yes" });
    }
}
```
