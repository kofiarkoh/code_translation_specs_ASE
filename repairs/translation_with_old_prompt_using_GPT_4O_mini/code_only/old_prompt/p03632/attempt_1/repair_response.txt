```rust
use std::io::{self, Write};

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let mut iter = input.split_whitespace();
    
    let a_start: usize = iter.next().unwrap().parse().unwrap();
    let a_end: usize = iter.next().unwrap().parse().unwrap();
    let b_start: usize = iter.next().unwrap().parse().unwrap();
    let b_end: usize = iter.next().unwrap().parse().unwrap();

    let mut a = [0; 101];
    let mut b = [0; 101];
    let mut c = [0; 101]; // Changed to 101 to match the C code's indexing

    for i in a_start..=a_end {
        a[i] = 1;
    }

    for i in b_start..=b_end {
        b[i] = 1;
    }

    let mut time = 0;

    for i in 0..=100 {
        c[i] = a[i] + b[i];
        if c[i] == 2 {
            time += 1;
        }
    }

    // Adjusting the output to match the C code's behavior
    if time == 0 {
        print!("{}", time);
    } else {
        print!("{}", time - 1);
    }
}
```
