// AOJ 1212: Mirror Illusion
// 2018.1.23 bal4u@uu

use std::io::{self, BufRead};

const SIZE: usize = 9;
const MV: [[i32; 2]; 4] = [[50, 50], [-50, 50], [-50, -50], [50, -50]];
const TR: [[usize; 4]; 2] = [[3, 2, 1, 0], [1, 0, 3, 2]];

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut mir = [[[0u8; SIZE]; SIZE]; SIZE]; // 4D array initialized to 0

    for line in reader.lines() {
        let line = line.unwrap();
        let mut parts = line.split_whitespace();
        let n: i32 = parts.next().unwrap().parse().unwrap();
        if n < 0 {
            break;
        }

        // Reset the mirror array
        for i in 0..SIZE {
            for j in 0..SIZE {
                for k in 0..SIZE {
                    for l in 0..SIZE {
                        mir[i][j][k][l] = 0;
                    }
                }
            }
        }

        for _ in 0..n {
            let s: String = parts.next().unwrap().to_string();
            let x: usize = parts.next().unwrap().parse().unwrap();
            let y: usize = parts.next().unwrap().parse().unwrap();
            if s.chars().next().unwrap() == 'x' {
                mir[x][y][x + 1][y] = 1;
            } else {
                mir[x][y][x][y + 1] = 1;
            }
        }

        let mut x = 50;
        let mut y = 0;
        let mut d = 0;
        let mut wall = 1;
        let mut v = 1;

        for i in 0.. {
            if i >= 2 {
                if x == 100 && y == 50 && d == 0 {
                    x = 75;
                    y = 25;
                    break;
                }
                if wall != 0 && (x == 0 || y == 0 || x == 800 || y == 800) {
                    break;
                }
            }

            x += MV[d][0];
            y += MV[d][1];
            wall = 1;

            if v != 0 {
                if mir[(x / 100) as usize][((y - 50) / 100) as usize][(x / 100) as usize][((y + 50) / 100) as usize] != 0 {
                    d = TR[v][d];
                    wall = 0;
                }
            } else {
                if mir[((x - 50) / 100) as usize][(y / 100) as usize][((x + 50) / 100) as usize][(y / 100) as usize] != 0 {
                    d = TR[v][d];
                    wall = 0;
                }
            }
            v = 1 - v; // Toggle v between 0 and 1
        }
        println!("{} {}", x, y);
    }
}
