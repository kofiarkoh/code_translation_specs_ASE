```rust
use std::io::{self, Write};

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let mut iter = input.split_whitespace();
    let n: usize = iter.next().unwrap().parse().unwrap();
    let m: usize = iter.next().unwrap().parse().unwrap();

    let mut s = vec![0; m];
    let mut c = vec![0; m];
    let d = [100, 10, 1];
    let mut digit = vec![-1; n];
    let mut num = 0;

    if m == 0 {
        num = match n {
            1 => 0,
            2 => 10,
            3 => 100,
            _ => num, // default case, num remains 0
        };
    } else {
        for i in 0..m {
            input.clear();
            io::stdin().read_line(&mut input).unwrap();
            let mut iter = input.split_whitespace();
            s[i] = iter.next().unwrap().parse().unwrap();
            c[i] = iter.next().unwrap().parse().unwrap();

            let index = s[i] - 1;
            let value = c[i] * d[3 - n + index];

            if digit[index] == -1 {
                digit[index] = value;
            } else if digit[index] / d[3 - n + index] != c[i] {
                num = -1;
            }
        }

        if num != -1 {
            for i in 0..n {
                if digit[i] != -1 {
                    num += digit[i];
                } else if n > 1 && i == 0 && digit[i] != 0 {
                    num += d[3 - n + i];
                }
            }
        }
    }

    if n > 1 && s[0] == 1 && c[0] == 0 {
        println!("-1");
    } else {
        print!("{}", num);
    }
}
```
