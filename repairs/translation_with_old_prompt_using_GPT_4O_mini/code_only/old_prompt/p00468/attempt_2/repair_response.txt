```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut input = stdin.lock();
    
    let mut student = [[0; 501]; 501];
    let mut list = [0; 501];
    
    loop {
        let mut n = String::new();
        let mut m = String::new();
        
        // Read n and m in a single line
        if input.read_line(&mut n).unwrap() == 0 {
            break; // End of input
        }
        
        let parts: Vec<&str> = n.trim().split_whitespace().collect();
        let n: i32 = parts[0].parse().unwrap_or(0);
        let m: i32 = parts.get(1).and_then(|s| s.parse().ok()).unwrap_or(0);
        
        if (n | m) == 0 {
            break;
        }
        
        for i in 0..=500 {
            list[i] = 0;
            for j in 0..=500 {
                student[i][j] = 0;
            }
        }
        
        for _ in 0..m {
            let mut line = String::new();
            input.read_line(&mut line).unwrap();
            let parts: Vec<usize> = line.trim().split_whitespace()
                .filter_map(|s| s.parse().ok()).collect();
            
            if parts.len() < 2 {
                continue; // Skip if not enough data
            }
            
            let p1 = parts[0];
            let p2 = parts[1];
            
            student[p1][p2] = 1;
            student[p2][p1] = 1;
        }
        
        for i in 2..=n {
            if student[1][i as usize] == 1 {
                list[i as usize] = 1;
            }
        }
        
        for i in 2..=n {
            if student[1][i as usize] == 1 {
                for j in 2..=n {
                    if student[i as usize][j as usize] == 1 {
                        list[j as usize] = 1;
                    }
                }
            }
        }
        
        let count: i32 = (2..=n).map(|i| list[i as usize]).sum();
        println!("{}", count);
    }
}
```
