```rust
use std::io::{self, Write};

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let n: usize = input.trim().parse().expect("Failed to parse input for n");

    let mut A = [0; 9];
    let mut next_A = [0; 9];
    let mut prev_A = [0; 9];

    for i in 0..n {
        input.clear();
        io::stdin().read_line(&mut input).unwrap();
        let value: i32 = input.trim().parse().expect("Failed to parse input for A[i]");
        A[i] = value;
        next_A[i] = value;
        prev_A[i] = value;
    }

    if n == 1 {
        println!("1");
    } else {
        // Search for previous permutation
        let mut flag = false;
        let mut i = n - 2;
        while i != usize::MAX {
            if prev_A[i] > prev_A[i + 1] {
                flag = true;
                break;
            }
            if i == 0 { break; } // Prevent underflow
            i -= 1;
        }

        if flag {
            let i = i as usize; // Cast to usize for indexing
            let mut j = n - 1;
            while j > i {
                if prev_A[j] < prev_A[i] {
                    prev_A.swap(j, i);
                    break;
                }
                j -= 1;
            }
            let mid = (n - i - 1) / 2;
            for k in 1..=mid {
                prev_A.swap(i + k, n - k);
            }
            // Print the previous permutation
            print!("{}", prev_A[0]);
            for i in 1..n {
                print!(" {}", prev_A[i]);
            }
            println!();
        }

        // Print the given sequence
        print!("{}", A[0]);
        for i in 1..n {
            print!(" {}", A[i]);
        }
        println!();

        // Search for next permutation
        flag = false;
        let mut i = n - 2;
        while i != usize::MAX {
            if next_A[i] < next_A[i + 1] {
                flag = true;
                break;
            }
            if i == 0 { break; } // Prevent underflow
            i -= 1;
        }

        if flag {
            let i = i as usize; // Cast to usize for indexing
            let mut j = n - 1;
            while j > i {
                if next_A[j] > next_A[i] {
                    next_A.swap(j, i);
                    break;
                }
                j -= 1;
            }
            let mid = (n - i - 1) / 2;
            for k in 1..=mid {
                next_A.swap(i + k, n - k);
            }
            // Print the next permutation
            print!("{}", next_A[0]);
            for i in 1..n {
                print!(" {}", next_A[i]);
            }
            println!();
        }
    }
}
```
