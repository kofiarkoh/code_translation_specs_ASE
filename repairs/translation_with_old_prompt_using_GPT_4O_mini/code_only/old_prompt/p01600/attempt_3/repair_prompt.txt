use std::io::{self, Read};

const ABS: fn(i32) -> i32 = |a| if a >= 0 { a } else { -a };
const MAX_N: usize = 1001;

fn in_number<R: Read>(reader: &mut R) -> i32 {
    let mut n = 0;
    let mut buffer = [0; 1];
    while reader.read(&mut buffer).unwrap() > 0 {
        let c = buffer[0] as char;
        if c.is_digit(10) {
            n = 10 * n + (c as i32 - '0' as i32);
        } else {
            break;
        }
    }
    n
}
