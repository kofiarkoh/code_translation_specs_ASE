// AOJ 2488 Tree Construction

use std::io::{self, Read};

const ABS: fn(i32) -> i32 = |a| if a >= 0 { a } else { -a };
const MAX_N: usize = 1001;

fn in_number<R: Read>(reader: &mut R) -> i32 {
    let mut n = 0;
    let mut buffer = [0; 1];
    while reader.read(&mut buffer).unwrap() > 0 {
        let c = buffer[0] as char;
        if c.is_digit(10) {
            n = 10 * n + (c as i32 - '0' as i32);
        } else {
            break;
        }
    }
    n
}

fn main() {
    let stdin = io::stdin();
    let n = in_number(&mut stdin.lock());
    let mut x = vec![0; MAX_N];
    let mut y = vec![0; MAX_N];
    let mut dp = vec![vec![0x10; MAX_N]; MAX_N];

    for i in 0..n {
        x[i] = in_number(&mut stdin.lock());
        y[i] = in_number(&mut stdin.lock());
    }

    for i in 0..n {
        dp[i][i] = 0;
    }

    for l in 0..n {
        for i in 0..n {
            if i + l < n {
                for j in i..(i + l) {
                    let t = dp[i][j] + dp[j + 1][i + l] + ABS(x[i] - x[j + 1]) + ABS(y[j] - y[i + l]);
                    if t < dp[i][i + l] {
                        dp[i][i + l] = t;
                    }
                }
            }
        }
    }
    println!("{}", dp[0][n as usize - 1]);
}
