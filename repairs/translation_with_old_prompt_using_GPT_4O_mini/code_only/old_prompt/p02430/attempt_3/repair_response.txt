use std::io::{self, BufRead};

const BUF_SIZE: usize = 20;
const DIVISOR: u64 = 1000000007;

fn get_str() -> Result<String, io::Error> {
    let mut buffer = String::new();
    let stdin = io::stdin();
    let mut handle = stdin.lock(); // Make handle mutable
    handle.read_line(&mut buffer)?;
    Ok(buffer)
}

fn get_int() -> i32 {
    let mut line = String::new();
    let stdin = io::stdin();
    let mut handle = stdin.lock(); // Make handle mutable
    handle.read_line(&mut line).unwrap();
    line.trim().parse::<i32>().unwrap_or(0)
}

fn get_int2() -> (i32, i32) {
    let mut line = String::new();
    let stdin = io::stdin();
    let mut handle = stdin.lock(); // Make handle mutable
    handle.read_line(&mut line).unwrap();
    let mut nums = line.trim().split_whitespace().map(|s| s.parse::<i32>().unwrap());
    (nums.next().unwrap_or(0), nums.next().unwrap_or(0))
}

fn main() {
    let (n, k) = get_int2();

    for pat in 0..(1 << n) {
        if pat.count_ones() != k as u32 {
            continue;
        }
        print!("{}:", pat);
        for i in 0..n {
            if pat & (1 << i) == 0 {
                continue;
            }
            print!(" {}", i);
        }
        println!();
    }
}
