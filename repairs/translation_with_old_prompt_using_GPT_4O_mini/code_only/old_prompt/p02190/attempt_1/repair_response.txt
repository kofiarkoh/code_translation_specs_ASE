```rust
use std::io::{self, BufRead};

fn comp(a: &i32, b: &i32) -> std::cmp::Ordering {
    // Emulating the C comparison function
    if a > b {
        std::cmp::Ordering::Greater
    } else if a == b {
        std::cmp::Ordering::Equal
    } else {
        std::cmp::Ordering::Less
    }
}

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    // Read the first line for n
    let n: usize = lines.next().unwrap().unwrap().trim().parse().unwrap();
    let mut a = Vec::with_capacity(n);

    // Read the next n lines for the array a
    for _ in 0..n {
        let value: i32 = lines.next().unwrap().unwrap().trim().parse().unwrap();
        a.push(value);
    }

    // Sort the array using Rust's sort_unstable, which is similar to qsort
    a.sort_unstable_by(comp);

    let mut ret = 1; // Start with 1 for the first unique element
    let mut prev = a[0];

    for &value in &a[1..] {
        if value != prev {
            ret += 1;
        }
        prev = value;
    }

    // Print the result
    println!("{}", ret);
}
```
