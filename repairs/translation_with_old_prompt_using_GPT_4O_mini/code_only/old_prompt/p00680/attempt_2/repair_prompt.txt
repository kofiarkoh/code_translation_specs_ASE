                if k > ep.cap {
                    k = ep.cap;
                }
                s = self.prevv[s];
            }
            *F -= k;

            ret += k as f64 * self.potential[T];
            let mut s = T;
            while s != S {
                let ep = &mut self.edge[self.prevv[s]][self.preve[s]];
                ep.cap -= k;
                self.edge[s][ep.rev].cap += k;
                s = self.prevv[s];
            }
        }
        ret
    }
}

fn gauss_jordan(n: usize) -> (Vec<f64>, bool) {
    let mut B = vec![vec![0.0; N]; N];
    let mut X = vec![0.0; N];

    for i in 0..n {
        for j in 0..n {
            B[i][j] = A[i][j];
        }
        B[i][n] = C[i] as f64;
    }

    for i in 0..n {
        let mut pivot = i;
        for j in i..n {
            if B[j][i].abs() > B[pivot][i].abs() {
                pivot = j;
            }
        }
        B.swap(i, pivot);
        if B[i][i].abs() < EPS {
            return (X, false); // error
        }

        for j in 0..=n {
            B[i][j] /= B[i][i];
        }
        for j in 0..n {
            if i != j {
                for k in i + 1..=n {
                    B[j][k] -= B[j][i] * B[i][k];
                }
            }
        }
    }
    for i in 0..n {
        X[i] = B[i][n];
    }
    (X, true)
}
