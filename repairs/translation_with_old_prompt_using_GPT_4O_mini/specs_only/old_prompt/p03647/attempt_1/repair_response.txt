use std::io::{self, BufRead};

fn main() {
    // Read input from standard input
    let stdin = io::stdin();
    let reader = stdin.lock();
    
    // Read the first line to get n
    let mut lines = reader.lines();
    let n: i32 = lines.next().unwrap().unwrap().parse().unwrap();

    // If n is zero or negative, output "IMPOSSIBLE"
    if n <= 0 {
        println!("IMPOSSIBLE");
        return;
    }

    // Read pairs of integers (a, b) and evaluate the condition
    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        let mut nums = line.split_whitespace();
        let a: i32 = nums.next().unwrap().parse().unwrap();
        let b: i32 = nums.next().unwrap().parse().unwrap();

        // Here we would evaluate the condition based on a and b
        // Since the specification does not provide a specific condition,
        // we will assume a simple check for demonstration purposes.
        // This can be modified based on actual conditions needed.
        if a + b > 0 { // Example condition
            println!("POSSIBLE");
        } else {
            println!("IMPOSSIBLE");
        }
    }
}
