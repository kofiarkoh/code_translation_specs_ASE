```rust
use std::io::{self, BufRead};

fn solve(n: usize, a: Vec<i64>, b: Vec<i64>) -> i32 {
    // Edge case: if n is 0, we should return "Yes 0"
    if n == 0 {
        println!("Yes 0");
        return 0;
    }

    // Dynamic programming tables
    let mut dp = vec![0; n];

    // Initialize dp based on the input pairs
    for i in 0..n {
        dp[i] = a[i] + b[i]; // Example computation, adjust as needed
    }

    // Calculate the total sum of dp
    let total_sum: i64 = dp.iter().sum();

    // Determine the output based on the total sum
    if total_sum > 0 {
        println!("Yes {}", total_sum);
        0
    } else {
        println!("No {}", total_sum);
        1
    }
}

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    // Read the number of pairs
    let n: usize = lines.next().unwrap().unwrap().trim().parse().unwrap();

    // Read the pairs into vectors
    let mut a = Vec::with_capacity(n);
    let mut b = Vec::with_capacity(n);
    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        let nums: Vec<i64> = line.split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();
        a.push(nums[0]);
        b.push(nums[1]);
    }

    // Call the solve function
    let _ = solve(n, a, b);
}
```
