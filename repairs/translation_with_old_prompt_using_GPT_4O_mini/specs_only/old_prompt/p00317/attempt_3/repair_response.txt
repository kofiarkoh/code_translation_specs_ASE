use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    // Read the number of words
    let mut lines = reader.lines();
    let n: usize = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse number of words");
    
    // Read the words into a vector
    let mut words: Vec<String> = Vec::with_capacity(n);
    for _ in 0..n {
        let word = lines.next().unwrap().unwrap();
        words.push(word);
    }

    // Sort the words for binary search
    words.sort();

    // Read the number of patterns
    let m: usize = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse number of patterns");

    // Process each pattern
    for _ in 0..m {
        let pattern = lines.next().unwrap().unwrap();
        let count = count_matching_words(&words, &pattern);
        println!("{}", count);
    }
}

// Function to count matching words based on the pattern
fn count_matching_words(words: &[String], pattern: &str) -> usize {
    let mut count = 0;

    for word in words {
        if matches_pattern(word, pattern) {
            count += 1;
        }
    }

    count
}

// Function to check if a word matches a given pattern with wildcards
fn matches_pattern(word: &str, pattern: &str) -> bool {
    let mut w_idx = 0;
    let mut p_idx = 0;
    let mut star_idx = None;
    let mut match_idx = 0;

    while w_idx < word.len() {
        if p_idx < pattern.len() && (pattern.as_bytes()[p_idx] == b'?' || pattern.as_bytes()[p_idx] == word.as_bytes()[w_idx]) {
            w_idx += 1;
            p_idx += 1;
        } else if p_idx < pattern.len() && pattern.as_bytes()[p_idx] == b'*' {
            star_idx = Some(p_idx);
            match_idx = w_idx;
            p_idx += 1;
        } else if let Some(star) = star_idx {
            p_idx = star + 1;
            w_idx = match_idx + 1;
            match_idx += 1;
        } else {
            return false;
        }
    }

    // Check for remaining characters in the pattern
    while p_idx < pattern.len() && pattern.as_bytes()[p_idx] == b'*' {
        p_idx += 1;
    }

    p_idx == pattern.len()
}
