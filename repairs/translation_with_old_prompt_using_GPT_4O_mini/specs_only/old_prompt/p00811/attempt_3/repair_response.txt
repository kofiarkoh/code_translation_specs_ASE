```rust
use std::io;

fn main() {
    let primes = generate_primes(100_000);
    
    loop {
        let (m, a, b) = read_input();
        if m == 0 && a == 0 && b == 0 {
            break; // Exit condition
        }
        
        let (p, q) = find_optimal_primes(&primes, m, a, b);
        println!("{} {}", p, q);
    }
}

// Function to generate all prime numbers up to a given limit using the Sieve of Eratosthenes
fn generate_primes(limit: usize) -> Vec<usize> {
    let mut is_prime = vec![true; limit + 1];
    is_prime[0] = false; // 0 is not prime
    is_prime[1] = false; // 1 is not prime

    for i in 2..=((limit as f64).sqrt() as usize) {
        if is_prime[i] {
            for j in (i * i..=limit).step_by(i) {
                is_prime[j] = false;
            }
        }
    }

    is_prime.iter()
        .enumerate()
        .filter_map(|(num, &is_prime)| if is_prime { Some(num) } else { None })
        .collect()
}

// Function to read input values
fn read_input() -> (usize, usize, usize) {
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let nums: Vec<usize> = input.split_whitespace()
        .filter_map(|s| s.parse().ok())
        .collect();
    
    match nums.as_slice() {
        [m, a, b] => (*m, *a, *b),
        _ => (0, 0, 0), // Default case if input is invalid
    }
}

// Function to find the optimal pair of primes based on the given constraints
fn find_optimal_primes(primes: &[usize], m: usize, a: usize, b: usize) -> (usize, usize) {
    let mut best_pair = (0, 0);
    let mut max_product = 0;

    // Iterate over primes for p and q based on the constraints
    for &p in primes.iter().filter(|&&p| p <= b) {
        for &q in primes.iter().filter(|&&q| q >= a) {
            let product = p * q;
            if product <= m && product > max_product {
                max_product = product;
                best_pair = (p, q);
            }
        }
    }

    // If no valid pair is found, return (0, 0)
    if max_product == 0 {
        return (0, 0);
    }

    best_pair // Return the best pair found
}
```
