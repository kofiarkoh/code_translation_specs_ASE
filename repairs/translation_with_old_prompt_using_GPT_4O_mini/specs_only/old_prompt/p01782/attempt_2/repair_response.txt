```rust
use std::collections::HashMap;
use std::io::{self, BufRead};

fn main() {
    // Read input from standard input
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read the number of rows in the character grid
    let n: usize = lines.next().unwrap().unwrap().parse().unwrap();
    
    // Ensure N is within the valid range
    if n == 0 || n > 52 {
        panic!("N must be a positive integer less than or equal to 52");
    }

    // Initialize a frequency map for characters
    let mut frequency: HashMap<char, usize> = HashMap::new();

    // Read the character grid and populate the frequency map
    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        if line.len() != 52 {
            panic!("Each row must contain exactly 52 characters");
        }
        
        for ch in line.chars() {
            // Increment the frequency of the character
            *frequency.entry(ch).or_insert(0) += 1;
        }
    }

    // Prepare the output string
    let mut output = String::new();

    // Collect uppercase letters first
    for ch in 'A'..='Z' {
        if let Some(&count) = frequency.get(&ch) {
            output.push_str(&ch.to_string().repeat(count));
        }
    }

    // Collect lowercase letters next
    for ch in 'a'..='z' {
        if let Some(&count) = frequency.get(&ch) {
            output.push_str(&ch.to_string().repeat(count));
        }
    }

    // Print the decoded message
    println!("{}", output);
}
```
