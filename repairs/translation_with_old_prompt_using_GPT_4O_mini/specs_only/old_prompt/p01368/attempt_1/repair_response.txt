```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    // Read the first line for N, M, L
    let mut first_line = String::new();
    reader.read_line(&mut first_line).unwrap();
    let mut params = first_line.split_whitespace()
        .map(|s| s.parse::<usize>().unwrap());
    
    let n = params.next().unwrap(); // Number of locations
    let m = params.next().unwrap(); // Number of edges
    let l = params.next().unwrap(); // Number of tasks

    // Initialize distance matrix
    let mut dist = vec![vec![usize::MAX; n]; n];
    for i in 0..n {
        dist[i][i] = 0; // Distance to self is zero
    }

    // Read edges
    for _ in 0..m {
        let mut edge_line = String::new();
        reader.read_line(&mut edge_line).unwrap();
        let mut edge_data = edge_line.split_whitespace()
            .map(|s| s.parse::<usize>().unwrap());
        
        let u = edge_data.next().unwrap(); // Start location
        let v = edge_data.next().unwrap(); // End location
        let d = edge_data.next().unwrap(); // Distance
        
        // Undirected graph
        dist[u][v] = dist[u][v].min(d);
        dist[v][u] = dist[v][u].min(d);
    }

    // Floyd-Warshall algorithm to compute shortest paths
    for k in 0..n {
        for i in 0..n {
            for j in 0..n {
                if dist[i][k] != usize::MAX && dist[k][j] != usize::MAX {
                    dist[i][j] = dist[i][j].min(dist[i][k] + dist[k][j]);
                }
            }
        }
    }

    // Read tasks
    let mut p = vec![0; l];
    let mut t = vec![0; l];
    for i in 0..l {
        let mut task_line = String::new();
        reader.read_line(&mut task_line).unwrap();
        let mut task_data = task_line.split_whitespace()
            .map(|s| s.parse::<usize>().unwrap());
        
        p[i] = task_data.next().unwrap(); // Starting location for task
        t[i] = task_data.next().unwrap(); // Starting time for task
    }

    // Bipartite matching setup
    let mut match_task = vec![None; l]; // Matches for tasks
    let mut seen = vec![false; l]; // Seen tasks in DFS

    // Function to perform DFS for bipartite matching
    fn can_match(
        task_index: usize,
        dist: &Vec<Vec<usize>>,
        p: &Vec<usize>,
        t: &Vec<usize>,
        match_task: &mut Vec<Option<usize>>,
        seen: &mut Vec<bool>,
    ) -> bool {
        for j in 0..match_task.len() {
            if !seen[j] && dist[p[task_index]][p[j]] != usize::MAX && dist[p[task_index]][p[j]] <= t[j] {
                seen[j] = true; // Mark this task as seen
                if match_task[j].is_none() || can_match(match_task[j].unwrap(), dist, p, t, match_task, seen) {
                    match_task[j] = Some(task_index);
                    return true;
                }
            }
        }
        false
    }

    // Count maximum matches
    let mut max_tasks = 0;
    for i in 0..l {
        seen.fill(false); // Reset seen for each task
        if can_match(i, &dist, &p, &t, &mut match_task, &mut seen) {
            max_tasks += 1; // Found a match
        }
    }

    // Output the result
    println!("{}", max_tasks);
}
```
