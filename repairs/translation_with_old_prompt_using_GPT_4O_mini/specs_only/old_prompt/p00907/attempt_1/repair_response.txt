```rust
use std::io::{self, BufRead};

fn main() {
    // Read input from standard input
    let stdin = io::stdin();
    let input = stdin.lock().lines().next().unwrap().unwrap();
    
    // Parse the input
    let mut values: Vec<f64> = input.split_whitespace()
        .filter_map(|s| s.parse::<f64>().ok())
        .collect();
    
    // The first value is `d`, the number of values excluding the outlier
    let d = values.remove(0) as usize;

    // Ensure d is within the valid range
    if d == 0 || d > 7 || values.len() != d + 3 {
        // If d is invalid, we do not process further
        return;
    }

    let n = d + 3; // Total number of values

    let mut outlier_index = n - 1; // Default to the last index
    let mut max_deviation = f64::MIN; // To track the maximum deviation

    // Calculate expected values and identify the outlier
    for i in 0..n {
        let mut sum = 0.0;
        let mut count = 0;

        // Calculate expected value based on other values
        for j in 0..n {
            if i != j {
                sum += values[j];
                count += 1;
            }
        }

        let expected_value = sum / count as f64; // Average of other values
        let deviation = (values[i] - expected_value).abs(); // Absolute deviation

        // Check if this deviation is the largest found
        if deviation > max_deviation {
            max_deviation = deviation;
            outlier_index = i; // Update the outlier index
        }
    }

    // Output the index of the outlier
    println!("{}", outlier_index);
}
```
