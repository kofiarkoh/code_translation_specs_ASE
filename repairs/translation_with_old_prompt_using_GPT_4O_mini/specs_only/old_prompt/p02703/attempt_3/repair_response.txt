```rust
use std::collections::BinaryHeap;
use std::io::{self, BufRead};

const MAX_CITIES: usize = 50;
const MAX_CONNECTIONS: usize = 49;

#[derive(Debug, Clone)] // Implement Clone for Connection
struct Connection {
    city: usize,
    cost: i32,
    time: i64,
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    let mut lines = reader.lines();
    let first_line = lines.next().unwrap().unwrap();
    let mut params = first_line.split_whitespace();
    
    let n: usize = params.next().unwrap().parse().unwrap(); // Number of cities
    let m: usize = params.next().unwrap().parse().unwrap(); // Number of train connections
    let s: i32 = params.next().unwrap().parse().unwrap(); // Initial budget

    let mut connections: Vec<Vec<Connection>> = vec![Vec::new(); n];

    for _ in 0..m {
        let line = lines.next().unwrap().unwrap();
        let mut conn_params = line.split_whitespace();
        let u: usize = conn_params.next().unwrap().parse().unwrap(); // City u
        let v: usize = conn_params.next().unwrap().parse().unwrap(); // City v
        let a: i32 = conn_params.next().unwrap().parse().unwrap(); // Cost
        let b: i64 = conn_params.next().unwrap().parse().unwrap(); // Travel time

        connections[u].push(Connection { city: v, cost: a, time: b });
        connections[v].push(Connection { city: u, cost: a, time: b }); // Assuming bidirectional
    }

    let mut ans = vec![vec![i64::MAX; (s + 1) as usize]; n]; // Minimum travel time for each city and budget
    let mut used = vec![vec![false; (s + 1) as usize]; n]; // Used budget states

    // Initialize the starting city (0) with zero time and budget
    for budget in 0..=s {
        ans[0][budget as usize] = 0;
    }

    let mut heap = BinaryHeap::new();
    heap.push((0, 0, s)); // (time, city, remaining_budget)

    while let Some((time, city, budget)) = heap.pop() {
        let time = -time; // Convert back to positive time

        if used[city][budget as usize] {
            continue;
        }
        used[city][budget as usize] = true;

        for conn in &connections[city] {
            let new_budget = budget - conn.cost;
            if new_budget >= 0 {
                let new_time = time + conn.time;
                if new_time < ans[conn.city][new_budget as usize] {
                    ans[conn.city][new_budget as usize] = new_time;
                    heap.push((-new_time, conn.city, new_budget));
                }
            }
        }
    }

    for city in 0..n {
        let min_time = ans[city].iter().cloned().min().unwrap_or(i64::MAX);
        if min_time == i64::MAX {
            println!("-1"); // If unreachable, print -1
        } else {
            println!("{}", min_time);
        }
    }
}
```
