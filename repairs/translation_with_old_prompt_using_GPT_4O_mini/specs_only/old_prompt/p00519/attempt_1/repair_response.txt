```rust
use std::collections::BinaryHeap;
use std::cmp::Ordering;
use std::io::{self, BufRead};

#[derive(Debug)]
struct Taxi {
    cost: u32,
    limit: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
struct Status {
    town: usize,
    remaining_rides: u32,
    cost: u32,
}

impl Ord for Status {
    fn cmp(&self, other: &Self) -> Ordering {
        // We want a min-heap, so we reverse the ordering
        other.cost.cmp(&self.cost)
    }
}

impl PartialOrd for Status {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    let mut lines = reader.lines();
    let first_line = lines.next().unwrap().unwrap();
    let mut params = first_line.split_whitespace();
    
    let n: usize = params.next().unwrap().parse().unwrap();
    let k: usize = params.next().unwrap().parse().unwrap();

    let mut taxis = Vec::with_capacity(n);
    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        let mut taxi_data = line.split_whitespace();
        let cost: u32 = taxi_data.next().unwrap().parse().unwrap();
        let limit: u32 = taxi_data.next().unwrap().parse().unwrap();
        taxis.push(Taxi { cost, limit });
    }

    let mut graph: Vec<Vec<(usize, u32)>> = vec![Vec::new(); n];
    for _ in 0..k {
        let line = lines.next().unwrap().unwrap();
        let mut connection_data = line.split_whitespace();
        let from: usize = connection_data.next().unwrap().parse().unwrap();
        let to: usize = connection_data.next().unwrap().parse().unwrap();
        let cost: u32 = connection_data.next().unwrap().parse().unwrap();
        graph[from].push((to, cost));
        graph[to].push((from, cost)); // Assuming undirected graph
    }

    let mut min_cost = vec![vec![u32::MAX; n]; n];
    let mut heap = BinaryHeap::new();

    // Start from town 0 with all rides available
    for rides in 0..=taxis[0].limit {
        min_cost[0][rides as usize] = 0;
        heap.push(Status { town: 0, remaining_rides: rides, cost: 0 });
    }

    while let Some(Status { town, remaining_rides, cost }) = heap.pop() {
        if town == n - 1 {
            println!("{}", cost);
            return;
        }

        if cost > min_cost[town][remaining_rides as usize] {
            continue;
        }

        // Explore neighbors
        for &(neighbor, edge_cost) in &graph[town] {
            let new_cost = cost + edge_cost;
            if new_cost < min_cost[neighbor][remaining_rides as usize] {
                min_cost[neighbor][remaining_rides as usize] = new_cost;
                heap.push(Status { town: neighbor, remaining_rides, cost: new_cost });
            }
        }

        // Use taxi service if rides are available
        if remaining_rides < taxis[town].limit {
            let new_rides = remaining_rides + 1;
            let new_cost = cost + taxis[town].cost;
            if new_cost < min_cost[town][new_rides as usize] {
                min_cost[town][new_rides as usize] = new_cost;
                heap.push(Status { town, remaining_rides: new_rides, cost: new_cost });
            }
        }
    }

    // If we exit the loop without finding a path to the last town, we do not specify output behavior
}
```
