```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read the number of height measurements
    let n: usize = match lines.next() {
        Some(Ok(line)) => line.trim().parse().unwrap_or(0),
        _ => 0,
    };

    // Ensure n is greater than 0
    if n == 0 {
        return; // Early exit if n is not valid
    }

    // Initialize counts for each height category
    let mut counts = [0; 6];

    // Read height measurements and categorize them
    for _ in 0..n {
        if let Some(Ok(line)) = lines.next() {
            if let Ok(height) = line.trim().parse::<f64>() {
                // Categorize heights into six ranges
                match height {
                    _ if height < 150.0 => counts[0] += 1, // Category 1
                    _ if height < 160.0 => counts[1] += 1, // Category 2
                    _ if height < 170.0 => counts[2] += 1, // Category 3
                    _ if height < 180.0 => counts[3] += 1, // Category 4
                    _ if height < 190.0 => counts[4] += 1, // Category 5
                    _ => counts[5] += 1,                    // Category 6
                }
            }
        }
    }

    // Print the histogram
    for (i, &count) in counts.iter().enumerate() {
        println!("{} {}", i + 1, "*".repeat(count));
    }
}
```
