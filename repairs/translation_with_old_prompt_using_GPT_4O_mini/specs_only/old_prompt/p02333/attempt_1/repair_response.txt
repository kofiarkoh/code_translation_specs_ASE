const MOD: i64 = 1_000_000_007; // Define the modulus as per the specification

// Global arrays for factorial and inverse factorial
static mut FACT: [i64; 2001] = [0; 2001];
static mut INVFACT: [i64; 2001] = [0; 2001];

// Function to compute factorials and their modular inverses
fn makefact(m: usize) {
    unsafe {
        FACT[0] = 1;
        for i in 1..=m {
            FACT[i] = FACT[i - 1] * i as i64 % MOD; // Compute factorial modulo MOD
        }
        INVFACT[m] = mod_inverse(FACT[m]); // Compute inverse of the last factorial
        for i in (1..=m).rev() {
            INVFACT[i - 1] = INVFACT[i] * i as i64 % MOD; // Compute inverses in reverse
        }
    }
}

// Function to compute modular inverse using Fermat's Little Theorem
fn mod_inverse(x: i64) -> i64 {
    let mut res = 1;
    let mut exp = MOD - 2; // MOD is prime, so we can use Fermat's theorem
    let mut base = x % MOD;
    while exp > 0 {
        if exp % 2 == 1 {
            res = res * base % MOD;
        }
        base = base * base % MOD;
        exp /= 2;
    }
    res
}

// Function to compute the combinatorial sum
fn compute_sum(n: usize, m: usize) -> i64 {
    let mut result = 0;
    for k in 0..=m {
        let binom = unsafe { (FACT[m] * INVFACT[k] % MOD) * INVFACT[m - k] % MOD }; // C(m, k)
        let power = mod_pow(k as i64, n as i64); // k^n
        result = (result + binom * power % MOD) % MOD; // Accumulate the result
    }
    result
}

// Function to compute modular exponentiation
fn mod_pow(base: i64, exp: i64) -> i64 {
    let mut res = 1;
    let mut b = base % MOD;
    let mut e = exp;
    while e > 0 {
        if e % 2 == 1 {
            res = res * b % MOD;
        }
        b = b * b % MOD;
        e /= 2;
    }
    res
}

// Main function to read input and output the result
fn main() {
    let n: usize = 3; // Example input for n
    let m: usize = 5; // Example input for m

    // Preconditions check
    if n < 0 || m < 0 || m > 2000 {
        panic!("Invalid input: n and m must be non-negative and m <= 2000");
    }

    unsafe {
        makefact(m); // Prepare factorials and inverses
    }

    let result = compute_sum(n, m); // Compute the sum
    println!("{}", result); // Output the result
}
