```rust
use std::io;

fn main() {
    // Read three integers from user input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let mut nums = input.split_whitespace()
        .filter_map(|s| s.parse::<i32>().ok())
        .collect::<Vec<i32>>();

    // Ensure we have exactly three integers
    if nums.len() != 3 {
        eprintln!("Expected three integers.");
        return;
    }

    let (a, b, mut c) = (nums[0], nums[1], nums[2]);

    // Calculate the difference
    let diff = a - b;

    // If the difference is positive, decrement c accordingly
    if diff > 0 {
        // Decrement c by diff, ensuring it does not go below zero
        c = if c >= diff { c - diff } else { 0 };
    }

    // Output the final value of c
    println!("{}", c);
}
```
