use std::collections::HashMap;

fn smallest_subarray_length(n: usize, k: usize, a: &[i32]) -> usize {
    // If N < K, immediately return 0 as no valid subarray can exist
    if n < k {
        return 0;
    }

    let mut count = HashMap::new();
    let mut left = 0;
    let mut min_length = n + 1; // Start with a value larger than any possible subarray length
    let mut unique_count = 0;

    for right in 0..n {
        // Include the current element in the window
        if a[right] >= 1 && a[right] <= k {
            let entry = count.entry(a[right]).or_insert(0);
            *entry += 1;
            if *entry == 1 {
                unique_count += 1; // We have found a new unique number in the range [1, K]
            }
        }

        // Try to shrink the window from the left as long as it contains all numbers from 1 to K
        while unique_count == k {
            // Update the minimum length found
            min_length = min_length.min(right - left + 1);

            // Remove the leftmost element from the window
            if a[left] >= 1 && a[left] <= k {
                let entry = count.entry(a[left]).or_insert(0);
                *entry -= 1;
                if *entry == 0 {
                    unique_count -= 1; // We lost a unique number from the range [1, K]
                }
            }
            left += 1; // Move the left pointer to the right
        }
    }

    // If min_length was updated, return it; otherwise, return 0
    if min_length == n + 1 {
        0
    } else {
        min_length
    }
}

fn main() {
    // Example usage
    let n = 7;
    let k = 3;
    let a = [1, 2, 1, 3, 2, 1, 3];
    let result = smallest_subarray_length(n, k, &a);
    println!("{}", result); // Output the result
}
