```rust
use std::collections::BinaryHeap;
use std::cmp::Ordering;

#[derive(Debug, Clone, Copy, Eq, PartialEq)]
struct Point {
    x: usize,
    y: usize,
}

#[derive(Debug, Clone, Copy, Eq, PartialEq)]
struct State {
    cost: u64,
    position: Point,
}

// Implementing Ord and PartialOrd for State to use in the priority queue
impl Ord for State {
    fn cmp(&self, other: &Self) -> Ordering {
        // Reverse ordering for min-heap
        other.cost.cmp(&self.cost)
    }
}

impl PartialOrd for State {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

fn dijkstra(m: usize, n: usize, k: usize, obstacles: &[(usize, usize)]) -> i64 {
    // Create a grid to mark obstacles
    let mut grid = vec![vec![false; n]; m];
    for &(x, y) in obstacles {
        grid[x - 1][y - 1] = true; // Marking obstacles (1-indexed to 0-indexed)
    }

    // Directions for moving in the grid (up, down, left, right)
    let directions = vec![
        Point { x: 0, y: 1 },  // right
        Point { x: 1, y: 0 },  // down
        Point { x: 0, y: usize::MAX }, // left (will be handled in bounds check)
        Point { x: usize::MAX, y: 0 }, // up (will be handled in bounds check)
    ];

    let start = Point { x: 0, y: 0 }; // Starting point (0,0)
    let goal = Point { x: m - 1, y: n - 1 }; // Goal point (m-1,n-1)

    // Priority queue for Dijkstra's algorithm
    let mut heap = BinaryHeap::new();
    let mut distances = vec![vec![u64::MAX; n]; m];
    distances[start.x][start.y] = 0;
    heap.push(State { cost: 0, position: start });

    while let Some(State { cost, position }) = heap.pop() {
        if position == goal {
            return cost as i64; // Return the cost if we reached the goal
        }

        // If the cost is greater than the recorded distance, skip
        if cost > distances[position.x][position.y] {
            continue;
        }

        // Explore neighbors
        for direction in &directions {
            let new_x = position.x.wrapping_add(direction.x);
            let new_y = position.y.wrapping_add(direction.y);

            // Check bounds and if the new position is an obstacle
            if new_x < m && new_y < n && !grid[new_x][new_y] {
                let new_cost = cost + 1; // Assuming each step has a cost of 1
                if new_cost < distances[new_x][new_y] {
                    distances[new_x][new_y] = new_cost;
                    heap.push(State { cost: new_cost, position: Point { x: new_x, y: new_y } });
                }
            }
        }
    }

    -1 // Return -1 if the goal is unreachable
}

fn main() {
    let m = 5; // Example grid dimensions
    let n = 5;
    let k = 2; // Number of obstacles
    let obstacles = vec![(2, 2), (3, 3)]; // Example obstacles

    let result = dijkstra(m, n, k, &obstacles);
    println!("{}", result); // Output the result
}
```
