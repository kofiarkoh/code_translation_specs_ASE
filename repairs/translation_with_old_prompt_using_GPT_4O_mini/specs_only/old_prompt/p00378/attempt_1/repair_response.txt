```rust
use std::io::{self, Write};

fn main() {
    // Read inputs A, B, and X from standard input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let inputs: Vec<i32> = input
        .trim()
        .split_whitespace()
        .map(|s| s.parse().expect("Invalid input"))
        .collect();

    // Unpack inputs, assuming they are provided in the correct order
    let (a, b, x) = (inputs[0], inputs[1], inputs[2]);

    // Calculate the total cost based on the given conditions
    let total_cost = if x == 0 {
        0 // If X is zero, the cost is zero
    } else if a > b {
        a * x // If A is greater than B, use A for cost calculation
    } else {
        b * x // Otherwise, use B for cost calculation
    };

    // Print the result
    println!("{}", total_cost);
}
```
