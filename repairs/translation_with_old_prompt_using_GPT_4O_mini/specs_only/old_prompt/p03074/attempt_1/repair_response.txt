```rust
use std::io::{self, Write};

fn main() {
    // Example input values; in a real scenario, these would be read from user input or another source.
    let num: usize = 10; // Length of the binary string
    let changes: usize = 2; // Number of '0's that can be changed to '1's
    let binary_string: &str = "1001101001"; // The binary string

    // Check for edge case where num is 0
    if num == 0 {
        println!("0");
        return;
    }

    // Initialize arrays to store the segments of '1's and cumulative counts
    let mut arr = vec![0; num];
    let mut cum = vec![0; num];

    // Process the binary string to populate arr and cum
    let mut count = 0;
    for (i, c) in binary_string.chars().enumerate() {
        if c == '1' {
            count += 1;
            arr[i] = count;
        } else {
            count = 0;
            arr[i] = 0;
        }
        cum[i] = if i == 0 { arr[i] } else { cum[i - 1] + arr[i] };
    }

    // Calculate the maximum length of contiguous '1's after allowed changes
    let mut max_length = 0;
    let mut zero_count = 0;
    let mut left = 0;

    for right in 0..num {
        if binary_string.chars().nth(right).unwrap() == '0' {
            zero_count += 1;
        }

        // If we exceed the allowed changes, move the left pointer
        while zero_count > changes {
            if binary_string.chars().nth(left).unwrap() == '0' {
                zero_count -= 1;
            }
            left += 1;
        }

        // Calculate the length of the current window
        max_length = max_length.max(right - left + 1);
    }

    // Output the maximum length of contiguous '1's
    println!("{}", max_length);
}
```
