```rust
use std::io::{self, BufRead};

fn main() {
    // Read input from standard input
    let stdin = io::stdin();
    let input = stdin.lock().lines().next().unwrap().unwrap();
    let parts: Vec<&str> = input.split_whitespace().collect();
    
    let num: usize = parts[0].parse().unwrap(); // Length of the binary string
    let changes: usize = parts[1].parse().unwrap(); // Number of '0's that can be changed to '1's
    let binary_string: &str = &parts[2]; // The binary string

    // Check for edge case where num is 0
    if num == 0 {
        println!("0");
        return;
    }

    // Calculate the maximum length of contiguous '1's after allowed changes
    let mut max_length = 0;
    let mut zero_count = 0;
    let mut left = 0;

    for right in 0..num {
        if binary_string.chars().nth(right).unwrap() == '0' {
            zero_count += 1;
        }

        // If we exceed the allowed changes, move the left pointer
        while zero_count > changes {
            if binary_string.chars().nth(left).unwrap() == '0' {
                zero_count -= 1;
            }
            left += 1;
        }

        // Calculate the length of the current window
        max_length = max_length.max(right - left + 1);
    }

    // Output the maximum length of contiguous '1's
    println!("{}", max_length);
}
```
