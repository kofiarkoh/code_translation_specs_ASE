use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let input: Vec<String> = stdin.lock().lines()
        .filter_map(Result::ok)
        .collect();

    if input.len() != 3 {
        // Return error code 1 if input does not consist of exactly three strings
        std::process::exit(1);
    }

    let a = &input[0];
    let b = &input[1];
    let c = &input[2];

    let count = count_valid_sequences(a, b, c);
    println!("{}", count);
}

fn count_valid_sequences(a: &str, b: &str, c: &str) -> usize {
    let max_len = b.len();
    let a_padded = pad_with_zeros(a, max_len);
    let b_padded = b.to_string();
    let c_len = c.len();

    // Initialize memoization table
    let mut memo = vec![vec![-1; c_len + 1]; max_len + 1];

    // Start the recursive counting
    count_sequences_helper(0, 0, &a_padded, &b_padded, c, &mut memo)
}

fn pad_with_zeros(s: &str, target_len: usize) -> String {
    let mut padded = s.to_string();
    while padded.len() < target_len {
        padded.insert(0, '0');
    }
    padded
}

fn count_sequences_helper(pos: usize, c_pos: usize, a: &str, b: &str, c: &str, memo: &mut Vec<Vec<i32>>) -> usize {
    if c_pos == c.len() {
        return 1; // Found a valid sequence
    }
    if pos == a.len() {
        return 0; // Reached the end of the range
    }
    if memo[pos][c_pos] != -1 {
        return memo[pos][c_pos] as usize; // Return cached result
    }

    let mut count = 0;
    let start_digit = a.chars().nth(pos).unwrap();
    let end_digit = b.chars().nth(pos).unwrap();
    let target_digit = c.chars().nth(c_pos).unwrap();

    for digit in (start_digit..=end_digit) {
        if digit == target_digit {
            count += count_sequences_helper(pos + 1, c_pos + 1, a, b, c, memo);
        } else if digit > target_digit {
            // If digit is greater than target_digit, we can only count sequences that start with this digit
            count += count_sequences_helper(pos + 1, c_pos, a, b, c, memo);
        } else {
            // If digit is less than target_digit, we can count all sequences from this position
            count += count_sequences_helper(pos + 1, c_pos, a, b, c, memo);
        }
    }

    memo[pos][c_pos] = count as i32; // Cache the result
    count
}
