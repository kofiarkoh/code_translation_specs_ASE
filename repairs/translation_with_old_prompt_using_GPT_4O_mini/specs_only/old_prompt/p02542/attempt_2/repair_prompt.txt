    adj: Vec<Vec<usize>>, // adjacency list
    edges: Vec<Edge>,
}

impl Graph {
    fn new(n: usize) -> Self {
        Graph {
            adj: vec![Vec::new(); n],
            edges: Vec::new(),
        }
    }

    fn add_edge(&mut self, from: usize, to: usize, capacity: i64, cost: i64) {
        let edge_index = self.edges.len();
        self.edges.push(Edge { to, capacity, flow: 0, cost });

fn successive_shortest_path(graph: &mut Graph, source: usize, sink: usize, flow: i64) -> i64 {
    let n = graph.adj.len();
    let mut dist = vec![i64::MAX; n];
    let mut prev = vec![None; n];
    let mut heap = BinaryHeap::new();
    
    dist[source] = 0;
    heap.push(State { cost: 0, node: source, flow });

        if node == sink {
