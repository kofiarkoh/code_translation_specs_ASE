```rust
use std::io::{self, Write};

fn main() {
    // Read input from standard input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");

    // Split the input into two parts and parse them as i64 (equivalent to long int)
    let nums: Vec<i64> = input
        .split_whitespace()
        .filter_map(|s| s.parse().ok())
        .collect();

    // Ensure we have exactly two numbers
    if nums.len() != 2 {
        // Undefined behavior on invalid input; we do not handle it as per specification
        return;
    }

    let a = nums[0];
    let b = nums[1];

    // Calculate the sum of a and b
    let sum = a + b;

    // Check if the sum is even
    if sum % 2 == 0 {
        // Calculate the average
        let average = sum / 2;
        // Print the average
        println!("{}", average);
    } else {
        // Print "IMPOSSIBLE" if the sum is odd
        println!("IMPOSSIBLE");
    }

    // Return 0 upon successful execution
    std::process::exit(0);
}
```
