fn welsh_powell(segments: &[Segment]) -> usize {
    let mut graph: HashMap<usize, HashSet<usize>> = HashMap::new();

            if i != j && seg1.intersects(seg2) {
                graph.entry(i).or_insert_with(HashSet::new).insert(j);

            }

    let mut colors = vec![None; segments.len()];
    let mut available_colors = vec![true; segments.len()];

        available_colors.fill(true);
