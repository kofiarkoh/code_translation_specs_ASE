```rust
use std::io::{self, BufRead};

#[derive(Debug)]
struct Rectangle {
    x1: i32,
    y1: i32,
    x2: i32,
    y2: i32,
    used: char, // '0' for unused, '1' for used
}

impl Rectangle {
    fn new(x1: i32, y1: i32, x2: i32, y2: i32) -> Self {
        Rectangle {
            x1,
            y1,
            x2,
            y2,
            used: '0', // Initially unused
        }
    }

    // Check if this rectangle overlaps with another
    fn overlaps(&self, other: &Rectangle) -> bool {
        self.x1 < other.x2 && self.x2 > other.x1 && self.y1 < other.y2 && self.y2 > other.y1
    }
}

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut rectangles: Vec<Rectangle> = Vec::new();

    // Read number of rectangles
    let mut n = String::new();
    handle.read_line(&mut n).unwrap();
    let n: usize = n.trim().parse().expect("Invalid number of rectangles");

    // Read rectangle coordinates
    for _ in 0..n {
        let mut coords = String::new();
        handle.read_line(&mut coords).unwrap();
        let coords: Vec<i32> = coords
            .trim()
            .split_whitespace()
            .map(|s| s.parse().expect("Invalid coordinate"))
            .collect();
        rectangles.push(Rectangle::new(coords[0], coords[1], coords[2], coords[3]));
    }

    // Sort rectangles based on their x1 coordinate
    rectangles.sort_by_key(|r| r.x1);

    // Process rectangles and print their order
    for rect in &mut rectangles {
        if rect.used == '0' { // Only process unused rectangles
            println!("{:?}", rect);
            rect.used = '1'; // Mark as used
        }
    }
}
```
