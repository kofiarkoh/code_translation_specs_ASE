```rust
use std::io;

fn main() {
    // Read three integers from the user
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    
    // Split the input into parts and parse them as integers
    let mut parts = input.split_whitespace();
    let a: i32 = parts.next().unwrap().parse().expect("Invalid input for base");
    let b: i32 = parts.next().unwrap().parse().expect("Invalid input for height");
    let _c: i32 = parts.next().unwrap().parse().expect("Invalid input for unused integer");

    // Calculate the area of the triangle
    // Area = (base * height) / 2
    // We use integer division, which automatically rounds down
    let area = (a * b) / 2;

    // Print the area as a rounded integer
    println!("{}", area);

    // Return 0 upon successful execution
    std::process::exit(0);
}
```
