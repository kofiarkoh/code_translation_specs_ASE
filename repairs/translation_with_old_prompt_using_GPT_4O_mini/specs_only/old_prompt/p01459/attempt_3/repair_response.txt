use std::collections::VecDeque;

const MAX_SIZE: usize = 103;

fn bfs_pathfinding(H: usize, W: usize, A: usize, B: usize, map: [[char; MAX_SIZE]; MAX_SIZE]) -> i32 {
    // Check preconditions
    if H == 0 || W == 0 || A == 0 || B == 0 {
        return -1;
    }

    let mut mk = [[[[false; 11]; 11]; 4]; MAX_SIZE]; // visited states
    let mut start = (0, 0);
    let mut goal = (0, 0);

    // Locate 'S' and 'G' in the grid
    for r in 0..H {
        for c in 0..W {
            if map[r][c] == 'S' {
                start = (r, c);
            } else if map[r][c] == 'G' {
                goal = (r, c);
            }
        }
    }

    // Check if 'S' is at the bottom row or adjacent to a wall below
    if start.0 == H - 1 || (start.0 < H - 1 && map[start.0 + 1][start.1] == '#') {
        return -1;
    }

    let directions = [(1, 0), (0, 1), (-1, 0), (0, -1)]; // down, right, up, left
    let mut que = VecDeque::new();
    que.push_back((start.0, start.1, 0, 0)); // (row, col, moves in A, moves in B)

    while let Some((r, c, a, b)) = que.pop_front() {
        // Check if we reached the goal
        if (r, c) == goal {
            return (a + b) as i32;
        }

        // Explore all directions
        for (d, (dr, dc)) in directions.iter().enumerate() {
            let mut nr = r;
            let mut nc = c;
            let mut na = a;
            let mut nb = b;

            // Move in the current direction up to A or B times
            let max_moves = if d % 2 == 0 { A } else { B };
            for _ in 0..max_moves {
                nr = (nr as isize + *dr as isize) as usize;
                nc = (nc as isize + *dc as isize) as usize;

                // Check bounds and obstacles
                if nr >= H || nc >= W || map[nr][nc] == '#' {
                    break;
                }

                // Update moves count
                if d % 2 == 0 {
                    na += 1;
                } else {
                    nb += 1;
                }

                // Ensure we do not exceed the bounds of the mk array
                if na < 11 && nb < 11 && !mk[nr][nc][na][nb] {
                    mk[nr][nc][na][nb] = true;
                    que.push_back((nr, nc, na, nb));
                }
            }
        }
    }

    // If no path was found
    -1
}

// Example usage
fn main() {
    let H = 5;
    let W = 5;
    let A = 2;
    let B = 2;
    let mut map = [['#'; MAX_SIZE]; MAX_SIZE]; // Initialize the map with default values
    map[1][1] = 'S';
    map[3][3] = 'G';
    map[1][2] = '.';
    map[1][3] = '.';
    map[2][1] = '.';
    map[3][2] = '.';

    let result = bfs_pathfinding(H, W, A, B, map);
    println!("{}", result); // Expected output: minimum moves or -1
}
