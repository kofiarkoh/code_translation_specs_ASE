```rust
use std::io::{self, Write};

fn main() {
    // Read five integers from standard input
    let mut inputs = String::new();
    io::stdin().read_line(&mut inputs).expect("Failed to read line");

    // Split the input string into parts and parse them as integers
    let numbers: Vec<i32> = inputs
        .split_whitespace()
        .map(|s| s.parse().expect("Failed to parse integer"))
        .collect();

    // Ensure exactly five integers are provided
    if numbers.len() != 5 {
        // Undefined behavior as per specification; we do nothing
        return;
    }

    // Check each number for zero and print the corresponding index (1-5)
    for (index, &number) in numbers.iter().enumerate() {
        if number == 0 {
            // Print the 1-based index of the first zero found
            println!("{}", index + 1);
            return; // Exit after printing the first zero's index
        }
    }
    // If no zeros are found, nothing is printed as per specification
}
```
