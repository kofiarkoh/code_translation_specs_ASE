```rust
use std::io::{self, BufRead};

const MAX: usize = 100001;

#[derive(Debug)]
struct Rect {
    height: i64,
    pos: usize,
}

fn large_histogram(w: usize, table: &mut [i64; MAX]) -> i64 {
    let mut sp = 0; // stack pointer
    let mut max_area = 0;
    table[w] = 0; // Set the sentinel value
    let mut stack: Vec<Rect> = Vec::new(); // Use a Vec for dynamic sizing

    for i in 0..=w {
        let r = Rect {
            height: table[i],
            pos: i,
        };

        if sp == 0 || (stack.last().unwrap().height < r.height) {
            stack.push(r);
            sp += 1;
        } else if stack.last().unwrap().height > r.height {
            let mut target = i;

            while sp > 0 && stack.last().unwrap().height >= r.height {
                let pre = stack.pop().unwrap(); // Pop from stack
                sp -= 1; // Decrement stack pointer
                let area = pre.height * (i as i64 - pre.pos as i64);
                if max_area < area {
                    max_area = area;
                }
                target = pre.pos;
            }

            r.pos = target;
            stack.push(r);
            sp += 1;
        }
    }

    max_area
}

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    let w: usize = lines.next().unwrap().unwrap().parse().unwrap();
    let mut table: [i64; MAX] = [0; MAX];

    for i in 0..w {
        table[i] = lines.next().unwrap().unwrap().parse().unwrap();
    }

    let max_area = large_histogram(w, &mut table);
    println!("{}", max_area);
}
```
