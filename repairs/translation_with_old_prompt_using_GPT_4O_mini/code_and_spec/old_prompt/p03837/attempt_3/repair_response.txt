```rust
use std::io::{self, BufRead};

fn min(a: i32, b: i32) -> i32 {
    if a < b {
        a
    } else {
        b
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    let mut lines = reader.lines();
    let first_line = lines.next().unwrap().unwrap();
    let mut nums = first_line.split_whitespace().map(|s| s.parse::<i32>().unwrap());
    let n = nums.next().unwrap() as usize; // Number of vertices
    let m = nums.next().unwrap() as usize; // Number of edges

    let mut a = vec![0; m];
    let mut b = vec![0; m];
    let mut c = vec![0; m];
    let mut dist = vec![vec![i32::MAX; n]; n];

    for i in 0..m {
        let line = lines.next().unwrap().unwrap();
        let mut edge_data = line.split_whitespace().map(|s| s.parse::<i32>().unwrap());
        a[i] = edge_data.next().unwrap() - 1; // Convert to 0-based index
        b[i] = edge_data.next().unwrap() - 1; // Convert to 0-based index
        c[i] = edge_data.next().unwrap();
    }

    for i in 0..n {
        dist[i][i] = 0; // Distance to self is 0
    }

    for i in 0..m {
        dist[a[i] as usize][b[i] as usize] = min(dist[a[i] as usize][b[i] as usize], c[i]);
        dist[b[i] as usize][a[i] as usize] = min(dist[b[i] as usize][a[i] as usize], c[i]);
    }

    for k in 0..n {
        for i in 0..n {
            for j in 0..n {
                dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]);
            }
        }
    }

    let mut ans = m as i32;

    for i in 0..m {
        let mut f = true;
        for j in 0..n {
            if dist[j][a[i] as usize] + c[i] == dist[j][b[i] as usize] {
                f = false;
                break;
            }
        }
        if !f {
            ans -= 1;
        }
    }

    println!("{}", ans);
}
```
