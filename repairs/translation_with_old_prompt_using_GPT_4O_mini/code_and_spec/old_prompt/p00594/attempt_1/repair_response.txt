```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut input = stdin.lock().lines();

    while let Some(line) = input.next() {
        let n: usize = line.unwrap().trim().parse().expect("Invalid input");
        if n == 0 {
            break;
        }

        let mut a: Vec<i32> = Vec::with_capacity(n);
        for _ in 0..n {
            let num: i32 = input.next().unwrap().unwrap().trim().parse().expect("Invalid input");
            a.push(num);
        }

        // Sort the array
        a.sort_unstable();

        let mut m = 0; // Length of the longest sequence of identical integers
        let mut k = 0; // The integer that has the longest sequence
        let mut i = 0;

        while i < n {
            let mut j = i + 1;
            while j < n && a[j] == a[i] {
                j += 1;
            }
            if m < j - i {
                m = j - i;
                k = a[i];
            }
            i = j; // Move to the next different integer
        }

        if m > n / 2 {
            println!("{}", k);
        } else {
            println!("NO COLOR");
        }
    }
}
```
