```rust
use std::io::{self, BufRead};

fn min(a: f64, b: f64) -> f64 {
    if a < b { a } else { b }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read n and m
    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let n: usize = parts.next().unwrap().parse().unwrap();
    let m: usize = parts.next().unwrap().parse().unwrap();

    // Initialize arrays
    let mut d = vec![f64::INFINITY; n + 1];
    let mut e = vec![usize::MAX; n + 1]; // Use usize::MAX to represent -1
    let mut c = vec![vec![0; n + 1]; n + 1];

    d[n] = 0.0;

    // Read edges
    for _ in 0..m {
        let edge_line = lines.next().unwrap().unwrap();
        let mut edge_parts = edge_line.split_whitespace();
        let i: usize = edge_parts.next().unwrap().parse().unwrap();
        let j: usize = edge_parts.next().unwrap().parse().unwrap();
        c[i][j] = 1;
    }

    // Calculate average distances
    for i in (1..n).rev() {
        let mut s = 0.0;
        let mut a = 0;
        let mut b = 0;

        for j in i..=n {
            if c[i][j] != 0 {
                s += d[j] + 1.0;
                if b == 0 || d[b] < d[j] {
                    b = j;
                }
                a += 1;
            }
        }

        d[i] = s / a as f64;
        if a > 1 {
            e[i] = b;
        }
    }

    let mut min = d[1];

    // Evaluate the effect of removing each edge
    for k in 1..=n {
        if e[k] == usize::MAX {
            continue;
        }

        // Reset distances
        for j in 0..=n {
            d[j] = f64::INFINITY;
        }
        d[n] = 0.0;

        for i in (1..n).rev() {
            let mut s = 0.0;
            let mut a = 0;

            for j in i..=n {
                if c[i][j] == 0 || (i == k && j == e[k]) {
                    continue;
                }
                s += d[j] + 1.0;
                a += 1;
            }

            d[i] = s / a as f64;
        }

        if min > d[1] {
            min = d[1];
        }
    }

    // Print the result formatted to ten decimal places
    println!("{:.10}", min);
}
```
