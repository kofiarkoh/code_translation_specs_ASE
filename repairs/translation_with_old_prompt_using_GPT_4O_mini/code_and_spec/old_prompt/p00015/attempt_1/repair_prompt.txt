void convert(char charNum[], int intNum[]) {
    int i, j = 0;
    for (i = (int)strlen(charNum) - 1; i >= 0 ; i --) {
        intNum[j] = (int)charNum[i] - 48;
        j ++;
    }
}

        count ++;
