use std::io::{self, BufRead};

const MAX: usize = 602;
const MAX_ROWS: usize = 60;
const MAX_COLS: usize = 12;

struct UnionFind {
    parent: [usize; MAX],
    rank: [usize; MAX],
}

impl UnionFind {
    fn new() -> Self {
        let mut parent = [0; MAX];
        let rank = [0; MAX];
        for i in 0..MAX {
            parent[i] = i;
        }
        UnionFind { parent, rank }
    }

    fn find(&mut self, x: usize) -> usize {
        if x != self.parent[x] {
            self.parent[x] = self.find(self.parent[x]);
        }
        self.parent[x]
    }

    fn union(&mut self, x: usize, y: usize) {
        let root_x = self.find(x);
        let root_y = self.find(y);
        if root_x != root_y {
            if self.rank[root_x] > self.rank[root_y] {
                self.parent[root_y] = root_x;
            } else {
                self.parent[root_x] = root_y;
                if self.rank[root_x] == self.rank[root_y] {
                    self.rank[root_y] += 1;
                }
            }
        }
    }
}

#[derive(Default, Clone)]
struct TreeNode {
    used: bool,
    m: usize,
    mx: usize,
    xl: isize,
    xr: isize,
    par: usize,
    ch: usize,
    sib: usize,
}

struct StabilityChecker {
    tbl: Vec<TreeNode>,
    tr: [isize; MAX],
    rel: [[usize; MAX_COLS]; MAX_ROWS],
    map: [[char; MAX_COLS]; MAX_ROWS],
    w: usize,
    h: usize,
    sz: usize,
}

impl StabilityChecker {
    fn new(w: usize, h: usize) -> Self {
        StabilityChecker {
            tbl: vec![TreeNode::default(); 100],
            tr: [-1; MAX],
            rel: [[0; MAX_COLS]; MAX_ROWS],
            map: [['.'; MAX_COLS]; MAX_ROWS],
            w,
            h,
            sz: 1,
        }
    }

    fn mktree(&mut self, r: usize, c: usize, this: usize) {
        if r == self.h - 1 {
            if self.map[r][c] == '.' {
                return;
            }
            if self.tbl[this].xl < 0 {
                self.tbl[this].xl = c as isize;
            } else if c as isize < self.tbl[this].xl {
                self.tbl[this].xl = c as isize;
            }
            if c > self.tbl[this].xr as usize {
                self.tbl[this].xr = c as isize;
            }
            return;
        }
        if self.map[r + 1][c] == '.' {
            return;
        }

        let k = self.rel[r + 1][c];
        let p = self.tr[k];
        if p as usize == this {
            return;
        }

        if self.tbl[this].xl < 0 {
            self.tbl[this].xl = c as isize;
        } else if c as isize < self.tbl[this].xl {
            self.tbl[this].xl = c as isize;
        }
        if c > self.tbl[this].xr as usize {
            self.tbl[this].xr = c as isize;
        }
        if self.tbl[this].par != 0 {
            return;
        }

        if self.tbl[p as usize].ch == 0 {
            self.tbl[p as usize].ch = this;
        } else {
            let mut s = self.tbl[p as usize].ch;
            while self.tbl[s].sib != 0 {
                s = self.tbl[s].sib;
            }
            self.tbl[s].sib = this;
        }
        self.tbl[this].par = p as usize;
    }

    fn check(&mut self, this: usize) -> bool {
        if self.tbl[this].used {
            return true;
        }
        let ch = self.tbl[this].ch;
        if ch != 0 && !self.check(ch) {
            return false;
        }

        let mut sib = self.tbl[this].sib;
        while sib != 0 {
            if !self.check(sib) {
                return false;
            }
            sib = self.tbl[sib].sib;
        }

        let x = 0.5 + self.tbl[this].mx as f64 / self.tbl[this].m as f64;
        if x <= self.tbl[this].xl as f64 || self.tbl[this].xr as f64 + 1.0 <= x {
            return false;
        }

        let par = self.tbl[this].par;
        self.tbl[par].m += self.tbl[this].m;
        self.tbl[par].mx += self.tbl[this].mx;
        self.tbl[this].used = true;
        true
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    while let Some(line) = lines.next() {
        let line = line.unwrap();
        let mut dimensions = line.split_whitespace();
        let w: usize = dimensions.next().unwrap().parse().unwrap();
        let h: usize = dimensions.next().unwrap().parse().unwrap();

        if w == 0 {
            break;
        }

        let mut union_find = UnionFind::new();
        let mut stability_checker = StabilityChecker::new(w, h);

        for r in 0..h {
            let grid_line = lines.next().unwrap().unwrap();
            stability_checker.map[r] = grid_line.chars().collect::<Vec<_>>().try_into().unwrap();
            for c in 0..w {
                let i = r * w + c;
                stability_checker.rel[r][c] = i;
                union_find.parent[i] = i;
            }
        }

        let mv: [(isize, isize); 4] = [(-1, 0), (0, 1), (1, 0), (0, -1)];
        for r in 0..h {
            for c in 0..w {
                if stability_checker.map[r][c] == '.' {
                    continue;
                }
                for (dr, dc) in mv.iter() {
                    let r2 = r as isize + dr;
                    let c2 = c as isize + dc;
                    if r2 >= 0 && r2 < h as isize && c2 >= 0 && c2 < w as isize {
                        if stability_checker.map[r2 as usize][c2 as usize] == stability_checker.map[r][c] {
                            union_find.union(stability_checker.rel[r][c], stability_checker.rel[r2 as usize][c2 as usize]);
                        }
                    }
                }
            }
        }

        stability_checker.sz = 1;
        stability_checker.tbl.iter_mut().for_each(|node| *node = TreeNode::default());
        stability_checker.tr.iter_mut().for_each(|tr| *tr = -1);

        for r in (0..h).rev() {
            for c in 0..w {
                if stability_checker.map[r][c] == '.' {
                    continue;
                }
                let k = stability_checker.rel[r][c] = union_find.find(stability_checker.rel[r][c]);
                let i = if stability_checker.tr[k] < 0 {
                    let idx = stability_checker.sz;
                    stability_checker.sz += 1;
                    stability_checker.tr[k] = idx as isize;
                    stability_checker.tbl[idx].xl = -1;
                    idx
                } else {
                    stability_checker.tr[k] as usize
                };
                stability_checker.tbl[i].m += 1;
                stability_checker.tbl[i].mx += c;
                stability_checker.mktree(r, c, i);
            }
        }
        stability_checker.tbl[1].par = 1;

        println!("{}", if stability_checker.check(1) { "STABLE" } else { "UNSTABLE" });
    }
}
