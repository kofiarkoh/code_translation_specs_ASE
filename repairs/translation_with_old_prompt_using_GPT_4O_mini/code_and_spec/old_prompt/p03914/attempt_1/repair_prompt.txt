#define Mod 1000000007

int main()

	scanf("%d %d", &N, &M);

	long long dp[302][302] = {}, tmp;
	for (k = 1, dp[1][1] = 1; k <= M; k++) {
		for (i = k; i >= 1; i--) {
			for (j = i; j >= 1; j--) {
				tmp = dp[i][j] % Mod;
				dp[i][j] = 0;

				dp[i][i] += tmp * j;
