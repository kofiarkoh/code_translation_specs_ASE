	for (i = 1; i <= N; i++) {
		sum += A[i];
		diff[i] = A[i] - A[i-1];
	}
	if (sum % (N * (N + 1) / 2) != 0) printf("NO\n");
	else {
		sum /= N * (N + 1) / 2;
		for (i = 1, diff[0] = 0; i <= N; i++) {
			diff[i] = (sum - diff[i]) / N;
			if (diff[i] < 0 || diff[i] > sum) break;
