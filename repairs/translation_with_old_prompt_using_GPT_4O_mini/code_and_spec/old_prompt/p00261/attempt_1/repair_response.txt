```rust
use std::io::{self, BufRead};

const A: usize = 0;
const X: usize = 1;
const Y: usize = 2;
const Z: usize = 3;
const W: usize = 4;
const B: usize = 5;
const NA: usize = 6;

const MAP: [[usize; 2]; 6] = [
    [X, Y],
    [NA, Z],
    [X, NA],
    [W, B],
    [B, Y],
    [Y, X],
];

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let path = line.unwrap();
        if path.chars().next() == Some('#') {
            break;
        }

        let mut city = A;
        let mut valid = true;

        for ch in path.chars() {
            if let Some(digit) = ch.to_digit(10) {
                let transition = MAP[city][digit as usize];
                if transition == NA {
                    valid = false;
                    break;
                }
                city = transition;
            } else {
                // Invalid character encountered, terminate processing for this string
                valid = false;
                break;
            }
        }

        if valid && city == B {
            println!("Yes");
        } else {
            println!("No");
        }
    }
}
```
