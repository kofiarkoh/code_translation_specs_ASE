#define A 0

#define B 5

    {X, Y},
    {NA, Z},

    {W, B},
    {B, Y},
    {Y, X}
};

int main(void)
{
    char path[110];
    int i;
    int city;
    
    while (1){
        scanf("%s", path);
        if (path[0] == '#'){
            break;
        }
        
        city = A;
        i = 0;
        while (path[i] != '\0'){
            if (map[city][path[i] - '0'] == NA){
                break;
            }
            city = map[city][path[i] - '0'];
            i++;
        }
        if (city == B){

            printf("No\n");
