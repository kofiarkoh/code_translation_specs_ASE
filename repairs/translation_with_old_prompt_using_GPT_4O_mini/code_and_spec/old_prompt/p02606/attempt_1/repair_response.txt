```rust
use std::io::{self, Write};

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    
    let nums: Vec<i32> = input
        .split_whitespace()
        .map(|s| s.parse().expect("Failed to parse integer"))
        .collect();
    
    let (l, r, d) = (nums[0], nums[1], nums[2]);

    // According to the specification, d must be greater than 0.
    // We assume valid input as per the spec, so we do not handle d == 0.
    let count = r / d - (l - 1) / d;

    // Output the result followed by a newline
    println!("{}", count);
}
```
