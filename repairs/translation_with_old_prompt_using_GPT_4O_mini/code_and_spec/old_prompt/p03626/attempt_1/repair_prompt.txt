        ans = 3;
        t = 1;
    }else{
        ans = 6;
        t = 0;

        if(S1[i] == S2[i]){
            if(t == 1){
                ans = (ans * 2) % m;
            }
            t = 1;
        }
        if(S1[i] == S1[i+1]){
            if(t == 1){
                ans = (ans * 2) % m;
