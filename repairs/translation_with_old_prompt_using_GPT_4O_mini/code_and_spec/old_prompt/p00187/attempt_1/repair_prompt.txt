	acx=cx-ax;
	acy=cy-ay;
	bunbo=(bx-ax)*(dy-cy)-(by-ay)*(dx-cx);
	if(bunbo){
		r=((dy-cy)*acx-(dx-cx)*acy);
		s=((by-ay)*acx-(bx-ax)*acy);
		if(bunbo<0){
			r=-r;
			s=-s;
			bunbo=-bunbo;

		if(r>=0&&r<=bunbo&&s>=0&&s<=bunbo){
			*X=ax+r/(double)bunbo*(bx-ax);
			*Y=ay+r/(double)bunbo*(by-ay);

			return 1;
		}
	}

}
