```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();
    
    // Read N and K
    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let n: usize = parts.next().unwrap().parse().unwrap();
    let k: usize = parts.next().unwrap().parse().unwrap();
    
    // Read the array A
    let mut a = vec![0; n + 1]; // Using 1-based indexing
    let second_line = lines.next().unwrap().unwrap();
    let values: Vec<i32> = second_line.split_whitespace()
                                       .map(|s| s.parse().unwrap())
                                       .collect();
    for i in 1..=n {
        a[i] = values[i - 1];
    }

    let mut b = vec![0; n + 2]; // Auxiliary array B
    for _ in 1..=k {
        let mut min = n as i32; // Change min to i32 to match a[i]
        for i in 1..=n + 1 {
            b[i] = 0;
        }
        for i in 1..=n {
            if a[i] >= i as i32 {
                b[1] += 1;
            } else {
                b[i - a[i] as usize] += 1;
            }
            if a[i] >= (n - i) as i32 {
                b[n + 1] -= 1;
            } else {
                b[i + a[i] as usize + 1] -= 1;
            }
        }
        let mut tmp = 0;
        for i in 1..=n {
            tmp += b[i];
            a[i] = tmp;
            if a[i] < min {
                min = a[i];
            }
        }
    }

    // Output the transformed array
    print!("{}", a[1]);
    for i in 2..=n {
        print!(" {}", a[i]);
    }
    println!(); // Ensure a newline at the end
}
```
