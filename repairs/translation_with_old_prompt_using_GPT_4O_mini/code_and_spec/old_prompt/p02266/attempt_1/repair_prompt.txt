#include <stdlib.h>

typedef struct _stack{
    int data;
    struct _stack *next;
}STACK;
typedef struct _pool{
    int area;
    int x;
    struct _pool *next;
}POOL;
void output(POOL *p){
    if(p==NULL)return;
    
    output(p->next);
    printf(" %d",p->area);
    return;
}

int main(void){
    char in;
    int x=1;
    STACK *stack,*tmp;
    stack=NULL;

        switch(in){
            case '\\':
                tmp=(STACK*)malloc(sizeof(STACK));
                tmp->data=x;
                if(stack==NULL){
                    tmp->next=NULL;
                }else{
                    tmp->next=stack;
                }
                stack=tmp;
                break;
            case '_':
                break;
            case '/':
                if(stack==NULL)break;
                
                if(pool==NULL){
                    pool=(POOL*)malloc(sizeof(POOL));
                    pool->area=x-stack->data;
                    pool->x=stack->data;
