```rust
use std::io::{self, BufRead};

const D_SIZE_MAX: usize = 205; // Maximum size

// Global variables
static mut SI_W: usize = 0; // Width
static mut SI_H: usize = 0; // Height
static mut SI_2_MASS: [[i32; D_SIZE_MAX]; D_SIZE_MAX] = [[0; D_SIZE_MAX]; D_SIZE_MAX]; // Grid
static mut SI_3_CNT: [[[i32; D_SIZE_MAX * 2]; D_SIZE_MAX]; D_SIZE_MAX] = [[[0; D_SIZE_MAX * 2]; D_SIZE_MAX]; D_SIZE_MAX]; // Count

// Output a single line
fn f_out_line(line: &str) {
    print!("{}", line);
}

// Get maximum value
fn f_get_max(val1: i32, val2: i32) -> i32 {
    if val1 > val2 {
        val1
    } else {
        val2
    }
}

// Move down and right
fn f_move(row: &mut isize, col: &mut isize) -> bool {
    *row -= 1; // Move up
    *col += 1; // Move right

    // Check out of bounds
    *row < 0 || *col >= unsafe { SI_W as isize }
}

// Get sum of values
fn f_get_add(row: usize, col: usize, row2: usize, col2: usize) -> i32 {
    let mut add = unsafe { SI_2_MASS[row][col] };
    if row == row2 && col == col2 {
        return add;
    }
    add += unsafe { SI_2_MASS[row2][col2] };
    add
}

// Main execution function
fn f_main() -> i32 {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    // Initialize count array
    unsafe {
        for i in 0..D_SIZE_MAX {
            for j in 0..D_SIZE_MAX {
                for k in 0..D_SIZE_MAX * 2 {
                    SI_3_CNT[i][j][k] = 0;
                }
            }
        }
    }

    // Read dimensions
    if let Some(Ok(line)) = lines.next() {
        let dimensions: Vec<usize> = line.split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();
        unsafe {
            SI_H = dimensions[0];
            SI_W = dimensions[1];
        }
    }

    // Read grid values
    for i in 0..unsafe { SI_H } {
        if let Some(Ok(line)) = lines.next() {
            let values: Vec<i32> = line.split_whitespace()
                .map(|s| s.parse().unwrap())
                .collect();
            for j in 0..unsafe { SI_W } {
                unsafe {
                    SI_2_MASS[i][j] = values[j];
                }
            }
        }
    }

    // Initialize count
    unsafe {
        SI_3_CNT[0][0][0] = SI_2_MASS[0][0];
    }

    // Loop through moves
    for i in 0..(unsafe { SI_W } + unsafe { SI_H } - 2) {
        let (mut li_row, mut li_col);
        if i < unsafe { SI_H } {
            li_row = i;
            li_col = 0;
        } else {
            li_row = unsafe { SI_H - 1 };
            li_col = i - li_row;
        }

        loop {
            let mut li_row2 = li_row;
            let mut li_col2 = li_col;
            let mut j = 0;

            loop {
                // Down, Down
                let li_add = f_get_add(li_row + 1, li_col, li_row2 + 1, li_col2);
                unsafe {
                    SI_3_CNT[li_row + 1][li_col][j] = f_get_max(SI_3_CNT[li_row + 1][li_col][j], SI_3_CNT[li_row][li_col][j] + li_add);
                }

                // Down, Right
                let li_add = f_get_add(li_row + 1, li_col, li_row2, li_col2 + 1);
                unsafe {
                    SI_3_CNT[li_row + 1][li_col][j + 1] = f_get_max(SI_3_CNT[li_row + 1][li_col][j + 1], SI_3_CNT[li_row][li_col][j] + li_add);
                }

                // Right, Down
                if j > 0 {
                    let li_add = f_get_add(li_row, li_col + 1, li_row2 + 1, li_col2);
                    unsafe {
                        SI_3_CNT[li_row][li_col + 1][j - 1] = f_get_max(SI_3_CNT[li_row][li_col + 1][j - 1], SI_3_CNT[li_row][li_col][j] + li_add);
                    }
                }

                // Right, Right
                let li_add = f_get_add(li_row, li_col + 1, li_row2, li_col2 + 1);
                unsafe {
                    SI_3_CNT[li_row][li_col + 1][j] = f_get_max(SI_3_CNT[li_row][li_col + 1][j], SI_3_CNT[li_row][li_col][j] + li_add);
                }

                // Move to next position
                if !f_move(&mut li_row2, &mut li_col2) {
                    break;
                }
                j += 1;
            }

            // Move to next position
            if !f_move(&mut li_row, &mut li_col) {
                break;
            }
        }
    }

    unsafe { SI_3_CNT[SI_H - 1][SI_W - 1][0] }
}

// Main function
fn main() {
    let result = f_main();
    println!("{}", result);
}
```
