```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut reader = stdin.lock(); // Make reader mutable

    // Read height and width
    let mut dimensions = String::new();
    reader.read_line(&mut dimensions).unwrap();
    let mut iter = dimensions.split_whitespace();
    let h: usize = iter.next().unwrap().parse().unwrap();
    let w: usize = iter.next().unwrap().parse().unwrap();

    // Initialize the grid
    let mut grid = vec![String::new(); h];
    for i in 0..h {
        reader.read_line(&mut grid[i]).unwrap();
    }

    // Check for isolation of '#' characters
    for i in 0..h {
        for j in 0..w {
            if grid[i].chars().nth(j).unwrap() == '#' {
                // Check adjacent cells
                let up = if i > 0 { grid[i - 1].chars().nth(j).unwrap() } else { '.' };
                let down = if i + 1 < h { grid[i + 1].chars().nth(j).unwrap() } else { '.' };
                let left = if j > 0 { grid[i].chars().nth(j - 1).unwrap() } else { '.' };
                let right = if j + 1 < w { grid[i].chars().nth(j + 1).unwrap() } else { '.' };

                if up == '#' || down == '#' || left == '#' || right == '#' {
                    println!("No");
                    return;
                }
            }
        }
    }

    println!("Yes");
}
```
