use std::io::{self, BufRead};
use std::cmp::Ordering;

const MAX_WORDS: usize = 50000;

fn read_input() -> (usize, usize, Vec<String>) {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let n: usize = lines.next().unwrap().unwrap().parse().expect("Failed to parse N");
    let m: usize = lines.next().unwrap().unwrap().parse().expect("Failed to parse M");
    let mut words = Vec::with_capacity(n);

    for _ in 0..n {
        let word = lines.next().unwrap().unwrap();
        words.push(word);
    }

    (n, m, words)
}

fn binary_search(words: &[&str], s: &str, n: usize, idx: bool, part: bool) -> usize {
    let mut low = 0;
    let mut high = n;

    if part {
        while low < high {
            let mid = (low + high) / 2;
            if words[mid].get(..s.len()).unwrap_or("") <= s {
                low = mid + 1;
            } else {
                high = mid;
            }
        }
        let k = low as isize - 1;

        low = 0;
        high = n;
        while low < high {
            let mid = (low + high) / 2;
            if words[mid].get(..s.len()).unwrap_or("") < s {
                low = mid + 1;
            } else {
                high = mid;
            }
        }
        return (k - low as isize + 1) as usize;
    } else {
        while low < high {
            let mid = (low + high) / 2;
            match words[mid].cmp(s) {
                Ordering::Equal => return 1,
                Ordering::Less => low = mid + 1,
                Ordering::Greater => high = mid,
            }
        }
    }
    0
}
