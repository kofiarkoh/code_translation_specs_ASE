use std::io::{self, BufRead};

struct UnionFind {
    par: Vec<usize>,
    root_cnt: Vec<usize>,
}

impl UnionFind {
    fn new(size: usize) -> Self {
        let mut par = Vec::with_capacity(size);
        let mut root_cnt = Vec::with_capacity(size);
        for i in 0..size {
            par.push(i);
            root_cnt.push(1);
        }
        UnionFind { par, root_cnt }
    }

    fn root(&mut self, x: usize) -> usize {
        if self.par[x] != x {
            self.par[x] = self.root(self.par[x]); // Path compression
        }
        self.par[x]
    }

    fn unite(&mut self, x: usize, y: usize) -> bool {
        let rx = self.root(x);
        let ry = self.root(y);
        if rx == ry {
            return false; // Already in the same set
        }
        self.par[rx] = ry; // Union
        true
    }

    fn same(&mut self, x: usize, y: usize) -> bool {
        self.root(x) == self.root(y)
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let n: usize = parts.next().unwrap().parse().unwrap();
    let m: usize = parts.next().unwrap().parse().unwrap();

    let mut a = Vec::with_capacity(m);
    let mut b = Vec::with_capacity(m);
    
    for _ in 0..m {
        let line = lines.next().unwrap().unwrap();
        let mut parts = line.split_whitespace();
        let ai: usize = parts.next().unwrap().parse().unwrap();
        let bi: usize = parts.next().unwrap().parse().unwrap();
        a.push(ai - 1); // Convert to 0-based index
        b.push(bi - 1); // Convert to 0-based index
    }

    let mut uf = UnionFind::new(n);
    let mut ans = vec![0; m];

    ans[m - 1] = (n * (n - 1)) / 2; // Initial number of connected pairs

    // Store roots in local variables to avoid multiple mutable borrows
    let root_a_last = uf.root(a[m - 1]);
    let root_b_last = uf.root(b[m - 1]);
    
    uf.root_cnt[root_b_last] += uf.root_cnt[root_a_last];
    uf.root_cnt[root_a_last] = 0;
    uf.unite(a[m - 1], b[m - 1]);

    if m > 1 {
        ans[m - 2] = (n * (n - 1)) / 2 - 1;
    }

    for i in (0..m - 1).rev() {
        if uf.same(a[i], b[i]) {
            ans[i] = ans[i + 1];
        } else {
            let root_a = uf.root(a[i]);
            let root_b = uf.root(b[i]);
            ans[i] = ans[i + 1] - (uf.root_cnt[root_b] * uf.root_cnt[root_a]);
            uf.root_cnt[root_b] += uf.root_cnt[root_a];
            uf.root_cnt[root_a] = 0;
            uf.unite(a[i], b[i]);
        }
    }

    for i in 0..m {
        println!("{}", ans[i]);
    }
}
