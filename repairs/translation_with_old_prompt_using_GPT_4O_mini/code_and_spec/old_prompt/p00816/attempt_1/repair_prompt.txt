int func(char* s, int n, int m)
{
    int i, sum=0;
    char* p=s;
    for(i=0;i<n;++i) if (m&(1<<i)) {
        char t=s[i+1];
        s[i+1]=0;
        sum+=atoi(p);
        s[i+1]=t;
        p=s+i+1;
    }
    return sum+atoi(p);
}

        for(i=(1<<(n-1));i<(1<<n);++i) {
