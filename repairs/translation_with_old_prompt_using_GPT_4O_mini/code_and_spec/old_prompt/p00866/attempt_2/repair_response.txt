```rust
use std::io::{self, BufRead};

#[derive(Debug)]
struct TimeEntry {
    t: i32,
    id: usize,
}

fn check(id: usize, t: &[i32], tbl: &mut Vec<TimeEntry>, a: &[[i32; 60]; 12]) {
    let c = [[0, 1, 2], [0, 2, 1], [1, 0, 2], [1, 2, 0], [2, 0, 1], [2, 1, 0]];
    for j in 0..60 {
        for i in 0..6 {
            let h = (t[c[i][0]] + j) % 60;
            let hh = h / 5;
            let m = (t[c[i][1]] + j) % 60;
            let s = (t[c[i][2]] + j) % 60;
            if a[hh as usize][m as usize] == h {
                tbl.push(TimeEntry {
                    t: (hh * 60 + m) * 60 + s,
                    id,
                });
            }
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut a = [[0; 60]; 12];
    let mut tbl = Vec::new();

    for i in 0..12 {
        for j in 0..60 {
            a[i][j] = (60 * i + j) / 12;
        }
    }

    for line in reader.lines() {
        let line = line.unwrap();
        let n: usize = line.trim().parse().unwrap();
        if n == 0 {
            break;
        }

        tbl.clear();
        for i in 0..n {
            let mut t = [0; 3];
            let line = reader.lines().next().unwrap().unwrap();
            let parts: Vec<i32> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
            t.copy_from_slice(&parts[..3]);
            check(i, &t, &mut tbl, &a);
        }

        tbl.sort_by(|a, b| a.t.cmp(&b.t));

        let mut min = i32::MAX;
        let mut max = i32::MIN;

        for i in 0..tbl.len() {
            let mut f = vec![0; n];
            let mut k = 0;
            for j in i..tbl.len() {
                if f[tbl[j].id] == 1 {
                    continue;
                }
                f[tbl[j].id] = 1;
                k += 1;
                if k == n {
                    let interval = tbl[j].t - tbl[i].t;
                    if interval < min {
                        min = tbl[i].t;
                        max = tbl[j].t;
                    }
                    break;
                }
            }
        }

        println!(
            "{:02}:{:02}:{:02} {:02}:{:02}:{:02}",
            min / 3600,
            (min / 60) % 60,
            min % 60,
            max / 3600,
            (max / 60) % 60,
            max % 60
        );
    }
}
```
