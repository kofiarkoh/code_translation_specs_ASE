#define MAX_K   (5000)

static int ais[MAX_K], bis[MAX_K];

int get_len(int an, int bn, int *ais, int *bis, int k0) {
  int ai, aj, bi, b;

  ai = 0;
  for (bi = k0; bi < bn; bi++) {
    if (ai >= an) return bi - k0;
    b = bis[bi];
    aj = ai;
    while (aj < an && ais[aj] != b) aj++;
    if (aj >= an) return bi - k0;
    ai = aj + 1;
  }

  return bn - k0;
}

int main(int argc, char **argv) {
