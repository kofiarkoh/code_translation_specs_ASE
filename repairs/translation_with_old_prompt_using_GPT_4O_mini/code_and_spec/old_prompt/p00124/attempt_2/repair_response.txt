```rust
use std::io::{self, BufRead};

#[derive(Debug)]
struct Team {
    name: String,
    score: i32,
}

fn main() {
    let stdin = io::stdin();
    let mut first_output = false;

    loop {
        let mut input = String::new();
        stdin.lock().read_line(&mut input).unwrap();
        let n: usize = input.trim().parse().unwrap();

        if n == 0 {
            break;
        }

        let mut teams: Vec<Team> = Vec::new();

        for _ in 0..n {
            let mut team_input = String::new();
            stdin.lock().read_line(&mut team_input).unwrap();
            let parts: Vec<&str> = team_input.trim().split_whitespace().collect();
            let name = parts[0].to_string();
            let win: i32 = parts[1].parse().unwrap();
            let draw: i32 = parts[2].parse().unwrap();
            let score = win * 3 + draw;

            teams.push(Team { name, score });
        }

        // Sort teams by score in descending order
        teams.sort_by(|a, b| b.score.cmp(&a.score));

        if first_output {
            println!(); // Print a blank line before the next output
        }
        for team in teams {
            println!("{},{}", team.name, team.score);
        }
        first_output = true;
    }
}
```
