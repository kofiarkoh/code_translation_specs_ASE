```rust
use std::io;

fn solve(n: i32, c: i32, low: i32) -> i32 {
    let mut ret = 0;
    let mut i = low;

    while i * i <= n {
        if n == i * i {
            ret += 1;
        } else if c < 4 {
            ret += solve(n - i * i, c + 1, i);
        }
        i += 1;
    }
    ret
}

fn main() {
    let mut input = String::new();
    while io::stdin().read_line(&mut input).unwrap() {
        let n: i32 = input.trim().parse().unwrap();
        if n == 0 {
            break;
        }
        let ret = solve(n, 1, 1);
        println!("{}", ret);
        input.clear(); // Clear the input buffer for the next read
    }
}
```
