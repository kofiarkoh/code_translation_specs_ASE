```rust
use std::io::{self, Write};

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    
    let mut nums = input.split_whitespace()
        .filter_map(|s| s.parse::<i32>().ok())
        .collect::<Vec<i32>>();

    if nums.len() != 3 {
        eprintln!("Expected three positive integers.");
        return;
    }

    let (a, b, c) = (nums[0], nums[1], nums[2]);

    // Preconditions: a, b, and c must be positive integers
    if a <= 0 || b <= 0 || c <= 0 {
        eprintln!("All inputs must be positive integers.");
        return;
    }

    let mut sum = 0;
    let mut l = -1; // Initialize l to -1 to indicate not found

    // Iterate from the smaller of a or b down to 1
    let max_divisor = a.min(b);
    for i in (1..=max_divisor).rev() {
        if a % i == 0 && b % i == 0 {
            sum += 1;
            if sum == c {
                l = i; // Found the c-th common divisor
                break;
            }
        }
    }

    // Output the result
    if l != -1 {
        println!("{}", l);
    } else {
        // If c-th common divisor does not exist, behavior is undefined
        // As per the spec, we may not output anything or handle it differently
        println!("No such common divisor found.");
    }
}
```
