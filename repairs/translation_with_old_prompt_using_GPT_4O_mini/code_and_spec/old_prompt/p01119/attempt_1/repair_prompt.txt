#define HASHSIZ 1000003
typedef struct { int n, b; } HA;
int ha1[HASHSIZ+5], *hae1 = ha1 + HASHSIZ;
HA  ha2[HASHSIZ+5], *hae2 = ha2 + HASHSIZ;

int lookup1(int n) {
	int *p = ha1 + n % HASHSIZ;
	while (*p) {
		if (*p == n) return 1;
		if (++p == hae1) p = ha1;
	}
	return 0;
}

void insert1(int n) {
	int *p = ha1 + n % HASHSIZ;
	while (*p) {
		if (*p == n) return;
		if (++p == hae1) p = ha1;
	}
	*p = n;
}

int lookup2(int n, int b) {
	HA *p = ha2 + n % HASHSIZ;
	while (p->n) {
		if (p->n == n) {
			if (p->b == b-1) { p->b = b; return 1; }
			return 0;
		}
		if (++p == ha2) p = ha2;
	}
	return 0;
}

void insert2(int n) {

	while (p->n) {
		if (p->n == n) return;
		if (++p == ha2) p = ha2;
