```rust
use std::collections::HashSet;
use std::io::{self, BufRead};

const HASHSIZ: usize = 1000003;
const INF: i32 = 0x7fffffff;

fn main() {
    let stdin = io::stdin();
    let mut input = stdin.lock().lines();

    while let Some(Ok(line)) = input.next() {
        let n: usize = line.trim().parse().unwrap();
        let m: usize = input.next().unwrap().unwrap().trim().parse().unwrap();

        let mut a = Vec::with_capacity(n);
        for _ in 0..n {
            let value: i32 = input.next().unwrap().unwrap().trim().parse().unwrap();
            a.push(value);
        }

        let mut w = Vec::with_capacity(m);
        for _ in 0..m {
            let value: i32 = input.next().unwrap().unwrap().trim().parse().unwrap();
            w.push(value);
        }

        let mut q = vec![0; 60000]; // 3^10
        let mut end = 1;

        for &weight in &w {
            let e = end;
            for j in 0..e {
                q[end] = q[j] + weight;
                end += 1;
                q[end] = q[j] - weight;
                end += 1;
            }
        }

        let mut ha1 = HashSet::new();
        for &value in &q[..end] {
            if value > 0 {
                ha1.insert(value);
            }
        }

        let mut ans = 0;
        let mut x = 0;

        for &ai in &a {
            if !ha1.contains(&ai) {
                if x == 0 {
                    let mut t;
                    ans = INF;
                    let mut ha2 = HashSet::new();
                    for &qi in &q[..end] {
                        t = (ai - qi).abs();
                        if t < ans {
                            ans = t;
                        }
                        ha2.insert(t);
                    }
                    x = 1;
                } else {
                    let mut t;
                    let mut r = 0;
                    x += 1;
                    ans = INF;
                    for &qi in &q[..end] {
                        t = (ai - qi).abs();
                        if ha2.contains(&t) {
                            if t < ans {
                                ans = t;
                            }
                            r = 1;
                        }
                    }
                    if r == 0 {
                        ans = -1;
                        break;
                    }
                }
            }
        }

        println!("{}", if ans == INF { -1 } else { ans });
    }
}
```
