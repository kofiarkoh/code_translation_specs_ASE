```rust
use std::io::{self, BufRead};

const N: usize = 1_000_000;
static mut A: [i32; N] = [0; N]; // Using unsafe for global mutable state, as Rust does not allow global mutable variables safely.

fn partition(p: usize, r: usize) -> usize {
    let x = unsafe { A[r] }; // Accessing global array A
    let mut i = p as isize - 1; // Convert to isize for indexing

    for j in p..r {
        if unsafe { A[j] } <= x {
            i += 1;
            unsafe {
                A.swap(i as usize, j); // Safe swap using Rust's built-in method
            }
        }
    }
    
    unsafe {
        A.swap((i + 1) as usize, r); // Swap the pivot into its correct position
    }
    
    (i + 1) as usize // Return the new index of the pivot
}

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    let n: usize = lines.next().unwrap().unwrap().parse().expect("Failed to parse n");
    
    for i in 1..=n {
        let value: i32 = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse value"); // Trim whitespace
        unsafe {
            A[i] = value; // Populate the global array A
        }
    }

    let x = partition(1, n);
    
    for i in 1..=n {
        if i == x {
            print!("[{}]", unsafe { A[i] });
        } else {
            print!("{}", unsafe { A[i] });
        }
        if i < n {
            print!(" ");
        }
    }
    println!(); // Print newline at the end
}
```
