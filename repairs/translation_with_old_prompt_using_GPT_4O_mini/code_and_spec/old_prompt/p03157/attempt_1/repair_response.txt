```rust
use std::io::{self, BufRead};

const MAX_SIZE: usize = 512;

fn dfs(y: usize, x: usize, mikisinn: usize, group: &mut [[usize; MAX_SIZE]; MAX_SIZE], grid: &Vec<Vec<char>>, H: usize, W: usize) {
    if group[y][x] == mikisinn {
        return;
    }
    group[y][x] = mikisinn;

    let directions: [(isize, isize); 4] = [(0, 1), (0, -1), (1, 0), (-1, 0)];
    for (dy, dx) in directions.iter() {
        let ny = (y as isize + dy) as usize;
        let nx = (x as isize + dx) as usize;
        if ny < H && nx < W && grid[y][x] != grid[ny][nx] {
            dfs(ny, nx, mikisinn, group, grid, H, W);
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut dimensions = first_line.split_whitespace();
    let H: usize = dimensions.next().unwrap().parse().unwrap();
    let W: usize = dimensions.next().unwrap().parse().unwrap();

    let mut grid: Vec<Vec<char>> = Vec::new();
    for _ in 0..H {
        let line = lines.next().unwrap().unwrap();
        grid.push(line.chars().collect());
    }

    let mut group = [[0; MAX_SIZE]; MAX_SIZE];
    let mut count = [[0; 2]; MAX_SIZE * MAX_SIZE];
    let mut asumisu = 0;
    let mut mizuhasu: u64 = 0;

    for i in 0..H {
        for j in 0..W {
            if group[i][j] == 0 {
                dfs(i, j, asumisu + 1, &mut group, &grid, H, W);
                asumisu += 1;
            }
        }
    }

    for i in 0..H {
        for j in 0..W {
            count[group[i][j]][(grid[i][j] == '.') as usize] += 1;
        }
    }

    for i in 1..=asumisu {
        mizuhasu += (count[i][0] as u64) * (count[i][1] as u64);
    }

    println!("{}", mizuhasu);
}
```
