int group[512][512];

void dfs(int y, int x, int mikisinn) {
	static const int d[4][2] = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
	int i;
	if (group[y][x] == mikisinn) return;
	group[y][x] = mikisinn;
	for (i = 0; i < 4; i++) {
		int ny = y + d[i][0], nx = x + d[i][1];
		if (0 <= ny && ny < H && 0 <= nx && nx < W && S[y][x] != S[ny][nx]) {
			dfs(ny, nx, mikisinn);
		}
	}
}
