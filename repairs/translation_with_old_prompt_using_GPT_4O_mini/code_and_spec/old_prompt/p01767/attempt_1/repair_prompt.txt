#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
	return *(int *)a - *(int *)b;
}

int upperBound(int data[], int n, int key) {
	int l = 0, r = n - 1;

	while (l <= r) {
		int m = (l + r) / 2;

		if (data[m] > key)
			r = m - 1;
		else
			l = m + 1;
	}

	return l;
}

int a[300000], b[300000];

int main(void) {

	int n, m;

	for (i = 0; i < n; i++)
		scanf("%d", a + i);

	for (i = 0; i < m; i++)

	qsort(a, n, sizeof(int), &compare);
