use std::io::{self, BufRead};
use std::f64;

const QMAX: usize = 1000;
const EPS: f64 = 1e-8;
const INF: f64 = f64::INFINITY; // Use f64::INFINITY instead of a hexadecimal representation
const MAX: usize = 105;

#[derive(Clone)]
struct Edge {
    to: usize,
    rev: usize,
    cap: i32,
    cost: f64,
}

struct MinHeap {
    que: Vec<(usize, f64)>,
}

impl MinHeap {
    fn new() -> Self {
        MinHeap { que: Vec::new() }
    }

    fn min_heapify(&mut self, i: usize) {
        let mut min = i;
        let l = self.left(i);
        let r = self.right(i);

        if l < self.que.len() && self.que[l].1 < self.que[min].1 {
            min = l;
        }
        if r < self.que.len() && self.que[r].1 < self.que[min].1 {
            min = r;
        }
        if min != i {
            self.que.swap(i, min);
            self.min_heapify(min);
        }
    }

    fn deq(&mut self) -> (usize, f64) {
        let min = self.que[0];
        self.que[0] = self.que.pop().unwrap();
        self.min_heapify(0);
        min
    }

    fn enq(&mut self, s: usize, t: f64) {
        self.que.push((s, t));
        let mut i = self.que.len() - 1;
        while i > 0 {
            let parent = self.parent(i);
            if self.que[parent].1 > self.que[i].1 {
                self.que.swap(i, parent);
                i = parent;
            } else {
                break;
            }
        }
    }

    fn parent(&self, i: usize) -> usize {
        i >> 1
    }

    fn left(&self, i: usize) -> usize {
        i << 1
    }

    fn right(&self, i: usize) -> usize {
        (i << 1) + 1
    }
}

struct Graph {
    edges: Vec<Vec<Edge>>,
    hi: Vec<usize>,
    lim: Vec<usize>,
    potential: Vec<f64>,
    min_cost: Vec<f64>,
    prevv: Vec<usize>,
    preve: Vec<usize>,
    V: usize,
}

impl Graph {
    fn new(v: usize) -> Self {
        Graph {
            edges: vec![Vec::new(); v],
            hi: vec![0; v],
            lim: vec![0; v],
            potential: vec![0.0; v],
            min_cost: vec![INF; v],
            prevv: vec![0; v],
            preve: vec![0; v],
            V: v,
        }
    }

    fn check_lim(&mut self, a: usize) {
        if self.hi[a] < self.lim[a] {
            return;
        }
        if self.lim[a] == 0 {
            self.lim[a] = 32;
            self.hi[a] = 0;
            self.edges[a].resize(self.lim[a], Edge { to: 0, rev: 0, cap: 0, cost: 0.0 });
        } else if self.hi[a] == self.lim[a] {
            let k = self.lim[a].min(128);
            self.lim[a] += k;
            self.edges[a].resize(self.lim[a], Edge { to: 0, rev: 0, cap: 0, cost: 0.0 });
        }
    }

    fn add_edge(&mut self, from: usize, to: usize, cap: i32, cost: f64) {
        self.check_lim(from);
        self.check_lim(to);
        let f = self.hi[from];
        let t = self.hi[to];
        self.edges[from].push(Edge { to, cap, cost, rev: t });
        self.edges[to].push(Edge { to: from, cap: 0, cost: -cost, rev: f });
        self.hi[from] += 1;
        self.hi[to] += 1;
    }

    fn min_cost_flow(&mut self, s: usize, t: usize, mut F: i32) -> f64 {
        let mut ret = 0.0;
        self.potential.fill(0.0);

        while F > 0 {
            self.min_cost.fill(INF);
            let mut heap = MinHeap::new();
            heap.enq(s, 0.0);
            self.min_cost[s] = 0.0;

            while !heap.que.is_empty() {
                let (u, cost) = heap.deq();
                if self.min_cost[u] < cost {
                    continue;
                }

                for i in 0..self.hi[u] {
                    let ep = &self.edges[u][i];
                    let ns = ep.to;
                    let nt = cost + ep.cost + self.potential[u] - self.potential[ns];
                    if ep.cap > 0 && self.min_cost[ns] > nt + EPS {
                        self.min_cost[ns] = nt;
                        self.prevv[ns] = u;
                        self.preve[ns] = i;
                        heap.enq(ns, nt);
                    }
                }
            }
            if self.min_cost[t] == INF {
                return -1.0;
            }
            for u in 0..self.V {
                self.potential[u] += self.min_cost[u];
            }

            let mut k = F;
            let mut v = t;
            while v != s {
                let ep = &self.edges[self.prevv[v]][self.preve[v]];
                if k > ep.cap {
                    k = ep.cap;
                }
                v = self.prevv[v];
            }
            F -= k;
            ret += k as f64 * self.potential[t];

            let mut v = t;
            while v != s {
                let (prev, edge_index) = (self.prevv[v], self.preve[v]);
                let ep = &mut self.edges[prev][edge_index];
                ep.cap -= k;

                // Use split_at_mut to avoid mutable borrow conflict
                let (before, after) = self.edges[v].split_at_mut(ep.rev);
                if let Some(rev_edge) = after.first_mut() {
                    rev_edge.cap += k;
                }
                v = prev;
            }
        }
        ret
    }
}

fn gauss_jordan(n: usize, a: &mut [[f64; MAX]; MAX], c: &mut [f64; MAX], x: &mut [f64; MAX]) -> bool {
    let mut b = [[0.0; MAX]; MAX];
    for i in 0..n {
        for j in 0..n {
            b[i][j] = a[i][j];
        }
        b[i][n] = c[i];
    }

    for i in 0..n {
        let mut pivot = i;
        for j in i..n {
            if b[j][i].abs() > b[pivot][i].abs() {
                pivot = j;
            }
        }
        if b[pivot][i].abs() < EPS {
            return false; // error
        }
        b.swap(i, pivot);
        for j in 0..=n {
            b[i][j] /= b[i][i];
        }
        for j in 0..n {
            if i != j {
                for k in i + 1..=n {
                    b[j][k] -= b[j][i] * b[i][k];
                }
            }
        }
    }
    for i in 0..n {
        x[i] = b[i][n];
    }
    true
}

fn read_int() -> i32 {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut input = String::new();
    handle.read_line(&mut input).unwrap();
    input.trim().parse().unwrap()
}

fn main() {
    let t = read_int();
    for _ in 0..t {
        let n = read_int() as usize;
        let s = read_int() as usize;
        let t = read_int() as usize;
        let f = read_int();

        let mut a = [[0.0; MAX]; MAX];
        let mut c = [0.0; MAX];
        let mut x = [0.0; MAX];
        for i in 0..n {
            for j in 0..n {
                a[i][j] = read_int() as f64;
            }
            c[i] = read_int() as f64;
        }

        if !gauss_jordan(n, &mut a, &mut c, &mut x) {
            println!("impossible");
            continue;
        }

        let mut graph = Graph::new(n);
        for i in 0..n {
            let m = read_int() as usize;
            let mut d = vec![0; m];
            for j in 0..m {
                d[j] = read_int() as usize;
            }
            for j in 0..m {
                graph.add_edge(i, d[j], read_int(), (x[i] - x[d[j]]).abs());
            }
        }

        let ans = graph.min_cost_flow(s, t, f);
        if ans < 0.0 {
            println!("impossible");
        } else {
            println!("{:.10}", ans);
        }
    }
}
