#define max(X, Y) ((X)>(Y)?(X):(Y))

{
    int w, d;
    while (scanf("%d%d", &w, &d), w) {
        int i, a[21]={0}, b[21]={0}, res;
        for(i=0;i<w;++i) { int t; scanf("%d", &t); a[t]++; }
