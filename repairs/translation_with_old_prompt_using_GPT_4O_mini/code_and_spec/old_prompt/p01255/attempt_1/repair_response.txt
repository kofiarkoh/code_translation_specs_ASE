```rust
use std::io::{self, BufRead};
use std::collections::HashSet;

const EPS: f64 = 1e-6;

#[derive(Clone, Copy, Debug)]
struct Point {
    x: f64,
    y: f64,
}

#[derive(Clone, Copy, Debug)]
struct Segment {
    start: Point,
    end: Point,
}

fn eq(a: f64, b: f64) -> bool {
    (a - b).abs() < EPS
}

fn add(p1: Point, p2: Point) -> Point {
    Point {
        x: p1.x + p2.x,
        y: p1.y + p2.y,
    }
}

fn sub(p1: Point, p2: Point) -> Point {
    Point {
        x: p1.x - p2.x,
        y: p1.y - p2.y,
    }
}

fn cross(p1: Point, p2: Point) -> f64 {
    p1.x * p2.y - p1.y * p2.x
}

fn ccw(p: Point, q: Point, r: Point) -> i32 {
    let dx = q.x - p.x;
    let dy = q.y - p.y;
    let dx_ = r.x - p.x;
    let dy_ = r.y - p.y;
    let t = dx * dy_ - dy * dx_;
    if eq(t, 0.0) {
        0
    } else if t < 0.0 {
        -1
    } else {
        1
    }
}

fn intersect(s1: Segment, s2: Segment) -> bool {
    ccw(s1.start, s1.end, s2.start) * ccw(s1.start, s1.end, s2.end) <= 0 &&
    ccw(s2.start, s2.end, s1.start) * ccw(s2.start, s2.end, s1.end) <= 0
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    while let Some(line) = lines.next() {
        let lim: usize = line.unwrap().trim().parse().unwrap();
        if lim == 0 {
            break;
        }

        let mut segments = Vec::new();
        let mut n = 0;

        for i in 0..lim {
            let s: usize = lines.next().unwrap().unwrap().trim().parse().unwrap();
            if s == 0 {
                continue;
            }
            let start_x: f64 = lines.next().unwrap().unwrap().trim().parse().unwrap();
            let start_y: f64 = lines.next().unwrap().unwrap().trim().parse().unwrap();
            let mut start_segment = Segment {
                start: Point { x: start_x, y: start_y },
                end: Point { x: 0.0, y: 0.0 },
            };

            for _ in 1..s {
                let end_x: f64 = lines.next().unwrap().unwrap().trim().parse().unwrap();
                let end_y: f64 = lines.next().unwrap().unwrap().trim().parse().unwrap();
                start_segment.end = Point { x: end_x, y: end_y };
                segments.push(start_segment);
                n += 1;
                start_segment.start = Point { x: end_x, y: end_y };
            }
        }

        let result = welsh_powell(&segments, n);
        println!("{}", result);
    }
}

fn welsh_powell(segments: &[Segment], n: usize) -> usize {
    let mut graph: Vec<HashSet<usize>> = vec![HashSet::new(); n];
    let mut colors = vec![0; n];

    for i in 0..n {
        for j in (i + 1)..n {
            if intersect(segments[i], segments[j]) {
                graph[i].insert(j);
                graph[j].insert(i);
            }
        }
    }

    let mut degree: Vec<usize> = graph.iter().map(|g| g.len()).collect();
    let mut vertices: Vec<usize> = (0..n).collect();
    vertices.sort_by(|&a, &b| degree[b].cmp(&degree[a]));

    let mut color_count = 0;

    for &u in &vertices {
        if colors[u] == 0 {
            color_count += 1;
            colors[u] = color_count;

            for &v in &graph[u] {
                if colors[v] == 0 {
                    colors[v] = color_count;
                }
            }
        }
    }

    color_count
}
```
