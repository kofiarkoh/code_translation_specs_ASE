#include <stdlib.h>

int A, B, C, D;

int nigeru(int x, int y, int isSrc) {
	int cost = 0;
	if (x < A) cost += A - x;
	if (C < x) cost += x - C;
	if (y < B) cost += B - y;
	if (D < y) cost += y - D;
	if (isSrc && cost > 0) cost--;
	return cost;
}
