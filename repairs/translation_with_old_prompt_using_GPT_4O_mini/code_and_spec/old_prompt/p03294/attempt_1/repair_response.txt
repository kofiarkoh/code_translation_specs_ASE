```rust
use std::io::{self, Write};

fn main() {
    let mut input = String::new();
    
    // Read the number of integers
    io::stdout().flush().unwrap(); // Ensure prompt is displayed before reading input
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let n: usize = input.trim().parse().expect("Please enter a valid positive integer");

    // Preconditions check
    if n == 0 {
        panic!("N must be a positive integer greater than 0");
    }

    let mut ans = 0;
    let mut a = Vec::with_capacity(n); // Allocate space for n integers

    // Read the integers and compute the sum
    for _ in 0..n {
        input.clear(); // Clear the input buffer for the next read
        io::stdin().read_line(&mut input).expect("Failed to read line");
        let value: i32 = input.trim().parse().expect("Please enter a valid integer");
        a.push(value);
        ans += value - 1; // Compute the sum of each integer minus one
    }

    // Output the result
    println!("{}", ans);
}
```
