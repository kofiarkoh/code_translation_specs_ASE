  scanf("%d",&n);
  for(i=0;i<n;i++){
    for(j=0;j<n;j++)scanf(" %c",&d[i+1][j+1]);
  }
  for(i=0;i<n;i++){
    for(j=0;j<n;j++){
      if(d[i+1][j+1]=='x')continue;
      d[y[t=0]=i+1][x[0]=j+1]='x';
      c++;
      for(r=1;r-t;t++){
	for(k=0;k<4;k++){
	  ny=y[t]+Y[k];
	  nx=x[t]+X[k];
	  if(d[ny][nx]-'o')continue;
