#define N	8
#define MD	1000000007

long long min(long long a, long long b) { return a < b ? a : b; }
long long max(long long a, long long b) { return a > b ? a : b; }

int cnt[N + 1];

void init() {
	int i;

	cnt[1] = 9;
	for (i = 2; i <= N; i++)
		cnt[i] = cnt[i - 1] * 10;
}

long long power(int a, int k) {
	long long p = 1, b = a;

	while (k) {
		if (k & 1)
			p = p * b % MD;
		b = b * b % MD;
		k >>= 1;
	}
	return p;
}
