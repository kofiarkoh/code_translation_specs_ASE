#define INF 1000

int D,N;

int e[100];
int r[100];

int memo[101][101];

	int ret=INF;
	int i;
	if(tkrk<=0)return 0;
	if(memo[tkrk][kkt]>0)return memo[tkrk][kkt]-1;
	memo[tkrk][kkt]=INF+1;
	for(i=0;i<N;i++) {
		if(kkt>=r[i]) {

			if(candidate<ret)ret=candidate;
		}
	}
	memo[tkrk][kkt]=ret+1;
	return ret;
}
