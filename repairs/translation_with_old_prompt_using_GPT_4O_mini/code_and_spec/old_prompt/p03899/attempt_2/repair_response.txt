```rust
use std::io::{self, BufRead};

#[derive(Copy, Clone)]
struct Node {
    k: i32,
    v: i64,
}

fn run() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut input = handle.lines().flat_map(Result::ok);
    
    let first_line = input.next().unwrap();
    let params: Vec<i32> = first_line.split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    
    let (n, m, k) = (params[0], params[1], params[2]);

    // Preconditions
    assert!(n >= 0 && m < n && k >= 0, "Invalid input parameters");

    let mut a = vec![0; n as usize];
    for i in 0..n {
        a[i as usize] = input.next().unwrap().parse().unwrap();
    }

    let mut dp = a.clone().into_iter().map(|x| x as i64).collect::<Vec<i64>>();
    let mut d = vec![Node { k: 0, v: 0 }; n as usize];

    for i in 2..=k {
        let mut f = 0;
        let mut b = 0;
        d[0] = Node { k: i - 2, v: dp[(i - 2) as usize] };

        for j in (i - 1)..n {
            if d[f].k == j - m - 1 {
                f += 1;
            }
            let v = d[f].v + (i as i64) * a[j as usize];

            while f <= b && d[b].v <= dp[j as usize] {
                b -= 1;
            }
            b += 1;
            d[b] = Node { k: j, v: dp[j as usize] };
            dp[j as usize] = v;
        }
    }

    // Correctly index dp using usize
    let ans = dp[(k - 1) as usize..].iter().cloned().max().unwrap_or(0);
    println!("{}", ans);
}

fn main() {
    run();
}
```
