```rust
use std::io::{self, BufRead};

const MV: [[i32; 2]; 4] = [[50, 50], [-50, 50], [-50, -50], [50, -50]];
const TR: [[usize; 4]; 2] = [[3, 2, 1, 0], [1, 0, 3, 2]];

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let line = line.unwrap();
        let mut parts = line.split_whitespace();
        let n: usize = parts.next().unwrap().parse().unwrap();

        if n < 0 {
            break; // Precondition: n must be non-negative
        }

        let mut mir = [[[0; 9]; 9]; 9]; // 4D array initialized to zero

        for _ in 0..n {
            let mirror_info: Vec<&str> = parts.collect();
            let s = mirror_info[0];
            let x: usize = mirror_info[1].parse().unwrap();
            let y: usize = mirror_info[2].parse().unwrap();

            if s == "x" {
                mir[x][y][x + 1][y] = 1;
            } else {
                mir[x][y][x][y + 1] = 1;
            }
        }

        let (mut x, mut y, mut d) = (50, 0, 0);
        let mut wall = 1;

        loop {
            if x == 100 && y == 50 && d == 0 {
                x = 75;
                y = 25;
                break;
            }
            if wall == 1 && (x == 0 || y == 0 || x == 800 || y == 800) {
                break; // Exit on wall collision
            }

            x += MV[d][0];
            y += MV[d][1];
            wall = 1;

            let x_index = (x / 100) as usize;
            let y_index = (y / 100) as usize;

            if (x_index < 9) && ((y - 50) / 100) < 9 && ((y + 50) / 100) < 9 {
                if mir[x_index][(y - 50) as usize / 100][x_index][(y + 50) as usize / 100] != 0 {
                    d = TR[1][d];
                    wall = 0;
                }
            } 
            if ((x - 50) / 100) < 9 && (y_index < 9) {
                if mir[(x - 50) as usize / 100][y_index][(x + 50) as usize / 100][y_index] != 0 {
                    d = TR[0][d];
                    wall = 0;
                }
            }
        }

        println!("{} {}", x, y);
    }
}
```
