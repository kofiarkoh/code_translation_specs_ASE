```rust
use std::io::{self, BufRead};

fn get_ints() -> Result<(i32, i32), io::Error> {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut line = String::new();
    handle.read_line(&mut line)?;
    let mut nums = line.split_whitespace()
        .filter_map(|s| s.parse::<i32>().ok());
    let a1 = nums.next().ok_or(io::Error::new(io::ErrorKind::InvalidInput, "Missing first integer"))?;
    let a2 = nums.next().ok_or(io::Error::new(io::ErrorKind::InvalidInput, "Missing second integer"))?;
    Ok((a1, a2))
}

fn main() -> Result<(), io::Error> {
    let (n, k) = get_ints()?;

    // Preconditions check
    if n < 0 || k < 0 || k > n {
        return Ok(()); // No output for invalid input
    }

    for pat in 0..(1 << n) {
        if pat.count_ones() != k as u32 {
            continue;
        }
        print!("{}:", pat);
        for i in 0..n {
            if pat & (1 << i) == 0 {
                continue;
            }
            print!(" {}", i);
        }
        println!();
    }
    Ok(())
}
```
