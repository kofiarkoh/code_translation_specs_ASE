```rust
use std::io::{self, Write};

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let mut iter = input.split_whitespace();
    
    let n: usize = iter.next().expect("N is required").parse().expect("Invalid input for N");
    let k: usize = iter.next().expect("K is required").parse().expect("Invalid input for K");

    // Preconditions
    assert!(n > 0, "N must be a positive integer");
    assert!(k >= 0, "K must be a non-negative integer");

    let mut multi = 1;
    let mut denomi = vec![0; n];
    let mut nume = 0.0;

    for i in (0..n).rev() {
        while k > multi * (i + 1) {
            multi *= 2;
        }
        denomi[i] += multi * n;
    }

    for j in 0..n {
        nume += denomi[0] as f64 / denomi[j] as f64;
    }

    // Output with twelve decimal places
    println!("{:.12}", nume / denomi[0] as f64);
}
```
