use std::collections::HashMap;
use std::io::{self, BufRead};

const SENTINEL: i64 = 1_000_000_000_000_000_000;

fn merge_sort(arr: &mut Vec<i64>, left: usize, right: usize) {
    if left + 1 < right {
        let mid = (left + right) / 2;
        merge_sort(arr, left, mid);
        merge_sort(arr, mid, right);
        merge(arr, left, mid, right);
    }
}

fn merge(arr: &mut Vec<i64>, left: usize, mid: usize, right: usize) {
    let n1 = mid - left;
    let n2 = right - mid;

    let mut left_part = vec![0; n1 + 1];
    let mut right_part = vec![0; n2 + 1];

    for i in 0..n1 {
        left_part[i] = arr[left + i];
    }
    for j in 0..n2 {
        right_part[j] = arr[mid + j];
    }

    left_part[n1] = SENTINEL;
    right_part[n2] = SENTINEL;

    let mut i = 0;
    let mut j = 0;

    for k in left..right {
        if left_part[i] <= right_part[j] {
            arr[k] = left_part[i];
            i += 1;
        } else {
            arr[k] = right_part[j];
            j += 1;
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut nums = first_line.split_whitespace().map(|s| s.parse::<i32>().unwrap());
    let n = nums.next().unwrap();
    let k = nums.next().unwrap();

    let mut counts = vec![0; (n + 1) as usize];
    let mut input_numbers = Vec::with_capacity(n as usize);

    for _ in 0..n {
        let number: i32 = lines.next().unwrap().unwrap().parse().unwrap();
        input_numbers.push(number);
        counts[number as usize] += 1;
    }

    merge_sort(&mut counts, 1, (n + 1) as usize);

    let mut ans = 0;
    let mut unique_count = 0;

    for i in 1..=(n - k) {
        if counts[i as usize] > 0 {
            ans += counts[i as usize];
            unique_count += 1;
        }
    }

    println!("{}", ans);
}
