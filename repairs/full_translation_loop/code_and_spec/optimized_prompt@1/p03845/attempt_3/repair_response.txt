```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    // Read the number of elements in the first list
    let mut lines = reader.lines();
    let n: usize = lines.next().unwrap().unwrap().parse().expect("Failed to parse the number of elements.");

    // Preconditions: n must be greater than zero
    assert!(n > 0, "The number of elements must be greater than zero.");

    // Read the list of integers
    let mut t = Vec::with_capacity(n);
    for _ in 0..n {
        let value: i32 = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse list element.");
        t.push(value);
    }

    // Read the number of updates
    let m: usize = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse the number of updates.");

    // Preconditions: m must be greater than or equal to zero
    assert!(m >= 0, "The number of updates must be greater than or equal to zero.");

    // Read the update pairs
    let mut p = Vec::with_capacity(m);
    let mut x = Vec::with_capacity(m);
    for _ in 0..m {
        let index: usize = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse index.");
        let new_value: i32 = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse new value.");
        p.push(index);
        x.push(new_value);
    }

    // Calculate the total sum of the first list
    let sum: i32 = t.iter().sum();

    // Output the modified sums based on the updates
    for i in 0..m {
        // Preconditions: p[i] must be a valid index for t
        assert!(p[i] > 0 && p[i] <= n, "Index out of bounds.");
        let modified_sum = sum - t[p[i] - 1] + x[i];
        println!("{}", modified_sum);
    }
}
```
