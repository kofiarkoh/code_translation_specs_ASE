// update kth value to v
void update(int k, long long v)
{
	int t;
	k += sz-1;
	seg[k] = v;
	while (k > 0) {
		if (!(k & 1)) k--;
		t = k, k >>= 1;
		v = seg[t++]; if (v < seg[t]) v = seg[t];
		seg[k] = v;
	}
}

// get max value in [a, b)
long long query(int a, int b, long long k, int l, int r) 
{
	int m;
	long long lmax, rmax;

	if (r <= a || b <= l) return 0;
    if (a <= l && r <= b) return seg[k];
	k <<= 1, m = (l + r) >> 1;
    lmax = query(a, b, ++k, l, m);
