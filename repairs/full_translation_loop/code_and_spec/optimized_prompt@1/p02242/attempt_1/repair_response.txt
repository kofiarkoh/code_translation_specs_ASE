```rust
use std::io::{self, BufRead, Write};

#[derive(Debug)]
struct Node {
    edge_to: Vec<usize>,
    edge_cost: Vec<usize>,
    edge_size: usize,
    cost: isize,
    visited: bool,
}

fn dijkstra_noheap(nodes: &mut [Node]) {
    let n = nodes.len();
    for node in nodes.iter_mut() {
        node.cost = -1;
        node.visited = false;
    }
    nodes[0].cost = 0;

    loop {
        let mut next: Option<&mut Node> = None;
        for node in nodes.iter_mut() {
            if node.visited || node.cost < 0 {
                continue;
            }
            if next.is_none() || node.cost < next.as_ref().unwrap().cost {
                next = Some(node);
            }
        }
        let next = match next {
            Some(n) => n,
            None => break,
        };
        next.visited = true;
        let u = next.clone();

        for i in 0..u.edge_size {
            let to = u.edge_to[i];
            let cost = u.cost + u.edge_cost[i] as isize;
            if nodes[to].cost < 0 || cost < nodes[to].cost {
                nodes[to].cost = cost;
            }
        }
    }
}

fn get_uint<R: BufRead>(reader: &mut R) -> Result<usize, io::Error> {
    let mut buffer = String::new();
    reader.read_line(&mut buffer)?;
    buffer.trim().parse::<usize>().map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "Invalid number"))
}

fn put_uint<W: Write>(writer: &mut W, n: usize) -> io::Result<()> {
    write!(writer, "{}", n)
}

fn main() -> io::Result<()> {
    let stdin = io::stdin();
    let mut reader = stdin.lock();
    let n = get_uint(&mut reader)?;

    let mut nodes = vec![Node {
        edge_to: Vec::new(),
        edge_cost: Vec::new(),
        edge_size: 0,
        cost: -1,
        visited: false,
    }; n];

    for _ in 0..n {
        let u = get_uint(&mut reader)?;
        let k = get_uint(&mut reader)?;
        nodes[u].edge_size = k;
        for _ in 0..k {
            let to = get_uint(&mut reader)?;
            let cost = get_uint(&mut reader)?;
            nodes[u].edge_to.push(to);
            nodes[u].edge_cost.push(cost);
        }
    }

    dijkstra_noheap(&mut nodes);

    let stdout = io::stdout();
    let mut writer = stdout.lock();
    for i in 0..n {
        put_uint(&mut writer, i)?;
        write!(writer, " ")?;
        put_uint(&mut writer, nodes[i].cost as usize)?;
        writeln!(writer)?;
    }

    Ok(())
}
```
