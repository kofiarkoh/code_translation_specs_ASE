```rust
use std::io::{self, BufRead};

#[derive(Debug)]
struct Rect {
    height: i64,
    pos: usize,
}

fn large_histogram(heights: &[i64]) -> i64 {
    let mut stack: Vec<Rect> = Vec::new();
    let mut max_area = 0;
    let mut extended_heights = heights.to_vec();
    extended_heights.push(0); // Append a zero height to flush the stack at the end

    for (i, &height) in extended_heights.iter().enumerate() {
        let mut target = i;

        while let Some(last) = stack.last() {
            if last.height < height {
                break;
            }

            let pre = stack.pop().unwrap();
            let area = pre.height * (i - pre.pos) as i64;
            if max_area < area {
                max_area = area;
            }
            target = pre.pos;
        }

        stack.push(Rect { height, pos: target });
    }

    max_area
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    let mut lines = reader.lines();
    let w: i64 = lines.next().unwrap().unwrap().parse().unwrap();

    // Preconditions check
    if w < 0 {
        panic!("Number of histogram bars must be non-negative.");
    }

    let mut heights = Vec::new();
    for _ in 0..w {
        let height: i64 = lines.next().unwrap().unwrap().parse().unwrap();
        // Preconditions check
        if height < 0 {
            panic!("Heights must be non-negative integers.");
        }
        heights.push(height);
    }

    let max_area = large_histogram(&heights);
    println!("{}", max_area);
}
```
