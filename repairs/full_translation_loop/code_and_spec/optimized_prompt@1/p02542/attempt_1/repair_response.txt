use std::collections::VecDeque;
use std::cmp::Ordering;

#[derive(Clone, Copy)]
struct Data {
    key: i64,
    id: usize,
}

struct MinHeap {
    obj: Vec<Data>,
    size: usize,
}

impl MinHeap {
    fn new() -> Self {
        MinHeap {
            obj: vec![Data { key: 0, id: 0 }; 1],
            size: 0,
        }
    }

    fn push(&mut self, x: Data) {
        self.size += 1;
        let mut i = self.size;
        self.obj.push(x);
        while i > 1 {
            let j = i / 2;
            if self.obj[i].key < self.obj[j].key {
                self.obj.swap(i, j);
                i = j;
            } else {
                break;
            }
        }
    }

    fn pop(&mut self) -> Data {
        let output = self.obj[1];
        self.obj[1] = self.obj[self.size];
        self.size -= 1;
        let mut i = 1;
        let mut j = 2;
        while j <= self.size {
            if j < self.size && self.obj[j + 1].key < self.obj[j].key {
                j += 1;
            }
            if self.obj[j].key < self.obj[i].key {
                self.obj.swap(i, j);
                i = j;
                j *= 2;
            } else {
                break;
            }
        }
        output
    }
}

struct Edge {
    id: usize,
    head: usize,
    tail: usize,
    flow: i32,
    cap: i32,
    cost: i32,
}

struct List {
    next: Option<Box<List>>,
    e: Edge,
}

const SUP: i64 = 1 << 60;
const INF: i64 = -SUP;

fn successive_shortest_path(n: usize, m: usize, adj: &Vec<Option<Box<List>>>, e: &mut Vec<Edge>, b: &mut Vec<i32>) -> i64 {
    let mut flag = vec![0; n + 1];
    let mut def = vec![0; n + 1];
    let mut dist = vec![0; n + 1];
    let mut pi = vec![SUP; n + 1];
    let mut prev = vec![None; n + 1];
    let mut ans = 0;

    let mut h_def = MinHeap::new();
    let mut h_dist = MinHeap::new();

    for i in 0..m {
        ans += e[i].cost as i64 * e[i].flow as i64;
    }

    for i in 1..=n {
        pi[i] = SUP;
    }

    for i in 1..=n {
        let mut p = &adj[i];
        def[i] = b[i];
        while let Some(node) = p {
            def[i] += node.e.flow - e[node.e.id ^ 1].flow;
            p = node.next.as_deref();
        }
        if def[i] < 0 {
            h_def.push(Data { key: def[i] as i64, id: i });
            if pi[i] == SUP {
                pi[i] = 0;
                for j in 1..=n {
                    let mut flag_0 = 0;
                    for k in 0..m {
                        if e[k].flow == e[k].cap {
                            continue;
                        }
                        let u = e[k].tail;
                        let w = e[k].head;
                        if pi[u] != SUP && pi[w] > pi[u] + e[k].cost as i64 {
                            flag_0 = 1;
                            pi[w] = pi[u] + e[k].cost as i64;
                        }
                    }
                    if flag_0 == 0 {
                        break;
                    }
                }
                if j > n {
                    return INF;
                }
            }
        }
    }

    let mut s;
    let mut t;
    let mut min;

    while h_def.size > 0 {
        let d = h_def.pop();
        s = d.id;

        for u in 1..=n {
            flag[u] = 0;
            dist[u] = SUP;
            prev[u] = None;
        }
        dist[s] = 0;
        h_dist = MinHeap::new();
        h_dist.push(Data { key: 0, id: s });

        while h_dist.size > 0 {
            let d = h_dist.pop();
            let u = d.id;
            if flag[u] != 0 {
                continue;
            }
            flag[u] = 1;

            let mut p = &adj[u];
            while let Some(node) = p {
                let w = node.e.head;
                if node.e.flow == node.e.cap {
                    continue;
                } else if dist[w] > dist[u] + node.e.cost as i64 + pi[u] - pi[w] {
                    dist[w] = dist[u] + node.e.cost as i64 + pi[u] - pi[w];
                    prev[w] = Some(node.e);
                    h_dist.push(Data { key: dist[w], id: w });
                }
                p = node.next.as_deref();
            }
        }

        t = (1..=n).find(|&t| dist[t] != SUP && def[t] > 0).unwrap_or(n + 1);
        if t > n {
            return SUP;
        }

        let mut u = t;
        min = (-def[s].min(def[t])) as i32;
        while let Some(edge) = prev[u] {
            min = min.min(edge.cap - edge.flow);
            u = edge.tail;
        }

        u = t;
        while let Some(edge) = prev[u] {
            let i = edge.id;
            if e[i ^ 1].flow > 0 {
                e[i ^ 1].flow -= min;
                e[i].cap -= min;
            } else {
                e[i].flow += min;
                e[i ^ 1].cap += min;
            }
            ans += e[i].cost as i64 * min as i64;
            u = edge.tail;
        }

        for i in 1..=n {
            if dist[i] != SUP {
                pi[i] += dist[i];
            }
        }

        def[s] += min;
        def[t] -= min;
        if def[s] < 0 {
            h_def.push(Data { key: def[s] as i64, id: s });
        }
    }

    ans
}

            S[i][j + 1] = c;
        }
    }

    let mut dist = vec![-1; 2501];
    let mut q = vec![0; 2501];

    let mut e = vec![Edge { id: 0, head: 0, tail: 0, flow: 0, cap: 0, cost: 0 }; 30000];

                    tail += 1;
                }

                e[m * 2 + 1].head = n * m + n;
                e[m * 2 + 1].cap = 0;
                e[m * 2 + 1].cost = dist[q[head]] as i32;

                d[m * 2 + 1].next = adj[q[head]].take().map(Box::new);

        e[m * 2 + 1].cost = 0;

    }
