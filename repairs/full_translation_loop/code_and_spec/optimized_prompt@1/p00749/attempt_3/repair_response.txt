use std::io::{self, BufRead};

const MAX: usize = 602;
const MAX_ROWS: usize = 60;
const MAX_COLS: usize = 12;

struct UnionFind {
    parent: [usize; MAX],
    rank: [usize; MAX],
}

impl UnionFind {
    fn new() -> Self {
        let mut parent = [0; MAX];
        let mut rank = [0; MAX];
        for i in 0..MAX {
            parent[i] = i;
        }
        UnionFind { parent, rank }
    }

    fn find(&mut self, x: usize) -> usize {
        if x != self.parent[x] {
            self.parent[x] = self.find(self.parent[x]);
        }
        self.parent[x]
    }

    fn union(&mut self, x: usize, y: usize) {
        let root_x = self.find(x);
        let root_y = self.find(y);
        if root_x != root_y {
            if self.rank[root_x] > self.rank[root_y] {
                self.parent[root_y] = root_x;
            } else {
                self.parent[root_x] = root_y;
                if self.rank[root_x] == self.rank[root_y] {
                    self.rank[root_y] += 1;
                }
            }
        }
    }
}

#[derive(Default, Clone)]
struct TreeNode {
    used: bool,
    m: usize,
    mx: usize,
    xl: isize,
    xr: isize,
    par: usize,
    ch: usize,
    sib: usize,
}

struct StabilityChecker {
    map: [[char; MAX_COLS]; MAX_ROWS],
    rel: [[usize; MAX_COLS]; MAX_ROWS],
    tbl: Vec<TreeNode>,
    tr: [isize; MAX],
    uf: UnionFind,
    w: usize,
    h: usize,
}

impl StabilityChecker {
    fn new() -> Self {
        StabilityChecker {
            map: [['.'; MAX_COLS]; MAX_ROWS],
            rel: [[0; MAX_COLS]; MAX_ROWS],
            tbl: vec![TreeNode::default(); 100],
            tr: [-1; MAX],
            uf: UnionFind::new(),
            w: 0,
            h: 0,
        }
    }

    fn mktree(&mut self, r: usize, c: usize, this: usize) {
        if r == self.h - 1 {
            if self.map[r][c] == '.' {
                return;
            }
            if self.tbl[this].xl < 0 {
                self.tbl[this].xl = c as isize;
            } else if c as isize < self.tbl[this].xl {
                self.tbl[this].xl = c as isize;
            }
            if c > self.tbl[this].xr as usize {
                self.tbl[this].xr = c as isize;
            }
            return;
        }
        if self.map[r + 1][c] == '.' {
            return;
        }

        let k = self.rel[r + 1][c];
        let p = self.tr[k];
        if p == this as isize {
            return;
        }

        if self.tbl[this].xl < 0 {
            self.tbl[this].xl = c as isize;
        } else if c as isize < self.tbl[this].xl {
            self.tbl[this].xl = c as isize;
        }
        if c > self.tbl[this].xr as usize {
            self.tbl[this].xr = c as isize;
        }
        if self.tbl[this].par != 0 {
            return;
        }

        if self.tbl[p as usize].ch == 0 {
            self.tbl[p as usize].ch = this;
        } else {
            let mut s = self.tbl[p as usize].ch;
            while self.tbl[s].sib != 0 {
                s = self.tbl[s].sib;
            }
            self.tbl[s].sib = this;
        }
        self.tbl[this].par = p as usize;
    }

    fn check(&mut self, this: usize) -> bool {
        if self.tbl[this].used {
            return true;
        }
        let ch = self.tbl[this].ch;
        if ch != 0 && !self.check(ch) {
            return false;
        }

        let mut sib = self.tbl[this].sib;
        while sib != 0 {
            if !self.check(sib) {
                return false;
            }
            sib = self.tbl[sib].sib;
        }

        let x = 0.5 + self.tbl[this].mx as f64 / self.tbl[this].m as f64;
        if x <= self.tbl[this].xl as f64 || self.tbl[this].xr as f64 + 1.0 <= x {
            return false;
        }

        let par = self.tbl[this].par;
        self.tbl[par].m += self.tbl[this].m;
        self.tbl[par].mx += self.tbl[this].mx;
        self.tbl[this].used = true;
        true
    }

    fn process(&mut self) {
        let stdin = io::stdin();
        let reader = stdin.lock();
        for line in reader.lines() {
            let line = line.unwrap();
            let dims: Vec<usize> = line.split_whitespace()
                .filter_map(|s| s.parse().ok())
                .collect();
            if dims.len() != 2 || dims[0] == 0 {
                break;
            }
            self.w = dims[0];
            self.h = dims[1];

            for r in 0..self.h {
                let row: Vec<char> = reader.lines().next().unwrap().unwrap().chars().collect();
                self.map[r][..self.w].copy_from_slice(&row[..self.w]);
                for c in 0..self.w {
                    let i = r * self.w + c;
                    self.uf.parent[i] = i;
                    self.rel[r][c] = i;
                }
            }

            for r in 0..self.h {
                for c in 0..self.w {
                    if self.map[r][c] == '.' {
                        continue;
                    }
                    for (dr, dc) in &[(0, 1), (1, 0), (0, -1), (-1, 0)] {
                        let r2 = r as isize + *dr;
                        let c2 = c as isize + *dc;
                        if r2 >= 0 && r2 < self.h as isize && c2 >= 0 && c2 < self.w as isize {
                            if self.map[r2 as usize][c2 as usize] == self.map[r][c] {
                                self.uf.union(self.rel[r][c], self.rel[r2 as usize][c2 as usize]);
                            }
                        }
                    }
                }
            }

            let mut sz = 1;
            self.tbl.iter_mut().for_each(|node| *node = TreeNode::default());
            self.tr.fill(-1);
            for r in (0..self.h).rev() {
                for c in 0..self.w {
                    if self.map[r][c] == '.' {
                        continue;
                    }
                    let k = self.rel[r][c] = self.uf.find(self.rel[r][c]);
                    let i = if self.tr[k] < 0 {
                        self.tr[k] = sz;
                        sz += 1;
                        let tp = &mut self.tbl[self.tr[k] as usize];
                        tp.xl = -1;
                        self.tr[k]
                    } else {
                        self.tr[k]
                    };
                    let tp = &mut self.tbl[i as usize];
                    tp.m += 1;
                    tp.mx += c;
                    self.mktree(r, c, i as usize);
                }
            }
            self.tbl[1].par = 1;

            println!("{}", if self.check(1) { "STABLE" } else { "UNSTABLE" });
        }
    }
}

fn main() {
    let mut checker = StabilityChecker::new();
    checker.process();
}
