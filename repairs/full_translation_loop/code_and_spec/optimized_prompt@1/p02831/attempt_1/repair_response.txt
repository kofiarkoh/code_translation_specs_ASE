```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut inputs = handle.lines().filter_map(Result::ok);

    // Read the two long integers from input
    let a: i64 = inputs.next().expect("Expected first input").trim().parse().expect("Invalid input for a");
    let b: i64 = inputs.next().expect("Expected second input").trim().parse().expect("Invalid input for b");

    // Preconditions check
    if a < 0 {
        panic!("First input must be a non-negative integer.");
    }
    if b <= 0 {
        panic!("Second input must be a positive integer.");
    }

    // Find the smallest multiple of `a` that is also a multiple of `b`
    for i in 1..=b + 1 {
        let multiple = a * i;
        if multiple % b == 0 {
            println!("{}", multiple);
            break;
        }
    }
}
```
