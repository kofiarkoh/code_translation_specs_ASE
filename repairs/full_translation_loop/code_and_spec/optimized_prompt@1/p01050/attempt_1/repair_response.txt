```rust
use std::io::{self, BufRead};

fn main() {
    // Read input from standard input
    let stdin = io::stdin();
    let input = stdin.lock().lines().next().unwrap().unwrap();

    // Ensure the input length does not exceed 99 characters
    if input.len() > 99 {
        // Behavior is not defined for inputs exceeding 99 characters
        return;
    }

    let mut frequency = [0; 256]; // Frequency array for ASCII characters
    let mut unique_count = 0; // Count of unique characters
    let mut consecutive_count = 0; // Count of consecutive characters

    // Count the frequency of each character in the input string
    for ch in input.chars() {
        frequency[ch as usize] += 1;
    }

    // Process the frequency array
    let mut remaining_length = input.len();
    while remaining_length > 0 {
        for i in 0..256 {
            if frequency[i] > 0 {
                unique_count += 1;
                remaining_length -= 1;
                consecutive_count += 1;
                frequency[i] -= 1;
            } else {
                if consecutive_count > 3 {
                    unique_count -= consecutive_count - 3;
                }
                consecutive_count = 0;
            }
        }
    }

    // Output the result
    println!("{}", unique_count);
}
```
