```rust
use std::io::{self, BufRead};

fn main() {
    // Read input from standard input
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read the number of blocks to place
    let num_blocks: usize = lines.next().unwrap().unwrap().parse().unwrap();

    // Initialize counts for orientations 1 to 6
    let mut orientation_counts = [0; 6];

    // Process each block's orientations
    for _ in 0..num_blocks {
        if let Some(Ok(line)) = lines.next() {
            let orientations: Vec<usize> = line
                .split_whitespace()
                .map(|s| s.parse().unwrap())
                .collect();

            // Each block has a top and front orientation
            if orientations.len() == 2 {
                let top = orientations[0];
                let front = orientations[1];

                // Increment the counts for the given orientations
                if (1..=6).contains(&top) {
                    orientation_counts[top - 1] += 1; // Adjust for 0-based index
                }
                if (1..=6).contains(&front) {
                    orientation_counts[front - 1] += 1; // Adjust for 0-based index
                }
            }
        }
    }

    // Output the counts as space-separated integers
    println!(
        "{}",
        orientation_counts
            .iter()
            .map(|count| count.to_string())
            .collect::<Vec<String>>()
            .join(" ")
    );
}
```
