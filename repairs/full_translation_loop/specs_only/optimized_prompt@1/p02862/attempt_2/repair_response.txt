use std::io;

const MOD: usize = 1_000_000_007; // Define a constant for modulo operations
const MAX: usize = 100; // Define a maximum for factorial calculations

// Precompute factorials and their modular inverses
fn precompute_factorials(max: usize) -> (Vec<usize>, Vec<usize>) {
    let mut fact = vec![1; max + 1];
    let mut inv_fact = vec![1; max + 1];

    for i in 2..=max {
        fact[i] = fact[i - 1] * i % MOD;
    }

    inv_fact[max] = mod_inverse(fact[max]);
    for i in (1..max).rev() {
        inv_fact[i] = inv_fact[i + 1] * (i + 1) % MOD;
    }

    (fact, inv_fact)
}

// Function to compute modular inverse using Fermat's Little Theorem
fn mod_inverse(x: usize) -> usize {
    let mut res = 1;
    let mut exp = MOD - 2;
    let mut base = x;

    while exp > 0 {
        if exp % 2 == 1 {
            res = res * base % MOD;
        }
        base = base * base % MOD;
        exp /= 2;
    }
    res
}

// Function to calculate the number of distinct paths
fn calculate_paths(x: usize, y: usize, fact: &[usize], inv_fact: &[usize]) -> usize {
    if (x + y) % 3 != 0 || x < 0 || y < 0 {
        return 0; // Return 0 if conditions are not met
    }

    let total_steps = x + y;
    let steps_to_x = x; // Number of steps to the right
    let steps_to_y = y; // Number of steps down

    // Calculate the number of distinct paths using combinatorial formula
    let paths = fact[total_steps] * inv_fact[steps_to_x] % MOD * inv_fact[steps_to_y] % MOD;
    paths
}

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let coords: Vec<isize> = input
        .trim()
        .split_whitespace()
        .filter_map(|s| s.parse().ok())
        .collect();

    if coords.len() != 2 {
        println!("0"); // Output 0 if input is not exactly two integers
        return;
    }

    let (x, y) = (coords[0], coords[1]);
    
    // Adjust for negative coordinates
    if x < 0 || y < 0 {
        println!("0");
        return;
    }

    let (fact, inv_fact) = precompute_factorials(MAX); // Precompute factorials

    let result = calculate_paths(x as usize, y as usize, &fact, &inv_fact);
    println!("{}", result); // Output the result
}
