fn compute_combinations(n: usize, keys: &str) -> i32 {
    const MODULO: i32 = 10007;

    // Preconditions: N must be non-negative and keys must have length N
    assert!(n >= 0);
    assert_eq!(keys.len(), n);
    
    // Initialize a schedule array to maintain counts of combinations
    let mut schedule = vec![0; 3]; // For 'J', 'O', 'I'
    
    // Map characters to indices
    let char_to_index = |c: char| match c {
        'J' => 0,
        'O' => 1,
        'I' => 2,
        _ => unreachable!(), // This should never happen due to preconditions
    };

    // Process each character in the input string
    for c in keys.chars() {
        let index = char_to_index(c);
        
        // Update the schedule based on the current character
        schedule[index] = (schedule[index] + 1) % MODULO;
    }

    // Calculate the total combinations
    let total_combinations: i32 = schedule.iter().sum();
    
    // Return the total combinations modulo 10007
    total_combinations % MODULO
}

fn main() {
    // Example usage
    let n = 5;
    let keys = "JOOIJ";
    let result = compute_combinations(n, keys);
    println!("{}", result);
}
