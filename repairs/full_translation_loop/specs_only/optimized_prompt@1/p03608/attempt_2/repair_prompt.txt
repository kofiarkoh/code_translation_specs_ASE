struct Graph {

    edges: Vec<(usize, usize, i32)>, // (from, to, weight)

impl Graph {
    fn new(nodes: usize, edges: Vec<(usize, usize, i32)>) -> Self {
        Graph { nodes, edges }
    }

    fn floyd_warshall(&self) -> Vec<Vec<i32>> {
        let mut dist = vec![vec![MAX_DISTANCE; self.nodes]; self.nodes];

        // Initialize distances based on edges
        for (from, to, weight) in &self.edges {
            dist[*from][*to] = (*weight).min(dist[*from][*to]);
            dist[*to][*from] = (*weight).min(dist[*to][*from]); // Assuming undirected graph
        }

        // Distance to self is zero
        for i in 0..self.nodes {
            dist[i][i] = 0;
        }

        // Floyd-Warshall algorithm
        for k in 0..self.nodes {
            for i in 0..self.nodes {
                for j in 0..self.nodes {
                    if dist[i][k] != MAX_DISTANCE && dist[k][j] != MAX_DISTANCE {
                        dist[i][j] = dist[i][j].min(dist[i][k] + dist[k][j]);
                    }
                }
            }
        }

        dist
    }
