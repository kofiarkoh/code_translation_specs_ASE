use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let line = line.unwrap();
        let mut parts = line.split_whitespace();
        let n: i32 = parts.next().unwrap().parse().unwrap();
        let x: i32 = parts.next().unwrap().parse().unwrap();

        // Termination condition
        if n == 0 && x == 0 {
            break;
        }

        // Count unique combinations of three distinct integers that sum to x
        let count = count_combinations(n, x);
        println!("{}", count);
    }
}

// Function to count unique combinations of three distinct integers from 1 to n that sum to x
fn count_combinations(n: i32, x: i32) -> i32 {
    let mut count = 0;

    // Iterate through all combinations of three distinct integers
    for i in 1..=n {
        for j in (i + 1)..=n {
            for k in (j + 1)..=n {
                if i + j + k == x {
                    count += 1;
                }
            }
        }
    }

    // Each combination is counted once, so we return the count directly
    count
}
