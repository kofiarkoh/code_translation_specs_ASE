use std::collections::HashMap;

pub fn min_expression_value(expression: &str) -> i64 {
    // Memoization to store already computed results for sub-expressions
    let mut memo: HashMap<String, i64> = HashMap::new();
    
    fn evaluate(expr: &str, memo: &mut HashMap<String, i64>) -> i64 {
        // Check if the result is already computed
        if let Some(&result) = memo.get(expr) {
            return result;
        }

        // Base case: if the expression is a single number
        if let Ok(num) = expr.trim().parse::<i64>() {
            return num;
        }

        let mut min_value = i64::MAX;

        // Iterate through the expression to find operators
        for i in 0..expr.len() {
            let c = expr.chars().nth(i).unwrap();
            if c == '+' || c == '-' || c == '*' {
                // Split the expression into left and right parts
                let left = &expr[..i];
                let right = &expr[i + 1..];

                // Evaluate both sides recursively
                let left_value = evaluate(left, memo);
                let right_value = evaluate(right, memo);

                // Calculate the result based on the operator
                let result = match c {
                    '+' => left_value + right_value,
                    '-' => left_value - right_value,
                    '*' => left_value * right_value,
                    _ => unreachable!(),
                };

                // Update the minimum value found
                min_value = min_value.min(result);
            }
        }

        // Store the computed result in the memoization map
        memo.insert(expr.to_string(), min_value);
        min_value
    }

    // Call the evaluate function on the full expression
    evaluate(expression, &mut memo)
}
