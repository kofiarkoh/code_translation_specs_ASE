use std::collections::HashMap;

fn longest_anagram_substring(s1: &str, s2: &str) -> usize {
    // Create a frequency map for characters in s2
    let mut s2_freq = HashMap::new();
    for c in s2.chars() {
        *s2_freq.entry(c).or_insert(0) += 1;
    }

    let mut max_length = 0;

    // Iterate over all possible substrings of s1
    for start in 0..s1.len() {
        let mut s1_freq = HashMap::new();
        for end in start..s1.len() {
            let c = s1.chars().nth(end).unwrap(); // Safe due to loop bounds
            *s1_freq.entry(c).or_insert(0) += 1;

            // Check if the current substring's frequency map is a valid anagram of any substring in s2
            if is_anagram(&s1_freq, &s2_freq) {
                max_length = max_length.max(end - start + 1);
            }
        }
    }

    max_length
}

// Helper function to determine if two frequency maps are anagrams
fn is_anagram(freq1: &HashMap<char, usize>, freq2: &HashMap<char, usize>) -> bool {
    // Check if freq1 matches freq2 for all characters
    for (key, &value) in freq1 {
        if value != *freq2.get(key).unwrap_or(&0) {
            return false;
        }
    }
    // Also check if freq2 has any extra characters not in freq1
    for (key, &value) in freq2 {
        if value != *freq1.get(key).unwrap_or(&0) {
            return false;
        }
    }
    true
}

    let result = longest_anagram_substring(s1, s2);
    println!("{}", result);
}
