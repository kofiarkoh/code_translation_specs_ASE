```rust
// AOJ 2251 Merry Christmas
// 2018.3.6 bal4u

use std::io::{self, Read};

const MAX: usize = 2002;
const INF: i32 = 0x1010;

fn bpm(u: usize, hi: &[usize], to: &[[usize; MAX]; MAX], seen: &mut [u8], match_arr: &mut [i32; MAX]) -> bool {
    for i in 0..hi[u] {
        let v = to[u][i];
        if seen[v] != 0 {
            continue;
        }
        seen[v] = 1;
        if match_arr[v] < 0 || bpm(match_arr[v] as usize, hi, to, seen, match_arr) {
            match_arr[u] = v as i32;
            match_arr[v] = u as i32;
            return true;
        }
    }
    false
}

fn in_number(handle: &mut dyn Read) -> i32 {
    let mut n = 0;
    let mut buffer = [0; 1];
    let mut c = handle.read(&mut buffer).unwrap();
    
    while c > 0 && buffer[0].is_ascii_digit() {
        n = (n << 3) + (n << 1) + (buffer[0] as i32 - b'0' as i32);
        c = handle.read(&mut buffer).unwrap();
    }
    n
}

fn main() {
    let stdin = io::stdin();
    let mut handle = stdin.lock();

    let mut hi = [0; MAX];
    let mut to = [[0; MAX]; MAX];
    let mut seen = [0; MAX];
    let mut match_arr = [-1; MAX];
    let mut dist = [[INF; 102]; 102];
    let mut p = [0; 1002];
    let mut t = [0; 1002];

    loop {
        let n = in_number(&mut handle);
        if n == 0 {
            break;
        }
        let m = in_number(&mut handle);
        let l = in_number(&mut handle);

        for k in 0..102 {
            for j in 0..102 {
                dist[k][j] = INF;
            }
            dist[k][k] = 0;
        }

        for _ in 0..m {
            let u = in_number(&mut handle);
            let v = in_number(&mut handle);
            let d = in_number(&mut handle);
            dist[u as usize][v as usize] = d;
            dist[v as usize][u as usize] = d;
        }

        for k in 0..n {
            for u in 0..n {
                if dist[u][k] != INF {
                    for v in 0..n {
                        if dist[k][v] != INF {
                            let d = dist[u][k] + dist[k][v];
                            if dist[u][v] > d {
                                dist[u][v] = d;
                            }
                        }
                    }
                }
            }
        }

        for k in 0..MAX {
            hi[k] = 0;
        }
        for k in 0..l {
            p[k] = in_number(&mut handle);
            t[k] = in_number(&mut handle);
        }
        for u in 0..l {
            for v in 0..l {
                if u != v {
                    if dist[p[u] as usize][p[v] as usize] <= t[v] - t[u] {
                        to[u][hi[u]] = v + l;
                        hi[u] += 1;
                    }
                }
            }
        }

        for k in 0..l {
            match_arr[k] = -1;
        }
        let mut k = l;
        for u in 0..l {
            for j in 0..l {
                seen[j] = 0;
            }
            if bpm(u, &hi, &to, &mut seen, &mut match_arr) {
                k -= 1;
            }
        }
        println!("{}", k);
    }
}
```
