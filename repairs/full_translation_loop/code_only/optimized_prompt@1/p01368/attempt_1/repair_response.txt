```rust
// AOJ 2251 Merry Christmas
// 2018.3.6 bal4u

use std::io::{self, BufRead};

const MAX: usize = 2002;
const INF: i32 = 0x1010;

let mut hi = [0; MAX];
let mut to = [[0; MAX]; MAX];
let mut seen = [0; MAX];
let mut match_arr = [-1; MAX];
let mut dist = [[INF; 102]; 102];
let mut p = [0; 1002];
let mut t = [0; 1002];

fn bpm(u: usize) -> bool {
    for i in 0..hi[u] {
        let v = to[u][i];
        if seen[v] != 0 {
            continue;
        }
        seen[v] = 1;
        if match_arr[v] < 0 || bpm(match_arr[v] as usize) {
            match_arr[u] = v;
            match_arr[v] = u as i32;
            return true;
        }
    }
    false
}

fn in_number() -> i32 {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut n = 0;
    let mut c = handle.bytes().next().unwrap().unwrap() as char;

    while c.is_digit(10) {
        n = (n << 3) + (n << 1) + (c as i32 - '0' as i32);
        c = handle.bytes().next().unwrap().unwrap() as char;
    }
    n
}

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();

    loop {
        let n = in_number();
        if n == 0 {
            break;
        }
        let m = in_number();
        let l = in_number();

        for k in 0..102 {
            for j in 0..102 {
                dist[k][j] = INF;
            }
            dist[k][k] = 0;
        }

        for _ in 0..m {
            let u = in_number();
            let v = in_number();
            let d = in_number();
            dist[u as usize][v as usize] = d;
            dist[v as usize][u as usize] = d;
        }

        for k in 0..n {
            for u in 0..n {
                if dist[u][k] != INF {
                    for v in 0..n {
                        if dist[k][v] != INF {
                            let d = dist[u][k] + dist[k][v];
                            if dist[u][v] > d {
                                dist[u][v] = d;
                            }
                        }
                    }
                }
            }
        }

        for k in 0..MAX {
            hi[k] = 0;
        }
        for k in 0..l {
            p[k] = in_number();
            t[k] = in_number();
        }
        for u in 0..l {
            for v in 0..l {
                if u != v {
                    if dist[p[u] as usize][p[v] as usize] <= t[v] - t[u] {
                        to[u][hi[u]] = v + l;
                        hi[u] += 1;
                    }
                }
            }
        }

        for k in 0..l {
            match_arr[k] = -1;
        }
        let mut k = l;
        for u in 0..l {
            for j in 0..l {
                seen[j] = 0;
            }
            if bpm(u) {
                k -= 1;
            }
        }
        println!("{}", k);
    }
}
```
