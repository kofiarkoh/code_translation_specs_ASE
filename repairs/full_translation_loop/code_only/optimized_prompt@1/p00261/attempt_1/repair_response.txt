```rust
use std::io::{self, BufRead};

const A: usize = 0;
const X: usize = 1;
const Y: usize = 2;
const Z: usize = 3;
const W: usize = 4;
const B: usize = 5;
const NA: usize = 6;

const MAP: [[usize; 2]; 6] = [
    [X, Y],
    [NA, Z],
    [X, NA],
    [W, B],
    [B, Y],
    [Y, X],
];

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let path = line.unwrap();
        if path.starts_with('#') {
            break;
        }

        let mut city = A;
        let mut i = 0;

        while i < path.len() {
            let index = path.as_bytes()[i] as usize - b'0' as usize;
            if MAP[city][index] == NA {
                break;
            }
            city = MAP[city][index];
            i += 1;
        }

        if city == B {
            println!("Yes");
        } else {
            println!("No");
        }
    }
}
```
