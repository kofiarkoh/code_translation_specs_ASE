// AOJ 1055 Huge Family

use std::io::{self, Read};

const MAX: usize = 100002;

struct UnionFind {
    id: [usize; MAX],
    size: [usize; MAX],
}

impl UnionFind {
    fn new(n: usize) -> Self {
        let mut id = [0; MAX];
        let size = [1; MAX];
        for i in 0..n {
            id[i] = i;
        }
        UnionFind { id, size }
    }

    fn root(&mut self, mut i: usize) -> usize {
        while i != self.id[i] {
            self.id[i] = self.id[self.id[i]]; // Path compression
            i = self.id[i];
        }
        i
    }

    fn connected(&mut self, p: usize, q: usize) -> bool {
        self.root(p) == self.root(q)
    }

    fn unite(&mut self, p: usize, q: usize) {
        let i = self.root(p);
        let j = self.root(q);
        if i == j {
            return;
        }
        if self.size[i] < self.size[j] {
            self.id[i] = j;
            self.size[j] += self.size[i];
        } else {
            self.id[j] = i;
            self.size[i] += self.size[j];
        }
    }
}

fn in_number() -> usize {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut n = 0;
    let mut buffer = [0; 1];
    let mut c = handle.read(&mut buffer).unwrap();
    while c > 0 && buffer[0] >= b'0' {
        n = (n << 3) + (n << 1) + (buffer[0] as usize & 0xf);
        c = handle.read(&mut buffer).unwrap();
    }
    n
}

fn check(k: usize, f: usize, cnt: &mut [usize], max: &mut [usize]) {
    if cnt[k] == 0 || f > max[k] {
        cnt[k] = 1;
        max[k] = f;
    } else if f == max[k] {
        cnt[k] += 1;
    }
}

fn main() {
    let mut f = [[0; 2]; MAX];
    let mut max = [0; MAX];
    let mut cnt = [0; MAX];

    loop {
        let n = in_number();
        if n == 0 {
            break;
        }

        let mut uf = UnionFind::new(n);

        for i in 0..n {
            let a = in_number();
            f[i][0] = in_number();
            let b = in_number();
            f[i][1] = in_number();
            uf.unite(i, a);
            uf.unite(i, b);
        }

        cnt.fill(0);
        for i in 0..n {
            let k = uf.root(i);
            check(k, f[i][0], &mut cnt, &mut max);
            check(k, f[i][1], &mut cnt, &mut max);
        }

        let mut ans = 1;
        for i in 0..n {
            if cnt[i] != 0 {
                ans = ans * (cnt[i] >> 1) % 10007;
            }
        }
        println!("{}", ans);
    }
}
