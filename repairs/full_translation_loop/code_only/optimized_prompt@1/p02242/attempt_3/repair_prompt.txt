fn dijkstra_noheap(node: &mut [Node], n: usize) {
    for i in 0..n {
        node[i].cost = -1;
        node[i].visited = 0;
    }
    node[0].cost = 0;

        for it in node.iter_mut() {

                continue;

            }
        }
    }
