```rust
use std::io::{self, BufRead};

const N: usize = 200000;
const INF: i64 = 0x3f3f3f3f3f3f3f3f;

fn min(a: i64, b: i64) -> i64 {
    if a < b { a } else { b }
}

fn update(ft: &mut [i64], mut i: usize, n: usize, x: i64) {
    while i < n {
        ft[i] = min(ft[i], x);
        i |= i + 1;
    }
}

fn query(ft: &[i64], mut i: usize) -> i64 {
    let mut x = INF;

    while i != usize::MAX {
        x = min(x, ft[i]);
        i &= i + 1;
        if i == 0 { break; } // Prevent underflow
        i -= 1;
    }
    x
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut params = first_line.split_whitespace().map(|s| s.parse::<i32>().unwrap());
    let n = params.next().unwrap() as usize;
    let q = params.next().unwrap();
    let mut i = params.next().unwrap() - 1; // Convert to 0-based index
    let mut j = params.next().unwrap() - 1; // Convert to 0-based index

    let mut dp = vec![INF; n];
    let mut ft1 = vec![INF; n];
    let mut ft2 = vec![INF; n];

    update(&mut ft1, j as usize, n, -j as i64);
    update(&mut ft2, n - 1 - j as usize, n, j as i64);

    let mut d = 0;

    for _ in 0..q {
        let i_ = lines.next().unwrap().unwrap().parse::<i32>().unwrap() - 1; // Convert to 0-based index
        d += (i - i_).abs();
        let x = min(query(&ft1, i_ as usize) + i_ as i64, query(&ft2, (n - 1 - i_) as usize) - i_ as i64) - (i - i_).abs();
        dp[i as usize] = min(dp[i as usize], x);
        update(&mut ft1, i as usize, n, x - i as i64);
        update(&mut ft2, n - 1 - i as usize, n, x + i as i64);
        i = i_;
    }

    let ans = dp.iter().cloned().fold(INF, min) + d as i64; // Ensure d is i64
    println!("{}", ans);
}
```
