use std::io::{self, BufRead};
use std::cmp::{max, min};

const INF: i32 = 1072114514;

fn zt(a: i32, b: i32) -> i32 {
    max(a, b) - min(a, b)
}

fn round(a: i32, b: i32) -> i32 {
    if (a % b) * 2 >= b {
        a / b + 1
    } else {
        a / b
    }
}

fn ceil(a: i32, b: i32) -> i32 {
    if a % b == 0 {
        a / b
    } else {
        a / b + 1
    }
}

fn gcd(mut a: i32, mut b: i32) -> i32 {
    while b != 0 {
        let c = a % b;
        a = b;
        b = c;
    }
    a
}

fn lcm(a: i32, b: i32) -> i32 {
    let c = gcd(a, b);
    a / c * b
}

fn nCr(a: i32, b: i32) -> i32 {
    let mut r = 1;
    for i in 1..=b {
        r *= a + 1 - i;
        r /= i;
    }
    r
}

fn nHr(a: i32, b: i32) -> i32 {
    nCr(a + b - 1, b)
}

fn fact(a: i32) -> i32 {
    let mut r = 1;
    for i in 1..=a {
        r *= i;
    }
    r
}

fn pow(a: i32, b: i32) -> i32 {
    let mut r = 1;
    for _ in 1..=b {
        r *= a;
    }
    r
}

fn dsum(mut x: i32) -> i32 {
    let mut r = 0;
    while x != 0 {
        r += x % 10;
        x /= 10;
    }
    r
}

fn dsumb(mut x: i32, b: i32) -> i32 {
    let mut r = 0;
    while x != 0 {
        r += x % b;
        x /= b;
    }
    r
}

fn sankaku(x: i32) -> i32 {
    (1 + x) * x / 2
}

fn shuffledget(x: &mut [i32], n: usize) {
    let mut b: Vec<usize> = (0..n).collect();
    let mut rng = rand::thread_rng();
    for i in (1..=n).rev() {
        let p = rand::random::<usize>() % i;
        b.swap(i - 1, p);
    }
    for i in 0..n {
        x[b[i]] = read_input();
    }
}

fn read_input() -> i32 {
    let stdin = io::stdin();
    let mut handle = stdin.lock(); // Make handle mutable
    let mut input = String::new();
    handle.read_line(&mut input).unwrap();
    input.trim().parse().unwrap()
}

fn search(x: i32, a: &[i32], n: usize) -> usize {
    let mut st = 0;
    let mut fi = n as isize - 1;
    while st <= fi {
        let te = (st + fi) / 2;
        if a[te as usize] < x {
            st = te + 1;
        } else {
            fi = te - 1;
        }
    }
    st as usize
}

fn prarr(arr: &[i32]) {

    println!("{}", output);
}

fn nCrbit(n: i64, r: i64) -> i64 {
    if n < 0 || r < 0 || n < r || (n | r) != n {
        return 0;
    }
    1
}

fn findcol(a: i64, b: i64, x: &[i64], y: &[i64], n: usize) -> i64 {
    let mut r = 0;
    for i in 0..n {
        if a - x[i] < 0 || b - y[i] < 0 {
            continue;
        }
        let p = (a - x[i]) + (b - y[i]);
        let q = b - y[i]; // Removed unnecessary parentheses
        r ^= nCrbit(p, q);
    }
    r
}

fn main() {

        y[i as usize] = read_input();

        let w = 1 << k;
        if findcol(CVAL - (r + w), r + w, &x.iter().map(|&v| v as i64).collect::<Vec<_>>(), &y.iter().map(|&v| v as i64).collect::<Vec<_>>(), n as usize) == 1 {
            r += w;
        }
    }
    
    println!("{}", CVAL - r);
}
