use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i64 = it.next().unwrap().parse().unwrap();
    let k: i64 = it.next().unwrap().parse().unwrap();

    // Memory proportional to N, as required by the specification.
    let mut s: Vec<i64> = vec![0; n as usize];

    for (idx, sj) in s.iter_mut().enumerate() {
        let j = (idx as i64) + 1;

        // Choose s_j as the smallest power of two such that K <= s_j * j.
        // Implemented via: s_j = 2^ceil(log2(ceil(K/j))).
        // This avoids per-term multiplication overflow in the loop under the spec's assumptions.
        let need = (k + j - 1) / j; // ceil(K / j), with positive integers per preconditions.

        let mut p: i64 = 1;
        while p < need {
            p *= 2;
        }
        *sj = p;
    }

    let mut ans: f64 = 0.0;
    for &p in &s {
        ans += 1.0f64 / (p as f64);
    }

    print!("{:.12}", ans);
}
