```rust
use std::io::{self, Read};

const MOD: i64 = 10007;

struct DSU {
    parent: Vec<usize>,
    size: Vec<usize>,
}

impl DSU {
    fn new(n: usize) -> Self {
        let mut parent = Vec::with_capacity(n + 1);
        let mut size = Vec::with_capacity(n + 1);
        for i in 0..=n {
            parent.push(i);
            size.push(1);
        }
        Self { parent, size }
    }

    fn find(&mut self, x: usize) -> usize {
        if self.parent[x] == x {
            return x;
        }
        let r = self.find(self.parent[x]);
        self.parent[x] = r;
        r
    }

    fn union(&mut self, a: usize, b: usize) {
        let mut ra = self.find(a);
        let mut rb = self.find(b);
        if ra == rb {
            return;
        }
        if self.size[ra] < self.size[rb] {
            std::mem::swap(&mut ra, &mut rb);
        }
        self.parent[rb] = ra;
        self.size[ra] += self.size[rb];
    }
}

fn main() {
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_whitespace();

    let mut out = String::new();

    loop {
        let Some(tok) = it.next() else { break };
        let n: usize = tok.parse().unwrap();
        if n == 0 {
            break;
        }

        // The sample indicates entity indices can include 0..n (inclusive),
        // even though there are n records. Therefore allocate DSU for n+1 nodes.
        let mut dsu = DSU::new(n + 1);

        // Store per-record endpoints and values; values are associated with the endpoints.
        let mut x = vec![0usize; n];
        let mut y = vec![0usize; n];
        let mut vx = vec![0i64; n];
        let mut vy = vec![0i64; n];

        for i in 0..n {
            let a: usize = it.next().unwrap().parse().unwrap();
            let b: usize = it.next().unwrap().parse().unwrap();
            let va: i64 = it.next().unwrap().parse().unwrap();
            let vb: i64 = it.next().unwrap().parse().unwrap();

            x[i] = a;
            y[i] = b;
            vx[i] = va;
            vy[i] = vb;

            dsu.union(a, b);
        }

        // Track per-component maximum and its occurrence count across all attributes.
        let mut maxv = vec![i64::MIN; n + 2];
        let mut cnt = vec![0i64; n + 2];

        // Add all attribute values to their component (by endpoint).
        for i in 0..n {
            let r1 = dsu.find(x[i]);
            let v1 = vx[i];
            if v1 > maxv[r1] {
                maxv[r1] = v1;
                cnt[r1] = 1;
            } else if v1 == maxv[r1] {
                cnt[r1] += 1;
            }

            let r2 = dsu.find(y[i]);
            let v2 = vy[i];
            if v2 > maxv[r2] {
                maxv[r2] = v2;
                cnt[r2] = 1;
            } else if v2 == maxv[r2] {
                cnt[r2] += 1;
            }
        }

        // Multiply F(C)=floor(T(C)/2) over all components that actually appear
        // (i.e., have at least one attribute contributed).
        let mut ans: i64 = 1;
        for node in 0..=n + 1 {
            if dsu.find(node) == node && cnt[node] > 0 {
                let f = cnt[node] / 2;
                ans = (ans * (f % MOD)) % MOD;
            }
        }

        out.push_str(&(ans.rem_euclid(MOD)).to_string());
        out.push('\n');
    }

    print!("{out}");
}
```
