```rust
use std::io::{self, Read};

fn time_to_sec(h: i32, m: i32, s: i32) -> usize {
    // Preconditions say inputs are valid times within a 24-hour day.
    // We keep this simple and deterministic.
    (h as usize) * 3600 + (m as usize) * 60 + (s as usize)
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut outputs: Vec<String> = Vec::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break; // Unspecified behavior for missing data; terminate quietly.
        }
        let n: usize = n_opt.unwrap().parse().unwrap();
        if n == 0 {
            break;
        }

        // One-second resolution over a 24-hour day.
        // We use a difference array over [0, 86400] to model half-open intervals [start, end).
        // Index 86400 is a sentinel boundary (end of day).
        let mut diff = vec![0i32; 86401];

        for _ in 0..n {
            let sh: i32 = it.next().unwrap().parse().unwrap();
            let sm: i32 = it.next().unwrap().parse().unwrap();
            let ss: i32 = it.next().unwrap().parse().unwrap();
            let eh: i32 = it.next().unwrap().parse().unwrap();
            let em: i32 = it.next().unwrap().parse().unwrap();
            let es: i32 = it.next().unwrap().parse().unwrap();

            let start = time_to_sec(sh, sm, ss);
            let end = time_to_sec(eh, em, es);

            // Inclusion is: at or after start and strictly before end.
            // If start == end, contributes to no seconds (no-op).
            if start < end {
                diff[start] += 1;
                diff[end] -= 1;
            }
        }

        let mut cur: i32 = 0;
        let mut maxv: i32 = 0;

        // Evaluate overlap for each second in the day: 0..86400 (86400 seconds).
        for t in 0..86400 {
            cur += diff[t];
            if cur > maxv {
                maxv = cur;
            }
        }

        outputs.push(maxv.to_string());
    }

    print!("{}", outputs.join("\n"));
}
```
