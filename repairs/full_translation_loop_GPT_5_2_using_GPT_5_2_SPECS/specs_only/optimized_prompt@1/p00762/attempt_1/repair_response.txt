```rust
use std::io::{self, Read};

const N: usize = 25; // fixed bounded grid; must be large enough per preconditions
const CENTER: i32 = (N as i32) / 2;

#[derive(Clone, Copy)]
struct Die {
    // Faces: values 1..6, opposite faces sum to 7.
    // We store current orientation by naming the value on each face.
    top: i32,
    bottom: i32,
    front: i32,
    back: i32,
    left: i32,
    right: i32,
}

impl Die {
    fn new_from_top_front(top: i32, front: i32) -> Self {
        // Preconditions guarantee a physically consistent orientation.
        // We reconstruct left/right by choosing the unique die orientation consistent with:
        // - opposite faces sum to 7
        // - given top and front
        // - a fixed right-hand convention for left/right
        let bottom = 7 - top;
        let back = 7 - front;

        // Determine left/right using a fixed canonical die model.
        // Canonical model:
        // top=1, bottom=6, front=2, back=5, left=3, right=4.
        // For any (top, front) that is physically possible, there is exactly one (left, right)
        // consistent with this model.
        let (left, right) = canonical_left_right(top, front);

        Die {
            top,
            bottom,
            front,
            back,
            left,
            right,
        }
    }

    fn roll_north(&mut self) {
        // Move to y-1: cube rolls so that front face becomes top.
        let old_top = self.top;
        self.top = self.front;
        self.front = self.bottom;
        self.bottom = self.back;
        self.back = old_top;
    }

    fn roll_south(&mut self) {
        // Move to y+1: cube rolls so that back face becomes top.
        let old_top = self.top;
        self.top = self.back;
        self.back = self.bottom;
        self.bottom = self.front;
        self.front = old_top;
    }

    fn roll_west(&mut self) {
        // Move to x-1: cube rolls so that right face becomes top.
        let old_top = self.top;
        self.top = self.right;
        self.right = self.bottom;
        self.bottom = self.left;
        self.left = old_top;
    }

    fn roll_east(&mut self) {
        // Move to x+1: cube rolls so that left face becomes top.
        let old_top = self.top;
        self.top = self.left;
        self.left = self.bottom;
        self.bottom = self.right;
        self.right = old_top;
    }
}

fn canonical_left_right(top: i32, front: i32) -> (i32, i32) {
    // Build the canonical die and enumerate all 24 orientations by BFS of rolls.
    // Then pick the one matching (top, front) and return (left, right).
    let start = Die {
        top: 1,
        bottom: 6,
        front: 2,
        back: 5,
        left: 3,
        right: 4,
    };

    let mut seen = std::collections::HashSet::<(i32, i32, i32)>::new();
    let mut q = std::collections::VecDeque::new();
    q.push_back(start);

    while let Some(d) = q.pop_front() {
        let key = (d.top, d.front, d.left);
        if !seen.insert(key) {
            continue;
        }
        if d.top == top && d.front == front {
            return (d.left, d.right);
        }
        let mut dn = d;
        dn.roll_north();
        q.push_back(dn);
        let mut ds = d;
        ds.roll_south();
        q.push_back(ds);
        let mut de = d;
        de.roll_east();
        q.push_back(de);
        let mut dw = d;
        dw.roll_west();
        q.push_back(dw);
    }

    // Preconditions say this cannot happen; keep a deterministic fallback.
    (3, 4)
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: i32 = n_opt.unwrap().parse().unwrap();
        if n == 0 {
            break;
        }

        let mut height = vec![vec![0i32; N]; N];
        let mut top_face = vec![vec![0i32; N]; N];

        for _ in 0..n {
            let top: i32 = it.next().unwrap().parse().unwrap();
            let front: i32 = it.next().unwrap().parse().unwrap();

            let mut die = Die::new_from_top_front(top, front);

            let mut x = CENTER;
            let mut y = CENTER;

            loop {
                let cur_h = height[y as usize][x as usize];

                // Eligible moves: adjacent cell with strictly lower height.
                // For each eligible direction, consider the face that would be exposed on top
                // after rolling into that cell. If that exposed value > 3, it is a candidate.
                // Choose the move with the highest such exposed value among eligible directions.
                // If no candidate, stop.
                let mut best_val = -1i32;
                let mut best_dir = 0i32; // 1=N,2=S,3=W,4=E

                // North
                if y - 1 >= 0 {
                    let nh = height[(y - 1) as usize][x as usize];
                    if nh < cur_h {
                        let mut d = die;
                        d.roll_north();
                        if d.top > 3 && d.top > best_val {
                            best_val = d.top;
                            best_dir = 1;
                        }
                    }
                }
                // South
                if (y + 1) < N as i32 {
                    let nh = height[(y + 1) as usize][x as usize];
                    if nh < cur_h {
                        let mut d = die;
                        d.roll_south();
                        if d.top > 3 && d.top > best_val {
                            best_val = d.top;
                            best_dir = 2;
                        }
                    }
                }
                // West
                if x - 1 >= 0 {
                    let nh = height[y as usize][(x - 1) as usize];
                    if nh < cur_h {
                        let mut d = die;
                        d.roll_west();
                        if d.top > 3 && d.top > best_val {
                            best_val = d.top;
                            best_dir = 3;
                        }
                    }
                }
                // East
                if (x + 1) < N as i32 {
                    let nh = height[y as usize][(x + 1) as usize];
                    if nh < cur_h {
                        let mut d = die;
                        d.roll_east();
                        if d.top > 3 && d.top > best_val {
                            best_val = d.top;
                            best_dir = 4;
                        }
                    }
                }

                if best_dir == 0 {
                    break;
                }

                match best_dir {
                    1 => {
                        die.roll_north();
                        y -= 1;
                    }
                    2 => {
                        die.roll_south();
                        y += 1;
                    }
                    3 => {
                        die.roll_west();
                        x -= 1;
                    }
                    4 => {
                        die.roll_east();
                        x += 1;
                    }
                    _ => break,
                }
            }

            let yi = y as usize;
            let xi = x as usize;
            height[yi][xi] += 1;
            top_face[yi][xi] = die.top;
        }

        let mut counts = [0i32; 6];
        for y in 0..N {
            for x in 0..N {
                let v = top_face[y][x];
                if (1..=6).contains(&v) {
                    counts[(v - 1) as usize] += 1;
                }
            }
        }

        println!(
            "{} {} {} {} {} {}",
            counts[0], counts[1], counts[2], counts[3], counts[4], counts[5]
        );
    }
}
```
