```rust
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let t1: i64 = it.next().unwrap().parse().unwrap();
    let t2: i64 = it.next().unwrap().parse().unwrap();

    let mut a1: i64 = it.next().unwrap().parse().unwrap();
    let mut a2: i64 = it.next().unwrap().parse().unwrap();
    let mut b1: i64 = it.next().unwrap().parse().unwrap();
    let mut b2: i64 = it.next().unwrap().parse().unwrap();

    // Equality check is on original inputs (swapping does not affect the truth of equality).
    let total_a = a1 * t1 + a2 * t2;
    let total_b = b1 * t1 + b2 * t2;
    if total_a == total_b {
        print!("infinity\n");
        return;
    }

    // Normalization: ensure a1 <= b1 by swapping the per-time-unit pairs if needed.
    if a1 > b1 {
        std::mem::swap(&mut a1, &mut b1);
        std::mem::swap(&mut a2, &mut b2);
    }

    // After normalization:
    // x = advantage of entity B over A in duration 1 (non-negative by construction)
    // y = advantage of entity A over B in duration 2 (can be negative, but finite path assumes divisor non-zero)
    let x = (b1 - a1) * t1;
    let y = (a2 - b2) * t2;

    // If the first-duration advantage strictly exceeds the second-duration disadvantage, output 0.
    // (This corresponds to x > y in the normalized formulation.)
    if x > y {
        print!("0\n");
        return;
    }

    // Finite-count computation.
    // Precondition: (y - x) != 0 (non-zero divisor requirement).
    let denom = y - x;
    let numer = x;

    // Compute q = floor(numer / denom) and remainder in a way consistent with i64 division.
    // The specification assumes inputs satisfy preconditions so that this path is meaningful.
    let q = numer / denom;
    let r = numer % denom;

    // Output: 2*q if exact divisibility, else 2*q + 1.
    let ans = if r == 0 { 2 * q } else { 2 * q + 1 };
    print!("{ans}\n");
}
```
