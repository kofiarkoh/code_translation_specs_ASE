```rust
use std::collections::VecDeque;
use std::io::{self, Read};
use std::process;

fn main() {
    // Read entire stdin; if dimensions/rows cannot be read, exit non-zero with no guaranteed output.
    let mut input = String::new();
    if io::stdin().read_to_string(&mut input).is_err() {
        process::exit(1);
    }

    let mut it = input.split_whitespace();

    let h: usize = match it.next().and_then(|s| s.parse::<usize>().ok()) {
        Some(v) => v,
        None => process::exit(1),
    };
    let w: usize = match it.next().and_then(|s| s.parse::<usize>().ok()) {
        Some(v) => v,
        None => process::exit(1),
    };

    // Preconditions are assumed; behavior is undefined if violated.
    // We still attempt to read exactly h rows; if any row missing, exit non-zero.
    let mut grid: Vec<Vec<u8>> = Vec::with_capacity(h);
    for _ in 0..h {
        let row = match it.next() {
            Some(s) => s.as_bytes(),
            None => process::exit(1),
        };
        // Preconditions guarantee row has at least w characters.
        grid.push(row[..w].to_vec());
    }

    let mut visited = vec![false; h * w];
    let mut q: VecDeque<(usize, usize)> = VecDeque::new();
    let mut ans: u64 = 0;

    for i in 0..h {
        for j in 0..w {
            let idx0 = i * w + j;
            if visited[idx0] {
                continue;
            }

            visited[idx0] = true;
            q.clear();
            q.push_back((i, j));

            let mut dot: u64 = 0;
            let mut nondot: u64 = 0;

            while let Some((x, y)) = q.pop_front() {
                if grid[x][y] == b'.' {
                    dot = dot.wrapping_add(1);
                } else {
                    nondot = nondot.wrapping_add(1);
                }

                let cur = grid[x][y];

                // 4-directional neighbors; traverse only if neighbor char differs.
                if x > 0 {
                    let nx = x - 1;
                    let ny = y;
                    let nidx = nx * w + ny;
                    if !visited[nidx] && grid[nx][ny] != cur {
                        visited[nidx] = true;
                        q.push_back((nx, ny));
                    }
                }
                if x + 1 < h {
                    let nx = x + 1;
                    let ny = y;
                    let nidx = nx * w + ny;
                    if !visited[nidx] && grid[nx][ny] != cur {
                        visited[nidx] = true;
                        q.push_back((nx, ny));
                    }
                }
                if y > 0 {
                    let nx = x;
                    let ny = y - 1;
                    let nidx = nx * w + ny;
                    if !visited[nidx] && grid[nx][ny] != cur {
                        visited[nidx] = true;
                        q.push_back((nx, ny));
                    }
                }
                if y + 1 < w {
                    let nx = x;
                    let ny = y + 1;
                    let nidx = nx * w + ny;
                    if !visited[nidx] && grid[nx][ny] != cur {
                        visited[nidx] = true;
                        q.push_back((nx, ny));
                    }
                }
            }

            ans = ans.wrapping_add(dot.wrapping_mul(nondot));
        }
    }

    println!("{}", ans);
}
```
