#[derive(Clone, Copy)]

fn mod_add(x: i64, y: i64) -> i64 {
    let mut v = x + y;
    if v >= MOD {
        v -= MOD;
    }
    v
}
fn mod_sub(x: i64, y: i64) -> i64 {
    let mut v = x - y;
    if v < 0 {
        v += MOD;
    }
    v
}

struct Fenwick {
    n: usize,
    bit: Vec<i64>,
}
impl Fenwick {
    fn new(n: usize) -> Self {
        Fenwick { n, bit: vec![0; n + 1] }
    }
    fn add(&mut self, mut idx: usize, delta: i64) {
        // idx is 1-based
        while idx <= self.n {
            self.bit[idx] = mod_add(self.bit[idx], delta);
            idx += idx & (!idx + 1);
        }
    }
    fn sum(&self, mut idx: usize) -> i64 {
        // sum over [1..=idx]
        let mut res = 0i64;
        while idx > 0 {
            res = mod_add(res, self.bit[idx]);
            idx &= idx - 1;
        }
        res
    }
}
