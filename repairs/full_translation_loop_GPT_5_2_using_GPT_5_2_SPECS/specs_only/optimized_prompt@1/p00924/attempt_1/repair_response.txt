use std::io::{self, Read};

fn compute_cost(bits: &[u8], runs: &[usize], start_bit: u8) -> u64 {
    // Working copy mutated in-place to model selecting and "moving" bits.
    let mut a = bits.to_vec();
    let n = a.len();

    let mut cost: u64 = 0;
    let mut pos: usize = 0;
    let mut cur = start_bit;

    for &len_run in runs {
        for _ in 0..len_run {
            // Find next occurrence at or to the right of current position.
            let mut j = pos;
            while j < n && a[j] != cur {
                j += 1;
            }
            // Spec: behavior undefined if missing required bits; we keep it total by
            // doing nothing special (would panic on out-of-bounds if we tried to swap).
            if j >= n {
                return cost;
            }

            cost = cost.wrapping_add((j - pos) as u64);

            // Move selected bit into position by shifting the intervening segment right by 1.
            // This preserves the multiset and models stable extraction of the chosen bit.
            let picked = a[j];
            for k in (pos + 1..=j).rev() {
                a[k] = a[k - 1];
            }
            a[pos] = picked;

            pos += 1;
        }
        cur ^= 1;
    }

    cost
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let m: usize = it.next().unwrap().parse().unwrap();

    let mut bits: Vec<u8> = Vec::with_capacity(n);
    for _ in 0..n {
        let tok = it.next().unwrap();
        let b = match tok.as_bytes().get(0).copied() {
            Some(b'0') => 0u8,
            Some(b'1') => 1u8,
            _ => tok.parse::<u8>().unwrap(),
        };
        bits.push(b);
    }

    let mut runs: Vec<usize> = Vec::with_capacity(m);
    for _ in 0..m {
        runs.push(it.next().unwrap().parse().unwrap());
    }

    // Determine feasibility by counts for each possible starting bit.
    let ones = bits.iter().filter(|&&b| b == 1).count();
    let zeros = n - ones;

    let mut need0_start0 = 0usize;
    let mut need1_start0 = 0usize;
    for (i, &len_run) in runs.iter().enumerate() {
        if i % 2 == 0 {
            need0_start0 += len_run;
        } else {
            need1_start0 += len_run;
        }
    }
    let need0_start1 = need1_start0;
    let need1_start1 = need0_start0;

    let feasible0 = zeros == need0_start0 && ones == need1_start0;
    let feasible1 = zeros == need0_start1 && ones == need1_start1;

    // Spec: choose appropriate starting bit (or cheaper of both when both feasible).
    // If neither feasible, behavior is undefined; we still compute both and take min.
    let ans = match (feasible0, feasible1) {
        (true, false) => compute_cost(&bits, &runs, 0),
        (false, true) => compute_cost(&bits, &runs, 1),
        (true, true) => {
            let c0 = compute_cost(&bits, &runs, 0);
            let c1 = compute_cost(&bits, &runs, 1);
            c0.min(c1)
        }
        (false, false) => {
            let c0 = compute_cost(&bits, &runs, 0);
            let c1 = compute_cost(&bits, &runs, 1);
            c0.min(c1)
        }
    };

    print!("{}", ans);
}
