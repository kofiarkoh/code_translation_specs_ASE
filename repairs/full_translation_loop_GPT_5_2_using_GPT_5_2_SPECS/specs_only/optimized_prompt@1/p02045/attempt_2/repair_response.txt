```rust
use std::io::{self, Read};
use std::process;

fn manhattan(a: (i64, i64), b: (i64, i64)) -> i64 {
    (a.0 - b.0).abs() + (a.1 - b.1).abs()
}

fn dist_to_rect(p: (i64, i64), xl: i64, xh: i64, yl: i64, yh: i64) -> i64 {
    // Manhattan distance from point to axis-aligned rectangle (0 if inside).
    let dx = if p.0 < xl {
        xl - p.0
    } else if p.0 > xh {
        p.0 - xh
    } else {
        0
    };
    let dy = if p.1 < yl {
        yl - p.1
    } else if p.1 > yh {
        p.1 - yh
    } else {
        0
    };
    dx + dy
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let n_tok = match it.next() {
            None => break, // initial read fails => terminate normally
            Some(s) => s,
        };
        let n_i64: i64 = match n_tok.parse() {
            Ok(v) => v,
            Err(_) => break, // invalid => terminate normally
        };
        if n_i64 <= 0 {
            break; // non-positive => terminate normally
        }
        let n: usize = n_i64 as usize;

        // If a dataset begins but is incomplete => exit non-zero and no further output.
        let xl: i64 = match it.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => process::exit(1),
        };
        let xh: i64 = match it.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => process::exit(1),
        };
        let yl: i64 = match it.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => process::exit(1),
        };
        let yh: i64 = match it.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => process::exit(1),
        };

        // Read first point, then process each step as we read the next point.
        let mut prev: (i64, i64) = {
            let x: i64 = match it.next().and_then(|s| s.parse().ok()) {
                Some(v) => v,
                None => process::exit(1),
            };
            let y: i64 = match it.next().and_then(|s| s.parse().ok()) {
                Some(v) => v,
                None => process::exit(1),
            };
            (x, y)
        };

        let mut total: i64 = 0;

        for _ in 0..n {
            let x: i64 = match it.next().and_then(|s| s.parse().ok()) {
                Some(v) => v,
                None => process::exit(1),
            };
            let y: i64 = match it.next().and_then(|s| s.parse().ok()) {
                Some(v) => v,
                None => process::exit(1),
            };
            let cur = (x, y);

            let direct = manhattan(prev, cur);

            // The intended model (per sample) is:
            // per-step cost = min( direct,
            //                      max(0, dist_to_rect(prev) - 1),
            //                      dist_to_rect(cur) )
            // i.e., you may either go directly, or "escape" to the rectangle (with 1 discount if outside),
            // or "enter" from the rectangle to the end point.
            let d_prev = dist_to_rect(prev, xl, xh, yl, yh);
            let d_cur = dist_to_rect(cur, xl, xh, yl, yh);
            let escape_only = if d_prev == 0 { 0 } else { (d_prev - 1).max(0) };
            let enter_only = d_cur;

            let step_cost = direct.min(escape_only.min(enter_only));
            total += step_cost;

            prev = cur;
        }

        out.push_str(&total.to_string());
        out.push('\n');
    }

    print!("{out}");
}
```
