fn dist_to_rect(p: (i64, i64), xl: i64, xh: i64, yl: i64, yh: i64) -> i64 {
    // Manhattan distance from point to axis-aligned rectangle (0 if inside).
    let dx = if p.0 < xl {
        xl - p.0
    } else if p.0 > xh {
        p.0 - xh
    } else {
        0
    };
    let dy = if p.1 < yl {
        yl - p.1
    } else if p.1 > yh {
        p.1 - yh
    } else {
        0
    };
    dx + dy
}

fn escape_cost(p: (i64, i64), xl: i64, xh: i64, yl: i64, yh: i64) -> i64 {
    // EscapeCost(p) = dist_to_rect(p) reduced by 1 if outside, but never below 0.
    let d = dist_to_rect(p, xl, xh, yl, yh);

        0
    } else {
        // One-unit discount when escape is needed.
        (d - 1).max(0)
    }
}

}
