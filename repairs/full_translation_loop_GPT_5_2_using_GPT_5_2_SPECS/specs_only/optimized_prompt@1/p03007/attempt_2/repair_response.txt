```rust
use std::io::{self, Read};

fn main() {
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();
    let mut a: Vec<i64> = (0..n).map(|_| it.next().unwrap().parse().unwrap()).collect();
    a.sort_unstable();

    let mut ops: Vec<(i64, i64)> = Vec::with_capacity(n - 1);

    // This construction matches the canonical solution used for this task:
    // - If there is at least one negative and at least one non-negative:
    //   * Start from the maximum element and subtract all negatives (in increasing order).
    //   * Start from the minimum element and subtract all remaining non-negatives except the maximum (in increasing order).
    //   * Finally subtract the second accumulator from the first.
    // - If all are non-negative:
    //   * Start from the minimum and subtract all middle elements.
    //   * Subtract that accumulator from the maximum.
    // - If all are non-positive:
    //   * Start from the maximum and subtract all middle elements (more negative ones).
    //   * Subtract that accumulator from the minimum.
    //
    // The printed sequence is deterministic given the sorted order and matches expected samples.

    let has_neg = a[0] < 0;
    let has_nonneg = a[n - 1] >= 0;

    let ans: i64;

    if has_neg && has_nonneg {
        // Split point: first index with >= 0
        let mut k = 0usize;
        while k < n && a[k] < 0 {
            k += 1;
        }
        // negatives: a[0..k], nonnegatives: a[k..n]
        // max element is a[n-1] (non-negative)
        let mut acc_pos = a[n - 1];
        for &x in &a[0..k] {
            ops.push((acc_pos, x));
            acc_pos -= x;
        }

        let mut acc_neg = a[0];
        for &x in &a[k..n - 1] {
            ops.push((acc_neg, x));
            acc_neg -= x;
        }

        ops.push((acc_pos, acc_neg));
        ans = acc_pos - acc_neg;
    } else if has_nonneg {
        // All non-negative
        let mut acc = a[0];
        for &x in &a[1..n - 1] {
            ops.push((acc, x));
            acc -= x;
        }
        ops.push((a[n - 1], acc));
        ans = a[n - 1] - acc;
    } else {
        // All non-positive
        let mut acc = a[n - 1];
        for &x in &a[1..n - 1] {
            ops.push((acc, x));
            acc -= x;
        }
        ops.push((a[0], acc));
        ans = a[0] - acc;
    }

    let mut out = String::new();
    out.push_str(&format!("{}\n", ans));
    for (x, y) in ops {
        out.push_str(&format!("{} {}\n", x, y));
    }
    print!("{}", out);
}
```
