```rust
use std::io::{self, Read};

const MOD: u64 = 1_000_000_007;

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let m: usize = it.next().unwrap().parse().unwrap();

    // Adjacency in input order; "first outgoing edge" is g[v][0] if any.
    let mut g: Vec<Vec<usize>> = vec![Vec::new(); n];
    let mut gr: Vec<Vec<usize>> = vec![Vec::new(); n];
    for _ in 0..m {
        let a: usize = it.next().unwrap().parse().unwrap();
        let b: usize = it.next().unwrap().parse().unwrap();
        let u = a - 1;
        let v = b - 1;
        g[u].push(v);
        gr[v].push(u);
    }

    // Kosaraju SCC.
    let mut used = vec![false; n];
    let mut order: Vec<usize> = Vec::with_capacity(n);

    for start in 0..n {
        if used[start] {
            continue;
        }
        let mut stack: Vec<(usize, usize)> = vec![(start, 0)];
        used[start] = true;
        while let Some((v, idx)) = stack.pop() {
            if idx < g[v].len() {
                stack.push((v, idx + 1));
                let to = g[v][idx];
                if !used[to] {
                    used[to] = true;
                    stack.push((to, 0));
                }
            } else {
                order.push(v);
            }
        }
    }

    let mut comp = vec![usize::MAX; n];
    let mut comps: Vec<Vec<usize>> = Vec::new();
    for &v in order.iter().rev() {
        if comp[v] != usize::MAX {
            continue;
        }
        let cid = comps.len();
        let mut members = Vec::new();
        let mut stack = vec![v];
        comp[v] = cid;
        while let Some(x) = stack.pop() {
            members.push(x);
            for &to in &gr[x] {
                if comp[to] == usize::MAX {
                    comp[to] = cid;
                    stack.push(to);
                }
            }
        }
        comps.push(members);
    }
    let k = comps.len();

    // Derive next relation:
    // "determined solely by the first outgoing edge of any vertex in that component"
    // and "the last such vertex encountered in vertex-index order determines next(i)".
    //
    // Narrow interpretation consistent with this wording:
    // - For each SCC i, scan vertices in increasing vertex index order.
    // - For each vertex v in SCC i, if it has any outgoing edge, consider its first outgoing edge.
    // - If that edge goes to a *different* SCC, it is a candidate; the last candidate encountered wins.
    // - If no such candidate exists, next(i) is undefined.
    //
    // This excludes self-looping within the SCC from defining next(i), matching the idea of a
    // component-level "next" relation to (at most) one *other* component.
    let mut next: Vec<Option<usize>> = vec![None; k];
    for i in 0..k {
        let mut in_comp = vec![false; n];
        for &v in &comps[i] {
            in_comp[v] = true;
        }

        let mut chosen: Option<usize> = None;
        for v in 0..n {
            if !in_comp[v] || g[v].is_empty() {
                continue;
            }
            let to = g[v][0];
            let cj = comp[to];
            if cj != i {
                chosen = Some(cj);
            }
        }
        next[i] = chosen;
    }

    // Aggregation per specification (multiplicative propagation):
    // - Each SCC starts with value 2.
    // - Process SCCs in increasing index order.
    // - When processing i: increment its value by 1.
    // - If next(i) is undefined: multiply into answer.
    // - Else: multiplicatively propagate into next(i).
    //
    // Note: next(i)=i cannot occur with the next-construction above; if it did, it would be treated
    // as terminal per the specification.
    let mut val: Vec<u64> = vec![2; k];
    let mut ans: u64 = 1;

    for i in 0..k {
        val[i] = (val[i] + 1) % MOD;
        match next[i] {
            None => ans = (ans * val[i]) % MOD,
            Some(j) if j == i => ans = (ans * val[i]) % MOD,
            Some(j) => val[j] = (val[j] * val[i]) % MOD,
        }
    }

    println!("{}", ans % MOD);
}
```
