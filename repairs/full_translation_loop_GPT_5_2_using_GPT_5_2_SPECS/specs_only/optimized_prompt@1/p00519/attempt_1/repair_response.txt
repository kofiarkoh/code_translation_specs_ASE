```rust
use std::cmp::Reverse;
use std::collections::BinaryHeap;
use std::io::{self, Read};
use std::process;

fn main() {
    // Read entire stdin; if anything goes wrong, exit nonzero with no guaranteed output.
    let mut input = String::new();
    if io::stdin().read_to_string(&mut input).is_err() {
        process::exit(1);
    }
    let mut it = input.split_whitespace();

    let n: usize = match it.next().and_then(|s| s.parse().ok()) {
        Some(v) => v,
        None => process::exit(1),
    };
    let m: usize = match it.next().and_then(|s| s.parse().ok()) {
        Some(v) => v,
        None => process::exit(1),
    };

    let mut fare: Vec<i64> = vec![0; n];
    let mut limit: Vec<usize> = vec![0; n];
    for i in 0..n {
        let f: i64 = match it.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => process::exit(1),
        };
        let l: usize = match it.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => process::exit(1),
        };
        fare[i] = f;
        limit[i] = l;
    }

    let mut adj: Vec<Vec<usize>> = vec![Vec::new(); n];
    for _ in 0..m {
        let a: usize = match it.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => process::exit(1),
        };
        let b: usize = match it.next().and_then(|s| s.parse().ok()) {
            Some(v) => v,
            None => process::exit(1),
        };
        // Preconditions guarantee indices are valid; behavior is undefined otherwise.
        adj[a].push(b);
        adj[b].push(a);
    }

    // State: (town, remaining ride allowance).
    // We run Dijkstra over this expanded state graph with nonnegative edge weights:
    // - Move along a road: cost 0, r decreases by 1 (only if r > 0)
    // - Pay taxi at current town: cost fare[t], r becomes limit[t] (always allowed)
    //
    // We terminate upon first pop of any state with town == n-1, which is minimal.
    let max_r: usize = 5000; // fixed internal capacity per specification
    let inf: i64 = i64::MAX / 4;

    // dist[t][r] stored as a flat vector for memory locality.
    let states = n.saturating_mul(max_r + 1);
    let mut dist: Vec<i64> = vec![inf; states];

    let idx = |t: usize, r: usize| -> usize { t * (max_r + 1) + r };

    let mut pq: BinaryHeap<(Reverse<i64>, usize, usize)> = BinaryHeap::new();

    // Start at town 0 with 0 remaining allowance and 0 cost (no taxi paid yet).
    dist[idx(0, 0)] = 0;
    pq.push((Reverse(0), 0, 0));

    while let Some((Reverse(c), t, r)) = pq.pop() {
        if c != dist[idx(t, r)] {
            continue;
        }

        if t + 1 == n {
            // First time we pop a destination-town state is minimal by Dijkstra.
            print!("{}", c);
            return;
        }

        // Transition: pay taxi at town t (always).
        let new_r = limit[t];
        if new_r <= max_r {
            let nc = c.saturating_add(fare[t]);
            let di = idx(t, new_r);
            if nc < dist[di] {
                dist[di] = nc;
                pq.push((Reverse(nc), t, new_r));
            }
        }

        // Transition: traverse roads if r > 0.
        if r > 0 {
            let nr = r - 1;
            for &u in &adj[t] {
                let di = idx(u, nr);
                if c < dist[di] {
                    dist[di] = c;
                    pq.push((Reverse(c), u, nr));
                }
            }
        }
    }

    // Destination unreachable: no output is guaranteed; terminate normally without printing.
}
```
