use std::io::{self, Read};

const MOD: u64 = 998_244_353;

fn mod_pow(mut a: u64, mut e: u64) -> u64 {
    let mut r = 1u64;
    while e > 0 {
        if (e & 1) != 0 {
            r = (r * a) % MOD;
        }
        a = (a * a) % MOD;
        e >>= 1;
    }
    r
}

fn mod_inv(a: u64) -> u64 {
    // MOD is prime per specification.
    mod_pow(a % MOD, MOD - 2)
}

#[derive(Clone, Debug)]
struct Poly {
    // Bits over GF(2), little-endian by bit index: bit i is coefficient of x^i.
    // Stored in 64-bit limbs.
    w: Vec<u64>,
}

impl Poly {
    fn zero() -> Self {
        Self { w: vec![0] }
    }

    fn is_zero(&self) -> bool {
        self.deg().is_none()
    }

    fn normalize(&mut self) {
        while self.w.len() > 1 && *self.w.last().unwrap() == 0 {
            self.w.pop();
        }
        if self.w.is_empty() {
            self.w.push(0);
        }
    }

    fn deg(&self) -> Option<usize> {
        for (i, &word) in self.w.iter().enumerate().rev() {
            if word != 0 {
                let msb = 63 - word.leading_zeros() as usize;
                return Some(i * 64 + msb);
            }
        }
        None
    }

    fn from_bin_str(s: &str) -> Self {
        // Assumption: s contains only '0'/'1'. Leftmost is highest-degree coefficient.
        let n = s.len();
        let mut w = vec![0u64; (n + 63) / 64];
        for (i, b) in s.bytes().enumerate() {
            if b == b'1' {
                let bit = n - 1 - i; // map to x^bit
                w[bit / 64] |= 1u64 << (bit % 64);
            }
        }
        let mut p = Self { w };
        p.normalize();
        p
    }

    fn xor_shifted(&mut self, other: &Poly, shift: usize) {
        // self ^= other << shift
        if other.is_zero() {
            return;
        }
        let word_shift = shift / 64;
        let bit_shift = shift % 64;

        let other_len = other.w.len();
        let needed = word_shift + other_len + if bit_shift != 0 { 1 } else { 0 };
        if self.w.len() < needed {
            self.w.resize(needed, 0);
        }

        if bit_shift == 0 {
            for i in 0..other_len {
                self.w[word_shift + i] ^= other.w[i];
            }
        } else {
            let bs = bit_shift as u32;
            let inv = 64 - bs;
            for i in 0..other_len {
                let lo = other.w[i] << bs;
                self.w[word_shift + i] ^= lo;
                let hi = other.w[i] >> inv;
                if hi != 0 {
                    self.w[word_shift + i + 1] ^= hi;
                }
            }
        }
        self.normalize();
    }

    fn rem_mod(&self, modulus: &Poly) -> Poly {
        // Remainder of self divided by modulus over GF(2).
        // If modulus is zero, behavior is mathematically undefined; we return self clone
        // to avoid panics while not inventing a defined contract.
        if modulus.is_zero() {
            return self.clone();
        }
        let mut r = self.clone();
        let md = modulus.deg().unwrap();
        while let Some(rd) = r.deg() {
            if rd < md {
                break;
            }
            let shift = rd - md;
            r.xor_shifted(modulus, shift);
        }
        r
    }

    fn gcd(mut a: Poly, mut b: Poly) -> Poly {
        // Euclidean algorithm over GF(2).
        // If both are zero, result is zero (degenerate/unspecified downstream per spec).
        while !b.is_zero() {
            let r = a.rem_mod(&b);
            a = b;
            b = r;
        }
        a
    }
}

fn parse_u64_mod_from_bin(s: &str) -> u64 {
    let mut v = 0u64;
    for &b in s.as_bytes() {
        v = (v * 2 + if b == b'1' { 1 } else { 0 }) % MOD;
    }
    v
}

fn main() {
    // Input tokens: count, initial binary string, then count binary strings.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let m: usize = match it.next() {
        Some(t) => t.parse().unwrap_or(0),
        None => return,
    };
    let a_str = match it.next() {
        Some(t) => t,
        None => return,
    };

    // GCD state: initialized from first subsequent polynomial if present.
    let mut g_opt: Option<Poly> = None;
    for _ in 0..m {
        if let Some(s) = it.next() {
            let p = Poly::from_bin_str(s);
            g_opt = Some(match g_opt {
                None => p,
                Some(g) => Poly::gcd(g, p),
            });
        } else {
            // Missing tokens: unspecified; stop processing.
            break;
        }
    }

    // If there are no subsequent polynomials, the specification does not define G.
    // Narrowest reasonable interpretation: treat G as 1 (divisible by 1 => all X),
    // because otherwise the set "divisible by G" is not meaningful.
    // This choice avoids inventing extra I/O and keeps behavior total.
    let g = g_opt.unwrap_or_else(|| {
        // Polynomial "1"
        Poly { w: vec![1] }
    });

    // If gcd is zero polynomial, output is unspecified by contract.
    // We avoid panics and produce a deterministic value: 0.
    if g.is_zero() {
        println!("0");
        return;
    }

    // Let d = deg(G). Over GF(2), X is divisible by G implies deg(X) >= d unless X=0.
    // Counting X < A divisible by G:
    // If A is interpreted as a binary integer, then for any fixed d,
    // the number of multiples of a degree-d polynomial below 2^n is 2^(n-d).
    // For a general bound A, the count is floor(A / 2^d) in integer sense,
    // because multiplication by a monic degree-d polynomial is a bijection on
    // polynomials of degree < n-d when considering truncation by degree.
    //
    // This reduces to: count = A >> d (integer shift), modulo MOD.
    // We compute A mod MOD and multiply by inv(2^d) only when A is a power-of-two aligned?
    // However, we need exact floor(A / 2^d), not A / 2^d in modular arithmetic.
    // So we compute the shifted integer value from the binary string directly.
    let d = g.deg().unwrap(); // degree >= 0

    // Compute floor(A / 2^d) modulo MOD by reading bits and skipping the last d bits.
    // If d >= len(A), result is 0.
    let n = a_str.len();
    let res = if d >= n {
        0u64
    } else {
        // Take prefix of length n-d as the shifted value.
        let prefix = &a_str[..(n - d)];
        parse_u64_mod_from_bin(prefix)
    };

    println!("{}", res % MOD);
}
