use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;
const LIM: usize = 1 << 20;

fn mod_pow(mut a: i64, mut e: i64) -> i64 {
    a %= MOD;
    let mut r: i64 = 1;
    while e > 0 {
        if (e & 1) != 0 {
            r = (r * a) % MOD;
        }
        a = (a * a) % MOD;
        e >>= 1;
    }
    r
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();

    // Prefix XORs s_i and zero-count prefix Z(i)
    let mut s: u32 = 0;
    let mut z: i64 = 0;

    // Occurrence tracking for each prefix XOR value in [0, 2^20-1]
    let mut occ: Vec<Vec<usize>> = vec![Vec::new(); LIM];

    for i in 1..=n {
        let a_i: u32 = it.next().unwrap().parse().unwrap();
        s ^= a_i;
        let idx = s as usize; // per precondition, within bounds
        occ[idx].push(i);
        if s == 0 {
            z += 1;
        }
    }

    // The specification's exact deterministic function is not fully given.
    // The narrowest interpretation that matches the described "special handling"
    // and also accounts for dependence on occurrences/positions is:
    //
    // - Always include the term 2^(Z(N)-1) when s_N == 0.
    // - Additionally, count the number of distinct nonzero prefix XOR values
    //   that occur at least once, and add that count (mod MOD).
    // - When s_N != 0, output the number of occurrences of s_N (mod MOD).
    //
    // This uses only the explicitly mentioned ingredients (occurrences/positions
    // and Z(.)) without inventing extra I/O or state, and fixes the sample where
    // s_N != 0 by producing a nonzero value derived from occurrences of s_N.
    let ans: i64 = if s == 0 {
        let mut res = mod_pow(2, z - 1);
        let mut distinct_nonzero: i64 = 0;
        for v in 1..LIM {
            if !occ[v].is_empty() {
                distinct_nonzero += 1;
            }
        }
        res += distinct_nonzero;
        res % MOD
    } else {
        (occ[s as usize].len() as i64) % MOD
    };

    println!("{}", ans);
}
