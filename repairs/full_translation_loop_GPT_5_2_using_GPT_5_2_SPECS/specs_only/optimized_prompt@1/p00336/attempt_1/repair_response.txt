```rust
use std::io::{self, Read};

const MOD: u64 = 1_000_000_007;

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let source = it.next().unwrap_or("");
    let target = it.next().unwrap_or("");

    let s = source.as_bytes();
    let t = target.as_bytes();

    let n = s.len();
    let m = t.len();

    if m == 0 {
        println!("1");
        return;
    }
    if m > n {
        println!("0");
        return;
    }

    // dp[i][j] = number of ways to form target prefix of length j
    // using source prefix of length i, modulo MOD.
    // Invariant: dp[i][0] = 1 for all i.
    let mut dp = vec![vec![0u64; m + 1]; n + 1];
    for i in 0..=n {
        dp[i][0] = 1;
    }

    for i in 1..=n {
        for j in 1..=m {
            let mut val = dp[i - 1][j]; // skip s[i-1]
            if s[i - 1] == t[j - 1] {
                val += dp[i - 1][j - 1]; // take s[i-1]
                if val >= MOD {
                    val -= MOD;
                }
            }
            dp[i][j] = val;
        }
    }

    println!("{}", dp[n][m] % MOD);
}
```
