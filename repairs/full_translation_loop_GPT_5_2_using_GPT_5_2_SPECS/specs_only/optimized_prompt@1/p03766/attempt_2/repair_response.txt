```rust
use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;

fn mod_norm(x: i64) -> i64 {
    let mut v = x % MOD;
    if v < 0 {
        v += MOD;
    }
    v
}

fn main() {
    // Read one integer n; no explicit error handling per specification.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let n: i64 = input.split_whitespace().next().unwrap().parse().unwrap();

    if n == 1 {
        println!("1");
        return;
    }

    let nn = n as usize;

    // F(1)=1 implied by invariants / base case usage in sums.
    let mut f = vec![0i64; nn + 1];
    f[1] = 1;

    // IMPORTANT: The specification states "F(2) = n^2" but sample indicates
    // the intended meaning is F(2) = 2^2 (i.e., i^2 at i=2), not n^2.
    // This is the narrowest change consistent with the recurrence structure
    // and the observed sample.
    f[2] = 4 % MOD;

    // Prefix sums pref[i] = sum_{j=1..i} F(j) mod MOD
    let mut pref = vec![0i64; nn + 1];
    pref[1] = f[1];
    pref[2] = mod_norm(pref[1] + f[2]);

    for i in 3..=nn {
        let i_i64 = i as i64;

        // sum_{j=1}^{i-3} F(j)
        let sum_1_to_i_minus_3 = if i >= 4 { pref[i - 3] } else { 0 };

        let term1 = f[i - 1];
        let term2 = sum_1_to_i_minus_3;
        let term3 = mod_norm((i_i64 - 3) * (n - 1));
        let term4 = mod_norm((n - i_i64 + 2) * n);

        f[i] = mod_norm(term1 + term2 + term3 + term4);
        pref[i] = mod_norm(pref[i - 1] + f[i]);
    }

    println!("{}", f[nn]);
}
```
