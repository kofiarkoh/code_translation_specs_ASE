```rust
use std::io::{self, Read};

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Tok {
    Num(i64),
    Op(u8), // b'+', b'-', b'*'
    LParen,
    RParen,
}

fn tokenize(s: &str) -> Vec<Tok> {
    let bytes = s.as_bytes();
    let mut i = 0usize;
    let mut out = Vec::new();
    while i < bytes.len() {
        match bytes[i] {
            b'0'..=b'9' => {
                let mut v: i64 = 0;
                while i < bytes.len() {
                    match bytes[i] {
                        b'0'..=b'9' => {
                            v = v * 10 + (bytes[i] - b'0') as i64;
                            i += 1;
                        }
                        _ => break,
                    }
                }
                out.push(Tok::Num(v));
            }
            b'+' | b'-' | b'*' => {
                out.push(Tok::Op(bytes[i]));
                i += 1;
            }
            b'(' => {
                out.push(Tok::LParen);
                i += 1;
            }
            b')' => {
                out.push(Tok::RParen);
                i += 1;
            }
            b' ' | b'\n' | b'\r' | b'\t' => {
                // Whitespace is not specified, but ignoring it is the narrowest harmless choice.
                i += 1;
            }
            _ => {
                // Undefined behavior per spec for malformed input; we choose to stop tokenizing.
                break;
            }
        }
    }
    out
}

fn apply_op(op: u8, a: i64, b: i64) -> i64 {
    match op {
        b'+' => a + b,
        b'-' => a - b,
        b'*' => a * b,
        _ => unreachable!(),
    }
}

fn precedence(op: u8, prec: &[u8; 3]) -> u8 {
    match op {
        b'+' => prec[0],
        b'-' => prec[1],
        b'*' => prec[2],
        _ => 0,
    }
}

fn eval_with_scheme(tokens: &[Tok], prec: &[u8; 3], right_assoc: &[bool; 3]) -> i64 {
    let mut vals: Vec<i64> = Vec::new();
    let mut ops: Vec<u8> = Vec::new(); // operators and '(' as b'('

    let mut reduce_top = |vals: &mut Vec<i64>, ops: &mut Vec<u8>| {
        let op = ops.pop().unwrap();
        let b = vals.pop().unwrap();
        let a = vals.pop().unwrap();
        vals.push(apply_op(op, a, b));
    };

    for &t in tokens {
        match t {
            Tok::Num(v) => vals.push(v),
            Tok::LParen => ops.push(b'('),
            Tok::RParen => {
                while let Some(&top) = ops.last() {
                    if top == b'(' {
                        break;
                    }
                    reduce_top(&mut vals, &mut ops);
                }
                // Pop '('
                if ops.last() == Some(&b'(') {
                    ops.pop();
                }
            }
            Tok::Op(op) => {
                let p_new = precedence(op, prec);
                let ra_new = match op {
                    b'+' => right_assoc[0],
                    b'-' => right_assoc[1],
                    b'*' => right_assoc[2],
                    _ => false,
                };

                while let Some(&top) = ops.last() {
                    if top == b'(' {
                        break;
                    }
                    let p_top = precedence(top, prec);
                    // Standard shunting-yard condition generalized for per-operator associativity:
                    // If new op is left-assoc: reduce while p_top >= p_new
                    // If new op is right-assoc: reduce while p_top >  p_new
                    let should_reduce = if ra_new { p_top > p_new } else { p_top >= p_new };
                    if !should_reduce {
                        break;
                    }
                    reduce_top(&mut vals, &mut ops);
                }
                ops.push(op);
            }
        }
    }

    while let Some(&top) = ops.last() {
        if top == b'(' {
            // Undefined per spec if mismatched; ignore.
            ops.pop();
            continue;
        }
        reduce_top(&mut vals, &mut ops);
    }

    vals.pop().unwrap()
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let expr = input.trim();
    let tokens = tokenize(expr);

    // 13 schemes:
    // - 6 strict precedence orders (all left-associative)
    // - 7 equal-precedence schemes with left/right associativity choices:
    //   * all three equal: 2 (L or R)
    //   * exactly one operator higher than the other two equal: 3 * 2 = 6
    // Total: 6 + 2 + 6 = 14, but we must output exactly the 13 predefined schemes.
    // Narrowest consistent interpretation: include all 6 strict orders, and for equal-precedence
    // cases include only:
    //   - all equal with left-assoc
    //   - all equal with right-assoc
    //   - exactly one operator higher with left-assoc for the lower tier (right-assoc unused)
    // This yields 6 + 2 + 3 = 11, still not 13.
    //
    // Alternative narrow interpretation matching common "unknown precedence" problems:
    // 13 schemes correspond to all total preorders of 3 operators (13 in count):
    // - 6 strict total orders
    // - 6 with a tie between two operators (and the third distinct)
    // - 1 with all three tied
    // Associativity is then fixed as left-to-right for tied operators.
    // This exactly yields 13 and matches the spec phrase "precedence/associativity schemes"
    // while keeping a single deterministic associativity choice.
    //
    // Therefore: evaluate 13 precedence schemes = all total preorders; associativity always left.
    let mut schemes: Vec<[u8; 3]> = Vec::new();

    // Helper: map ranks to (+,-,*) in order.
    // Rank values: higher number => higher precedence.
    // For strict orders, use ranks 3,2,1.
    // For ties, use ranks like 2,2,1 etc.
    // For all tied, use 1,1,1.
    //
    // Enumerate all total preorders of 3 labeled items:
    // 1) all tied
    schemes.push([1, 1, 1]);

    // 2) one highest, other two tied lower: 3 cases
    // '+' highest
    schemes.push([2, 1, 1]);
    // '-' highest
    schemes.push([1, 2, 1]);
    // '*' highest
    schemes.push([1, 1, 2]);

    // 3) one lowest, other two tied higher: 3 cases
    // '+' lowest
    schemes.push([1, 2, 2]);
    // '-' lowest
    schemes.push([2, 1, 2]);
    // '*' lowest
    schemes.push([2, 2, 1]);

    // 4) all distinct: 6 permutations of ranks 3,2,1
    // (+,-,*)
    schemes.push([3, 2, 1]);
    schemes.push([3, 1, 2]);
    schemes.push([2, 3, 1]);
    schemes.push([1, 3, 2]);
    schemes.push([2, 1, 3]);
    schemes.push([1, 2, 3]);

    // Associativity: fixed left-to-right for all operators.
    let right_assoc = [false, false, false];

    let mut best = i64::MIN;
    for prec in schemes {
        let v = eval_with_scheme(&tokens, &prec, &right_assoc);
        if v > best {
            best = v;
        }
    }

    print!("{best}");
}
```
