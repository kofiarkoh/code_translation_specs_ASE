```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, then mimic scanf("%s", s) by taking the first whitespace-delimited token.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let s = match input.split_whitespace().next() {
        Some(t) => t.as_bytes(),
        None => {
            // If no token is present, C's scanf would fail and `s` would be uninitialized (UB).
            // We choose to do nothing meaningful; print 0 to keep output defined.
            // (This case is outside well-defined C behavior anyway.)
            println!("0");
            return;
        }
    };

    // C: int b[256] = {0}; and b[s[l]]++ where s[l] is a (possibly signed) char.
    // To preserve the typical byte-indexing intent, treat input as raw bytes 0..=255.
    let mut b = [0i32; 256];
    for &ch in s {
        b[ch as usize] += 1;
    }

    let mut l: i32 = s.len() as i32;
    let mut a: i32 = 0;
    let mut c: i32 = 0;

    while l != 0 {
        for i in 0..256usize {
            if b[i] != 0 {
                a = a.wrapping_add(1);
                l = l.wrapping_sub(1);
                c = c.wrapping_add(1);
                b[i] = b[i].wrapping_sub(1);
            } else {
                if c > 3 {
                    a = a.wrapping_sub(c - 3);
                }
                c = 0;
            }
        }
    }

    // C prints "%d\n"
    println!("{}", a);
}
```
