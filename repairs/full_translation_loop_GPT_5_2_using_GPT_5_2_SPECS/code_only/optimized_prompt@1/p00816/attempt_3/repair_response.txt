```rust
use std::io::{self, Read};

fn atoi_like_c(bytes: &[u8]) -> i32 {
    // Emulate C's atoi (wrapping on overflow to avoid panics / keep deterministic).
    let mut i = 0usize;
    while i < bytes.len() && bytes[i].is_ascii_whitespace() {
        i += 1;
    }

    let mut sign: i32 = 1;
    if i < bytes.len() {
        if bytes[i] == b'-' {
            sign = -1;
            i += 1;
        } else if bytes[i] == b'+' {
            i += 1;
        }
    }

    let mut any = false;
    let mut val: i32 = 0;
    while i < bytes.len() && bytes[i].is_ascii_digit() {
        any = true;
        let d = (bytes[i] - b'0') as i32;
        val = val.wrapping_mul(10).wrapping_add(d);
        i += 1;
    }

    if !any { 0 } else { val.wrapping_mul(sign) }
}

fn func(s: &[u8], n: usize, m: u32) -> i32 {
    let mut sum: i32 = 0;
    let mut start: usize = 0;
    for i in 0..n {
        if (m & (1u32 << i)) != 0 {
            let end = i + 1;
            sum = sum.wrapping_add(atoi_like_c(&s[start..end]));
            start = end;
        }
    }
    sum.wrapping_add(atoi_like_c(&s[start..n]))
}

fn next_token<'a>(buf: &'a [u8], idx: &mut usize) -> Option<&'a [u8]> {
    while *idx < buf.len() && buf[*idx].is_ascii_whitespace() {
        *idx += 1;
    }
    if *idx >= buf.len() {
        return None;
    }
    let start = *idx;
    while *idx < buf.len() && !buf[*idx].is_ascii_whitespace() {
        *idx += 1;
    }
    Some(&buf[start..*idx])
}

fn main() {
    let mut input = Vec::new();
    io::stdin().read_to_end(&mut input).unwrap();
    let mut idx = 0usize;

    let mut out = String::new();

    loop {
        let t_tok = match next_token(&input, &mut idx) {
            Some(x) => x,
            None => break,
        };
        let target = atoi_like_c(t_tok);

        let paper_tok = match next_token(&input, &mut idx) {
            Some(x) => x,
            None => break,
        };

        if target == 0 {
            break;
        }

        let n = paper_tok.len();

        // The original C uses char paper[10], so n is expected <= 9.
        // If n==0, C would have UB in shifts; we avoid hanging.
        if n == 0 || n > 31 {
            out.push_str("error\n");
            continue;
        }

        let mut res: u32 = 0;
        let mut best: i32 = -1;
        let mut cnt: i32 = 0;

        let start_mask: u32 = 1u32 << (n - 1);
        let end_mask: u32 = 1u32 << n;

        for mask in start_mask..end_mask {
            let t = func(paper_tok, n, mask);
            if t <= target {
                if best < t {
                    best = t;
                    res = mask;
                    cnt = 1;
                } else if best == t {
                    cnt += 1;
                }
            }
        }

        if cnt != 1 {
            out.push_str(if cnt == 0 { "error\n" } else { "rejected\n" });
        } else {
            out.push_str(&best.to_string());
            out.push(' ');
            for i in 0..n {
                out.push(paper_tok[i] as char);
                if (res & (1u32 << i)) != 0 {
                    if i == n - 1 {
                        out.push('\n');
                    } else {
                        out.push(' ');
                    }
                }
            }
        }
    }

    print!("{out}");
}
```
