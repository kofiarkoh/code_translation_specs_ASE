```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, parse as i32 like C's "%d".
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let counter: i32 = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };

    let mut count: i32 = 0;

    while counter > count {
        let mut x: i32 = it.next().unwrap().parse().unwrap();

        let cter: i32 = 10;
        let mut c: i32 = 0;

        // Match C output formatting exactly.
        print!("Case {}:\n", count + 1);

        while c < cter {
            // C uses signed int arithmetic; overflow is UB in C, but many platforms wrap.
            // Use wrapping ops to emulate typical two's-complement behavior.
            x = x.wrapping_mul(x);
            x = x.wrapping_div(100);
            x = x.wrapping_rem(10000);

            c += 1;
            print!("{}\n", x);
        }

        count += 1;
    }
}
```
