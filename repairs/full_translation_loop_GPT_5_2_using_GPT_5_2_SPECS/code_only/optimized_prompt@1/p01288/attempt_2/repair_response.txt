```rust
use std::io::{self, Read};

fn main() {
    // Fast byte parser; emulate scanf behavior enough for this problem.
    let mut input = Vec::new();
    io::stdin().read_to_end(&mut input).unwrap();
    let mut idx: usize = 0;

    #[inline]
    fn skip_ws(a: &[u8], i: &mut usize) {
        while *i < a.len() && a[*i].is_ascii_whitespace() {
            *i += 1;
        }
    }

    #[inline]
    fn read_i32(a: &[u8], i: &mut usize) -> Option<i32> {
        skip_ws(a, i);
        if *i >= a.len() {
            return None;
        }
        let mut sign = 1i32;
        if a[*i] == b'-' {
            sign = -1;
            *i += 1;
        }
        let mut val: i32 = 0;
        let mut any = false;
        while *i < a.len() {
            let c = a[*i];
            if c.is_ascii_digit() {
                any = true;
                val = val * 10 + (c - b'0') as i32;
                *i += 1;
            } else {
                break;
            }
        }
        if any { Some(val * sign) } else { None }
    }

    #[inline]
    fn read_op_char(a: &[u8], i: &mut usize) -> Option<u8> {
        // In the original C, scanf("%c %d\n", ...) reads the next character,
        // which could be a newline. However, typical inputs intend to read the
        // operation letter; if we don't skip whitespace, we can get stuck with
        // '\n' and then fail to parse the integer, causing an infinite loop.
        // So we skip leading whitespace here to match intended behavior and
        // avoid timeouts.
        skip_ws(a, i);
        if *i >= a.len() {
            None
        } else {
            let c = a[*i];
            *i += 1;
            Some(c)
        }
    }

    let mut out = String::new();

    loop {
        let n = match read_i32(&input, &mut idx) {
            Some(x) => x as usize,
            None => break,
        };
        let qn = match read_i32(&input, &mut idx) {
            Some(x) => x as usize,
            None => break,
        };
        if n == 0 {
            break;
        }

        // Use N+1 because the C code writes P[N] (loop i<=N).
        let mut p = vec![0usize; n + 1];
        let mut m = vec![0i32; n + 1];
        let mut q = vec![0u8; qn];
        let mut v = vec![0usize; qn];

        for i in 0..n {
            p[i] = i;
            m[i] = 0;
        }
        if n >= 1 {
            m[1] = 1;
        }

        for i in 2..=n {
            let t = read_i32(&input, &mut idx).unwrap_or(0);
            // Indices are assumed valid by the C code; keep as usize.
            p[i] = t as usize;
        }

        for i in 0..qn {
            let ch = read_op_char(&input, &mut idx).unwrap_or(0);
            let val = read_i32(&input, &mut idx).unwrap_or(0) as usize;
            q[i] = ch;
            v[i] = val;
            if ch == b'M' && val <= n {
                m[val] += 1;
            }
        }

        let mut r: i64 = 0;
        for i in (0..qn).rev() {
            if q[i] == b'M' {
                let vi = v[i];
                if vi <= n {
                    m[vi] -= 1;
                }
            } else {
                let mut t = v[i];
                // Follow parent pointers until we find an active marker.
                // The C code assumes this terminates.
                while t <= n && m[t] == 0 {
                    t = p[t];
                }
                r += t as i64;
            }
        }

        out.push_str(&r.to_string());
        out.push('\n');
    }

    print!("{out}");
}
```
