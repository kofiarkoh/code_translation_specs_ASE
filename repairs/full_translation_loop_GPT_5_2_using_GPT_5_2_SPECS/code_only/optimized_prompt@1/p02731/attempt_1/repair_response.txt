```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin (scanf("%d", &l) reads an int, skipping leading whitespace).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    // Parse as i32 to match typical C `int` behavior.
    // If parsing fails, C would have undefined/unspecified behavior; we unwrap to fail fast.
    let l: i32 = input.split_whitespace().next().unwrap().parse().unwrap();

    let y: f64 = (l as f64) / 3.0;
    let x: f64 = y * y * y;

    // Match printf("%.12lf\n", x);
    println!("{:.12}", x);
}
```
