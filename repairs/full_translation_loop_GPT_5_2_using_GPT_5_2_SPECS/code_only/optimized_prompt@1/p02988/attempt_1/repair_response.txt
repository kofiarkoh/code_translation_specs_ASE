```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, parse as whitespace-separated integers (like scanf("%d")).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return, // No input: C would have undefined behavior; we just exit.
    };

    // C code uses num[1000] and indexes from 1..=n (1-based).
    // We'll allocate 1000 elements and keep index 0 unused to preserve semantics.
    let mut num = vec![0i32; 1000];

    // for (i = 1; i <= n; i++) scanf("%d", &num[i]);
    // If n is out of bounds, C would have UB; we avoid panics by only writing in-bounds.
    for i in 1..=n {
        if let Some(s) = it.next() {
            if i >= 0 && (i as usize) < num.len() {
                num[i as usize] = s.parse().unwrap();
            } else {
                // Out-of-bounds write would be UB in C; ignore to keep Rust safe.
                // (No external spec to define intended behavior.)
            }
        } else {
            // Missing input: C would leave num[i] uninitialized (UB on read later).
            // We keep default 0; this is a best-effort safe approximation.
            break;
        }
    }

    let mut count: i32 = 0;

    // for (i = 1; i < n-1; i++)
    // Note: if n <= 2, loop does not execute.
    for i in 1..(n - 1) {
        let i_us = i as usize;
        // Accesses num[i], num[i+1], num[i+2]; ensure indices are in-bounds.
        if i_us + 2 >= num.len() {
            break; // Avoid panic; C would be UB.
        }

        if num[i_us] < num[i_us + 1] {
            if num[i_us + 1] < num[i_us + 2] {
                count += 1;
            }
        } else if num[i_us] > num[i_us + 1] {
            if num[i_us + 1] > num[i_us + 2] {
                count += 1;
            }
        }
    }

    print!("{}\n", count);
}
```
