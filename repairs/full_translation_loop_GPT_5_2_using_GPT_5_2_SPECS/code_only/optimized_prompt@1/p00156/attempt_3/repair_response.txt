```rust
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut sc = Scanner::new(&input);

    // C globals: int x[200000], y[200000];
    let mut x = vec![0i32; 200_000];
    let mut y = vec![0i32; 200_000];

    let dx = [0i32, 1, 0, -1];
    let dy = [1i32, 0, -1, 0];

    let mut out = String::new();

    loop {
        let w = match sc.next_i32() {
            Some(v) => v,
            None => break,
        };
        let h = match sc.next_i32() {
            Some(v) => v,
            None => break,
        };

        if w == 0 && h == 0 {
            break;
        }

        // C: int c[101][101] = {0};
        let mut c = vec![vec![0i32; 101]; 101];
        // C: char m[101][101];
        let mut m = vec![vec![b'\0'; 101]; 101];

        // In C, `t` is only assigned when '&' is found; otherwise UB.
        // We avoid UB by tracking whether start exists; if not, we skip BFS.
        let mut start_found = false;
        let mut t: i32 = 0;

        for i in 0..h as usize {
            for j in 0..w as usize {
                let ch = sc.next_non_ws_byte().unwrap();
                m[i][j] = ch;
                if ch == b'&' {
                    m[i][j] = b'.';
                    y[0] = i as i32;
                    x[0] = j as i32;
                    t = 0;
                    c[i][j] = 1;
                    start_found = true;
                }
            }
        }

        let mut r: i32 = if start_found { 1 } else { 0 };

        // C: for(r=1; r-t; t++) { ... }  (condition is r - t != 0)
        while r - t != 0 {
            for dir in 0..4 {
                let nx = x[t as usize] + dx[dir];
                let ny = y[t as usize] + dy[dir];

                if nx < 0 || nx >= w || ny < 0 || ny >= h {
                    continue;
                }

                let cx = x[t as usize] as usize;
                let cy = y[t as usize] as usize;
                let nxu = nx as usize;
                let nyu = ny as usize;

                if m[cy][cx] == m[nyu][nxu] {
                    if c[cy][cx] < c[nyu][nxu] || c[nyu][nxu] == 0 {
                        c[nyu][nxu] = c[cy][cx];
                        x[r as usize] = nx;
                        y[r as usize] = ny;
                        r += 1;
                    }
                } else {
                    let cand = c[cy][cx] + 1;
                    if cand < c[nyu][nxu] || c[nyu][nxu] == 0 {
                        c[nyu][nxu] = cand;
                        x[r as usize] = nx;
                        y[r as usize] = ny;
                        r += 1;
                    }
                }
            }
            t += 1;
        }

        // C: min = w; then checks borders with integer division by 2.
        let mut minv: i32 = w;

        for i in 0..w as usize {
            let top = c[0][i] / 2;
            if top < minv {
                minv = top;
            }
            let bottom = c[(h - 1) as usize][i] / 2;
            if bottom < minv {
                minv = bottom;
            }
        }
        for i in 0..h as usize {
            let left = c[i][0] / 2;
            if left < minv {
                minv = left;
            }
            let right = c[i][(w - 1) as usize] / 2;
            if right < minv {
                minv = right;
            }
        }

        out.push_str(&format!("{minv}\n"));
    }

    print!("{out}");
}

struct Scanner<'a> {
    s: &'a [u8],
    i: usize,
}
impl<'a> Scanner<'a> {
    fn new(input: &'a str) -> Self {
        Self {
            s: input.as_bytes(),
            i: 0,
        }
    }

    fn skip_ws(&mut self) {
        while self.i < self.s.len() && self.s[self.i].is_ascii_whitespace() {
            self.i += 1;
        }
    }

    fn next_i32(&mut self) -> Option<i32> {
        self.skip_ws();
        if self.i >= self.s.len() {
            return None;
        }
        let mut sign = 1i32;
        if self.s[self.i] == b'-' {
            sign = -1;
            self.i += 1;
        }
        let mut val: i32 = 0;
        let mut any = false;
        while self.i < self.s.len() {
            let b = self.s[self.i];
            if !b.is_ascii_digit() {
                break;
            }
            any = true;
            // C int overflow is UB, but contest inputs won't rely on it; keep simple.
            val = val * 10 + (b - b'0') as i32;
            self.i += 1;
        }
        if any { Some(val * sign) } else { None }
    }

    // Mimic scanf(" %c"): skip whitespace, then read one byte.
    fn next_non_ws_byte(&mut self) -> Option<u8> {
        self.skip_ws();
        if self.i >= self.s.len() {
            return None;
        }
        let b = self.s[self.i];
        self.i += 1;
        Some(b)
    }
}
```
