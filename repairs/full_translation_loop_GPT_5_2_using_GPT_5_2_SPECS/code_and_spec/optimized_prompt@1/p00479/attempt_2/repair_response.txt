```rust
use std::io::{self, Read};

fn main() {
    // Fast integer parser to avoid potential timeouts from heavy formatting/parsing.
    let mut input = Vec::new();
    io::stdin().read_to_end(&mut input).unwrap();

    let mut idx = 0usize;
    let len = input.len();

    fn next_i64(buf: &[u8], idx: &mut usize) -> i64 {
        let len = buf.len();
        while *idx < len && buf[*idx].is_ascii_whitespace() {
            *idx += 1;
        }
        let mut sign = 1i64;
        if *idx < len && buf[*idx] == b'-' {
            sign = -1;
            *idx += 1;
        }
        let mut x = 0i64;
        while *idx < len {
            let c = buf[*idx];
            if !c.is_ascii_digit() {
                break;
            }
            x = x * 10 + (c - b'0') as i64;
            *idx += 1;
        }
        sign * x
    }

    if idx >= len {
        return; // no input; unspecified behavior, but avoid looping
    }

    let n = next_i64(&input, &mut idx);
    let k = next_i64(&input, &mut idx);

    // Pre-allocate output: each line is small; 4 bytes per line is a reasonable lower bound.
    let mut out = String::with_capacity((k.max(0) as usize).saturating_mul(4));

    for _ in 0..k {
        let mut a = next_i64(&input, &mut idx);
        let mut b = next_i64(&input, &mut idx);

        // d(p) = min(p, n + 1 - p)
        let na = n + 1 - a;
        if na < a {
            a = na;
        }
        let nb = n + 1 - b;
        if nb < b {
            b = nb;
        }

        let m = if a < b { a } else { b };
        let c = ((m - 1).rem_euclid(3)) + 1;

        out.push_str(c.to_string().as_str());
        out.push('\n');
    }

    print!("{out}");
}
```
