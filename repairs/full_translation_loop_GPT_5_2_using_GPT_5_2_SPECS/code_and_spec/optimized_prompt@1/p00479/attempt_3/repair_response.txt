```rust
use std::io::{self, Read};

fn main() {
    // Read all input.
    let mut buf = Vec::new();
    io::stdin().read_to_end(&mut buf).unwrap();
    let mut i = 0usize;

    // Fast integer parser; returns None if no more tokens.
    fn next_i64(buf: &[u8], i: &mut usize) -> Option<i64> {
        while *i < buf.len() && buf[*i].is_ascii_whitespace() {
            *i += 1;
        }
        if *i >= buf.len() {
            return None;
        }
        let mut sign = 1i64;
        if buf[*i] == b'-' {
            sign = -1;
            *i += 1;
        }
        let mut x = 0i64;
        while *i < buf.len() {
            let c = buf[*i];
            if !c.is_ascii_digit() {
                break;
            }
            x = x * 10 + (c - b'0') as i64;
            *i += 1;
        }
        Some(sign * x)
    }

    let n = match next_i64(&buf, &mut i) {
        Some(v) => v,
        None => return,
    };
    let k = match next_i64(&buf, &mut i) {
        Some(v) => v,
        None => return,
    };

    // If k is negative (outside preconditions), avoid an infinite loop.
    if k <= 0 {
        return;
    }

    // Build output as bytes for speed.
    let mut out: Vec<u8> = Vec::with_capacity(k as usize * 2);

    for _ in 0..k {
        let mut a = next_i64(&buf, &mut i).unwrap_or(0);
        let mut b = next_i64(&buf, &mut i).unwrap_or(0);

        // d(p) = min(p, n + 1 - p)
        let na = n + 1 - a;
        if na < a {
            a = na;
        }
        let nb = n + 1 - b;
        if nb < b {
            b = nb;
        }

        let m = if a < b { a } else { b };
        let c = ((m - 1).rem_euclid(3)) + 1; // in {1,2,3}

        // c is 1..3, so single digit.
        out.push(b'0' + c as u8);
        out.push(b'\n');
    }

    print!("{}", String::from_utf8_lossy(&out));
}
```
