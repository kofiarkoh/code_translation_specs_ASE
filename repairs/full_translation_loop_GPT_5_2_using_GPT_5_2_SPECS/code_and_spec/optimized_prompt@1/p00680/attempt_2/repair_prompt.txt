        to: from,

// Solves AX = C using Gauss-Jordan elimination with partial pivoting.
// Returns Some(X) on success, None if treated as singular (|pivot| < EPS).
fn gauss_jordan(a: &[Vec<i32>], c: &[i32]) -> Option<Vec<f64>> {
    let n = a.len();
    let mut b = vec![vec![0.0f64; n + 1]; n];
    for i in 0..n {
        for j in 0..n {
            b[i][j] = a[i][j] as f64;
        }
        b[i][n] = c[i] as f64;
    }

    for i in 0..n {
        let mut pivot = i;
        for j in i..n {
            if b[j][i].abs() > b[pivot][i].abs() {
                pivot = j;
            }
        }
        b.swap(i, pivot);

        if b[i][i].abs() < EPS {
            return None; // matches C's "return 0" error
        }

        // Normalize row i for columns i+1..=n (C does not explicitly set b[i][i]=1)
        let div = b[i][i];
        for j in (i + 1)..=n {
            b[i][j] /= div;
        }

        // Eliminate column i from other rows, updating columns i+1..=n
        for j in 0..n {
            if j == i {
                continue;
            }
            let factor = b[j][i];
            if factor == 0.0 {
                continue;
            }
            for k in (i + 1)..=n {
                b[j][k] -= factor * b[i][k];
            }
        }
    }

    let mut x = vec![0.0f64; n];
    for i in 0..n {
        x[i] = b[i][n];
    }
    Some(x)
}
