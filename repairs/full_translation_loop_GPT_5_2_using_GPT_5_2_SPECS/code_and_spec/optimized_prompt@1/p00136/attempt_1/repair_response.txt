```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin (no explicit error handling per spec).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut it = input.split_whitespace();

    let n: i32 = it.next().unwrap().parse().unwrap();

    let mut ca: usize = 0;
    let mut cb: usize = 0;
    let mut cc: usize = 0;
    let mut cd: usize = 0;
    let mut ce: usize = 0;
    let mut cf: usize = 0;

    // Spec: if n is negative, process no heights and still print histogram of zeros.
    if n > 0 {
        for _ in 0..n {
            let h: f64 = it.next().unwrap().parse().unwrap();
            if h < 165.0 {
                ca += 1;
            } else if h >= 165.0 && h < 170.0 {
                cb += 1;
            } else if h >= 170.0 && h < 175.0 {
                cc += 1;
            } else if h >= 175.0 && h < 180.0 {
                cd += 1;
            } else if h >= 180.0 && h < 185.0 {
                ce += 1;
            } else {
                // h >= 185.0
                cf += 1;
            }
        }
    }

    // Preserve exact output formatting: "k:" then k markers, then newline; six lines.
    let mut out = String::new();

    out.push_str("1:");
    out.push_str(&"*".repeat(ca));
    out.push('\n');

    out.push_str("2:");
    out.push_str(&"*".repeat(cb));
    out.push('\n');

    out.push_str("3:");
    out.push_str(&"*".repeat(cc));
    out.push('\n');

    out.push_str("4:");
    out.push_str(&"*".repeat(cd));
    out.push('\n');

    out.push_str("5:");
    out.push_str(&"*".repeat(ce));
    out.push('\n');

    out.push_str("6:");
    out.push_str(&"*".repeat(cf));
    out.push('\n');

    print!("{out}");
}
```
