```rust
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Rect {
    height: i64,
    pos: i64,
}

fn large_histogram(table: &mut Vec<i64>, w: i64) -> i64 {
    let mut sp: usize = 0;
    let mut max_area: i64 = 0;
    let mut stack: Vec<Rect> = Vec::with_capacity((w as usize) + 1);

    // Matches C: table[w] = 0 sentinel to flush the stack.
    table.push(0);

    for i in 0..=w {
        let mut r = Rect {
            height: table[i as usize],
            pos: i,
        };

        if sp == 0 || stack[sp - 1].height < r.height {
            stack.push(r);
            sp += 1;
        } else if stack[sp - 1].height > r.height {
            let mut target = i;
            while sp != 0 && stack[sp - 1].height >= r.height {
                sp -= 1;
                let pre = stack[sp];
                // Per spec, overflow is undefined; we use normal i64 arithmetic.
                let area = pre.height * (i - pre.pos);
                if max_area < area {
                    max_area = area;
                }
                target = pre.pos;
            }
            r.pos = target;
            if sp < stack.len() {
                stack[sp] = r;
            } else {
                stack.push(r);
            }
            sp += 1;
        }
        // If equal height, do nothing (matches C behavior: neither branch taken).
    }

    max_area
}

fn main() {
    // Read all input; no explicit error handling (matches spec/C behavior expectations).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let w: i64 = it.next().unwrap().parse().unwrap();

    let mut table: Vec<i64> = Vec::with_capacity((w as usize) + 1);
    for _ in 0..w {
        let h: i64 = it.next().unwrap().parse().unwrap();
        table.push(h);
    }

    let temp = large_histogram(&mut table, w);
    let mut max_out: i64 = 0;
    if max_out < temp {
        max_out = temp;
    }

    print!("{max_out}\n");
}
```
