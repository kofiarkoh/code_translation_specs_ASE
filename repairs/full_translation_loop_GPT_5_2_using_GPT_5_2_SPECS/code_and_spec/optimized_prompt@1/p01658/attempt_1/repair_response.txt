```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // C code reads: scanf("%d%d",&M,&N); where M=rows, N=cols
    let m: usize = it.next().unwrap().parse().unwrap(); // rows
    let n: usize = it.next().unwrap().parse().unwrap(); // cols

    // Spec says columns <= 10^6, but C code uses A[106]. We follow the spec and allocate by n.
    // Note: This is a mismatch with the C code's fixed-size buffer; spec is treated as source-of-truth.
    let mut a: i64 = 0;
    let mut row: Vec<i64> = vec![0; n];

    for i in 0..m {
        for j in 0..n {
            let mut v: i64 = it.next().unwrap().parse().unwrap();
            if i != 0 {
                v ^= 1;
            }
            row[j] = v;
        }

        if n == 0 {
            continue;
        }

        let mut l: isize = 0;
        let mut r: isize = (n as isize) - 1;

        while l <= r {
            let li = l as usize;
            let ri = r as usize;

            if row[li] == 1 {
                a += 1;
                l += 1;
                // Mirrors C: A[++l] ^= 1; (note: this can go out of bounds when l was r==n-1)
                if (l as usize) < n {
                    row[l as usize] ^= 1;
                }
            } else if row[ri] == 1 {
                a += 1;
                r -= 1;
                // Mirrors C: A[--r] ^= 1; (note: this can go out of bounds when r was l==0)
                if r >= 0 {
                    row[r as usize] ^= 1;
                }
            } else {
                l += 1;
                // Mirrors C: A[++l] ^= 1; (same potential out-of-bounds)
                if (l as usize) < n {
                    row[l as usize] ^= 1;
                }
            }
        }
    }

    print!("{a}\n");
}
```
