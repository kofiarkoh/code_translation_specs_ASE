```rust
use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;

fn add_bit(n: usize, bit: &mut [i64], mut i: usize, k: i64) {
    // Fenwick tree point update: BIT[i] += k (mod MOD)
    let mut val = k % MOD;
    if val < 0 {
        val += MOD;
    }
    while i <= n {
        bit[i] += val;
        bit[i] %= MOD;
        i += i & (!i + 1); // i += lowbit(i)
    }
}

fn sum_bit(bit: &[i64], mut r: usize) -> i64 {
    // Fenwick tree prefix sum: sum_{1..r} BIT (mod MOD)
    let mut sum = 0i64;
    while r > 0 {
        sum += bit[r];
        sum %= MOD;
        r &= r - 1; // r -= lowbit(r)
    }
    sum
}

fn main() {
    // Read single integer n; no explicit error handling per spec.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let n: i64 = input.split_whitespace().next().unwrap().parse().unwrap();

    if n == 1 {
        print!("1\n");
        return;
    }

    // Preconditions say 1 <= n <= 1_000_000; behavior otherwise is undefined in C.
    let n_usize = n as usize;

    // BIT is 1-indexed; allocate n+1.
    let mut bit = vec![0i64; n_usize + 1];

    // tmp corresponds to F(2) initially (per spec: F(2) = n^2 mod MOD).
    let mut tmp = (n % MOD) * (n % MOD) % MOD;

    // C code seeds BIT with F(1)=n and F(2)=tmp.
    // This matches the recurrence usage of prefix sums over F(j).
    add_bit(n_usize, &mut bit, 1, n % MOD);
    add_bit(n_usize, &mut bit, 2, tmp);

    for i in 3..=n_usize {
        let i_i64 = i as i64;

        // sum_{j=1}^{i-3} F(j)
        let s = sum_bit(&bit, i - 3);

        // (i-3)(n-1) + (n-i+2)n
        let term1 = ((i_i64 - 3) % MOD) * ((n - 1) % MOD) % MOD;
        let term2 = ((n - i_i64 + 2) % MOD) * (n % MOD) % MOD;

        tmp = (tmp + s + term1 + term2) % MOD;
        add_bit(n_usize, &mut bit, i, tmp);
    }

    print!("{tmp}\n");
}
```
