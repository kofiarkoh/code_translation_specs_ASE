1. Functional Description  
Computes, for each input case, a maximum-total-length ride through an undirected weighted graph such that no edge is used more than once. Outputs the maximum achievable total length and one corresponding sequence of visited vertices (a trail) that attains that length.

2. Inputs  
- Repeated test cases read from standard input.  
- For each test case:  
  - An integer count of vertices.  
  - An integer count of edges.

- A test case with vertex count equal to zero terminates processing.

For each non-terminating test case, writes to standard output:  
- The maximum total length among all trails that do not reuse any edge.

- The reported trail never uses the same input edge more than once.

#include <stdio.h>
#include <string.h>

typedef struct { int a, b, d, f; } T;
T tbl[25]; int n, m;
int ans[12], tmp[12], len, num;

void combi(int k, int a, int w)
{
	int i, b;

	for (b = 1; b <= n; b++) {
		if (b == a) continue;
		for (i = 0; i < m; i++) {
			if ((tbl[i].a == a && tbl[i].b == b || tbl[i].a == b && tbl[i].b == a) && !tbl[i].f) {
				tbl[i].f = 1, tmp[k] = b;
				combi(k+1, b, w + tbl[i].d);
				tbl[i].f = 0;
			}
		}
	}
	if (w > len) num = k, len = w, memcpy(ans, tmp, sizeof(int)*num);
}

int main()
{
	int i, a, b;

	while (scanf("%d%d", &n, &m) && n) {
		for (i = 0; i < m; i++) scanf("%d%d%d", &tbl[i].a, &tbl[i].b, &tbl[i].d), tbl[i].f = 0;
		for (len = num = 0, a = 1; a <= n; a++) for (b = 1; b <= n; b++) {
			if (b == a) continue;
			for (i = 0; i < m; i++) {
				if ((tbl[i].a == a && tbl[i].b == b || tbl[i].b == a && tbl[i].a == b) && !tbl[i].f) {
					tbl[i].f = 1;
					tmp[0] = a, tmp[1] = b, combi(2, b, tbl[i].d);
					tbl[i].f = 0;
				}
			}
		}
		printf("%d\n", len);

    }

}
