int d[201][201];
int r[9];
int res;
bool used[9];
int N, M, R;

void dfs(int c, int las, int dist){
    if(c == R + 1){
        if(res > dist){
            res = dist;
        }
        return;
    }

    for(int i = 1; i <= R; i++){
        if(!used[i]){

            if(las == -1){
                dfs(c+1, i, 0);
            }else{
                dfs(c+1, i, dist + d[r[las]][r[i]]);
            }
            used[i] = false;
        }
    }
}

    scanf("%d%d%d", &N, &M, &R);

    for(int i = 1; i <= N; i++){
