use std::io::{self, Read};

fn digits_pow10(x: u32) -> u64 {
    // Returns 10^(number of decimal digits of x), with the C behavior that 0 -> 10.
    // (C loop starts k=10 and stops immediately for x=0.)
    if x < 10 {
        10
    } else if x < 100 {
        100
    } else if x < 1_000 {
        1_000
    } else if x < 10_000 {
        10_000
    } else if x < 100_000 {
        100_000
    } else if x < 1_000_000 {
        1_000_000
    } else if x < 10_000_000 {
        10_000_000
    } else if x < 100_000_000 {
        100_000_000
    } else if x < 1_000_000_000 {
        1_000_000_000
    } else {
        10_000_000_000
    }
}

fn main() {
    // Fast input: read all, split whitespace.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };

    // Store as u32 (spec: non-negative; C uses int). Use u64 for concatenation arithmetic.
    let mut a: Vec<u32> = Vec::with_capacity(n);
    for _ in 0..n {
        let v: u32 = it.next().unwrap().parse().unwrap();
        a.push(v);
    }

    // Precompute 10^digits for each element to avoid O(N^2 * digits) work (timeout fix).
    let mut pow10: Vec<u64> = Vec::with_capacity(n);
    for &v in &a {
        pow10.push(digits_pow10(v));
    }

    // Sentinel as in C (but using u64). Must be larger than any valid concatenation under preconditions.
    let mut r: [u64; 3] = [9_999_999_999; 3];

    for i in 0..n {
        let ai = a[i] as u64;
        for j in 0..n {
            if i == j {
                continue;
            }
            let val = ai * pow10[j] + a[j] as u64;

            // Maintain three smallest values with multiplicity.
            let mut x = val;
            for idx in 0..3 {
                if r[idx] > x {
                    std::mem::swap(&mut r[idx], &mut x);
                }
            }
        }
    }

    // Output as integer with newline (C prints %d; here we print the numeric value).
    println!("{}", r[2]);
}
