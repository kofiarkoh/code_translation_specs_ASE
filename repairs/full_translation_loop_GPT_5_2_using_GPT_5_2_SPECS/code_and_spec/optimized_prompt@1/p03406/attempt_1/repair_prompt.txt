#define N	16
#define M	16
#define N_	(1 << N)
#define MD	1000000007

int vv[N_], ff[N_], gg[N_];

void init() {
	int i;

	ff[0] = gg[0] = 1;
	for (i = 1; i < N_; i++) {

long long choose(int n, int k) {
	return k > n ? 0 : (long long) ff[n] * gg[k] % MD * gg[n - k] % MD;
}
