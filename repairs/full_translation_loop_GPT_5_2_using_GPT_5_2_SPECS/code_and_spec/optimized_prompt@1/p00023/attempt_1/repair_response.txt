```rust
use std::io::{self, Read};

fn sq(x: f64) -> f64 {
    x * x
}

fn main() {
    // Read entire stdin (no explicit error handling per spec).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = it.next().unwrap().parse().unwrap();
    let mut out = String::new();

    for _ in 0..n {
        let ax: f64 = it.next().unwrap().parse().unwrap();
        let ay: f64 = it.next().unwrap().parse().unwrap();
        let ar: f64 = it.next().unwrap().parse().unwrap();
        let bx: f64 = it.next().unwrap().parse().unwrap();
        let by: f64 = it.next().unwrap().parse().unwrap();
        let br: f64 = it.next().unwrap().parse().unwrap();

        let dx = ax - bx;
        let dy = ay - by;
        let d2 = sq(dx) + sq(dy);

        let sum_r = ar + br;
        let diff_r = ar - br;

        if d2 > sq(sum_r) {
            out.push_str("0\n");
        } else if sq(sum_r) > d2 && d2 > sq(diff_r) {
            out.push_str("1\n");
        } else if d2 < sq(diff_r) {
            if ar > br {
                out.push_str("2\n");
            } else {
                out.push_str("-2\n");
            }
        } else {
            out.push_str("1\n");
        }
    }

    print!("{out}");
}
```
