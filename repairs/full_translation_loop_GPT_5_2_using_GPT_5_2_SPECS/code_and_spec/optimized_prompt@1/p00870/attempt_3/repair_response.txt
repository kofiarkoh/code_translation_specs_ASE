```rust
use std::io::{self, Read};

fn main() {
    // Read all input; parse as whitespace-separated tokens (scanf-like).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let Some(n_tok) = it.next() else { break };
        let n: usize = n_tok.parse().unwrap();
        let m: usize = it.next().unwrap().parse().unwrap();
        if n == 0 && m == 0 {
            break;
        }

        let mut patterns: Vec<Vec<u8>> = Vec::with_capacity(n);
        let mut lens: Vec<usize> = Vec::with_capacity(n);
        for _ in 0..n {
            let s = it.next().unwrap();
            let b = s.as_bytes().to_vec();
            lens.push(b.len());
            patterns.push(b);
        }

        let mut text: Vec<u8> = Vec::new();
        for _ in 0..m {
            let frag = it.next().unwrap();
            text.extend_from_slice(frag.as_bytes());
        }
        let l_s = text.len();

        let states = 1usize << n;
        let full = states - 1;

        // The original C program uses a boolean reachability table (not counts).
        // It also implicitly allows starting from ANY position with empty mask because it
        // does not require flag[i][0] to be set (it only skips when j!=0 and flag[i][j]==0).
        // Although the provided SPEC says "start at the beginning", the C code's behavior
        // is to count any position where all patterns can be placed contiguously.
        // We follow the C code as the source-of-truth for edge behavior.
        let mut flag = vec![vec![false; states]; l_s + 1];

        for i in 0..l_s {
            for mask in 0..states {
                if mask != 0 && !flag[i][mask] {
                    continue;
                }
                for k in 0..n {
                    let bit = 1usize << k;
                    if (mask & bit) != 0 {
                        continue;
                    }
                    let l = lens[k];
                    if i + l <= l_s && text[i..i + l] == patterns[k][..] {
                        flag[i + l][mask | bit] = true;
                    }
                }
            }
        }

        let mut cnt: i32 = 0;
        for i in 0..=l_s {
            if flag[i][full] {
                cnt += 1;
            }
        }

        out.push_str(&cnt.to_string());
        out.push('\n');
    }

    print!("{out}");
}
```
