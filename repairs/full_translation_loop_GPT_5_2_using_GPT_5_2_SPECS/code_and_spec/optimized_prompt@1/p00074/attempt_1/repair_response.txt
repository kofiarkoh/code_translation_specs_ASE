```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin, then parse whitespace-separated integers.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    loop {
        let h = match it.next() {
            Some(v) => match v.parse::<i32>() {
                Ok(n) => n,
                Err(_) => break, // non-integer => terminate with no further output
            },
            None => break, // input exhausted
        };
        let m = match it.next() {
            Some(v) => match v.parse::<i32>() {
                Ok(n) => n,
                Err(_) => break,
            },
            None => break, // incomplete triple
        };
        let s = match it.next() {
            Some(v) => match v.parse::<i32>() {
                Ok(n) => n,
                Err(_) => break,
            },
            None => break, // incomplete triple
        };

        // Sentinel: stop processing when all three are -1.
        if h == -1 && m == -1 && s == -1 {
            break;
        }

        // Preserve C's integer arithmetic semantics for typical inputs (no overflow assumed by spec).
        let cur_sec = h * 3600 + m * 60 + s;
        let mut remaining = 2 * 3600 - cur_sec;

        println!(
            "{:02}:{:02}:{:02}",
            remaining / 3600,
            (remaining % 3600) / 60,
            remaining % 60
        );

        remaining *= 3;

        println!(
            "{:02}:{:02}:{:02}",
            remaining / 3600,
            (remaining % 3600) / 60,
            remaining % 60
        );
    }
}
```
