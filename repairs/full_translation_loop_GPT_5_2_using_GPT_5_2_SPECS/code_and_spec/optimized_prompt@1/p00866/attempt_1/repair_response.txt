```rust
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Entry {
    t: i32,
    id: usize,
}

fn check(id: usize, t: [i32; 3], a: &[[i32; 60]; 12], tbl: &mut Vec<Entry>) {
    // Permutations of indices for assigning observed values to (hour, minute, second) hands.
    const C: [[usize; 3]; 6] = [
        [0, 1, 2],
        [0, 2, 1],
        [1, 0, 2],
        [1, 2, 0],
        [2, 0, 1],
        [2, 1, 0],
    ];

    for j in 0..60 {
        for perm in C.iter() {
            let h = (t[perm[0]] + j) % 60;
            let hh = (h / 5) as usize; // hour in 0..11
            let m = (t[perm[1]] + j) % 60;
            let s = (t[perm[2]] + j) % 60;

            // a[hh][m] gives the expected hour-hand minute-mark (0..59) at hour hh and minute m.
            if a[hh][m as usize] == h {
                let tt = (((hh as i32) * 60 + m) * 60 + s) as i32;
                tbl.push(Entry { t: tt, id });
            }
        }
    }
}

fn main() {
    // Read all input as tokens.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // Precompute mapping a[hour][minute] = (60*hour + minute)/12
    // (integer division), matching the C code.
    let mut a = [[0i32; 60]; 12];
    for hh in 0..12 {
        for m in 0..60 {
            a[hh][m] = ((60 * hh + m) / 12) as i32;
        }
    }

    let mut out = String::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: i32 = n_opt.unwrap().parse().unwrap();
        if n <= 0 {
            break;
        }
        let n_usize = n as usize;

        let mut tbl: Vec<Entry> = Vec::new();

        for id in 0..n_usize {
            let t0: i32 = it.next().unwrap().parse().unwrap();
            let t1: i32 = it.next().unwrap().parse().unwrap();
            let t2: i32 = it.next().unwrap().parse().unwrap();
            check(id, [t0, t1, t2], &a, &mut tbl);
        }

        tbl.sort_by_key(|e| e.t);

        // Find minimum-length interval [min_t, max_t] that contains at least one feasible time per watch.
        // Tie-breaking: earliest start time (as in C: only update on strictly smaller duration).
        let mut best_d: i32 = i32::MAX;
        let mut best_min: i32 = 0;
        let mut best_max: i32 = 0;

        for i in 0..tbl.len() {
            let mut seen = vec![false; n_usize];
            let mut k = 0usize;

            for j in i..tbl.len() {
                let id = tbl[j].id;
                if seen[id] {
                    continue;
                }
                seen[id] = true;
                k += 1;
                if k == n_usize {
                    let d = tbl[j].t - tbl[i].t;
                    if d < best_d {
                        best_d = d;
                        best_min = tbl[i].t;
                        best_max = tbl[j].t;
                    }
                    break;
                }
            }
        }

        out.push_str(&format!(
            "{:02}:{:02}:{:02} {:02}:{:02}:{:02}\n",
            best_min / 3600,
            (best_min / 60) % 60,
            best_min % 60,
            best_max / 3600,
            (best_max / 60) % 60,
            best_max % 60
        ));
    }

    print!("{out}");
}
```
