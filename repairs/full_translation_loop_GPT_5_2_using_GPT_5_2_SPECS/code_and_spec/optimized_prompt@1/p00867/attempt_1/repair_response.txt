```rust
use std::io::{self, Read};

const MAX: usize = 1002; // matches C fixed capacity and grid bounds

#[derive(Copy, Clone, Default, PartialEq, Eq)]
struct Point {
    x: usize,
    y: usize,
}

#[derive(Copy, Clone, Default)]
struct Seg {
    s: Point,
    e: Point,
}

#[derive(Copy, Clone, Default)]
struct Comp {
    n: u8,            // number of segments in component (C: char)
    u: [Seg; 5],      // up to 5 segments stored (C assumes this)
    p: u8,            // shared-endpoint incidences count (C: char)
}

struct UnionFind {
    id: Vec<usize>,
    sz: Vec<usize>,
}
impl UnionFind {
    fn new(n: usize) -> Self {
        let mut id = Vec::with_capacity(n);
        let mut sz = Vec::with_capacity(n);
        for i in 0..n {
            id.push(i);
            sz.push(1);
        }
        Self { id, sz }
    }
    fn root(&mut self, mut i: usize) -> usize {
        while i != self.id[i] {
            let p = self.id[i];
            self.id[i] = self.id[p];
            i = self.id[i];
        }
        i
    }
    fn unite(&mut self, p: usize, q: usize) {
        let i = self.root(p);
        let j = self.root(q);
        if i == j {
            return;
        }
        if self.sz[i] < self.sz[j] {
            self.id[i] = j;
            self.sz[j] += self.sz[i];
        } else {
            self.id[j] = i;
            self.sz[i] += self.sz[j];
        }
    }
}

fn p_on_seg(p: Point, u: Seg) -> bool {
    // Strict interior only: exclude endpoints (matches C)
    if p == u.s || p == u.e {
        return false;
    }
    let (mut x1, mut y1, mut x2, mut y2) = (u.s.x as i32, u.s.y as i32, u.e.x as i32, u.e.y as i32);
    let (px, py) = (p.x as i32, p.y as i32);

    // C swaps endpoints if x1 > x2 (and swaps y accordingly)
    if x1 > x2 {
        std::mem::swap(&mut x1, &mut x2);
        std::mem::swap(&mut y1, &mut y2);
    }

    if !(x1 <= px && px <= x2) {
        return false;
    }
    let y_in_range = if y1 <= y2 {
        y1 <= py && py <= y2
    } else {
        y2 <= py && py <= y1
    };
    if !y_in_range {
        return false;
    }

    // collinearity check
    (py - y1) * (x2 - x1) == (y2 - y1) * (px - x1)
}

fn check_08(
    comp: &Comp,
    n: usize,
    uf: &mut UnionFind,
    u: &[Seg],
    ans: &mut [i32; 10],
) {
    // If any endpoint of any segment (representative roots only, as in C) lies strictly
    // in the interior of any segment in this component => digit 8 else digit 0.
    for i in 1..=n {
        if uf.root(i) == i {
            for j in 0..(comp.n as usize) {
                if p_on_seg(u[i].s, comp.u[j]) {
                    ans[8] += 1;
                    return;
                }
                if p_on_seg(u[i].e, comp.u[j]) {
                    ans[8] += 1;
                    return;
                }
            }
        }
    }
    ans[0] += 1;
}

fn check_37(
    comp: &Comp,
    n: usize,
    uf: &mut UnionFind,
    u: &[Seg],
    f: &[[u8; MAX]; MAX],
    ans: &mut [i32; 10],
) {
    // Find a segment in component whose both endpoints are shared (>1).
    // If any endpoint of any representative-root segment lies strictly on that segment => digit 3 else 7.
    for i in 0..(comp.n as usize) {
        let seg = comp.u[i];
        if f[seg.s.x][seg.s.y] > 1 && f[seg.e.x][seg.e.y] > 1 {
            for j in 1..=n {
                if uf.root(j) == j {
                    if p_on_seg(u[j].s, seg) {
                        ans[3] += 1;
                        return;
                    }
                    if p_on_seg(u[j].e, seg) {
                        ans[3] += 1;
                        return;
                    }
                }
            }
        }
    }
    ans[7] += 1;
}

fn check_256(comp: &Comp, f: &[[u8; MAX]; MAX], ans: &mut [i32; 10]) {
    // This follows the C logic exactly; it assumes the component matches the expected 5-segment pattern.
    // If the structural assumptions do not hold, behavior is not inferable (SPEC §5/§7).
    let mut p0 = [Point::default(); 2];
    let mut p1 = [Point::default(); 2];
    let mut p2 = [Point::default(); 2];

    let mut j = 0usize;
    for i in 0..(comp.n as usize) {
        let seg = comp.u[i];
        if f[seg.s.x][seg.s.y] == 1 {
            p0[j] = seg.s;
            p1[j] = seg.e;
            j += 1;
        } else if f[seg.e.x][seg.e.y] == 1 {
            p0[j] = seg.e;
            p1[j] = seg.s;
            j += 1;
        }
    }

    for i in 0..(comp.n as usize) {
        let seg = comp.u[i];

        if seg.s != p0[0] && seg.e == p1[0] {
            p2[0] = seg.s;
        } else if seg.e != p0[0] && seg.s == p1[0] {
            p2[0] = seg.e;
        }

        if seg.s != p0[1] && seg.e == p1[1] {
            p2[1] = seg.s;
        } else if seg.e != p0[1] && seg.s == p1[1] {
            p2[1] = seg.e;
        }
    }

    // Complex-number-like operations from C:
    // p = vmul(vsub(p0,p1), (0,1)) => rotate by +90 degrees: (dx,dy) -> (-dy, dx)
    // k = dot(vsub(p2,p1), p)
    let dx0 = p0[0].x as i32 - p1[0].x as i32;
    let dy0 = p0[0].y as i32 - p1[0].y as i32;
    let px0 = -dy0;
    let py0 = dx0;
    let vx0 = p2[0].x as i32 - p1[0].x as i32;
    let vy0 = p2[0].y as i32 - p1[0].y as i32;
    let k1 = vx0 * px0 + vy0 * py0;

    let dx1 = p0[1].x as i32 - p1[1].x as i32;
    let dy1 = p0[1].y as i32 - p1[1].y as i32;
    let px1 = -dy1;
    let py1 = dx1;
    let vx1 = p2[1].x as i32 - p1[1].x as i32;
    let vy1 = p2[1].y as i32 - p1[1].y as i32;
    let k2 = vx1 * px1 + vy1 * py1;

    if k1 > 0 && k2 > 0 {
        ans[2] += 1;
    } else if k1 < 0 && k2 < 0 {
        ans[5] += 1;
    } else {
        ans[6] += 1;
    }
}

fn main() {
    // Read all input, parse as non-negative integers (matches C in()).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: usize = n_opt.unwrap().parse().unwrap();
        if n == 0 {
            break;
        }

        // Fixed-size arrays to match C behavior/capacity.
        let mut ans = [0i32; 10];
        let mut t = vec![Comp::default(); n + 1];
        let mut f = [[0u8; MAX]; MAX];
        let mut uid = [[0u16; MAX]; MAX];
        let mut u = vec![Seg::default(); n + 1];

        let mut uf = UnionFind::new(n + 1);

        for i in 1..=n {
            let x1: usize = it.next().unwrap().parse().unwrap();
            let y1: usize = it.next().unwrap().parse().unwrap();
            let x2: usize = it.next().unwrap().parse().unwrap();
            let y2: usize = it.next().unwrap().parse().unwrap();

            u[i] = Seg {
                s: Point { x: x1, y: y1 },
                e: Point { x: x2, y: y2 },
            };

            if f[x1][y1] != 0 {
                uf.unite(uid[x1][y1] as usize, i);
            } else {
                uid[x1][y1] = i as u16;
            }
            f[x1][y1] = f[x1][y1].wrapping_add(1);

            if f[x2][y2] != 0 {
                uf.unite(uid[x2][y2] as usize, i);
            } else {
                uid[x2][y2] = i as u16;
            }
            f[x2][y2] = f[x2][y2].wrapping_add(1);
        }

        for i in 1..=n {
            let j = uf.root(i);
            let k = t[j].n as usize;
            t[j].n = t[j].n.wrapping_add(1);
            if k < 5 {
                t[j].u[k] = u[i];
            }
            if f[u[i].s.x][u[i].s.y] > 1 {
                t[j].p = t[j].p.wrapping_add(1);
            }
            if f[u[i].e.x][u[i].e.y] > 1 {
                t[j].p = t[j].p.wrapping_add(1);
            }
        }

        let mut s: i32 = 0;
        for i in 1..=n {
            if t[i].n >= 2 {
                s += t[i].n as i32;
                if t[i].n == 2 && t[i].p == 2 {
                    ans[4] += 1;
                } else if t[i].n == 4 && t[i].p == 6 {
                    ans[9] += 1;
                } else if t[i].n == 4 && t[i].p == 8 {
                    check_08(&t[i], n, &mut uf, &u, &mut ans);
                } else if t[i].n == 5 && t[i].p == 8 {
                    check_256(&t[i], &f, &mut ans);
                } else if t[i].n == 3 && t[i].p == 4 {
                    check_37(&t[i], n, &mut uf, &u, &f, &mut ans);
                }
            }
        }

        // Matches C: s counts segments in components with n>=2, then add counts of digits
        // where only 1 per component was added (4,3,8) to convert to segment counts.
        s += ans[4] + ans[3] + ans[8];
        ans[1] = n as i32 - s;

        out.push_str(&format!("{}", ans[0]));
        for i in 1..10 {
            out.push_str(&format!(" {}", ans[i]));
        }
        out.push('\n');
    }

    print!("{out}");
}
```
