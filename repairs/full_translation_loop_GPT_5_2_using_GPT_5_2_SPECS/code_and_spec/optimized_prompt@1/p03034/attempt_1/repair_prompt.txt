5. Postconditions  
Let the sequence be indexed from 0 to n−1. For each step size d such that 1 ≤ d and 2d < n−1, define a process starting with left index 0 and right index n−1. After k inward moves (k ≥ 1), the selected pair is at indices (k·d, n−1−k·d). A move is considered valid only while the next move would keep both indices strictly inside the ends and not cause the two indices to coincide or the left index to reach the right end. #include<stdio.h>
#define ll long long
#define max(p,q)((p)>(q)?(p):(q))

ll a[100010];
int main(){
	int n;
	scanf("%d",&n);
	for(int i=0;i<n;i++)scanf("%lld",a+i);
	ll ans=0;
	for(int d=1;d*2<n-1;d++){
		int s=0,g=n-1;
		ll crr=0;
		while(1){
			int ns=s+d;
			int ng=g-d;
			if(ng<=d||ns==ng||ns==g)break;
			s=ns;
			g=ng;
			crr+=a[s]+a[g];
			ans=max(ans,crr);
		}
	}
	printf("%lld",ans);
}
