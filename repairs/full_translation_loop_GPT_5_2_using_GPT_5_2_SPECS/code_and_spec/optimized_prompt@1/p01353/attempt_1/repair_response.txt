```rust
use std::cmp::Ordering;
use std::io::{self, Read};

#[derive(Clone, Copy, Debug)]
struct T {
    a: i64,
    d: i64,
}

fn main() {
    // Fast input: read all bytes, parse non-negative integers.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let bytes = input.as_bytes();
    let mut idx = 0usize;

    fn next_u64(bytes: &[u8], idx: &mut usize) -> u64 {
        while *idx < bytes.len() && (bytes[*idx] < b'0' || bytes[*idx] > b'9') {
            *idx += 1;
        }
        let mut n: u64 = 0;
        while *idx < bytes.len() && (b'0'..=b'9').contains(&bytes[*idx]) {
            n = n * 10 + (bytes[*idx] - b'0') as u64;
            *idx += 1;
        }
        n
    }

    let n = next_u64(bytes, &mut idx) as usize;

    // Fixed-capacity arrays in C; in Rust use Vec sized n+1.
    // Values are non-negative in input, but we store in i64 for arithmetic.
    let mut h = vec![0i64; n + 1];
    let mut a = vec![0i64; n + 1];
    let mut d = vec![0i64; n + 1];
    let mut s = vec![0i64; n + 1];

    for i in 0..=n {
        h[i] = next_u64(bytes, &mut idx) as i64;
        a[i] = next_u64(bytes, &mut idx) as i64;
        d[i] = next_u64(bytes, &mut idx) as i64;
        s[i] = next_u64(bytes, &mut idx) as i64;
    }

    let mut tbl: Vec<T> = Vec::new();
    tbl.reserve(n);

    let mut turn: i64 = 0;
    let mut ans: i64 = 0;

    for i in 1..=n {
        let k = a[i] - d[0];
        if k <= 0 {
            continue; // Only positive deltas influence computation (SPEC invariant).
        }
        if a[0] <= d[i] {
            // Infeasible/unsafe condition (SPEC postcondition).
            print!("-1\n");
            return;
        }

        turn += k;
        if s[i] > s[0] {
            ans += k;
        }

        // Derived count:
        // ((h[i] - d[i] + a[0] - 1) / (a[0] - d[i]))
        // SPEC assumes divisor is strictly positive whenever feasible.
        let denom = a[0] - d[i];
        let numer = h[i] - d[i] + a[0] - 1;
        let cnt = numer / denom;

        tbl.push(T { a: cnt, d: k });
    }

    // C's qsort comparator returns (a->a * b->d > b->a * a->d),
    // which effectively sorts ascending by ratio a/d (ties treated as equal).
    // We implement a total order using cross-multiplication; on ties, keep stable order.
    tbl.sort_by(|x, y| {
        let lhs = (x.a as i128) * (y.d as i128);
        let rhs = (y.a as i128) * (x.d as i128);
        if lhs < rhs {
            Ordering::Less
        } else if lhs > rhs {
            Ordering::Greater
        } else {
            Ordering::Equal
        }
    });

    for t in tbl {
        ans += turn * t.a - t.d;
        if h[0] <= ans {
            // Unsafe condition during aggregation (SPEC postcondition).
            print!("-1\n");
            return;
        }
        turn -= t.d;
    }

    print!("{ans}\n");
}
```
