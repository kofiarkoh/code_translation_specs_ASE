```rust
use std::io::{self, Read};

const FH: usize = 0; // lower half
const SH: usize = 1; // upper half
const ALL: usize = 2;
const MAX: usize = 200_005;

struct Scanner {
    buf: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut s = String::new();
        io::stdin().read_to_string(&mut s).unwrap();
        Self { buf: s.into_bytes(), idx: 0 }
    }
    #[inline]
    fn next_usize(&mut self) -> Option<usize> {
        let n = self.buf.len();
        while self.idx < n && self.buf[self.idx] < b'0' {
            self.idx += 1;
        }
        if self.idx >= n {
            return None;
        }
        let mut v: usize = 0;
        while self.idx < n {
            let c = self.buf[self.idx];
            if c < b'0' {
                break;
            }
            v = v * 10 + (c - b'0') as usize;
            self.idx += 1;
        }
        Some(v)
    }
}

struct State {
    // We keep fixed-capacity arrays and indices, mirroring the C code.
    que: [[i32; MAX]; 2],
    qt: [usize; 2],
    qe: [usize; 2],

    minq: [[i32; MAX]; 2],
    at: [usize; 2],
    ae: [usize; 2],

    maxq: [[i32; MAX]; 2],
    zt: [usize; 2],
    ze: [usize; 2],
}

impl State {
    fn new() -> Self {
        Self {
            que: [[0; MAX]; 2],
            qt: [0; 2],
            qe: [0; 2],
            minq: [[0; MAX]; 2],
            at: [0; 2],
            ae: [0; 2],
            maxq: [[0; MAX]; 2],
            zt: [0; 2],
            ze: [0; 2],
        }
    }

    #[inline]
    fn reset(&mut self) {
        self.qt = [0, 0];
        self.qe = [0, 0];
        self.at = [0, 0];
        self.ae = [0, 0];
        self.zt = [0, 0];
        self.ze = [0, 0];
    }

    #[inline]
    fn size(&self, id: usize) -> usize {
        self.qe[id] - self.qt[id]
    }
    #[inline]
    fn min_sz(&self, id: usize) -> usize {
        self.ae[id] - self.at[id]
    }
    #[inline]
    fn max_sz(&self, id: usize) -> usize {
        self.ze[id] - self.zt[id]
    }

    fn add(&mut self, id: usize, a: i32) {
        if id == ALL {
            self.add(SH, a);
            if self.size(FH) < self.size(SH) {
                let v = self.del(SH);
                self.add(FH, v);
            }
            return;
        }

        self.que[id][self.qe[id]] = a;
        self.qe[id] += 1;

        if self.size(id) == 1 {
            self.minq[id][self.ae[id]] = a;
            self.ae[id] += 1;
            self.maxq[id][self.ze[id]] = a;
            self.ze[id] += 1;
        } else if id == FH {
            // FH: monotone stacks without popping
            if self.minq[id][self.ae[id] - 1] > a {
                self.minq[id][self.ae[id]] = a;
                self.ae[id] += 1;
            }
            if self.maxq[id][self.ze[id] - 1] < a {
                self.maxq[id][self.ze[id]] = a;
                self.ze[id] += 1;
            }
        } else {
            // SH: monotone queues with popping from back
            while self.min_sz(id) > 0 && self.minq[id][self.ae[id] - 1] > a {
                self.ae[id] -= 1;
            }
            while self.max_sz(id) > 0 && self.maxq[id][self.ze[id] - 1] < a {
                self.ze[id] -= 1;
            }
            self.minq[id][self.ae[id]] = a;
            self.ae[id] += 1;
            self.maxq[id][self.ze[id]] = a;
            self.ze[id] += 1;
        }
    }

    fn del(&mut self, id: usize) -> i32 {
        if id == ALL {
            let a = self.del(FH);
            if self.size(FH) < self.size(SH) {
                let v = self.del(SH);
                self.add(FH, v);
            }
            return a;
        }

        if id == FH {
            self.qe[id] -= 1;
            let a = self.que[id][self.qe[id]];
            if self.minq[id][self.ae[id] - 1] == a {
                self.ae[id] -= 1;
            }
            if self.maxq[id][self.ze[id] - 1] == a {
                self.ze[id] -= 1;
            }
            a
        } else {
            let a = self.que[id][self.qt[id]];
            self.qt[id] += 1;
            if self.minq[id][self.at[id]] == a {
                self.at[id] += 1;
            }
            if self.maxq[id][self.zt[id]] == a {
                self.zt[id] += 1;
            }
            a
        }
    }

    #[inline]
    fn gmin(&self, id: usize, th: usize) -> i32 {
        if id == FH {
            self.minq[id][self.ae[id] - th]
        } else {
            self.minq[id][self.at[id] + th - 1]
        }
    }

    #[inline]
    fn gmax(&self, id: usize, th: usize) -> i32 {
        if id == FH {
            self.maxq[id][self.ze[id] - th]
        } else {
            self.maxq[id][self.zt[id] + th - 1]
        }
    }
}

fn push_i32(out: &mut Vec<u8>, mut x: i32) {
    if x == 0 {
        out.extend_from_slice(b"0\n");
        return;
    }
    if x < 0 {
        out.push(b'-');
        x = -x;
    }
    let mut buf = [0u8; 12];
    let mut i = 0usize;
    let mut v = x as u32;
    while v > 0 {
        buf[i] = b'0' + (v % 10) as u8;
        i += 1;
        v /= 10;
    }
    while i > 0 {
        i -= 1;
        out.push(buf[i]);
    }
    out.push(b'\n');
}

fn main() {
    let mut sc = Scanner::new();
    let mut st = State::new();
    let mut out: Vec<u8> = Vec::with_capacity(1 << 20);

    while let Some(q) = sc.next_usize() {
        if q == 0 {
            break;
        }
        st.reset();

        for _ in 0..q {
            let op = sc.next_usize().unwrap();
            let a: i32;
            match op {
                0 => {
                    let x = sc.next_usize().unwrap() as i32;
                    st.add(ALL, x);
                    continue;
                }
                1 => a = st.del(ALL),
                2 => a = st.gmin(FH, 1),
                3 => a = st.gmin(SH, 1),
                4 => {
                    let k = sc.next_usize().unwrap();
                    a = st.gmin(FH, k);
                }
                5 => {
                    let k = sc.next_usize().unwrap();
                    a = st.gmin(SH, k);
                }
                6 => a = st.gmax(FH, 1),
                7 => a = st.gmax(SH, 1),
                8 => {
                    let k = sc.next_usize().unwrap();
                    a = st.gmax(FH, k);
                }
                _ => {
                    let k = sc.next_usize().unwrap();
                    a = st.gmax(SH, k);
                }
            }
            push_i32(&mut out, a);
        }
        out.extend_from_slice(b"end\n");
    }

    print!("{}", String::from_utf8_lossy(&out));
}
```
