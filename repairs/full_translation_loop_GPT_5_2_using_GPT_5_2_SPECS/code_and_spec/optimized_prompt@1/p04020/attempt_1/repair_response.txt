```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // Per spec: undefined behavior on malformed/missing input; we unwrap.
    let n: usize = it.next().unwrap().parse().unwrap();

    // Fixed capacity per spec (<= 100000). Undefined if violated; we don't enforce.
    // Use 1-based indexing to mirror the C code and its semantics exactly.
    let mut a: Vec<i64> = vec![0; n + 1];
    for i in 1..=n {
        a[i] = it.next().unwrap().parse().unwrap();
    }

    let mut ans: i64 = 0;

    // Process from first to last, allowing at most one cross-position pair per boundary,
    // only when successor has > 1 at the moment and current has an odd leftover.
    for i in 1..n {
        ans += a[i] / 2;
        if a[i + 1] > 1 && (a[i] % 2 != 0) {
            a[i + 1] -= 1;
            ans += 1;
        }
    }
    ans += a[n] / 2;

    print!("{ans}\n");
}
```
