```rust
use std::io::{self, Read};

fn main() {
    // Read entire stdin
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();

    // We use 0-based vectors but keep the same indexing scheme as the C code:
    // A[1..=N+1], B[0..=N+1]
    let mut a = vec![0i64; n + 2];
    let mut b = vec![0i64; n + 2];

    for i in 1..=n + 1 {
        a[i] = it.next().unwrap().parse().unwrap();
    }
    for i in 1..=n {
        b[i] = it.next().unwrap().parse().unwrap();
    }

    b[0] = 0;
    b[n + 1] = 0;

    let mut s: i64 = 0;

    for i in 1..=n + 1 {
        // First, use capacity B[i-1] on A[i] as much as possible.
        let take_left = if a[i] > b[i - 1] { b[i - 1] } else { a[i] };
        s += take_left;
        a[i] = if a[i] > b[i - 1] { a[i] - b[i - 1] } else { 0 };

        // Then, use capacity B[i] on the remaining A[i] as much as possible.
        let take_right = if a[i] > b[i] { b[i] } else { a[i] };
        s += take_right;

        // Update remaining capacity B[i] after taking from A[i].
        b[i] = if a[i] > b[i] { 0 } else { b[i] - a[i] };
    }

    print!("{s}\n");
}
```
