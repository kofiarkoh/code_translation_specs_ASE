use std::io::{self, BufRead};
use std::cmp::{max};

const LLINF: i64 = 4154118101919364364;
const MOD: i64 = 1000000007;

#[derive(Debug)]
struct Sd {
    val: i64,
    node: i64,
}

fn make_pl() -> (Vec<i64>, Vec<i64>, Vec<i64>) {
    let mut fl = vec![false; 2222];
    let mut pl = Vec::new();
    let mut dpl = Vec::new();
    let mut trpl = Vec::new();

    for i in 2..2222 {
        if fl[i] {
            continue;
        }
        pl.push(i.try_into().unwrap());
        dpl.push((i * i).try_into().unwrap());
        trpl.push((i * i * i).try_into().unwrap());
        for j in (2 * i..2222).step_by(i) {
            fl[j] = true;
        }
    }
    (pl, dpl, trpl)
}

fn llsqr(x: i64) -> i64 {
    let (mut st, mut fi) = (1, 1_000_000_000);
    while st <= fi {
        let te = (st + fi) / 2;
        if te * te < x {
            st = te + 1;
        } else {
            fi = te - 1;
        }
    }
    st
}

fn main() {

    let reader = stdin.lock();
    let mut lines = reader.lines();

    let n: usize = lines.next().unwrap().unwrap().parse().unwrap();
    if n == 0 {
        println!("0");
        return;
    }

    let (pl, dpl, trpl) = make_pl();

                w /= trpl[j];

            let l = llsqr(temp_t);
            if l * l != temp_t {
                r += 1;
                continue;
            }

        a.push(w);
    }

    if a.is_empty() {
        println!("{}", r);
        return;

    let mut tfl = vec![false; a.len()];

            t += 1;
            w = 1;

        } else {
            w += 1;
        }

    t += 1;
    dat.push(Sd { val: LLINF, node: 0 });

    for i in 0..t {
        if tfl[i] {
            continue;
        }
        tfl[i] = true;
        if dat[i].val == 1 {
            r += 1;
            continue;

                h /= pl[j];

            if dat[i].val % pl[j] == 0 {
                if dat[i].val % dpl[j] == 0 {
                    w *= pl[j];
                    if w > 10_000_000_000 {
                        w = -1;
                        break;
                    }
                } else {
                    w *= pl[j];
                    if w > 10_000_000_000 {
                        w = -1;
                        break;
                    }
                    w *= pl[j];
                    if w > 10_000_000_000 {
                        w = -1;
                        break;
                    }
                }
            }
        }

        if w == -1 {
            r += dat[i].node;

            if w > 10_000_000_000 {
