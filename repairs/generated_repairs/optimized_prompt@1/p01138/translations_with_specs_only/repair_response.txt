fn main() {
    use std::io::{self, BufRead};

    // Read input from standard input
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let line = line.unwrap();
        let mut parts = line.split_whitespace();

        // Read the number of intervals
        let num_intervals: usize = parts.next().unwrap().parse().unwrap();
        if num_intervals == 0 {
            break; // Terminate on zero input
        }

        // Initialize an array to count overlaps for each second of the day
        let mut time_array = vec![0; 86400]; // 86,400 seconds in a day

        for _ in 0..num_intervals {
            // Read departure and arrival times
            let dep_h: usize = parts.next().unwrap().parse().unwrap();
            let dep_m: usize = parts.next().unwrap().parse().unwrap();
            let dep_s: usize = parts.next().unwrap().parse().unwrap();
            let arr_h: usize = parts.next().unwrap().parse().unwrap();
            let arr_m: usize = parts.next().unwrap().parse().unwrap();
            let arr_s: usize = parts.next().unwrap().parse().unwrap();

            // Convert times to seconds since start of the day
            let dep_time = dep_h * 3600 + dep_m * 60 + dep_s;
            let arr_time = arr_h * 3600 + arr_m * 60 + arr_s;

            // Increment the start of the interval
            time_array[dep_time] += 1;

            // Decrement the end of the interval (exclusive)
            if arr_time < 86400 {
                time_array[arr_time] -= 1;
            } else {
                // Ensure we don't decrement out of bounds
                time_array[86399] -= 1;
            }
        }

        // Calculate the maximum overlap
        let mut max_overlap = 0;
        let mut current_overlap = 0;

        for count in time_array {
            current_overlap += count;
            if current_overlap > max_overlap {
                max_overlap = current_overlap;
            }
        }

        // Output the maximum number of overlapping intervals
        println!("{}", max_overlap);
    }
}
