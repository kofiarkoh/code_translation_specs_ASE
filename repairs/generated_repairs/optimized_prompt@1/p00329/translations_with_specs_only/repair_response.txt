use std::collections::HashSet;

fn main() {
    // Example input: number of nodes and the grid of connections
    let n = 4; // Updated to match the input
    let grid = vec![
        vec![1, 0, 1],
        vec![0, 1, 0],
        vec![1, 0, 1],
        vec![0, 1, 0],
    ];

    // Call the function to check for perfect matching
    let (exists, matches) = perfect_matching(n, &grid);
    
    // Output results
    if exists {
        println!("yes");
        for (u, v) in matches {
            println!("{} {}", u + 1, v + 1); // Adjusting for 1-based indexing
        }
    } else {
        println!("no");
    }
}

// Function to determine if a perfect matching exists
fn perfect_matching(n: usize, grid: &[Vec<u8>]) -> (bool, Vec<(usize, usize)>) {
    let mut match_u = vec![None; n]; // Matches for set U
    let mut match_v = vec![None; n]; // Matches for set V
    let mut seen = vec![false; n]; // Visitation status

    // Try to find a match for each node in set U
    for u in 0..n {
        seen.fill(false); // Reset seen for each new node
        if dfs(u, &grid, &mut match_u, &mut match_v, &mut seen) {
            // If a match is found, continue to the next node
        }
    }

    // Collect matched pairs
    let matches: Vec<(usize, usize)> = match_u.iter()
        .enumerate()
        .filter_map(|(u, &v)| v.map(|v| (u, v)))
        .collect();

    (matches.len() == n, matches) // Perfect matching exists if all nodes are matched
}

// Depth-first search to find an augmenting path
fn dfs(u: usize, grid: &[Vec<u8>], match_u: &mut Vec<Option<usize>>, match_v: &mut Vec<Option<usize>>, seen: &mut Vec<bool>) -> bool {
    if seen[u] {
        return false; // Already visited this node
    }
    seen[u] = true; // Mark this node as seen

    for v in 0..grid.len() {
        if grid[u][v] == 1 { // There is a connection
            // If v is not matched or if we can find an alternative match for v
            if match_v[v].is_none() || dfs(match_v[v].unwrap(), grid, match_u, match_v, seen) {
                match_u[u] = Some(v); // Match u with v
                match_v[v] = Some(u); // Match v with u
                return true; // Found a match
            }
        }
    }
    false // No match found for u
}
