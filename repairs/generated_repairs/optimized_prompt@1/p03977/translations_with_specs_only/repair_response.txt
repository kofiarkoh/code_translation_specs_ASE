```rust
use std::io::{self, BufRead};

fn main() {
    // Read input from standard input
    let stdin = io::stdin();
    let input = stdin.lock().lines();

    // Initialize a vector to store results
    let mut results = Vec::new();

    // Process each line of input
    for line in input {
        let line = line.unwrap();
        let mut sum = 0;

        // Process each number in the line
        for number in line.split_whitespace() {
            if let Ok(num) = number.parse::<u32>() {
                sum += num; // Accumulate the sum of numbers
            }
        }

        // Store the result for the current line
        results.push(sum.to_string());
    }

    // Output results, ensuring each result is on a new line
    for result in results {
        println!("{}", result);
    }
}
```
