```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();
    
    let first_line = lines.next().unwrap().unwrap();
    let mut nums = first_line.split_whitespace().map(|s| s.parse::<usize>());
    let n = match nums.next() {
        Some(Ok(num)) => num,
        _ => {
            println!("0");
            return;
        }
    };
    let k = match nums.next() {
        Some(Ok(num)) => num,
        _ => {
            println!("0");
            return;
        }
    };

    if n < k {
        println!("0");
        return;
    }

    let mut a = Vec::with_capacity(n);
    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        match line.trim().parse::<usize>() {
            Ok(num) => a.push(num),
            Err(_) => {
                println!("0");
                return;
            }
        }
    }

    let mut f = vec![0; k + 1];
    let mut s = 0; // Count of unique integers from 1 to K
    let mut ans = n + 1; // Initialize to a value larger than any possible subarray length
    let (mut i, mut j) = (0, 0);

    // Expand the right boundary of the window
    while s < k && j < n {
        let t = a[j];
        if t <= k {
            if f[t] == 0 {
                s += 1;
            }
            f[t] += 1;
        }
        j += 1;
    }

    // If we didn't find a valid window
    if s < k {
        println!("0");
        return;
    }

    // Contract the left boundary of the window
    while i < j && (a[i] > k || f[a[i]] > 1) {
        if a[i] <= k {
            f[a[i]] -= 1;
        }
        i += 1;
    }
    ans = j - i;

    // Continue to expand the right boundary and adjust the left boundary
    while j < n {
        let t = a[j];
        if t <= k {
            f[t] += 1;
        }
        j += 1;

        while i < j && (a[i] > k || f[a[i]] > 1) {
            if a[i] <= k {
                f[a[i]] -= 1;
            }
            i += 1;
        }

        if j - i < ans {
            ans = j - i;
        }
    }

    println!("{}", if ans == n + 1 { 0 } else { ans });
}
```
