```rust
fn main() {
    let n = 100; // Example input, replace with actual input as needed
    if n < 2 {
        panic!("Input must be greater than or equal to 2");
    }
    
    let mut prime_counts = vec![0; n + 1]; // Array to store counts of prime factors

    // Perform prime factorization for each integer from 2 to n
    for i in 2..=n {
        let mut num = i;
        let mut factor = 2;

        // Count prime factors
        while num > 1 {
            while num % factor == 0 {
                prime_counts[factor] += 1; // Update count for this prime factor
                num /= factor; // Reduce num by the prime factor
            }
            factor += 1; // Move to the next potential prime factor
        }
    }

    // Calculate the result based on the counts of prime factors
    let result = prime_counts.iter().filter(|&&count| count > 0).sum::<usize>() as i32; // Sum of counts as the result

    // Output the result
    println!("{}", result);
}
```
