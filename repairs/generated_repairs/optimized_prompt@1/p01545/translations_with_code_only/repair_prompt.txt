const MAX_SIZE: usize = 262145;

struct SegmentTree {
    seg: [i64; MAX_SIZE],
    sz: usize,
}

impl SegmentTree {
    fn new(n: usize) -> Self {
        let mut sz = 1;
        while sz < n {
            sz <<= 1;
        }
        SegmentTree {
            seg: [0; MAX_SIZE],
            sz,
        }

    // update kth value to v
    fn update(&mut self, mut k: usize, v: i64) {
        k += self.sz - 1;
        self.seg[k] = v;
        while k > 0 {
            if k & 1 == 0 {
                k -= 1;
            }
            let t = k;
            k >>= 1;
            let mut max_val = self.seg[t + 1];
            if self.seg[t] > max_val {
                max_val = self.seg[t];
            }
            self.seg[k] = max_val;
        }
    }

    // get max value in [a, b)
    fn query(&self, a: usize, b: usize, mut k: usize, l: usize, r: usize) -> i64 {
        if r <= a || b <= l {
            return 0;

        }
        lmax
    }
}
