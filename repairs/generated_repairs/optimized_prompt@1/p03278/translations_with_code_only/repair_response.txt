use std::io::{self, BufRead};

const MOD: usize = 1_000_000_007;

fn calc(x: i64) -> i64 {
    if x & 1 == 1 {
        0
    } else {
        pw[(x >> 1) as usize]
    }
}

fn add_edge(u: usize, v: usize, lst: &mut Vec<i32>, to: &mut Vec<usize>, pre: &mut Vec<i32>, tot: &mut usize) {
    to[*tot] = v;
    pre[*tot] = lst[u];
    lst[u] = *tot as i32;
    *tot += 1;
}

fn init_dfs(u: usize, fa: i32, lst: &Vec<i32>, to: &Vec<usize>, sz: &mut Vec<i64>, pre: &Vec<i32>) {
    sz[u] = 1;
    let mut i = lst[u];
    while i != -1 {
        let v = to[i as usize];
        if v as i32 == fa {
            i = pre[i as usize];
            continue;
        }
        init_dfs(v, u as i32, lst, to, sz, pre);
        sz[u] += sz[v];
        i = pre[i as usize];
    }
}

fn dfs(u: usize, fa: i32, lst: &Vec<i32>, to: &Vec<usize>, dp: &mut Vec<Vec<[i64; 2]>>, dp_sz: &mut Vec<i64>, sz: &Vec<i64>, pre: &Vec<i32>) {
    dp[u][0][0] = 1;
    let mut i = lst[u];
    while i != -1 {
        let v = to[i as usize];
        if v as i32 == fa {
            i = pre[i as usize];
            continue;
        }
        dfs(v, u as i32, lst, to, dp, dp_sz, sz, pre);
        let mut tmp = [[0; 2]; 5005];
        for x in (0..=dp_sz[u]).rev() {
            for y in (0..=dp_sz[v]).rev() {
                tmp[x + y as usize][0] = (tmp[x + y as usize][0] + dp[u][x as usize][0] * dp[v][y as usize][0]) % MOD;
                tmp[x + y as usize][0] = (tmp[x + y as usize][0] + dp[u][x as usize][1] * dp[v][y as usize][1]) % MOD;
                tmp[x + y as usize][1] = (tmp[x + y as usize][1] + dp[u][x as usize][0] * dp[v][y as usize][1]) % MOD;
                tmp[x + y as usize][1] = (tmp[x + y as usize][1] + dp[u][x as usize][1] * dp[v][y as usize][0]) % MOD;
            }
        }
        for x in 0..=dp_sz[u] {
            dp[u][x as usize] = tmp[x as usize];
        }
        dp_sz[u] += dp_sz[v];
        i = pre[i as usize];
    }
    if u != 0 {
        for x in (0..=dp_sz[u]).rev() {
            dp[u][sz[u] as usize][0] = (dp[u][sz[u] as usize][0] + dp[u][x as usize][1] * calc(sz[u] - x)) % MOD;
            dp[u][sz[u] as usize][1] = (dp[u][sz[u] as usize][1] + dp[u][x as usize][0] * calc(sz[u] - x)) % MOD;
        }
    }
    dp_sz[u] += 1;
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();
    
    let n: usize = lines.next().unwrap().unwrap().parse().unwrap();
    
    let mut lst = vec![-1; 5005];
    let mut to = vec![0; 10005];
    let mut pre = vec![-1; 10005];
    let mut tot = 0;
    let mut sz = vec![0; 5005];
    let mut dp_sz = vec![0; 5005];
    let mut dp = vec![vec![[0; 2]; 5005]; 5005];
    let mut pw = vec![1; 5005];
    let mut ans = 0;

    for i in 1..=n {
        pw[i] = (pw[i - 1] * (i * 2 - 1) % MOD) % MOD;
    }

    for _ in 1..n {
        let line = lines.next().unwrap().unwrap();
        let mut nums = line.split_whitespace().map(|x| x.parse::<usize>().unwrap());
        let u = nums.next().unwrap() - 1;
        let v = nums.next().unwrap() - 1;
        add_edge(u, v, &mut lst, &mut to, &mut pre, &mut tot);
        add_edge(v, u, &mut lst, &mut to, &mut pre, &mut tot);
    }

    init_dfs(0, -1, &lst, &to, &mut sz, &pre);
    dfs(0, -1, &lst, &to, &mut dp, &mut dp_sz, &sz, &pre);

    for i in 0..=n {
        ans = (ans + dp[0][i][0] * calc(n as i64 - i as i64)) % MOD;
        ans = (ans - dp[0][i][1] * calc(n as i64 - i as i64)) % MOD;
    }

    println!("{}", (ans + MOD) % MOD);
}
