```rust
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut lines = input.lines();

    let n: usize = lines.next().unwrap().parse().unwrap();
    let s: Vec<char> = lines.next().unwrap().chars().collect();
    let t: Vec<char> = lines.next().unwrap().chars().collect();

    let mut ans = 0;
    let mut i = n as isize - 1;

    while i >= 0 {
        let mut j = n as isize - 1; // Start j from the end of t
        while j >= 0 && s[i as usize] != t[j as usize] {
            j -= 1;
        }
        if j >= 0 {
            while j >= 0 && i >= 0 && s[i as usize] == t[j as usize] {
                i -= 1;
                j -= 1;
            }
            if j < 0 {
                ans = 2 * n as isize - (i + 1);
                break;
            }
        } else {
            ans = 2 * n as isize - (i + 1);
            break;
        }
        i -= 1; // Decrement i for the next iteration of the outer loop
    }

    println!("{}", ans);
}
```
