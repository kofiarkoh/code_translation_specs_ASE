use std::io::{self, BufRead, Write};
use std::io::Read;

fn get_uint() -> i32 {
    let stdin = io::stdin();
    let mut handle = stdin.lock();
    let mut n = 0;
    let mut c = handle.bytes().next().unwrap().unwrap() as char;

    if c < '0' || c > '9' {
        return c as i32;
    }

    while c >= '0' && c <= '9' {
        n = 10 * n + (c as i32 - '0' as i32);
        c = handle.bytes().next().unwrap().unwrap() as char;
    }
    n
}

fn put_uint(mut n: i32) {
    if n == 0 {
        print!("0");
        return;
    }
    let mut buf = Vec::new();
    while n > 0 {
        buf.push((n % 10 + '0' as i32) as u8 as char);
        n /= 10;
    }
    buf.reverse();
    print!("{}", buf.iter().collect::<String>());
}

#[derive(Default, Clone)]
struct Node {
    edge_to: [i32; 100],
    edge_cost: [i32; 100],
    edge_size: i32,
    cost: i32,
    visited: i32,
}

fn dijkstra_noheap(node: &mut [Node], n: usize) {
    for i in 0..n {
        node[i].cost = -1;
        node[i].visited = 0;
    }
    node[0].cost = 0;

    loop {
        let mut next: Option<&mut Node> = None;
        for it in node.iter_mut() {
            if it.visited != 0 || it.cost < 0 {
                continue;
            }
            if next.is_none() || it.cost < next.as_ref().unwrap().cost {
                next = Some(it);
            }
        }
        if next.is_none() {
            break;
        }
        let next_node = next.unwrap();
        next_node.visited = 1;
        let u = next_node.clone(); // Clone the node to avoid borrowing issues
        for i in 0..u.edge_size {
            let to = u.edge_to[i as usize];
            let cost = u.cost + u.edge_cost[i as usize];
            if node[to as usize].cost < 0 || cost < node[to as usize].cost {
                node[to as usize].cost = cost;
            }
        }
    }
}

fn main() {
    let n = get_uint() as usize;
    let mut node = vec![Node::default(); n];

    for _ in 0..n {
        let u = get_uint();
        let k = {
            let size = get_uint();
            node[u as usize].edge_size = size;
            size
        };
        for j in 0..k {
            node[u as usize].edge_to[j as usize] = get_uint();
            node[u as usize].edge_cost[j as usize] = get_uint();
        }
    }

    dijkstra_noheap(&mut node, n);
    for i in 0..n {
        put_uint(i as i32);
        print!(" ");
        put_uint(node[i].cost);
        println!();
    }
}
