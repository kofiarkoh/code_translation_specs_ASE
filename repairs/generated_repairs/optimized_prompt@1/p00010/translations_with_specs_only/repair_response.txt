use std::io::{self, BufRead};

fn main() {
    // Read the number of triangles from standard input
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read the first line to get the number of triangles
    let num_triangles: usize = lines.next().unwrap().unwrap().parse().unwrap();

    // Process each triangle
    for _ in 0..num_triangles {
        // Read the coordinates for the three vertices
        let line = lines.next().unwrap().unwrap();
        let coords: Vec<f64> = line.split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();

        // Ensure we have exactly six coordinates
        assert_eq!(coords.len(), 6);

        // Extract the points
        let (x1, y1) = (coords[0], coords[1]);
        let (x2, y2) = (coords[2], coords[3]);
        let (x3, y3) = (coords[4], coords[5]);

        // Calculate the circumcenter and circumradius
        let (px, py, r) = calculate_circumcenter_and_radius(x1, y1, x2, y2, x3, y3);

        // Output the results formatted to three decimal places
        println!("{:.3} {:.3} {:.3}", px, py, r);
    }
}

// Function to calculate the circumcenter and circumradius of a triangle
fn calculate_circumcenter_and_radius(x1: f64, y1: f64, x2: f64, y2: f64, x3: f64, y3: f64) -> (f64, f64, f64) {
    // Calculate the midpoints of the segments
    let mid_ab_x = (x1 + x2) / 2.0;
    let mid_ab_y = (y1 + y2) / 2.0;
    let mid_bc_x = (x2 + x3) / 2.0;
    let mid_bc_y = (y2 + y3) / 2.0;

    // Calculate the slopes of the segments
    let slope_ab = (y2 - y1) / (x2 - x1);
    let slope_bc = (y3 - y2) / (x3 - x2);

    // Check for vertical lines to avoid division by zero
    if slope_ab.is_infinite() || slope_bc.is_infinite() {
        return (f64::NAN, f64::NAN, f64::NAN);
    }

    // Calculate the perpendicular slopes
    let perp_slope_ab = -1.0 / slope_ab;
    let perp_slope_bc = -1.0 / slope_bc;

    // Calculate the circumcenter using the intersection of the two perpendicular bisectors
    let px = (perp_slope_ab * mid_ab_x - perp_slope_bc * mid_bc_x + mid_bc_y - mid_ab_y) /
             (perp_slope_ab - perp_slope_bc);
    let py = perp_slope_ab * (px - mid_ab_x) + mid_ab_y;

    // Calculate the circumradius using the distance from the circumcenter to one of the vertices
    let r = ((px - x1).powi(2) + (py - y1).powi(2)).sqrt();

    (px, py, r)
}
