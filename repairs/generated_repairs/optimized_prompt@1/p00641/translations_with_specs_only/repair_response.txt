// Define the maximum number of entities allowed
const MAX_ENTITIES: usize = 1000; // Adjust as necessary based on the specification

// Union-Find structure to manage relationships
struct UnionFind {
    parent: Vec<usize>,
    max_value: Vec<i32>,
    count: Vec<usize>,
}

impl UnionFind {
    // Create a new Union-Find structure
    fn new(size: usize) -> Self {
        let mut parent = Vec::with_capacity(size);
        let mut max_value = Vec::with_capacity(size);
        let mut count = Vec::with_capacity(size);
        
        for i in 0..size {
            parent.push(i);
            max_value.push(0); // Initialize max values to 0
            count.push(1); // Each entity is its own group initially
        }
        
        UnionFind { parent, max_value, count }
    }

    // Find the root of the set containing `x`
    fn find(&mut self, x: usize) -> usize {
        if self.parent[x] != x {
            self.parent[x] = self.find(self.parent[x]); // Path compression
        }
        self.parent[x]
    }

    // Union the sets containing `x` and `y`
    fn union(&mut self, x: usize, y: usize, value: i32) {
        let root_x = self.find(x);
        let root_y = self.find(y);
        
        if root_x != root_y {
            // Union by size
            if self.count[root_x] < self.count[root_y] {
                self.parent[root_x] = root_y;
                self.max_value[root_y] = self.max_value[root_y].max(value);
                self.count[root_y] += self.count[root_x];
            } else {
                self.parent[root_y] = root_x;
                self.max_value[root_x] = self.max_value[root_x].max(value);
                self.count[root_x] += self.count[root_y];
            }
        } else {
            // Update max value if they are already in the same group
            self.max_value[root_x] = self.max_value[root_x].max(value);
        }
    }

    // Calculate the final result based on the groups
    fn calculate_result(&self) -> i32 {
        let mut result = 1;
        let mut seen = std::collections::HashSet::new();

        for i in 0..self.parent.len() {
            let root = self.find(i);
            if seen.insert(root) { // Only process each root once
                result = (result * self.count[root] as i32) % 10007; // Product of counts modulo 10007
            }
        }

        result
    }
}
