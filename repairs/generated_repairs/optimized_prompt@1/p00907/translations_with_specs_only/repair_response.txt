fn main() {
    use std::io::{self, Write};

    // Read the number of values to process
    let mut input = String::new();
    io::stdout().flush().unwrap(); // Ensure prompt is displayed
    io::stdin().read_line(&mut input).expect("Failed to read line");
    
    // Parse the number of values
    let n: usize = match input.trim().parse() {
        Ok(num) if num > 0 && num <= 7 => num,
        _ => return, // Terminate if input is invalid or zero
    };

    let mut values = Vec::new();
    
    // Read the floating-point values
    for _ in 0..n {
        let mut value_input = String::new();
        io::stdin().read_line(&mut value_input).expect("Failed to read line");
        
        // Parse the floating-point number
        if let Ok(value) = value_input.trim().parse::<f64>() {
            values.push(value);
        } else {
            return; // Terminate if input is invalid
        }
    }

    // Identify the outlier
    let outlier_index = identify_outlier(&values);
    
    // Output the index of the outlier or the last processed value
    println!("{}", outlier_index);
}

// Function to identify the outlier based on the specified criterion
fn identify_outlier(values: &[f64]) -> usize {
    let n = values.len();
    
    // If there's only one value, it is trivially the outlier
    if n == 1 {
        return 0;
    }

    // Calculate the mean of the values
    let mean: f64 = values.iter().sum::<f64>() / n as f64;

    // Calculate the deviation threshold (using a simple criterion)
    let threshold = mean * 1.5; // Example criterion for outlier detection

    // Check for outliers
    for (index, &value) in values.iter().enumerate() {
        if (value - mean).abs() > threshold {
            return index; // Return the index of the outlier
        }
    }

    // If no outlier is found, return the index of the last processed value
    n - 1
}
