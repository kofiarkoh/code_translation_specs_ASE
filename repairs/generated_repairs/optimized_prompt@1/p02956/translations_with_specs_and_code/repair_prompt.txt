        ans = (ans + (MOD - pow[n - i - 1]) * (i - j as usize) % MOD) % MOD;

        for k in (j + 1)..=i {
            add_bit(&mut bit, m, data[k].id as usize, 1);
        }
        i = j + 1;
    }

    for i in 0..n {
        data[i].key ^= data[i].id;
        data[i].id ^= data[i].key;
        data[i].key ^= data[i].id;
    }

    merge_sort(&mut data);

    let mut i = 0;
    while i < n {
        let tmp = data[i].key;
        let mut j = i;
        while j < n && data[j].key == tmp {
            j += 1;
        }
        ans = (ans + (MOD - pow[i]) * (j - i) % MOD) % MOD;
        i = j - 1;
    }

    for i in (0..n).rev() {
        let tmp = data[i].key;
        let mut j = i;
        while j >= 0 && data[j].key == tmp {
            j = j.wrapping_sub(1);
        }
        ans = (ans + (MOD - pow[n - i - 1]) * (i - j as usize) % MOD) % MOD;
        i = j + 1;
    }

    println!("{}", ans);
}
