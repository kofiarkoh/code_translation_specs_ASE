fn main() {
    // Read input values for width and height
    let (width, height) = (4, 3); // Example input values

    // Create a grid of dimensions (height + 2) x (width * 2 + 4)
    let mut grid = vec![vec![0; width * 2 + 4]; height + 2];

    // Mark cells based on specific movement rules
    mark_cells(&mut grid, width, height);

    // Count the number of unmarked cells adjacent to marked cells
    let count = count_adjacent_unmarked(&grid, width, height);

    // Print the result
    println!("{}", count);
}

// Function to mark cells based on movement rules
fn mark_cells(grid: &mut Vec<Vec<i32>>, width: usize, height: usize) {
    // Example marking logic based on input
    let input = vec![
        vec![0, 1, 0, 0],
        vec![1, 0, 1, 0],
        vec![0, 0, 0, 1],
    ];

    for y in 0..height {
        for x in 0..width {
            grid[y + 1][x + 2] = input[y][x]; // Adjusting index for grid
        }
    }
}

// Function to count unmarked cells adjacent to marked cells
fn count_adjacent_unmarked(grid: &Vec<Vec<i32>>, width: usize, height: usize) -> i32 {
    let mut count = 0;

    // Directions for adjacent cells (up, down, left, right)
    let directions = [(-1, 0), (1, 0), (0, -1), (0, 1)];

    for y in 1..=height {
        for x in 0..(width * 2 + 4) {
            if grid[y][x] == 0 { // Check if the cell is unmarked
                // Check adjacent cells
                for &(dy, dx) in &directions {
                    let adj_y = (y as isize + dy) as usize;
                    let adj_x = (x as isize + dx) as usize;

                    // Ensure we stay within grid boundaries
                    if adj_y < grid.len() && adj_x < grid[0].len() {
                        if grid[adj_y][adj_x] == 1 { // Check if adjacent cell is marked
                            count += 1;
                            break; // No need to check further adjacent cells
                        }
                    }
                }
            }
        }
    }

    count
}
