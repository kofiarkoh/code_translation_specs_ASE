```rust
// AOJ 3130 Calculation Training

use std::io::{self, Write};

fn main() {
    // Read input values
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let mut iter = input.split_whitespace();

    // Parse inputs
    let n: i64 = iter.next().expect("Missing N").parse().expect("Invalid N");
    let mut a: i32 = iter.next().expect("Missing A").parse().expect("Invalid A");
    let mut b: i32 = iter.next().expect("Missing B").parse().expect("Invalid B");

    // Preconditions check
    assert!(n >= 0, "N must be a non-negative long integer");

    // Calculate the number of iterations based on N modulo 12
    let n = (n % 12) as usize;

    // Perform the iterative arithmetic operations
    for _ in 0..n {
        a -= b; // A = A - B
        b += a; // B = A + B
    }

    // Output the final values of A and B
    println!("{} {}", a, b);
}
```
