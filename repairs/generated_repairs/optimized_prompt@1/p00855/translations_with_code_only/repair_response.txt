fn p(mut n: i32) -> i32 {
    if n < 2 {
        return 0; // Handle cases where n is less than 2
    }
    let mut t = 0; // Count of non-divisors
    let mut i = 2; // Start i at 2
    while i * i <= n {
        if n % i != 0 {
            t += 1; // Increment t if n is not divisible by i
        }
        i += 1; // Increment i
    }
    t + (if n > 1 { 1 } else { 0 }) // Include n itself if n > 1
}

fn main() {
    use std::io::{self, BufRead};

    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut l: i32;
    let mut u: i32;

    for line in handle.lines() {
        let line = line.unwrap();
        l = line.trim().parse().unwrap(); // Read l from input
        u = l; // Initialize u to l

        while p(l) > 0 {
            l -= 1; // Decrement l while p(l) is greater than 0
        }
        while p(u) > 0 {
            u += 1; // Increment u while p(u) is greater than 0
        }
        println!("{}", u - l); // Print the difference
    }
}
