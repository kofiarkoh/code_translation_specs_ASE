use std::collections::BinaryHeap;
use std::io::{self, BufRead};

const INF: usize = 0x10101010;

#[derive(Debug)]
struct Edge {
    to: usize,
    weight: usize,
}

fn dijkstra(start: usize, graph: &[Vec<Edge>], dist: &mut [usize]) {
    let mut heap = BinaryHeap::new();
    dist[start] = 0;
    heap.push((0, start)); // (distance, node)

    while let Some((d, u)) = heap.pop() {
        let d = d.wrapping_neg() as usize; // Convert back to positive distance
        if d > dist[u] {
            continue;
        }
        for edge in &graph[u] {
            let new_dist = dist[u].wrapping_add(edge.weight);
            if new_dist < dist[edge.to] {
                dist[edge.to] = new_dist;
                heap.push((new_dist.wrapping_neg() as isize, edge.to));
            }
        }
    }
}

fn topological_sort(start: usize, goal: usize, graph: &[Vec<Edge>], dist: &[usize]) -> Vec<usize> {
    let mut mk = vec![false; graph.len()];
    let mut q = vec![goal];
    let mut ord = Vec::new();
    let mut hi2 = vec![Vec::new(); graph.len()];

    while let Some(s) = q.pop() {
        if mk[s] {
            continue;
        }
        for edge in &graph[s] {
            if dist[s] == dist[edge.to].wrapping_add(edge.weight) {
                q.push(edge.to);
                hi2[edge.to].push(s);
            }
        }
        mk[s] = true;
    }

    let mut deg = vec![0; graph.len()];
    for s in 0..graph.len() {
        for &e in &hi2[s] {
            deg[e] += 1;
        }
    }

    q.push(start);
    while let Some(s) = q.pop() {
        ord.push(s);
        for &e in &hi2[s] {
            if deg[e] > 0 {
                deg[e] -= 1;
                if deg[e] == 0 {
                    q.push(e);
                }
            }
        }
    }

    ord
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines().map(|line| line.unwrap());

    let s: usize = lines.next().unwrap().parse().unwrap();
    let r: usize = lines.next().unwrap().parse().unwrap();

    let mut graph = vec![Vec::new(); s];
    let mut edges = Vec::new();

    for _ in 0..r {
        let line: Vec<usize> = lines.next().unwrap().split_whitespace().map(|x| x.parse().unwrap()).collect();
        let u = line[0] - 1;
        let v = line[1] - 1;
        let weight = line[2];

        graph[u].push(Edge { to: v, weight });
        graph[v].push(Edge { to: u, weight });
    }

    let a: usize = lines.next().unwrap().parse().unwrap() - 1;
    let b: usize = lines.next().unwrap().parse().unwrap() - 1;
    let q: usize = lines.next().unwrap().parse().unwrap();

    let mut queries = Vec::new();
    for _ in 0..q {
        let line: Vec<usize> = lines.next().unwrap().split_whitespace().map(|x| x.parse().unwrap()).collect();
        queries.push((line[0] - 1, line[1] - 1));
    }

    let mut dist = vec![INF; s];
    dijkstra(a, &graph, &mut dist);
    let ord = topological_sort(a, b, &graph, &dist);

    let mut dp = vec![0u64; s];
    let mut results = Vec::new();

    for i in 0..((q + 63) / 64) {
        dp.fill(0);
        for j in 0..64 {
            if (i * 64 + j) < q {
                dp[queries[i * 64 + j].0] |= 1 << j;
            }
        }
        for &u in &ord {
            for edge in &graph[u] {
                dp[edge.to] |= dp[u];
            }
        }
        for j in 0..64 {
            if (i * 64 + j) < q {
                results.push(if dp[queries[i * 64 + j].1] & (1 << j) != 0 { "Yes" } else { "No" });
            }
        }
    }

    for result in results {
        println!("{}", result);
    }
}
