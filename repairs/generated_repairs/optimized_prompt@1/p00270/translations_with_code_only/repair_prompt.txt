        to[u][k] = v;
        w[u][k] = _w;
        let k = hi[v] - 1;
        to[v][k] = u;
        w[v][k] = _w;
    }

    let a = in_int() - 1;
    let b = in_int() - 1;
    let Q = in_int();
    for i in 0..Q {
        c[i] = in_int() - 1;
        d[i] = in_int() - 1;
    }

    let mut heap = MinHeap::new();
    dijkstra(a, &hi, &to, &w, &mut dist, &mut heap);
    tsort(a, b, &hi, &to, &w, &dist, &mut ord, &mut deg, &mut hi2, &mut to2, &mut mk);

    let ii = (Q + 63) >> 6;
    for i in 0..ii {
        let mut dp = vec![0u64; S];
        for j in 0..64 {
            if (i << 6) + j < Q {
                dp[c[(i << 6) + j]] |= 1 << j;
            }
        }
        for &u in &ord {
            for &k in &to2[u] {
                dp[k] |= dp[u];
            }
        }
        for j in 0..64 {
            if (i << 6) + j < Q {
                println!("{}", if dp[d[(i << 6) + j]] & (1 << j) != 0 { "Yes" } else { "No" });
            }
        }
    }
}
