```rust
use std::io::{self, Write};

fn main() {
    // Input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let mut nums = input.split_whitespace();
    
    // Ensure we have two numbers
    let x: i32 = match nums.next() {
        Some(num) => num.parse().expect("Invalid input for x"),
        None => {
            eprintln!("Expected input for x");
            return;
        }
    };
    
    let y: i32 = match nums.next() {
        Some(num) => num.parse().expect("Invalid input for y"),
        None => {
            eprintln!("Expected input for y");
            return;
        }
    };

    // Solve
    let ans: i64 = if x >= 0 {
        if y >= x {
            -x as i64 + y as i64
        } else if y > 0 {
            x as i64 - y as i64 + 2
        } else if y >= -x {
            x as i64 + y as i64 + 1
        } else {
            -y as i64 - x as i64 + 1
        }
    } else {
        if y >= -x {
            x as i64 + y as i64 + 1
        } else if y > 0 {
            -x as i64 - y as i64 + 1
        } else if y >= x {
            -x as i64 + y as i64
        } else {
            x as i64 - y as i64 + 2
        }
    };

    // Output
    print!("{}", ans);
    io::stdout().flush().expect("Failed to flush stdout");
}
```
