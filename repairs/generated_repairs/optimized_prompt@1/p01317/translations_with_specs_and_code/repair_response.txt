```rust
use std::io::{self, BufRead};
use std::cmp::min;

const MN: usize = 205;
const INF: i32 = 100_000_000;

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    while let Some(line) = lines.next() {
        let line = line.unwrap();
        let mut nums = line.split_whitespace().map(|s| s.parse::<usize>());
        let n = nums.next().unwrap().unwrap();
        let m = nums.next().unwrap().unwrap();

        if n == 0 {
            break;
        }

        let mut s = vec![vec![INF; MN]; MN];
        let mut l = vec![vec![INF; MN]; MN];
        let mut dp = vec![vec![INF; 205]; 1005];
        let mut de = vec![0; 1005];

        for i in 0..n {
            l[i][i] = 0;
            s[i][i] = 0;
        }

        for _ in 0..m {
            let edge_line = lines.next().unwrap().unwrap();
            let mut edge_parts = edge_line.split_whitespace();
            let x: usize = edge_parts.next().unwrap().parse::<usize>().unwrap() - 1;
            let y: usize = edge_parts.next().unwrap().parse::<usize>().unwrap() - 1;
            let cost: i32 = edge_parts.next().unwrap().parse::<i32>().unwrap();
            let edge_type: char = edge_parts.next().unwrap().chars().next().unwrap();

            if edge_type == 'L' {
                l[x][y] = min(l[x][y], cost);
                l[y][x] = min(l[y][x], cost);
            } else {
                s[x][y] = min(s[x][y], cost);
                s[y][x] = min(s[y][x], cost);
            }
        }

        let r: usize = lines.next().unwrap().unwrap().parse::<usize>().unwrap();
        for i in 0..r {
            de[i] = lines.next().unwrap().unwrap().parse::<usize>().unwrap() - 1;
        }

        for k in 0..n {
            for i in 0..n {
                for j in 0..n {
                    s[i][j] = min(s[i][j], s[i][k] + s[k][j]);
                    l[i][j] = min(l[i][j], l[i][k] + l[k][j]);
                }
            }
        }

        dp[0][de[0]] = 0;
        for k in 0..r - 1 {
            for i in 0..n {
                for j in 0..n {
                    dp[k + 1][i] = min(dp[k + 1][i], dp[k][j] + l[de[k]][j] + s[j][i] + l[i][de[k + 1]]);
                    if i == j {
                        dp[k + 1][i] = min(dp[k + 1][i], dp[k][j] + l[de[k]][de[k + 1]]);
                    }
                }
            }
        }

        let mut min_cost = INF;
        for i in 0..n {
            min_cost = min(min_cost, dp[r - 1][i]);
        }
        println!("{}", if min_cost == INF { -1 } else { min_cost });
    }
}
```
