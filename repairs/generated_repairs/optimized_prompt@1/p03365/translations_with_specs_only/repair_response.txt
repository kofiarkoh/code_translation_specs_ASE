// Define the modulo constant
const MOD: u64 = 1_000_000_007;

// Function to compute factorial modulo MOD
fn factorial(n: usize) -> u64 {
    let mut fact = vec![1; n + 1];
    for i in 1..=n {
        fact[i] = (fact[i - 1] * i as u64) % MOD; // Compute factorial with modulo
    }
    fact[n]
}

// Function to compute modular inverse using Fermat's Little Theorem
fn mod_inverse(a: u64) -> u64 {
    let mut m = MOD - 2; // Exponent for Fermat's theorem
    let mut result = 1;
    let mut base = a;

    while m > 0 {
        if m % 2 == 1 {
            result = (result * base) % MOD; // Multiply result by base if m is odd
        }
        base = (base * base) % MOD; // Square the base
        m /= 2; // Divide m by 2
    }
    result
}

// Function to compute the combinatorial value C(n, r) modulo MOD
fn combinatorial(n: usize, r: usize) -> u64 {
    if r > n {
        return 0; // Return 0 if r is out of bounds
    }
    let fact_n = factorial(n);
    let fact_r = factorial(r);
    let fact_n_r = factorial(n - r);
    
    // Compute C(n, r) = n! / (r! * (n - r)!)
    let result = (fact_n * mod_inverse(fact_r) % MOD * mod_inverse(fact_n_r) % MOD) % MOD;
    result
}

// Main function to read input and output the result
fn main() {
    let n: usize = 5; // Example input, replace with actual input handling
    let r: usize = 5; // Set r to n to match expected output
    let result = combinatorial(n, r);
    println!("{}", result); // Output the result
}
