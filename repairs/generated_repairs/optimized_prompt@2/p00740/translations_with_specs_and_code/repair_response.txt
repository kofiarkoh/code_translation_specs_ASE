```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    
    for line in reader.lines() {
        let line = line.unwrap();
        let mut nums = line.split_whitespace()
            .filter_map(|s| s.parse::<i32>().ok());
        
        let n = match nums.next() {
            Some(num) if num > 0 => num,
            _ => break, // Terminate if the first integer is zero or invalid
        };
        
        let mut p = match nums.next() {
            Some(num) => num,
            _ => continue, // Skip if the second integer is missing
        };

        let mut i = 0;
        let mut w = p;
        let mut c = vec![0; n as usize]; // Array to hold counts of eliminations

        loop {
            if w > 0 {
                w -= 1;
                c[i as usize] += 1;

                if w == 0 {
                    let mut found = false;
                    for j in 0..n {
                        if j != i && c[j as usize] > 0 {
                            found = true;
                            break;
                        }
                    }
                    if !found {
                        println!("{}", i);
                        break;
                    }
                }
            } else {
                w += c[i as usize];
                c[i as usize] = 0;
            }
            i += 1;
            if i >= n { // Prevent overflow
                i = 0;
            }
        }
    }
}
```
