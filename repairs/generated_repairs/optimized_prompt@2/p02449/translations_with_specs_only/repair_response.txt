    use std::io::{self, Write};

    // Read the number of elements in the sequence
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");

    if n < 1 || n > 9 {
        return; // No output if n is out of bounds
    }

    let mut sequence = Vec::new();
    input.clear();
    io::stdin().read_line(&mut input).expect("Failed to read line");

// Function to compute the previous permutation
fn previous_permutation(seq: &[i32]) -> Option<Vec<i32>> {
    let mut perm = seq.to_vec();
    let len = perm.len();

    // Find the largest index k such that perm[k] > perm[k + 1]
    let mut k = len as isize - 2;
    while k >= 0 && perm[k as usize] <= perm[(k + 1) as usize] {
        k -= 1;
    }

    // If no such index exists, this is the last permutation
    if k < 0 {
        return None;
    }

    // Find the largest index l such that perm[k] > perm[l]
    let mut l = len as isize - 1;
    while perm[k as usize] <= perm[l as usize] {
        l -= 1;
    }

    // Swap perm[k] and perm[l]
    perm.swap(k as usize, l as usize);

    // Reverse the sequence from perm[k + 1] to the end
    perm[(k + 1) as usize..].reverse();

    Some(perm)
}

// Function to compute the next permutation
fn next_permutation(seq: &[i32]) -> Option<Vec<i32>> {
    let mut perm = seq.to_vec();
    let len = perm.len();

    // Find the largest index k such that perm[k] < perm[k + 1]
    let mut k = len as isize - 2;
    while k >= 0 && perm[k as usize] >= perm[(k + 1) as usize] {
        k -= 1;
    }

    // If no such index exists, this is the last permutation
    if k < 0 {
        return None;
    }

    // Find the largest index l such that perm[k] < perm[l]
    let mut l = len as isize - 1;
    while perm[k as usize] >= perm[l as usize] {
        l -= 1;
    }

    perm.swap(k as usize, l as usize);

    perm[(k + 1) as usize..].reverse();

    Some(perm)
}

// Function to print a vector in the required format
fn print_vector(vec: &[i32]) {
    let output: String = vec.iter().map(|x| x.to_string()).collect::<Vec<String>>().join(" ");
    println!("{}", output);
