```rust
use std::io::{self, Read, Write};

fn main() {
    let mut buf = [0u8; 10000];
    let mut out = Vec::with_capacity(20000);
    let mut r = [0u8; 10];
    let mut sz = 0;
    let mut outsz = 0;

    // Read input from stdin
    let stdin = io::stdin();
    let mut handle = stdin.lock();
    handle.read(&mut buf).unwrap();

    // Skip leading non-digit characters
    while sz < buf.len() && buf[sz] < b'0' {
        sz += 1;
    }

    while sz < buf.len() && buf[sz] >= b'0' {
        let mut N = 0;
        let mut D = 0;

        // Read N
        loop {
            let k = buf[sz];
            sz += 1;
            if k < b'0' {
                break;
            }
            N = N * 10 + (k - b'0') as i32;
        }

        // Read D
        loop {
            let k = buf[sz];
            sz += 1;
            if k < b'0' {
                break;
            }
            D = D * 10 + (k - b'0') as i32;
        }

        // Calculate ans based on the parity of N
        let ans = if N & 1 != 0 {
            127 * (N - 1) + D
        } else {
            127 * N - D
        };

        // Convert ans to string and store in r
        let mut ri = 10;
        let mut temp_ans = ans;
        if temp_ans == 0 {
            ri -= 1;
            r[ri] = b'0';
        } else {
            while temp_ans > 0 {
                ri -= 1;
                r[ri] = (temp_ans % 10) as u8 + b'0';
                temp_ans /= 10;
            }
        }

        // Append the result to out
        while ri < 10 {
            out.push(r[ri]);
            ri += 1;
        }
        out.push(b'\n');
        outsz += 1;
    }

    // Write output to stdout
    io::stdout().write_all(&out).unwrap();
}
```
