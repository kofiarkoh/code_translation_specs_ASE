```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut handle = stdin.lock(); // Changed to mutable

    // Read the length of the strings
    let mut input = String::new();
    handle.read_line(&mut input).expect("Failed to read line");
    let n: usize = input.trim().parse().expect("Input must be a positive integer");

    // Read the two strings
    let mut strings = String::new();
    handle.read_line(&mut strings).expect("Failed to read line");
    let parts: Vec<&str> = strings.split_whitespace().collect();
    let s1 = parts[0];
    let s2 = parts[1];

    // Ensure the strings are of the correct length
    assert_eq!(s1.len(), n);
    assert_eq!(s2.len(), n);

    let mut ans: i64 = 1;
    let mut t: i32;

    // Initial comparison of the first characters
    if s1.chars().nth(0) == s2.chars().nth(0) {
        ans = 3;
        t = 1;
    } else {
        ans = 6;
        t = 0;
    }

    let m: i64 = 1000000007;

    // Iterate through the strings starting from the second character
    for i in 1..n {
        if s1.chars().nth(i) == s2.chars().nth(i) {
            if t == 1 {
                ans = (ans * 2) % m;
            }
            t = 1;
        }
        if i < n - 1 && s1.chars().nth(i) == s1.chars().nth(i + 1) {
            if t == 1 {
                ans = (ans * 2) % m;
            } else {
                ans = (ans * 3) % m;
            }
            t = 0;
        }
    }

    // Print the result
    println!("{}", ans);
}
```
