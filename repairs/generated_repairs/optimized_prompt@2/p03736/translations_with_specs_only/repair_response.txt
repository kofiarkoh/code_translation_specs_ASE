// Define a function to compute the minimum cost based on the given specification.
fn compute_minimum_cost(n: usize, q: usize, i: usize, j: usize, queries: Vec<usize>) -> i64 {
    // Preconditions check
    if n == 0 {
        panic!("n must be a positive integer.");
    }
    if q == 0 {
        panic!("q must be a non-negative integer.");
    }
    if i >= n || j >= n {
        panic!("Initial indices i and j must be within the bounds of the array (0 to n-1).");
    }
    
    // Initialize the minimum values to a predefined maximum value
    let mut min_values = vec![i64::MAX; n];
    min_values[i] = 0; // Starting point for index i
    min_values[j] = 0; // Starting point for index j

    // Initialize the current index and total cost
    let mut current_index = i;
    let mut total_cost = 0;

    // Process each query
    for &query in &queries {
        // Check if the query index is within bounds
        if query >= n {
            panic!("Query index must be within the bounds of the array (0 to n-1).");
        }

        // Calculate the distance from the current index to the query index
        let distance = (current_index as isize - query as isize).abs() as i64;
        total_cost += distance;

        // Update the minimum value for the queried index
        min_values[query] = min_values[query].min(total_cost);

        // Update the current index to the queried index
        current_index = query;
    }

    // Return the minimum cost calculated after processing all queries
    *min_values.iter().min().unwrap_or(&0)
}

// Main function to execute the program
fn main() {
    let n = 5;
    let q = 3;
    let i = 0;
    let j = 1;
    let queries = vec![2, 3, 4];
    let result = compute_minimum_cost(n, q, i, j, queries);
    println!("{}", result);
}
