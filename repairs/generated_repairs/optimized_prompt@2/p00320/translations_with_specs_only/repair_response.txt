use std::io::{self, BufRead};
use std::collections::HashSet;

fn main() {
    // Read input from standard input
    let stdin = io::stdin();
    let input = stdin.lock().lines().collect::<Result<Vec<_>, _>>().unwrap();

    // Check if we have exactly six pairs of integers
    if input.len() != 6 {
        eprintln!("Error: Expected exactly six pairs of integers.");
        std::process::exit(1);
    }

    let mut dimensions = Vec::new();

    // Parse the input into pairs of integers
    for line in input {
        let nums: Vec<i32> = line.split_whitespace()
            .filter_map(|s| s.parse().ok())
            .collect();
        if nums.len() != 2 || nums.iter().any(|&n| n < 0) {
            eprintln!("Error: Each line must contain exactly two non-negative integers.");
            std::process::exit(1);
        }
        dimensions.push((nums[0], nums[1]));
    }

    // Check for valid configurations
    if can_form_equal_lengths(&dimensions) {
        println!("yes");
    } else {
        println!("no");
    }
}

// Function to check if a valid configuration exists
fn can_form_equal_lengths(dimensions: &[(i32, i32)]) -> bool {
    // Generate all permutations of the dimensions
    let mut seen = HashSet::new();
    for perm in permutations(dimensions) {
        // Check if any orientation of the permutation can form equal lengths
        if check_equal_lengths(&perm) {
            return true;
        }
    }
    false
}

// Function to generate permutations of the given dimensions
fn permutations<T: Clone>(items: &[(T, T)]) -> Vec<Vec<(T, T)>> {
    let mut result = Vec::new();
    let n = items.len();
    let mut indices: Vec<usize> = (0..n).collect();
    let mut stack = vec![0];

    while let Some(i) = stack.pop() {
        if i == n {
            result.push(indices.iter().map(|&j| items[j].clone()).collect());
        } else {
            for j in i..n {
                indices.swap(i, j);
                stack.push(i + 1);
                indices.swap(i, j); // backtrack
            }
        }
    }
    result
}

// Function to check if any orientation of the dimensions can form equal lengths
fn check_equal_lengths(perm: &[(i32, i32)]) -> bool {
    // Check all combinations of orientations
    let orientations = [
        (0, 1), // Original orientation
        (1, 0), // Rotated orientation
    ];

    for &o1 in &orientations {
        for &o2 in &orientations {
            let length1 = perm[0].0 + perm[1].1; // Corrected indexing
            let length2 = perm[2].0 + perm[3].1; // Corrected indexing
            let length3 = perm[4].0 + perm[5].1; // Corrected indexing
            if length1 == length2 && length2 == length3 {
                return true;
            }
        }
    }
    false
}
