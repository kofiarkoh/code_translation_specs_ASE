```rust
use std::io::{self, BufRead};

fn min(p: i32, q: i32) -> i32 {
    if p < q { p } else { q }
}

fn f(p: &[i32], n: usize, t: &mut [i32], x: &[i32]) {
    for i in 0..n {
        let mut cnt = 0;
        for j in 0..n {
            if p[j] > p[i] {
                cnt += 1;
            }
        }
        t[i] = x[cnt];
    }
}

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();
    
    let n: usize = lines.next().unwrap().unwrap().parse().expect("Failed to parse n");
    let mut x = vec![0; n];
    for i in 0..n {
        x[i] = lines.next().unwrap().unwrap().parse().expect("Failed to parse x[i]");
    }
    
    let mut a = vec![vec![0; n]; 3]; // Changed from 110 to n
    for i in 0..n {
        let nums: Vec<i32> = lines.next().unwrap().unwrap().split_whitespace()
            .map(|s| s.parse().expect("Failed to parse nums")).collect();
        a[0][i] = nums[0];
        a[1][i] = nums[1];
        a[2][i] = nums[2];
    }
    
    let mut ans = i32::MAX; // Changed from 1000 to i32::MAX
    for k in 0..3 {
        let mut s = vec![0; n];
        for kk in 0..3 {
            if kk != k {
                let mut t = vec![0; n];
                f(&a[kk], n, &mut t, &x);
                for i in 0..n {
                    s[i] += t[i];
                }
            }
        }

        let ori = a[k][0];
        for i in 1..n {
            for d in 0..2 {
                if a[k][i] + d > ori {
                    a[k][0] = a[k][i] + d;
                    let mut t = vec![0; n];
                    f(&a[k], n, &mut t, &x);
                    let mut cnt = 0;
                    for j in 0..n {
                        if s[j] + t[j] > s[0] + t[0] {
                            cnt += 1;
                        }
                    }
                    if cnt <= 7 {
                        ans = min(ans, a[k][0] - ori);
                    }
                }
            }
        }
        a[k][0] = ori;
    }
    
    if ans == i32::MAX {
        println!("Saiko");
    } else {
        println!("{}", ans);
    }
}
```
