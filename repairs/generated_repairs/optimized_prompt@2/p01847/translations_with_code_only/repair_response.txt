use std::io::{self, BufRead};

fn in_value() -> i32 {
    let mut n = 0;
    let mut c = std::io::stdin().bytes().next().unwrap().unwrap() as char;

    if c == '-' {
        c = std::io::stdin().bytes().next().unwrap().unwrap() as char;
        while c.is_digit(10) {
            n = 10 * n + (c as u8 - b'0') as i32;
            c = std::io::stdin().bytes().next().unwrap().unwrap() as char;
        }
        return -n;
    }

    while c.is_digit(10) {
        n = 10 * n + (c as u8 - b'0') as i32;
        c = std::io::stdin().bytes().next().unwrap().unwrap() as char;
    }
    n
}

const INF: i32 = 20002;

fn main() {
    let stdin = io::stdin();
    let mut input = stdin.lock().lines();

    let mut x = vec![0; 102];
    let mut y = vec![0; 102];

    while let Ok(line) = input.next() {
        if let Ok(n) = line.unwrap().parse::<i32>() {
            for i in 0..n {
                x[i as usize] = in_value();
                y[i as usize] = in_value();
            }
            x[n as usize] = x[0];
            y[n as usize] = y[0];

            let mut xmin = INF;
            let mut xmax = -INF;
            let mut ymin = INF;
            let mut ymax = -INF;

            for _ in 0..4 {
                let a = in_value();
                let b = in_value();
                if a < xmin { xmin = a; }
                if a > xmax { xmax = a; }
                if b < ymin { ymin = b; }
                if b > ymax { ymax = b; }
            }

            let mut win: i64 = 0;
            let mut cur: i64 = 0;

            for i in 0..n {
                if x[i as usize] == x[(i + 1) as usize] {
                    cur += (x[i as usize] as i64) * (y[i as usize] - y[(i + 1) as usize]) as i64;
                    let a = x[i as usize].min(xmax).max(xmin);
                    let b = y[i as usize].min(ymax).max(ymin);
                    let c = y[(i + 1) as usize].min(ymax).max(ymin);
                    win += (a as i64) * (b - c) as i64;
                }
            }
            println!("{}", win - cur);
        }
    }
}
