    fn min_heapify(&mut self, i: usize) {
        let l = self.left(i);
        let r = self.right(i);
        let mut min = i;

        if l < self.qsize && self.que[l].t < self.que[min].t {
            min = l;

        }
    }

    fn deq(&mut self) {
        if self.qsize > 0 {
            self.que[0] = self.que[self.qsize - 1].clone();
            self.qsize -= 1;
            self.min_heapify(0);
        }
    }

    fn enq(&mut self, s: i16, t: i32, c: i16) {
        let i = self.qsize;
        self.que.push(QueueEntry { s, t, c });

            } else {

            }
        }
    }
