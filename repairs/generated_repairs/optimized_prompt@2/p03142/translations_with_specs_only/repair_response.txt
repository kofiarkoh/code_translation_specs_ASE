```rust
use std::collections::{HashMap, HashSet};
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read number of nodes and edges
    let first_line = lines.next().expect("Input format invalid").unwrap();
    let mut parts = first_line.split_whitespace();
    let n: usize = parts.next().expect("Input format invalid").parse().expect("Input format invalid");
    let m: usize = parts.next().expect("Input format invalid").parse().expect("Input format invalid");

    // Initialize parent and incoming edge count
    let mut parent: Vec<usize> = vec![0; n + 1]; // 1-indexed
    let mut incoming_edges: Vec<usize> = vec![0; n + 1]; // 1-indexed
    let mut edges: HashMap<usize, HashSet<usize>> = HashMap::new();

    // Read edges
    for _ in 0..m {
        let edge_line = lines.next().expect("Input format invalid").unwrap();
        let mut edge_parts = edge_line.split_whitespace();
        let from: usize = edge_parts.next().expect("Input format invalid").parse().expect("Input format invalid");
        let to: usize = edge_parts.next().expect("Input format invalid").parse().expect("Input format invalid");

        // Add edge
        edges.entry(from).or_insert_with(HashSet::new).insert(to);
        incoming_edges[to] += 1;
        parent[to] = from; // Set parent for the child node
    }

    // Output parent nodes in order from 1 to N
    for i in 1..=n {
        if incoming_edges[i] == 0 {
            println!("0"); // No parent
        } else {
            // Find the correct parent for nodes with incoming edges
            println!("{}", parent[i]);
        }
    }
}
```
