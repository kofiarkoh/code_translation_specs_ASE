```rust
use std::collections::VecDeque;
use std::fmt;

#[derive(Debug)]
struct Edge {
    source: usize,
    destination: usize,
    cost: i32,
}

#[derive(Debug)]
struct Graph {
    nodes: usize,
    edges: Vec<Edge>,
}

impl Graph {
    fn new(nodes: usize, edges: Vec<Edge>) -> Self {
        Graph { nodes, edges }
    }

    fn min_cost(&self, start: usize, target: usize) -> String {
        // Initialize distances with "infinity" (using i32::MAX for simplicity)
        let mut dist = vec![i32::MAX; self.nodes];
        dist[start] = 0;

        // Relax edges up to (nodes - 1) times
        for _ in 0..self.nodes - 1 {
            for edge in &self.edges {
                if dist[edge.source] != i32::MAX {
                    let new_cost = dist[edge.source] + edge.cost;
                    if new_cost < dist[edge.destination] {
                        dist[edge.destination] = new_cost;
                    }
                }
            }
        }

        // Check for negative cycles affecting the target node
        for _ in 0..self.nodes - 1 {
            for edge in &self.edges {
                if dist[edge.source] != i32::MAX {
                    let new_cost = dist[edge.source] + edge.cost;
                    if new_cost < dist[edge.destination] {
                        dist[edge.destination] = new_cost;
                    }
                }
            }
        }

        // If the target node's distance is still "infinity", return "inf"
        if dist[target] == i32::MAX {
            return "inf".to_string();
        }

        // Return the minimum cost
        dist[target].to_string()
    }
}

fn main() {
    // Example usage
    let nodes = 5;
    let edges = vec![
        Edge { source: 0, destination: 1, cost: 2 },
        Edge { source: 1, destination: 2, cost: 3 },
        Edge { source: 1, destination: 3, cost: 6 },
        Edge { source: 3, destination: 4, cost: 1 },
        Edge { source: 2, destination: 4, cost: 5 },
        Edge { source: 2, destination: 0, cost: -6 }, // Negative cycle
        Edge { source: 1, destination: 4, cost: 10 },
        Edge { source: 3, destination: 2, cost: -1 },
    ];

    let graph = Graph::new(nodes, edges);
    let result = graph.min_cost(0, 4);
    println!("{}", result); // Output the result
}
```
