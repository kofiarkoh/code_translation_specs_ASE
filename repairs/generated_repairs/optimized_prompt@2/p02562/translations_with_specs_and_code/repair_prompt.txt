            }
        }

        let mut t = 0;
        for t_candidate in 1..=n {
            if dist[t_candidate] != SUP && def[t_candidate] > 0 {
                t = t_candidate;
                break;
            }
        }
        if t == 0 {
            return SUP;
        }

        let mut min_flow = (-def[s]).min(def[t]);
        let mut u = t;
        while let Some(prev_edge) = prev[u] {
            if min_flow > prev_edge.cap - prev_edge.flow {
                min_flow = prev_edge.cap - prev_edge.flow;
            }
            u = prev_edge.tail;
        }

        u = t;
        while let Some(prev_edge) = prev[u] {
            let i = prev_edge.id;
            if e[i ^ 1].flow > 0 {
                e[i ^ 1].flow -= min_flow;
                e[i].cap -= min_flow;
            } else {
                e[i].flow += min_flow;
                e[i ^ 1].cap += min_flow;
            }
            ans += e[i].cost as i64 * min_flow as i64;
            u = prev_edge.tail;
        }

        for i in 1..=n {
            if dist[i] != SUP {
                pi[i] += dist[i];
            }
        }

        def[s] += min_flow;
        def[t] -= min_flow;
        if def[s] < 0 {
            h_def.push(Data { key: def[s] as i64, id: s });
        }
    }

    ans
}
