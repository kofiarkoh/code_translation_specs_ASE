fn max_k(r: i64, b: i64) -> i64 {
    // Preconditions: r and b must be non-negative integers.
    // If either r or b is negative, we return -1 as per the postconditions.
    if r < 0 || b < 0 {
        return -1;
    }

    // We need to find the maximum k such that the condition is satisfied.
    // The condition is that k must be less than or equal to the minimum of r and b.
    let max_possible_k = (r + b) / 2;

    // Return the maximum k found.
    max_possible_k
}

fn main() {
    // Example usage of max_k function
    let r = 100000;
    let b = 200000;
    let result = max_k(r, b);
    println!("{}", result);
}
