```rust
use std::io::{self, BufRead};

const MAX_STRINGS: usize = 1024;
const MAX_LENGTH: usize = 80;

struct StringContainer {
    strings: [[u8; MAX_LENGTH]; MAX_STRINGS],
    count: usize,
}

impl StringContainer {
    fn new() -> Self {
        StringContainer {
            strings: [[0; MAX_LENGTH]; MAX_STRINGS],
            count: 0,
        }
    }

    fn contains(&self, s: &[u8]) -> bool {
        for i in 0..self.count {
            if self.strings[i] == s {
                return true;
            }
        }
        false
    }

    fn add(&mut self, s: &[u8]) {
        if self.count < MAX_STRINGS {
            self.strings[self.count].copy_from_slice(s);
            self.count += 1;
        }
    }
}

fn reverse(s: &mut [u8], n: usize) {
    for i in 0..n / 2 {
        s.swap(i, n - 1 - i);
    }
}

fn swap(s1: &mut [u8], s2: &[u8]) {
    s1.copy_from_slice(s2);
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let n: usize = lines.next().unwrap().unwrap().parse().unwrap();
    
    for _ in 0..n {
        let mut container = StringContainer::new();
        let mut s = lines.next().unwrap().unwrap().into_bytes();
        let len = s.len();
        let t = s.clone();

        for i in 1..len {
            if !container.contains(&s) {
                container.add(&s);
            }
            reverse(&mut s[..i], i);
            if !container.contains(&s) {
                container.add(&s);
            }
            reverse(&mut s[i..len], len - i);
            if !container.contains(&s) {
                container.add(&s);
            }
            reverse(&mut s[..i], i);
            s.copy_from_slice(&t);
            swap(&mut s[..], &s[i..]);
            if !container.contains(&s) {
                container.add(&s);
            }
            reverse(&mut s[len - i..], len - i);
            if !container.contains(&s) {
                container.add(&s);
            }
            reverse(&mut s[len - i..len], i);
            if !container.contains(&s) {
                container.add(&s);
            }
            reverse(&mut s[..len - i], len - i);
            if !container.contains(&s) {
                container.add(&s);
            }
            s.copy_from_slice(&t);
        }
        println!("{}", container.count);
    }
}
```
