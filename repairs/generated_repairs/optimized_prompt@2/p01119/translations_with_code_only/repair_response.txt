// AOJ 1634 Balance Scale

use std::io::{self, BufRead, Write};

const HASHSIZ: usize = 1000003;
const INF: i32 = 0x7fffffff;

#[derive(Default, Copy, Clone)]
struct HA {
    n: i32,
    b: i32,
}

struct HashTable1 {
    table: [i32; HASHSIZ],
}

impl HashTable1 {
    fn new() -> Self {
        HashTable1 {
            table: [0; HASHSIZ],
        }
    }

    fn lookup(&self, n: i32) -> bool {
        let mut p = (n % HASHSIZ as i32) as usize;
        while self.table[p] != 0 {
            if self.table[p] == n {
                return true;
            }
            p = (p + 1) % HASHSIZ;
        }
        false
    }

    fn insert(&mut self, n: i32) {
        let mut p = (n % HASHSIZ as i32) as usize;
        while self.table[p] != 0 {
            if self.table[p] == n {
                return;
            }
            p = (p + 1) % HASHSIZ;
        }
        self.table[p] = n;
    }
}

struct HashTable2 {
    table: [HA; HASHSIZ],
}

impl HashTable2 {
    fn new() -> Self {
        HashTable2 {
            table: [HA::default(); HASHSIZ],
        }
    }

    fn lookup(&self, n: i32, b: i32) -> bool {
        let mut p = (n % HASHSIZ as i32) as usize;
        while self.table[p].n != 0 {
            if self.table[p].n == n {
                if self.table[p].b == b {
                    return true;
                }
                return false;
            }
            p = (p + 1) % HASHSIZ;
        }
        false
    }

    fn insert(&mut self, n: i32, b: i32) {
        let mut p = (n % HASHSIZ as i32) as usize;
        while self.table[p].n != 0 {
            if self.table[p].n == n {
                return;
            }
            p = (p + 1) % HASHSIZ;
        }
        self.table[p].n = n;
        self.table[p].b = b;
    }
}

fn read_non_negative_integer() -> i32 {
    let stdin = io::stdin();
    let mut input = String::new();
    stdin.lock().read_line(&mut input).unwrap();
    input.trim().parse().unwrap()
}

fn output_integer(n: i32) {
    let mut stdout = io::stdout();
    if n == 0 {
        write!(stdout, "0\n").unwrap();
    } else {
        let mut num = n;
        let mut buffer = Vec::new();
        if num < 0 {
            write!(stdout, "-").unwrap();
            num = -num;
        }
        while num > 0 {
            buffer.push((num % 10) as u8 + b'0');
            num /= 10;
        }
        buffer.reverse();
        stdout.write_all(&buffer).unwrap();
        writeln!(stdout).unwrap();
    }
}

fn abs(x: i32) -> i32 {
    if x >= 0 { x } else { -x }
}

fn main() {
    let mut q = vec![0; 60000];
    let mut a = vec![0; 102];
    let mut w = vec![0; 12];

    loop {
        let n = read_non_negative_integer();
        if n == 0 { break; }
        let m = read_non_negative_integer();
        for i in 0..n {
            a[i as usize] = read_non_negative_integer();
        }
        for i in 0..m {
            w[i as usize] = read_non_negative_integer();
        }

        let mut end = 1;
        for i in 0..m {
            let e = end;
            for j in 0..e {
                q[end] = q[j] + w[i as usize];
                end += 1;
                q[end] = q[j] - w[i as usize];
                end += 1;
            }
        }

        let mut ha1 = HashTable1::new();
        for i in 0..end {
            if q[i] > 0 {
                ha1.insert(q[i]);
            }
        }

        let mut x = 0;
        let mut ans = 0;
        let mut ha2 = HashTable2::new(); // Move ha2 initialization outside the loop
        for i in 0..n {
            if !ha1.lookup(a[i as usize]) {
                if x == 0 {
                    let mut t;
                    ans = INF;
                    for j in 0..end {
                        t = abs(a[i as usize] - q[j]);
                        if t < ans {
                            ans = t;
                        }
                        ha2.insert(t, 1); // Pass the correct b value
                    }
                    x = 1;
                } else {
                    let mut t;
                    let mut r = 0;
                    x += 1;
                    ans = INF;
                    for j in 0..end {
                        t = abs(a[i as usize] - q[j]);
                        if ha2.lookup(t, x) {
                            if t < ans {
                                ans = t;
                            }
                            r = 1;
                        }
                    }
                    if r == 0 {
                        ans = -1;
                        break;
                    }
                }
            }
        }
        output_integer(ans);
    }
}
