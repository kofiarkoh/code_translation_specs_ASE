```rust
fn main() {
    use std::io::{self, BufRead};

    // Read input from standard input
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read the number of pairs
    let n: usize = lines.next().unwrap().unwrap().parse().expect("Invalid input for N");

    // Preconditions: N must be greater than zero
    assert!(n > 0, "N must be greater than zero");

    // Vector to hold the pairs
    let mut pairs: Vec<(i64, i32)> = Vec::with_capacity(n);

    // Read the pairs of integers
    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        let nums: Vec<i64> = line.split_whitespace()
            .map(|s| s.parse().expect("Invalid input for pair"))
            .collect();
        // Each pair consists of a long long (i64) and an integer (i32)
        pairs.push((nums[0], nums[1] as i32));
    }

    // Initialize cumulative sum and max value
    let mut cumulative_sum: i64 = 0; // Change to i64 to match the type of first
    let mut max_value: i64 = i64::MIN; // Start with the minimum possible value

    // Process each pair
    for (first, second) in pairs {
        cumulative_sum += second as i64; // Update cumulative sum as i64
        let computed_value = first + cumulative_sum; // Compute value based on the pair
        max_value = max_value.max(computed_value); // Update max value if computed value is greater
    }

    // Output the maximum computed value
    println!("{}", max_value);
}
```
