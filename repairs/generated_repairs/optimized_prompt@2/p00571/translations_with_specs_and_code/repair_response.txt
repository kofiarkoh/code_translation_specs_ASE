```rust
use std::io::{self, BufRead};

const INF: i64 = 0x1010101010101010;

#[derive(Debug)]
struct Pair {
    a: i64,
    b: i64,
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read the number of pairs
    let n: usize = lines.next().unwrap().unwrap().trim().parse().expect("Invalid input for N");
    let mut pairs: Vec<Pair> = Vec::with_capacity(n);

    // Read the pairs
    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        let mut nums = line.split_whitespace();
        let a: i64 = nums.next().unwrap().parse().expect("Invalid input for long long");
        let b: i64 = nums.next().unwrap().parse().expect("Invalid input for integer");
        pairs.push(Pair { a, b });
    }

    // Sort pairs based on the first element, and then the second element
    pairs.sort_by(|x, y| {
        if x.a == y.a {
            x.b.cmp(&y.b)
        } else {
            x.a.cmp(&y.a)
        }
    });

    let mut s: Vec<i64> = vec![0; n + 1];
    let mut min = INF;
    let mut max = i64::MIN;

    for i in 1..=n {
        let t = s[i - 1] - pairs[i - 1].a;
        if t < min {
            min = t;
        }
        s[i] = s[i - 1] + pairs[i - 1].b; 
        let t = s[i] - pairs[i - 1].a - min;
        if t > max {
            max = t;
        }
    }

    // Output the maximum computed value
    println!("{}", max);
}
```
