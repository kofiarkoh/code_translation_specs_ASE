use std::io::{self, BufRead};

fn dfs(a: &mut Vec<i32>, n: usize) {
    let mut t = a[n];
    while t != -1 {
        let temp = a[t as usize];
        a[t as usize] = -1;
        t = temp;
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read n and k
    let first_line = lines.next().unwrap().unwrap();
    let mut nums = first_line.split_whitespace().map(|s| s.parse::<i32>());
    let n = match nums.next() {
        Some(Ok(num)) => num as usize,
        _ => return, // Handle parse error for n
    };
    let k = match nums.next() {
        Some(Ok(num)) => num,
        _ => return, // Handle parse error for k
    };

    // Preconditions check
    if n < 1 || n > 100_000 {
        return; // Behavior is not defined for invalid n
    }

    // Initialize the array
    let mut a = vec![0; n + 1]; // Using 1-based indexing
    for i in 1..=n {
        a[i] = match lines.next().unwrap().unwrap().parse::<i32>() {
            Ok(num) => num,
            Err(_) => return, // Handle parse error for array elements
        };
    }

    if n == k as usize {
        let x = a[1] - 1;
        let mut flag = true;
        for i in 1..=n {
            flag &= (a[i] - i as i32 + n as i32) % n as i32 == x;
        }
        println!("{}", if flag { "Yes" } else { "No" });
    } else {
        let mut cnt = 0;
        for i in 1..=n {
            if a[i] != -1 {
                dfs(&mut a, i);
                cnt += 1;
            }
        }
        println!("{}", if (n - cnt) % 2 != 0 && k % 2 != 0 { "No" } else { "Yes" });
    }
}
