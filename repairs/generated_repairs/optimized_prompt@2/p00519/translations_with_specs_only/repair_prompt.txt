fn min_cost_to_destination(
    towns: Vec<Town>,
    connections: Vec<Connection>,
    start: usize,
    destination: usize,
) -> Option<usize> {
    let n = towns.len();
    let mut graph: HashMap<usize, Vec<usize>> = HashMap::new();

    // Build the graph from connections
    for conn in connections {
        graph.entry(conn.town1).or_default().push(conn.town2);
        graph.entry(conn.town2).or_default().push(conn.town1);
    }

    // Min cost array, initialized to a large value
    let mut status_mincost = vec![vec![usize::MAX; towns[start].max_rides + 1]; n];
    status_mincost[start][towns[start].max_rides] = 0;

    let mut heap = BinaryHeap::new();
    heap.push(State {
        cost: 0,
        town: start,
        rides_left: towns[start].max_rides,
    });

    while let Some(State { cost, town, rides_left }) = heap.pop() {
        // If we reached the destination, return the cost
        if town == destination {
            return Some(cost);
        }

        // If we have no rides left, continue
        if rides_left == 0 {
            continue;
        }

        // Explore neighbors
        if let Some(neighbors) = graph.get(&town) {
            for &neighbor in neighbors {
                let new_cost = cost + towns[neighbor].fare;
                let new_rides_left = rides_left - 1;

                // Check if we found a cheaper way to reach this neighbor with the remaining rides
                if new_cost < status_mincost[neighbor][new_rides_left] {
                    status_mincost[neighbor][new_rides_left] = new_cost;
                    heap.push(State {
                        cost: new_cost,
                        town: neighbor,
                        rides_left: new_rides_left,
                    });
                }
            }
        }
    }

    // If we exit the loop without finding a path to the destination, return None
    None
}
