```rust
use std::io::{self, BufRead};

#[derive(Debug, Clone)]
struct Node {
    parent: i32,
    left: i32,
    right: i32,
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read the number of nodes
    let n: usize = lines.next().unwrap().unwrap().trim().parse().unwrap();
    let mut tree: Vec<Node> = vec![Node { parent: -1, left: -1, right: -1 }; n];
    let mut depths: Vec<i32> = vec![-1; n];

    // Read the nodes and their children
    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        let mut parts = line.split_whitespace();
        let id: usize = parts.next().unwrap().parse().unwrap();
        let num: usize = parts.next().unwrap().parse().unwrap();

        let mut last_child = -1;
        for (j, child_str) in parts.enumerate().skip(2) {
            let child: i32 = child_str.parse().unwrap();
            if j == 2 {
                tree[id].left = child;
            } else {
                tree[last_child as usize].right = child;
            }
            tree[child as usize].parent = id as i32;
            last_child = child;
        }
    }

    // Find the root node
    let root = tree.iter().position(|node| node.parent == -1).unwrap();

    // Calculate depths
    get_depths(root, 0, &tree, &mut depths);

    // Print the tree
    for i in 0..n {
        print_tree(i, &tree, &depths);
    }
}

fn get_depths(node: usize, depth: i32, tree: &[Node], depths: &mut Vec<i32>) {
    depths[node] = depth;
    if tree[node].left != -1 {
        get_depths(tree[node].left as usize, depth + 1, tree, depths);
    }
    if tree[node].right != -1 {
        get_depths(tree[node].right as usize, depth, tree, depths);
    }
}

fn print_tree(i: usize, tree: &[Node], depths: &[i32]) {
    let mut children = Vec::new();
    let mut current = tree[i].left;

    while current != -1 {
        children.push(current);
        current = tree[current as usize].right;
    }

    let node_type = if tree[i].parent == -1 {
        "root"
    } else if tree[i].left == -1 {
        "leaf"
    } else {
        "internal node"
    };

    let children_str = children.iter().map(|&c| c.to_string()).collect::<Vec<String>>().join(", ");
    println!("node {}: parent = {}, depth = {}, {}, [{}]", i, tree[i].parent, depths[i], node_type, children_str);
}
```
