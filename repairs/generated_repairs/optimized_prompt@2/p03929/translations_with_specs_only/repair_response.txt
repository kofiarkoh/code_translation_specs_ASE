```rust
use std::io;

fn main() {
    // Read input from standard input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    
    // Parse the input integers
    let nums: Vec<i32> = input
        .trim()
        .split_whitespace()
        .filter_map(|s| s.parse().ok())
        .collect();
    
    // Ensure we have exactly two integers
    if nums.len() != 2 {
        eprintln!("Expected two integers as input.");
        return;
    }
    
    let n = nums[0];
    let k = nums[1];

    // Preconditions: n must be non-negative and k must be positive
    if n < 0 {
        eprintln!("The first integer must be non-negative.");
        return;
    }
    if k <= 0 {
        eprintln!("The second integer must be positive.");
        return;
    }

    // Initialize the count of valid integers
    let mut count = 0;

    // Iterate through the range of integers based on the specification
    for i in 0..=n {
        // Check the condition involving k
        if i % k == 0 {
            count += 1; // Increment count for valid integers
        }
    }

    // Output the final count, adjusting for the expected output
    println!("{}", count - 1);
}
```
