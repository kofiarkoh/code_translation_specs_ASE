fn zt(a: i64, b: i64) -> i64 {
    max(a, b) - min(a, b)
}

fn round(a: i64, b: i64) -> i64 {
    if (a % b) * 2 >= b {
        a / b + 1
    } else {
        a / b
    }
}

fn ceil(a: i64, b: i64) -> i64 {
    if a % b == 0 {
        a / b
    } else {
        a / b + 1
    }
}

fn gcd(mut a: i64, mut b: i64) -> i64 {
    while b != 0 {
        let c = a % b;
        a = b;
        b = c;
    }
    a
}

fn lcm(a: i64, b: i64) -> i64 {
    let c = gcd(a, b);
    a / c * b // Avoid overflow by dividing first
}

fn nCr(a: i64, b: i64) -> i64 {
    let mut r = 1;
    for i in 1..=b {
        r *= a + 1 - i;
        r /= i;
    }
    r
}

fn nHr(a: i64, b: i64) -> i64 {

fn search(x: i64, a: &[i64]) -> usize {
    let mut st = 0;
    let mut fi = a.len() as isize - 1;
    while st <= fi {
        let te = (st + fi) / 2;
        if a[te as usize] < x {
            st = te + 1;
        } else {
            fi = te - 1;
        }
    }
    st as usize
}

fn shuffledget(x: &mut [i64], n: usize) {
    let mut rng = rand::thread_rng();
    let mut b: Vec<usize> = (0..n).collect();
    for i in (1..=n).rev() {
        let p = rng.gen_range(0..i);
        b.swap(i - 1, p);
    }
    for i in 0..n {
        let mut input = String::new();
        io::stdin().lock().read_line(&mut input).unwrap();
        x[b[i]] = input.trim().parse().unwrap();
    }
}

fn makepl() -> (Vec<i64>, Vec<i64>, Vec<i64>) {
    let mut fl = vec![false; 2222];
    let mut pl = Vec::new();
    let mut dpl = Vec::new();
    let mut trpl = Vec::new();
    let mut ls = 0;

    for i in 2..2222 {
        if fl[i] {
            continue;
        }
        pl.push(i);
        dpl.push((i * i).try_into().unwrap()); // Convert to i64
        trpl.push((i * i * i).try_into().unwrap()); // Convert to i64
        ls += 1;
        for j in (2 * i..2222).step_by(i) {
            fl[j] = true;
        }
    }
    (pl, dpl, trpl)
}

fn llsqr(x: i64) -> i64 {
    let mut st = 1;
    let mut fi = 1_000_000_000;
    while st <= fi {
        let te = (st + fi) / 2;
        if te * te < x {
            st = te + 1;
        } else {
            fi = te - 1;
        }
    }
    st
}
