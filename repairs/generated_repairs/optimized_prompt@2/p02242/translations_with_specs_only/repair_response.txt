```rust
use std::collections::HashMap;
use std::io::{self, Write};

struct Graph {
    edges: HashMap<usize, Vec<(usize, usize)>>, // (destination, cost)
    costs: Vec<usize>,
    visited: Vec<bool>,
}

impl Graph {
    fn new(num_nodes: usize) -> Self {
        let costs = vec![usize::MAX; num_nodes]; // Initialize costs to "infinity"
        let visited = vec![false; num_nodes]; // Track visited nodes
        let edges = HashMap::new(); // Initialize edges

        Graph { edges, costs, visited }
    }

    fn add_edge(&mut self, from: usize, to: usize, cost: usize) {
        self.edges.entry(from).or_insert_with(Vec::new).push((to, cost));
    }

    fn dijkstra(&mut self, source: usize) {
        if source >= self.costs.len() {
            return; // Prevent out-of-bounds access
        }
        self.costs[source] = 0; // Cost to source is zero

        for _ in 0..self.costs.len() {
            let (current_node, current_cost) = self.find_min_cost_node();
            if current_node.is_none() {
                break; // No reachable nodes left
            }
            let current_node = current_node.unwrap();
            self.visited[current_node] = true;

            if let Some(neighbors) = self.edges.get(&current_node) {
                for &(neighbor, edge_cost) in neighbors {
                    if neighbor < self.visited.len() && !self.visited[neighbor] {
                        let new_cost = current_cost + edge_cost;
                        if new_cost < self.costs[neighbor] {
                            self.costs[neighbor] = new_cost; // Update cost if new cost is lower
                        }
                    }
                }
            }
        }
    }

    fn find_min_cost_node(&self) -> (Option<usize>, usize) {
        let mut min_cost = usize::MAX;
        let mut min_node = None;

        for (node, &cost) in self.costs.iter().enumerate() {
            if !self.visited[node] && cost < min_cost {
                min_cost = cost;
                min_node = Some(node);
            }
        }

        (min_node, min_cost)
    }

    fn print_costs(&self) {
        for (node, &cost) in self.costs.iter().enumerate() {
            if cost == usize::MAX {
                // Indicate unreachable nodes
                println!("Node {}: Unreachable", node);
            } else {
                println!("Node {}: Cost {}", node, cost);
            }
        }
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let num_nodes: usize = input.trim().parse().expect("Invalid number of nodes");

    let mut graph = Graph::new(num_nodes);

    for _ in 0..num_nodes {
        input.clear();
        io::stdin().read_line(&mut input).unwrap();
        let parts: Vec<usize> = input.trim().split_whitespace().map(|s| s.parse().unwrap()).collect();
        let node_index = parts[0];
        let num_edges = parts[1];

        for _ in 0..num_edges {
            input.clear();
            io::stdin().read_line(&mut input).unwrap();
            let edge_parts: Vec<usize> = input.trim().split_whitespace().map(|s| s.parse().unwrap()).collect();
            let destination = edge_parts[0];
            let cost = edge_parts[1];
            graph.add_edge(node_index, destination, cost);
        }
    }

    // Check if the source node is valid before running Dijkstra's algorithm
    if num_nodes > 0 {
        graph.dijkstra(0);
    }
    graph.print_costs();
}
```
