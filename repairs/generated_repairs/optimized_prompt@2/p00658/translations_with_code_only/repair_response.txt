```rust
use std::io::{self, BufRead};

const DEBUG_MODE: usize = 0;

const GAMEOVER_CLEAR: i32 = 1;
const GAMEOVER_HOLE: i32 = 2;
const GAMEOVER_BLOCK: i32 = 3;
const GAMEOVER_WGOAL: i32 = 4;

static mut WIDTH: usize = 0;
static mut HEIGHT: usize = 0;
static mut MAP: [[char; 51]; 51] = [['\0'; 51]; 51];

static mut PX: usize = 0;
static mut PY: usize = 0;
static mut GX: usize = 0;
static mut GY: usize = 0;
static mut GAMEOVER: i32 = 0;

fn game_init() {
    unsafe {
        for y in 0..HEIGHT {
            for x in 0..WIDTH {
                if MAP[y][x] == 'S' {
                    PX = x;
                    PY = y;
                    MAP[y][x] = '.';
                } else if MAP[y][x] == 'G' {
                    GX = x;
                    GY = y;
                }
            }
        }
        for x in 0..WIDTH {
            MAP[HEIGHT][x] = '#';
        }
        GAMEOVER = 0;
    }
}

fn moveto(to: usize) {
    unsafe {
        let old_px = PX;
        if to < 1 || to > WIDTH {
            return;
        }
        let to = to - 1;
        let add = if to < PX { -1 } else { 1 };
        if MAP[PY][to] != '.' {
            return;
        }
        if PX == to {
            return;
        }
        let mut i = PX;
        while i != to {
            if MAP[PY + 1][i] == '.' {
                break;
            }
            i = (i as isize + add) as usize;
        }
        if i == to && MAP[PY + 1][to] != '.' {
            PX = to;
            // countdown
            match MAP[PY + 1][old_px] {
                '3' => MAP[PY + 1][old_px] = '2',
                '2' => MAP[PY + 1][old_px] = '1',
                '1' => MAP[PY + 1][old_px] = '.',
                _ => {}
            }
        }
    }
}

fn push_internal(x: usize, y: usize, isright: bool) {
    unsafe {
        let block = MAP[y][x];
        if block == 'C' || block == '.' {
            return;
        }
        MAP[y][x] = '.';
        if isright {
            if x < WIDTH - 1 && MAP[y][x + 1] != '.' {
                MAP[y][x] = block;
                return;
            }
            let mut i = x + 1;
            while i < WIDTH {
                if !(block == 'I' || MAP[y + 1][i] == 'I') || MAP[y + 1][i] == '.' {
                    break;
                }
                if i < WIDTH - 1 && MAP[y][i + 1] != '.' {
                    break;
                }
                i += 1;
            }
            if i < WIDTH {
                MAP[y][i] = block;
            }
        } else {
            if x > 0 && MAP[y][x - 1] != '.' {
                MAP[y][x] = block;
                return;
            }
            let mut i = x as isize - 1;
            while i >= 0 {
                if !(block == 'I' || MAP[y + 1][i as usize] == 'I') || MAP[y + 1][i as usize] == '.' {
                    break;
                }
                if i > 0 && MAP[y][i as usize - 1] != '.' {
                    break;
                }
                i -= 1;
            }
            if i >= 0 {
                MAP[y][i as usize] = block;
            }
        }
    }
}

fn push(isright: bool) {
    unsafe {
        if isright {
            if PX >= WIDTH - 1 {
                return;
            }
            if MAP[PY][PX + 1] == '.' {
                return;
            }
            let mut i = PX + 1;
            while i < WIDTH && MAP[PY][i] != '.' {
                if MAP[PY][i] == 'C' {
                    return;
                }
                i += 1;
            }
            for i in (PX + 1..i).rev() {
                push_internal(i, PY, true);
            }
        } else {
            if PX == 0 {
                return;
            }
            if MAP[PY][PX - 1] == '.' {
                return;
            }
            let mut i = PX as isize - 1;
            while i >= 0 && MAP[PY][i as usize] != '.' {
                if MAP[PY][i as usize] == 'C' {
                    return;
                }
                i -= 1;
            }
            for i in (i + 1..PX).rev() {
                push_internal(i.try_into().unwrap(), PY, false);
            }
        }
    }
}

fn pull(isright: bool) {
    unsafe {
        let old_px = PX;
        if PX == 0 || PX >= WIDTH - 1 {
            return;
        }
        if isright {
            if MAP[PY][PX + 1] == '.' && MAP[PY][PX - 1] != '.' && MAP[PY][PX - 1] != 'C' {
                MAP[PY][PX] = MAP[PY][PX - 1];
                MAP[PY][PX - 1] = '.';
                PX += 1;
            } else {
                return;
            }
        } else {
            if MAP[PY][PX - 1] == '.' && MAP[PY][PX + 1] != '.' && MAP[PY][PX + 1] != 'C' {
                MAP[PY][PX] = MAP[PY][PX + 1];
                MAP[PY][PX + 1] = '.';
                PX -= 1;
            } else {
                return;
            }
        }
        // countdown
        match MAP[PY + 1][old_px] {
            '3' => MAP[PY + 1][old_px] = '2',
            '2' => MAP[PY + 1][old_px] = '1',
            '1' => MAP[PY + 1][old_px] = '.',
            _ => {}
        }
    }
}

fn climb(isright: bool) {
    unsafe {
        let old_px = PX;
        if PY == 0 {
            return;
        }
        if isright {
            if PX < WIDTH - 1 && MAP[PY - 1][PX] == '.' && MAP[PY - 1][PX + 1] == '.' && MAP[PY][PX + 1] != '.' {
                PX += 1;
                PY -= 1;
            } else {
                return;
            }
        } else {
            if PX > 0 && MAP[PY - 1][PX] == '.' && MAP[PY - 1][PX - 1] == '.' && MAP[PY][PX - 1] != '.' {
                PX -= 1;
                PY -= 1;
            } else {
                return;
            }
        }
        // countdown
        match MAP[PY + 2][old_px] {
            '3' => MAP[PY + 2][old_px] = '2',
            '2' => MAP[PY + 2][old_px] = '1',
            '1' => MAP[PY + 2][old_px] = '.',
            _ => {}
        }
    }
}

fn getdown(isright: bool) {
    unsafe {
        let old_px = PX;
        if isright {
            if PX < WIDTH - 1 && MAP[PY][PX + 1] == '.' && MAP[PY + 1][PX + 1] == '.' {
                PX += 1;
                PY += 1;
            } else {
                return;
            }
        } else {
            if PX > 0 && MAP[PY][PX - 1] == '.' && MAP[PY + 1][PX - 1] == '.' {
                PX -= 1;
                PY += 1;
            } else {
                return;
            }
        }
        // countdown
        match MAP[PY][old_px] {
            '3' => MAP[PY][old_px] = '2',
            '2' => MAP[PY][old_px] = '1',
            '1' => MAP[PY][old_px] = '.',
            _ => {}
        }
    }
}

fn gameover_hantei() -> i32 {
    unsafe {
        if GAMEOVER != 0 {
            return GAMEOVER;
        }
        if MAP[PY + 1][PX] == 'G' {
            GAMEOVER = GAMEOVER_CLEAR;
        } else if MAP[PY + 1][PX] == 'B' {
            GAMEOVER = GAMEOVER_HOLE;
        } else if MAP[PY][PX] != '.' {
            GAMEOVER = GAMEOVER_BLOCK;
        } else if MAP[GY][GX] != 'G' {
            GAMEOVER = GAMEOVER_WGOAL;
        }
        GAMEOVER
    }
}

fn auto_syori() {
    unsafe {
        let mut moved = true;
        while moved {
            moved = false;
            for y in 0..HEIGHT {
                for x in 0..WIDTH {
                    if MAP[y + 1][x] == 'B' && MAP[y][x] != '.' {
                        MAP[y][x] = '.';
                        moved = true;
                    }
                }
            }
            while MAP[PY + 1][PX] == '.' {
                PY += 1;
                moved = true;
            }
            for y in (0..HEIGHT - 1).rev() {
                for x in 0..WIDTH {
                    if MAP[y][x] != '.' {
                        let mut y2 = y;
                        while true {
                            if (x == 0 || MAP[y2 + 1][x - 1] == '.') && MAP[y2 + 1][x] == '.' && (x >= WIDTH - 1 || MAP[y2 + 1][x + 1] == '.') {
                                moved = true;
                            } else {
                                break;
                            }
                            y2 += 1;
                        }
                        if y2 != y {
                            MAP[y2][x] = MAP[y][x];
                            MAP[y][x] = '.';
                        }
                    }
                }
            }
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    loop {
        let line = lines.next().unwrap().unwrap();
        let mut dimensions = line.split_whitespace();
        let height: usize = dimensions.next().unwrap().parse().unwrap();
        let width: usize = dimensions.next().unwrap().parse().unwrap();
        if height == 0 && width == 0 {
            break;
        }
        unsafe {
            HEIGHT = height;
            WIDTH = width;
        }
        for i in 0..height {
            let line = lines.next().unwrap().unwrap();
            unsafe {
                MAP[i] = line.chars().collect::<Vec<char>>()[..51].try_into().unwrap();
            }
        }
        let command_num: usize = lines.next().unwrap().unwrap().parse().unwrap();
        game_init();
        for _ in 0..command_num {
            let command_line = lines.next().unwrap().unwrap();
            let mut command_parts = command_line.split_whitespace();
            let command = command_parts.next().unwrap();
            let command2 = command_parts.next().unwrap();
            if gameover_hantei() == 0 {
                match command {
                    "MOVETO" => moveto(command2.parse::<usize>().unwrap()),
                    "PUSH" => push(command2.chars().next().unwrap() == 'R'),
                    "PULL" => pull(command2.chars().next().unwrap() == 'R'),
                    "GETDOWN" => getdown(command2.chars().next().unwrap() == 'R'),
                    "CLIMB" => climb(command2.chars().next().unwrap() == 'R'),
                    _ => return,
                }
                if gameover_hantei() == 0 {
                    auto_syori();
                }
                #[cfg(DEBUG_MODE)]
                {
                    let block = MAP[PY][PX];
                    if block == '.' {
                        MAP[PY][PX] = 'S';
                    }
                    for j in 0..HEIGHT {
                        println!("{}", MAP[j].iter().collect::<String>());
                    }
                    MAP[PY][PX] = block;
                }
            }
            #[cfg(DEBUG_MODE)]
            {
                if gameover_hantei() != 0 {
                    break;
                }
            }
        }
        match gameover_hantei() {
            GAMEOVER_CLEAR => println!("Game Over : Cleared"),
            GAMEOVER_HOLE => println!("Game Over : Death by Hole"),
            GAMEOVER_BLOCK => println!("Game Over : Death by Block"),
            GAMEOVER_WGOAL => println!("Game Over : Death by Walking Goal"),
            _ => println!("Game Over : Gave Up"),
        }
    }
}
```
