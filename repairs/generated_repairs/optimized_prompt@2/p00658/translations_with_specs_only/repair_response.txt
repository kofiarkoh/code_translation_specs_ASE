// Define the possible outcomes of the game
#[derive(Debug)]
enum GameOutcome {
    Cleared,
    DeathByHole,
    DeathByBlock,
    DeathByWalkingGoal,
    GaveUp,
}

// Define the possible commands the player can execute
#[derive(Debug)]
enum Command {
    MoveUp,
    MoveDown,
    MoveLeft,
    MoveRight,
    Push,
    Pull,
    Climb,
    GetDown,
}

// Define the structure for the game state
struct Game {
    height: usize,
    width: usize,
    map: Vec<Vec<char>>, // The grid representation of the game map
    player_pos: (usize, usize), // Player's current position
    goal_pos: (usize, usize), // Goal position
}

impl Game {
    // Initialize a new game with given dimensions and map
    fn new(height: usize, width: usize, map: Vec<Vec<char>>) -> Option<Self> {
        if height == 0 || width == 0 || height > 51 || width > 51 {
            return None; // Invalid dimensions
        }

        let mut player_pos = (0, 0);
        let mut goal_pos = (0, 0);
        
        // Find the player's start position and goal position
        for i in 0..height {
            for j in 0..width {
                if map[i][j] == 'S' {
                    player_pos = (i, j);
                } else if map[i][j] == 'G' {
                    goal_pos = (i, j);
                }
            }
        }

        Some(Game {
            height,
            width,
            map,
            player_pos,
            goal_pos,
        })
    }

    // Execute a sequence of commands
    fn execute_commands(&mut self, commands: Vec<Command>) -> GameOutcome {
        for command in commands {
            match command {
                Command::MoveUp => self.move_player(-1, 0),
                Command::MoveDown => self.move_player(1, 0),
                Command::MoveLeft => self.move_player(0, -1),
                Command::MoveRight => self.move_player(0, 1),
                Command::Push => self.push_block(),
                Command::Pull => self.pull_block(),
                Command::Climb => self.climb(),
                Command::GetDown => self.get_down(),
            }

            // Check for game-over conditions
            if self.player_pos == self.goal_pos {
                return GameOutcome::Cleared;
            }
        }
        GameOutcome::GaveUp // Default outcome if no other conditions are met
    }

    // Move the player in the specified direction
    fn move_player(&mut self, delta_y: isize, delta_x: isize) {
        let new_y = (self.player_pos.0 as isize + delta_y) as usize;
        let new_x = (self.player_pos.1 as isize + delta_x) as usize;

        // Check bounds and update position if valid
        if new_y < self.height && new_x < self.width {
            if self.map[new_y][new_x] != '#' { // Ensure the player doesn't move into a block
                self.player_pos = (new_y, new_x);
            }
        }
    }

// Example usage
fn main() {
    let height = 5;
    let width = 5;
    let map = vec![
        vec!['#', '#', '#', '#', '#'],
        vec!['#', 'S', '.', '.', '#'],
        vec!['#', '.', '.', '.', '#'],
        vec!['#', '.', '.', 'G', '#'],
        vec!['#', '#', '#', '#', '#'],
    ];

    if let Some(mut game) = Game::new(height, width, map) {
        let commands = vec![
            Command::MoveDown,
            Command::MoveDown,
            Command::MoveRight,
            Command::GetDown,
            Command::Climb,
            Command::MoveRight,
        ];
        let outcome = game.execute_commands(commands);
        match outcome {
            GameOutcome::GaveUp => println!("Game Over : Gave Up"),
            GameOutcome::Cleared => println!("Game Over : Cleared"),
            _ => println!("{:?}", outcome),
        }
    } else {
        println!("Invalid game dimensions.");
    }
}
