use std::io::{self, BufRead};

const MAX: usize = 602;
const MAX_ROWS: usize = 60;
const MAX_COLS: usize = 12;

struct UnionFind {
    parent: [usize; MAX],
    rank: [usize; MAX],
}

impl UnionFind {
    fn new() -> Self {
        let mut parent = [0; MAX];
        let mut rank = [0; MAX];
        for i in 0..MAX {
            parent[i] = i;
        }
        UnionFind { parent, rank }
    }

    fn find(&mut self, x: usize) -> usize {
        if x != self.parent[x] {
            self.parent[x] = self.find(self.parent[x]);
        }
        self.parent[x]
    }

    fn union(&mut self, x: usize, y: usize) {
        let root_x = self.find(x);
        let root_y = self.find(y);
        if root_x != root_y {
            if self.rank[root_x] > self.rank[root_y] {
                self.parent[root_y] = root_x;
            } else {
                self.parent[root_x] = root_y;
                if self.rank[root_x] == self.rank[root_y] {
                    self.rank[root_y] += 1;
                }
            }
        }
    }
}

#[derive(Clone)]
struct TreeNode {
    used: bool,
    m: usize,
    mx: usize,
    xl: isize,
    xr: isize,
    par: usize,
    ch: usize,
    sib: usize,
}

impl TreeNode {
    fn new() -> Self {
        TreeNode {
            used: false,
            m: 0,
            mx: 0,
            xl: -1,
            xr: -1,
            par: 0,
            ch: 0,
            sib: 0,
        }
    }
}

fn mktree(r: usize, c: usize, this: usize, h: usize, map: &Vec<Vec<char>>, tbl: &mut Vec<TreeNode>, rel: &Vec<Vec<usize>>, tr: &mut Vec<isize>) {
    if r == h - 1 {
        if map[r][c] == '.' {
            return;
        }
        if tbl[this].xl < 0 {
            tbl[this].xl = c as isize;
        } else if (c as isize) < tbl[this].xl {
            tbl[this].xl = c as isize;
        }
        if (c as isize) > tbl[this].xr {
            tbl[this].xr = c as isize;
        }
        return;
    }
    if map[r + 1][c] == '.' {
        return;
    }

    let k = rel[r + 1][c];
    let p = tr[k] as usize;
    if p == this || p >= tbl.len() {
        return;
    }

    if tbl[this].xl < 0 {
        tbl[this].xl = c as isize;
    } else if (c as isize) < tbl[this].xl {
        tbl[this].xl = c as isize;
    }
    if (c as isize) > tbl[this].xr {
        tbl[this].xr = c as isize;
    }
    if tbl[this].par != 0 {
        return;
    }

    if tbl[p].ch == 0 {
        tbl[p].ch = this;
    } else {
        let mut s = tbl[p].ch;
        while tbl[s].sib != 0 {
            s = tbl[s].sib;
        }
        tbl[s].sib = this;
    }
    tbl[this].par = p;
}

fn check(this: usize, tbl: &mut Vec<TreeNode>) -> bool {
    if tbl[this].used {
        return true;
    }
    let ch = tbl[this].ch;
    if ch != 0 && !check(ch, tbl) {
        return false;
    }

    let mut sib = tbl[this].sib;
    while sib != 0 {
        if !check(sib, tbl) {
            return false;
        }
        sib = tbl[sib].sib;
    }

    let x = 0.5 + tbl[this].mx as f64 / tbl[this].m as f64;
    if x <= tbl[this].xl as f64 || tbl[this].xr as f64 + 1.0 <= x {
        return false;
    }

    let par = tbl[this].par;
    if par < tbl.len() {
        tbl[par].m += tbl[this].m;
        tbl[par].mx += tbl[this].mx;
    }
    tbl[this].used = true;
    true
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let mut map = vec![vec!['.'; MAX_COLS]; MAX_ROWS];
    let mut rel = vec![vec![0; MAX_COLS]; MAX_ROWS];
    let mut uf = UnionFind::new();
    let mut tbl = vec![TreeNode::new(); 100];
    let mut tr = vec![-1; MAX];

    while let Some(line) = lines.next() {
        let line = line.unwrap();
        let mut dimensions = line.split_whitespace();
        let w: usize = dimensions.next().unwrap().parse().unwrap();
        let h: usize = dimensions.next().unwrap().parse().unwrap();

        if w == 0 {
            break;
        }

        for r in 0..h {
            let row = lines.next().unwrap().unwrap();
            for c in 0..w {
                map[r][c] = row.chars().nth(c).unwrap();
                let i = r * w + c;
                rel[r][c] = i;
                uf.parent[i] = i;
            }
        }

        for r in 0..h {
            for c in 0..w {
                if map[r][c] == '.' {
                    continue;
                }
                for (dr, dc) in &[(0, 1), (1, 0), (0, -1), (-1, 0)] {
                    let r2 = r as isize + *dr;
                    let c2 = c as isize + *dc;
                    if r2 >= 0 && r2 < h as isize && c2 >= 0 && c2 < w as isize {
                        if map[r2 as usize][c2 as usize] == map[r][c] {
                            uf.union(rel[r][c], rel[r2 as usize][c2 as usize]);
                        }
                    }
                }
            }
        }

        let mut sz = 1;
        for r in (0..h).rev() {
            for c in 0..w {
                if map[r][c] == '.' {
                    continue;
                }
                let k = rel[r][c];
                let k = uf.find(k);
                let i = if tr[k] < 0 {
                    tr[k] = sz;
                    sz += 1;
                    tbl[tr[k] as usize].xl = -1;
                    tr[k] as usize
                } else {
                    tr[k] as usize
                };
                tbl[i].m += 1;
                tbl[i].mx += c;
                mktree(r, c, i, h, &map, &mut tbl, &rel, &mut tr);
            }
        }
        tbl[1].par = 1;

        println!("{}", if check(1, &mut tbl) { "STABLE" } else { "UNSTABLE" });
    }
}
