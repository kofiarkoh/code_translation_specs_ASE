use std::io::{self, BufRead, Write};

const MAX: usize = 1005;

fn read_input() -> (usize, Vec<Vec<u8>>) {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();
    
    let n: usize = lines.next().unwrap().unwrap().trim().parse::<usize>().unwrap() - 1;
    let mut grid = vec![vec![0; n + 1]; n + 1];

    for i in 0..=n {
        let line = lines.next().unwrap().unwrap();
        for (j, c) in line.chars().enumerate() {
            if c == '1' {
                grid[j][i] = 1;
            }
        }
    }
    (n, grid)
}

fn output(s: &str) {
    let stdout = io::stdout();
    let mut handle = stdout.lock();
    writeln!(handle, "{}", s).unwrap();
}

fn output_number(n: isize) {
    let stdout = io::stdout();
    let mut handle = stdout.lock();
    writeln!(handle, "{}", n).unwrap();
}

struct BipartiteMatcher {
    hi: Vec<usize>,
    to: Vec<Vec<usize>>,
    seen: Vec<usize>,
    match_: Vec<isize>,
    mark: usize,
}

impl BipartiteMatcher {
    fn new(n: usize) -> Self {
        BipartiteMatcher {
            hi: vec![0; n],
            to: vec![vec![]; n],
            seen: vec![0; n],
            match_: vec![-1; n * 2],
            mark: 1,
        }
    }

    fn bpm(&mut self, u: usize) -> bool {
        self.seen[u] = self.mark;
        for &v in &self.to[u][..self.hi[u]] {
            let w = self.match_[v];
            if w < 0 || (self.seen[w as usize] != self.mark && self.bpm(w as usize)) {
                self.match_[v] = u as isize;
                self.match_[u] = v as isize;
                return true;
            }
        }
        false
    }

    fn bipartite_matching(&mut self, m: usize, n: usize) -> usize {
        self.match_.fill(-1);
        let mut max = 0;
        for u in 0..m {
            if self.match_[u] < 0 {
                self.mark += 1;
                if self.bpm(u) {
                    max += 1;
                }
            }
        }
        max
    }
}

fn main() {
    let (n, grid) = read_input();
    let mut matcher = BipartiteMatcher::new(n);

    for i in 0..n {
        for j in 0..n {
            if grid[j][i] == 1 {
                matcher.to[j].push(n + i);
                matcher.hi[j] += 1;
            }
        }
    }

    if matcher.bipartite_matching(n, n) < n {
        output("no");
    } else {
        output("yes");
        for i in 0..n {
            let matched_index = matcher.match_[i] as usize;
            matcher.match_[matched_index] = -1;
            matcher.match_[i] = -1;
            matcher.mark += 1;
            matcher.bpm(i);
            matcher.hi[i] = 0;
            matcher.to[i].push(matcher.match_[i] as usize);
            output_number(matcher.match_[i] - n as isize + 1);
        }
    }
}
