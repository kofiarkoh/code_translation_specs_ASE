// AOJ 1370 Hidden Anagrams

use std::io::{self, BufRead};

const HASHSIZ: usize = 10007;

#[derive(Default)]
struct Hash {
    f: [i16; 26],
}

struct HashTable {
    table: [Hash; HASHSIZ + 2],
}

impl HashTable {
    fn new() -> Self {
        HashTable {
            table: Default::default(),
        }
    }

    fn insert(&mut self, f: &mut [i16; 26]) {
        f[0] += 1;
        let mut k: u64 = 0;
        for i in (0..26).step_by(2) {
            k = (k << 4) + f[i as usize] as u64;
        }
        let mut tp_index = (k % HASHSIZ as u64) as usize;
        while self.table[tp_index].f[0] != 0 {
            if self.table[tp_index].f == *f {
                f[0] -= 1;
                return; // Already exists
            }
            tp_index = (tp_index + 1) % (HASHSIZ + 2);
        }
        self.table[tp_index].f.copy_from_slice(f);
        f[0] -= 1;
    }

    fn lookup(&mut self, f: &mut [i16; 26]) -> bool {
        f[0] += 1;
        let mut k: u64 = 0;
        for i in (0..26).step_by(2) {
            k = (k << 4) + f[i as usize] as u64;
        }
        let mut tp_index = (k % HASHSIZ as u64) as usize;
        while self.table[tp_index].f[0] != 0 {
            if self.table[tp_index].f == *f {
                f[0] -= 1;
                return true;
            }
            tp_index = (tp_index + 1) % (HASHSIZ + 2);
        }
        f[0] -= 1;
        false
    }
}

fn main() {
    let stdin = io::stdin();
    let mut handle = stdin.lock();

    let mut sa = String::new();
    let mut sb = String::new();
    handle.read_line(&mut sa).unwrap();
    handle.read_line(&mut sb).unwrap();

    let wa = sa.trim().len() as i32;
    let wb = sb.trim().len() as i32;

    let mut f = [0i16; 26];

    let mut p: Vec<_> = sa.chars().filter(|&c| c.is_ascii_lowercase()).collect();
    for c in &mut p {
        *c = (*c as u8 - b'a') as char;
        f[*c as usize] += 1;
    }

    let mut w = wa;
    if w > wb {
        w = wb;
    }

    let mut hash_table = HashTable::new();

    for k in (1..=w).rev() {
        f = [0; 26];
        for c in &p[..k as usize] {
            f[*c as usize] += 1;
        }
        hash_table.insert(&mut f);

        let mut i = wa - k;
        let mut q = p.iter();
        while i > 0 {
            f[*q.next().unwrap() as usize] -= 1;
            f[*q.next().unwrap() as usize] += 1;
            hash_table.insert(&mut f);
            i -= 1;
        }

        f = [0; 26];
        let sb_chars: Vec<_> = sb.chars().filter(|&c| c.is_ascii_lowercase()).collect();
        for c in &sb_chars[..k as usize] {
            f[*c as usize] += 1;
        }
        if hash_table.lookup(&mut f) {
            println!("{}", k);
            return;
        }

        let mut i = wb - k;
        let mut q = sb_chars.iter();
        while i > 0 {
            f[*q.next().unwrap() as usize] -= 1;
            f[*q.next().unwrap() as usize] += 1;
            if hash_table.lookup(&mut f) {
                println!("{}", k);
                return;
            }
            i -= 1;
        }
    }
}
