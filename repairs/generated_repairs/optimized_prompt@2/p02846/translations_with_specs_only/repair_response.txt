use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let lines = stdin.lock().lines();

    let mut inputs: Vec<i64> = Vec::new();
    for line in lines.take(6) { // Changed to take 6 lines to match the number of inputs
        if let Ok(value) = line {
            inputs.extend(value.split_whitespace().filter_map(|s| s.parse::<i64>().ok())); // Specified i64 type
        }
    }

    let t1 = inputs[0]; // First time factor
    let t2 = inputs[1]; // Second time factor
    let a1 = inputs[2]; // First parameter of set A
    let a2 = inputs[3]; // Second parameter of set A
    let b1 = inputs[4]; // First parameter of set B
    let b2 = inputs[5]; // Second parameter of set B

    // Check if sets A and B are equivalent
    if (a1 == b1 && a2 == b2) || (a1 == b2 && a2 == b1) {
        // If A and B are equivalent, output "1"
        println!("1");
    } else {
        // Calculate the output based on the differences
        let result = calculate_difference(a1, a2, b1, b2);
        println!("{}", result);
    }
}

// Function to calculate the output based on differences between A and B
fn calculate_difference(a1: i64, a2: i64, b1: i64, b2: i64) -> i64 {
    // Example calculation: return the sum of differences
    (a1 - b1).abs() + (a2 - b2).abs() // Using absolute differences as a simple example
}
