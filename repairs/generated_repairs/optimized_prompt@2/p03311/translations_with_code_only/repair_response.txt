use std::io::{self, BufRead};

fn cmpnum(n1: &i64, n2: &i64) -> std::cmp::Ordering {
    n1.cmp(n2)
}

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    // Read the first line for the size of the array
    let n: usize = lines.next().unwrap().unwrap().trim().parse().unwrap();
    let mut a: Vec<i64> = Vec::with_capacity(n);

    // Read the array elements
    for i in 0..n {
        let mut input = lines.next().unwrap().unwrap();
        let mut value: i64 = input.trim().parse().unwrap();
        value -= (i as i64) + 1; // Adjust value as per the original C code
        a.push(value);
    }

    // Sort the array using the custom comparator
    a.sort_by(cmpnum);

    let mut all = 0;
    for i in 1..n {
        all += a[i] - a[0];
    }

    let mut sadness = all;
    let mut min = all;
    let mut level = a[0];
    let mut width: usize;
    let mut height: i64;

    for i in 0..n {
        if i + 1 < n && a[i + 1] == level {
            continue;
        }
        width = i + 1;
        if n - width * 2 <= 0 {
            break;
        }
        height = a[i + 1] - a[i];

        if height * (n - width * 2) as i64 <= sadness {
            level = a[i + 1];
            sadness -= height * (n - width * 2) as i64;
            min = sadness;
        } else {
            height = sadness / (n - width * 2) as i64;
            level += height;
            sadness -= height * (n - width * 2) as i64;
            if sadness <= min {
                min = sadness;
            }
            sadness = -(sadness - (n - width * 2) as i64);
            if sadness <= min {
                min = sadness;
            }
            break;
        }
    }

    // Print the result
    println!("{}", min);
}
