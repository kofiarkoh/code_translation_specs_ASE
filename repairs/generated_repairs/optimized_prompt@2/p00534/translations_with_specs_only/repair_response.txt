fn minimum_cost(n: usize, m: usize, seq_a: &[i32], seq_b: &[i32]) -> i32 {
    // Preconditions check
    if n < 1 || m < 1 {
        panic!("Input lengths must be greater than or equal to 1");
    }

    // Initialize the cost matrix with a high value
    let mut cost_matrix = vec![vec![i32::MAX; m + 1]; n + 1];
    
    // Set the first column to zero (base case)
    for i in 0..=n {
        cost_matrix[i][0] = 0;
    }

    // Set the first row to zero (base case)
    for j in 0..=m {
        cost_matrix[0][j] = 0;
    }

    // Fill the cost matrix based on the product of elements and their indices
    for i in 1..=n {
        for j in 1..=m {
            let cost = seq_a[i - 1] * seq_b[j - 1] * (i as i32) * (j as i32);
            cost_matrix[i][j] = cost_matrix[i][j].min(cost_matrix[i - 1][j - 1] + cost);
        }
    }

    // The minimum cost associated with the last elements of both sequences
    cost_matrix[n][m]
}

fn main() {
    // Example usage
    let n = 3;
    let m = 3;
    let seq_a = vec![1, 2, 3];
    let seq_b = vec![4, 5, 6];

    let result = minimum_cost(n, m, &seq_a, &seq_b);
    println!("{}", result);
}
