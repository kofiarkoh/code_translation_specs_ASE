fn binomial_coefficient(n: usize, r: usize) -> Vec<u64> {
    let mut result = vec![0; r + 1];
    result[0] = 1; // Initialize the result to represent 1

    for k in 1..=n {
        for j in (1..=r).rev() {
            if j <= k {
                result[j] += result[j - 1]; // Update the binomial coefficient
            }
        }
    }

    result
}
