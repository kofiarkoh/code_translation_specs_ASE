```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut student = [[0; 501]; 501];
    let mut list = [0; 501];
    
    loop {
        let mut input = String::new();
        stdin.lock().read_line(&mut input).unwrap();
        let n: i32 = input.trim().parse().unwrap_or_else(|_| {
            eprintln!("Invalid input for n");
            std::process::exit(1);
        });
        
        input.clear();
        stdin.lock().read_line(&mut input).unwrap();
        let m: i32 = input.trim().parse().unwrap_or_else(|_| {
            eprintln!("Invalid input for m");
            std::process::exit(1);
        });
        
        if n == 0 && m == 0 {
            break;
        }
        
        for i in 0..=500 {
            list[i] = 0;
            for j in 0..=500 {
                student[i][j] = 0;
            }
        }
        
        for _ in 0..m {
            input.clear();
            stdin.lock().read_line(&mut input).unwrap();
            let nums: Vec<i32> = input.split_whitespace()
                                       .map(|s| s.parse().unwrap_or_else(|_| {
                                           eprintln!("Invalid input in edge list");
                                           std::process::exit(1);
                                       }))
                                       .collect();
            if nums.len() < 2 {
                eprintln!("Not enough numbers in edge list");
                std::process::exit(1);
            }
            let p1 = nums[0];
            let p2 = nums[1];
            if p1 < 1 || p1 > n || p2 < 1 || p2 > n {
                eprintln!("Node indices out of bounds");
                std::process::exit(1);
            }
            student[p1 as usize][p2 as usize] = 1;
            student[p2 as usize][p1 as usize] = 1;
        }
        
        for i in 2..=n {
            if student[1][i as usize] == 1 {
                list[i as usize] = 1;
            }
        }
        
        for i in 2..=n {
            if student[1][i as usize] == 1 {
                for j in 2..=n {
                    if student[i as usize][j as usize] == 1 {
                        list[j as usize] = 1;
                    }
                }
            }
        }
        
        let count: i32 = (2..=n).map(|i| list[i as usize]).sum();
        println!("{}", count);
    }
}
```
