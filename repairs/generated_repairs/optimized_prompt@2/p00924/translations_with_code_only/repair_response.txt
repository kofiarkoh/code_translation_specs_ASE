use std::io::{self, BufRead};
use std::convert::TryInto;

fn solve(bit: &mut [char], runlen: &mut [i32], len: usize, runlenlen: usize, start: char) -> i32 {
    if len == 1 {
        return 0;
    }
    if runlen[0] == 0 {
        return solve(bit, &mut runlen[1..], len, runlenlen - 1, if start == '1' { '0' } else { '1' });
    }
    if len == 2 {
        if bit[0] == start {
            return 0;
        } else {
            return 1;
        }
    } else if bit[0] == start {
        runlen[0] -= 1;
        return solve(&mut bit[1..], runlen, len - 1, runlenlen, start);
    } else if start == '0' {
        if let Some(pos) = bit.iter().position(|&c| c == '0') {
            runlen[0] -= 1;
            bit[pos] = '1';
            return (pos as i32) + solve(&mut bit[1..], runlen, len - 1, runlenlen, start);
        }
    } else if start == '1' {
        if let Some(pos) = bit.iter().position(|&c| c == '1') {
            runlen[0] -= 1;
            bit[pos] = '0';
            return (pos as i32) + solve(&mut bit[1..], runlen, len - 1, runlenlen, start);
        }
    }
    0 // Fallback return, should not reach here
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let n: usize = parts.next().unwrap().parse().unwrap();
    let m: usize = parts.next().unwrap().parse().unwrap();

    let mut b: Vec<char> = Vec::with_capacity(n);
    let mut c1 = 0;

    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        let ch: char = line.chars().next().unwrap();
        b.push(ch);
        if ch == '1' {
            c1 += 1;
        }
    }

    let mut p: Vec<i32> = vec![0; m]; // Initialize with zeros
    let mut ce = 0;

    for i in 0..m {
        let line = lines.next().unwrap().unwrap();
        let value: i32 = line.parse().unwrap();
        p[i] = value; // Assign directly to the vector
        if i % 2 == 0 {
            ce += value;
        }
    }

    let mut ret = 0;

    if c1 * 2 == n {
        let mut b1 = b.clone();
        let mut p1 = p.clone();
        let ret1 = solve(&mut b, &mut p, n, m, '1');
        let ret2 = solve(&mut b1, &mut p1, n, m, '0');
        ret = ret1.min(ret2);
    } else if ce == c1.try_into().unwrap() {
        ret = solve(&mut b, &mut p, n, m, '1');
    } else {
        ret = solve(&mut b, &mut p, n, m, '0');
    }

    println!("{}", ret);
}
