```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    // Read the first line for n
    let n: usize = match lines.next() {
        Some(Ok(line)) => line.trim().parse().unwrap_or(0),
        _ => return,
    };

    // Initialize the array with a size of n
    let mut num = Vec::with_capacity(n);

    // Read the next n numbers into the array
    for _ in 0..n {
        if let Some(Ok(line)) = lines.next() {
            if let Ok(value) = line.trim().parse() {
                num.push(value);
            }
        }
    }

    let mut count = 0;

    // Iterate through the array to count the conditions
    for i in 0..(n - 2) {
        if num[i] < num[i + 1] {
            if num[i + 1] < num[i + 2] {
                count += 1;
            }
        } else if num[i] > num[i + 1] {
            if num[i + 1] > num[i + 2] {
                count += 1;
            }
        }
    }

    // Print the result
    println!("{}", count);
}
```
