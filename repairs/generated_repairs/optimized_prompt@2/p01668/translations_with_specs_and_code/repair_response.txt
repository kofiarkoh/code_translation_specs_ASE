use std::io::{self, BufRead};

const MOD_BY: i32 = 1_000_000_007;

fn add(a: i32, b: i32) -> i32 {
    let mut r = a + b;
    if r >= MOD_BY {
        r -= MOD_BY;
    }
    r
}

struct SequenceCounter {
    a: Vec<i32>,
    b: Vec<i32>,
    c: Vec<i32>,
    n: usize,
    m: usize,
    exist_pos: usize,
    memo: Vec<Vec<Vec<Vec<i32>>>>,
    number_of_number: Vec<Vec<Vec<i32>>>,
}

impl SequenceCounter {
    fn new(a: &str, b: &str, c: &str) -> Self {
        let alen = a.len();
        let blen = b.len();
        let n = blen.max(alen);
        let m = c.len();
        
        let mut a_vec = vec![0; n];
        let mut b_vec = vec![0; n];
        let mut c_vec = vec![0; m];

        for (i, ch) in a.chars().rev().enumerate() {
            if i < alen {
                a_vec[i] = ch.to_digit(10).unwrap() as i32;
            }
        }
        for (i, ch) in b.chars().rev().enumerate() {
            if i < blen {
                b_vec[i] = ch.to_digit(10).unwrap() as i32;
            }
        }
        for (i, ch) in c.chars().rev().enumerate() {
            if i < m {
                c_vec[i] = ch.to_digit(10).unwrap() as i32;
            }
        }

        let exist_pos = blen.saturating_sub(alen);

        Self {
            a: a_vec,
            b: b_vec,
            c: c_vec,
            n,
            m,
            exist_pos,
            memo: vec![vec![vec![vec![-1; 2]; 2]; 10002]; 10002],
            number_of_number: vec![vec![vec![-1; 2]; 2]; 10002],
        }
    }

    fn get_number_of_number(&mut self, pos: usize, ue_flag: bool, sita_flag: bool) -> i32 {
        if pos >= self.n {
            return 1;
        }
        if self.number_of_number[pos][ue_flag as usize][sita_flag as usize] != -1 {
            return self.number_of_number[pos][ue_flag as usize][sita_flag as usize];
        }

        let imin = if sita_flag { self.a[pos] } else { 0 };
        let imax = if ue_flag { self.b[pos] } else { 9 };
        let mut ret = 0;

        for i in imin..=imax {
            ret = add(ret, self.get_number_of_number(pos + 1, ue_flag && (i == imax), sita_flag && (i == imin)));
        }

        self.number_of_number[pos][ue_flag as usize][sita_flag as usize] = ret;
        ret
    }

    fn search(&mut self, pos: usize, seq_pos: usize, ue_flag: bool, sita_flag: bool) -> i32 {
        if seq_pos >= self.m {
            return self.get_number_of_number(pos, ue_flag, sita_flag);
        }
        if pos >= self.n {
            return 0;
        }
        if self.memo[pos][seq_pos][ue_flag as usize][sita_flag as usize] != -1 {
            return self.memo[pos][seq_pos][ue_flag as usize][sita_flag as usize];
        }

        let imin = if sita_flag { self.a[pos] } else { 0 };
        let imax = if ue_flag { self.b[pos] } else { 9 };
        let mut ret = 0;

        for i in imin..=imax {
            if seq_pos == 0 {
                ret = add(ret, self.search(pos + 1, 0, ue_flag && (i == imax), sita_flag && (i == imin)));
            }
            if (!sita_flag || i != imin || self.exist_pos <= pos) && seq_pos < self.m && self.c[seq_pos] == i {
                ret = add(ret, self.search(pos + 1, seq_pos + 1, ue_flag && (i == imax), sita_flag && (i == imin)));
            }
        }

        self.memo[pos][seq_pos][ue_flag as usize][sita_flag as usize] = ret;
        ret
    }
}

fn main() {
    let stdin = io::stdin();
    let input = stdin.lock().lines().collect::<Result<Vec<_>, _>>().unwrap();

    if input.len() < 3 {
        std::process::exit(1);
    }

    let a = &input[0];
    let b = &input[1];
    let c = &input[2];

    let mut counter = SequenceCounter::new(a, b, c);
    let result = counter.search(0, 0, true, true);
    println!("{}", result);
}
