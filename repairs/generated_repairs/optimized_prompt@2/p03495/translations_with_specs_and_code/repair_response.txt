```rust
use std::io::{self, BufRead};

const SENTINEL: i64 = 1_000_000_000_000_000_000;

fn merge(arr: &mut Vec<i64>, left: usize, mid: usize, right: usize) {
    let n1 = mid - left;
    let n2 = right - mid;

    let mut left_arr = vec![0; n1 + 1];
    let mut right_arr = vec![0; n2 + 1];

    for i in 0..n1 {
        left_arr[i] = arr[left + i];
    }
    for j in 0..n2 {
        right_arr[j] = arr[mid + j];
    }

    left_arr[n1] = SENTINEL;
    right_arr[n2] = SENTINEL;

    let mut i = 0;
    let mut j = 0;

    for k in left..right {
        if left_arr[i] <= right_arr[j] {
            arr[k] = left_arr[i];
            i += 1;
        } else {
            arr[k] = right_arr[j];
            j += 1;
        }
    }
}

fn merge_sort(arr: &mut Vec<i64>, left: usize, right: usize) {
    if left + 1 < right {
        let mid = (left + right) / 2;
        merge_sort(arr, left, mid);
        merge_sort(arr, mid, right);
        merge(arr, left, mid, right);
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let first_line = lines.next().unwrap().unwrap();
    let mut nums = first_line.split_whitespace();
    let n: usize = nums.next().unwrap().parse().unwrap();
    let k: usize = nums.next().unwrap().parse().unwrap();

    let mut a = vec![0; n];
    let mut c = vec![0; SENTINEL as usize + 1]; // Adjusted size for c

    for i in 0..n {
        let line = lines.next().unwrap().unwrap();
        a[i] = line.trim().parse().expect("Failed to parse number");
        if a[i] < 0 || a[i] as usize >= c.len() {
            panic!("Input number out of bounds");
        }
        c[a[i] as usize] += 1; // Cast to usize for indexing
    }

    merge_sort(&mut c, 1, c.len());

    let mut ans = 0;
    let mut t = 0;

    for i in 1..=n - k {
        if c[i] > 0 {
            ans += c[i];
            t += 1;
        }
    }

    println!("{}", ans);
}
```
