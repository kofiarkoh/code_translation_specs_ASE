```rust
use std::io::{self, BufRead};

const MODULO: i64 = 1_000_000_007;
const M: usize = 1 << 20;

fn mod_value(a: i64) -> i64 {
    let mut a = a % MODULO;
    if a < 0 {
        a += MODULO;
    }
    a
}

fn power(a: i64, n: i64) -> i64 {
    if n == 0 {
        1
    } else if n % 2 == 0 {
        mod_value(power(mod_value(a * a), n / 2))
    } else {
        mod_value(a * power(a, n - 1))
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let n: usize = lines.next().unwrap().unwrap().parse().expect("Failed to parse n");
    let mut a = vec![0; n + 1];
    let mut s = vec![0; n + 1];
    let mut onum = vec![0; n + 1];

    for i in 1..=n {
        a[i] = lines.next().unwrap().unwrap().parse().expect("Failed to parse a[i]");
        s[i] = s[i - 1] ^ a[i];
        onum[i] = onum[i - 1] + if s[i] == 0 { 1 } else { 0 };
    }

    let mut prev = vec![0; M];
    let mut dp = vec![0; M];
    let mut sum = vec![0; M];
    let mut osum = vec![1; M];

    for i in 1..=n {
        let x = s[i] as usize;
        if x < M {
            dp[x] = mod_value(sum[x] * (onum[i] - onum[prev[x]]) as i64 + osum[x]);
            osum[x] = mod_value(osum[x] + sum[x] * (onum[i] - onum[prev[x]]) as i64);
            sum[x] = mod_value(sum[x] + dp[x]);
            prev[x] = i;
        }
    }

    let mut ans = 0;
    if s[n] == 0 {
        ans = power(2, onum[n] - 1);
        for i in 1..M {
            ans = mod_value(ans + sum[i]);
        }
    } else if s[n] < M as i64 {
        ans = dp[s[n] as usize];
    }

    println!("{}", ans);
}
```
