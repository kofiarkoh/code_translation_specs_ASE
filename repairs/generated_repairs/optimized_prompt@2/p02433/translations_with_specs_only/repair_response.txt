```rust
use std::io::{self, BufRead};

struct Node {
    value: i32,
    next: Option<Box<Node>>,
}

struct LinkedList {
    head: Option<Box<Node>>,
    cursor: Option<*mut Node>, // Using raw pointer for cursor to allow pointer arithmetic
}

impl LinkedList {
    fn new() -> Self {
        LinkedList {
            head: None,
            cursor: None,
        }
    }

    fn insert(&mut self, value: i32) {
        let new_node = Box::new(Node {
            value,
            next: self.head.take(),
        });
        self.head = Some(new_node);
        // Move cursor to the new head
        self.cursor = self.head.as_deref_mut().map(|node| Box::as_mut(node) as *mut _);
    }

    fn move_cursor(&mut self, direction: &str) {
        match direction {
            "next" => {
                if let Some(current) = self.cursor {
                    unsafe {
                        if let Some(next_node) = (*current).next.as_deref_mut() {
                            self.cursor = Some(Box::as_mut(next_node) as *mut _);
                        } else {
                            eprintln!("Error: Attempted to move cursor beyond the end of the list.");
                            std::process::exit(1);
                        }
                    }
                } else {
                    eprintln!("Error: Cursor is not pointing to a valid node.");
                    std::process::exit(1);
                }
            }
            "prev" => {
                // Moving cursor to previous is not implemented as per specification
                eprintln!("Error: Moving cursor to previous is not supported.");
                std::process::exit(1);
            }
            _ => {
                eprintln!("Error: Invalid move command.");
                std::process::exit(1);
            }
        }
    }

    fn erase(&mut self) {
        if let Some(current) = self.cursor {
            unsafe {
                if let Some(head) = self.head.as_mut() {
                    if head.as_mut() as *mut _ == current {
                        // Erasing head
                        self.head = head.next.take();
                        self.cursor = self.head.as_deref_mut().map(|node| Box::as_mut(node) as *mut _);
                    } else {
                        // Find the previous node to erase the current node
                        let mut prev = Some(head);
                        while let Some(next) = prev.and_then(|node| node.next.as_deref_mut()) {
                            if next.as_mut() as *mut _ == current {
                                prev.as_mut().unwrap().next = next.next.take();
                                self.cursor = prev.as_deref_mut().map(|node| Box::as_mut(node) as *mut _);
                                return;
                            }
                            prev = Some(next);
                        }
                    }
                }
            }
        }
    }

    fn print(&self) {
        let mut current = self.head.as_deref();
        while let Some(node) = current {
            println!("{}", node.value);
            current = node.next.as_deref();
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut commands = reader.lines().filter_map(Result::ok);

    let command_count: usize = commands.next().unwrap().parse().unwrap();
    let mut list = LinkedList::new();

    for _ in 0..command_count {
        if let Some(command) = commands.next() {
            let parts: Vec<&str> = command.split_whitespace().collect();
            match parts.as_slice() {
                ["insert", value] => {
                    let value: i32 = value.parse().unwrap();
                    list.insert(value);
                }
                ["move", direction] => {
                    list.move_cursor(direction);
                }
                ["erase"] => {
                    list.erase();
                }
                _ => {
                    eprintln!("Error: Invalid command.");
                    std::process::exit(1);
                }
            }
        }
    }

    list.print();
}
```
