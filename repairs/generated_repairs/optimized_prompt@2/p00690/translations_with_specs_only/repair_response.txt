```rust
use std::collections::{HashMap, HashSet};
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let line = line.unwrap();
        let mut iter = line.split_whitespace();
        
        // Read number of stations and connections
        let num_stations: usize = iter.next().unwrap().parse().unwrap();
        let num_connections: usize = iter.next().unwrap().parse().unwrap();

        // Terminate if the number of stations is zero
        if num_stations == 0 {
            return;
        }

        // Create a graph representation
        let mut graph: HashMap<usize, Vec<(usize, usize)>> = HashMap::new();

        // Read connections
        for _ in 0..num_connections {
            let connection_line = reader.lines().next().unwrap().unwrap();
            let mut conn_iter = connection_line.split_whitespace();
            let station_a: usize = conn_iter.next().unwrap().parse().unwrap();
            let station_b: usize = conn_iter.next().unwrap().parse().unwrap();
            let distance: usize = conn_iter.next().unwrap().parse().unwrap();

            // Validate station identifiers
            if station_a < 1 || station_a > num_stations || station_b < 1 || station_b > num_stations {
                continue; // Ignore invalid connections
            }

            // Add undirected connections
            graph.entry(station_a).or_default().push((station_b, distance));
            graph.entry(station_b).or_default().push((station_a, distance));
        }

        // Find the longest path using DFS
        let (max_length, path) = find_longest_path(&graph, num_stations);
        
        // Output the results
        println!("{}", max_length);
        println!("{:?}", path);
    }
}

// Function to perform DFS and find the longest path
fn find_longest_path(graph: &HashMap<usize, Vec<(usize, usize)>>, num_stations: usize) -> (usize, Vec<usize>) {
    let mut max_length = 0;
    let mut longest_path = Vec::new();

    // Visited set to track visited stations
    let mut visited = HashSet::new();

    // Perform DFS from each station
    for &start_station in graph.keys() {
        let (length, path) = dfs(start_station, &graph, &mut visited, vec![start_station], 0);
        if length > max_length {
            max_length = length;
            longest_path = path;
        }
        visited.clear(); // Clear visited for the next starting station
    }

    (max_length, longest_path)
}

// DFS helper function
fn dfs(
    current: usize,
    graph: &HashMap<usize, Vec<(usize, usize)>>,
    visited: &mut HashSet<usize>,
    path: Vec<usize>,
    current_length: usize,
) -> (usize, Vec<usize>) {
    visited.insert(current);
    let mut max_length = current_length;
    let mut longest_path = path.clone();

    // Explore neighbors
    if let Some(neighbors) = graph.get(&current) {
        for &(neighbor, distance) in neighbors {
            if !visited.contains(&neighbor) {
                let mut new_path = path.clone();
                new_path.push(neighbor);
                let (length, mut updated_path) = dfs(neighbor, graph, visited, new_path, current_length + distance);
                
                if length > max_length {
                    max_length = length;
                    longest_path = updated_path;
                }
            }
        }
    }

    visited.remove(&current); // Backtrack
    (max_length, longest_path)
}
```
