```rust
use std::io::{self, BufRead};

fn main() {
    // Read five integers from standard input
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut inputs = Vec::new();

    for line in handle.lines().take(1) { // Change to take only 1 line
        if let Ok(value) = line {
            // Parse the input as integers and push to the vector
            inputs.extend(value.trim().split_whitespace().filter_map(|s| s.parse::<i32>().ok()));
        }
    }

    // Check for the first zero and output its position
    for (index, &value) in inputs.iter().enumerate() {
        if value == 0 {
            // Output the position (1-based index)
            println!("{}", index + 1);
            return; // Exit after finding the first zero
        }
    }
    
    // If no zero is found, output 0 as per specification
    println!("0");
}
```
