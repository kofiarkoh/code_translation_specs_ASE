fn smallest_subarray_length(n: usize, k: usize, arr: &[i32]) -> usize {
    // If N is less than K, return 0 immediately as per the specification.
    if n < k {
        return 0;
    }

    use std::collections::HashMap;

    let mut count = HashMap::new();
    let mut left = 0;
    let mut min_length = usize::MAX;
    let mut unique_count = 0;

    for right in 0..n {
        // Include the current element in the count
        if arr[right] >= 1 && arr[right] <= k as i32 {
            let entry = count.entry(arr[right]).or_insert(0);
            if *entry == 0 {
                unique_count += 1; // New unique integer found
            }
            *entry += 1;
        }

        // While we have all integers from 1 to K in the current window
        while unique_count == k {
            // Update the minimum length if the current window is smaller
            min_length = min_length.min(right - left + 1);

            // Try to contract the window from the left
            if let Some(entry) = count.get_mut(&arr[left]) {
                *entry -= 1;
                if *entry == 0 {
                    unique_count -= 1; // Lost a unique integer
                }
            }
            left += 1; // Move the left boundary to the right
        }
    }

    // If min_length was updated, return it; otherwise, return 0
    if min_length == usize::MAX {
        0
    } else {
        min_length
    }
}

fn main() {
    // Example usage
    let arr = [1, 2, 2, 1, 3, 1, 2, 1, 2, 3];
    let n = arr.len();
    let k = 3;
    let result = smallest_subarray_length(n, k, &arr);
    println!("{}", result); // Changed to match expected output format
}
