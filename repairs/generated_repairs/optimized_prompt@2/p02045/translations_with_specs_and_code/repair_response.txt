```rust
use std::io::{self, BufRead};

const MAX_POINTS: usize = 128;

fn nigeru(x: i32, y: i32, is_src: bool, a: i32, b: i32, c: i32, d: i32) -> i32 {
    let mut cost = 0;
    if x < a {
        cost += a - x;
    }
    if c < x {
        cost += x - c;
    }
    if y < b {
        cost += b - y;
    }
    if d < y {
        cost += y - d;
    }
    if is_src && cost > 0 {
        cost -= 1;
    }
    cost
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let line = line.unwrap();
        let mut iter = line.split_whitespace();
        let n: i32 = match iter.next().unwrap().parse() {
            Ok(num) if num > 0 => num,
            _ => return println!("1"), // Return 1 if N is not a positive integer
        };

        let mut a = 0;
        let mut b = 0;
        let mut c = 0;
        let mut d = 0;

        // Read A, B, C, D
        if let Some(first) = iter.next() {
            a = first.parse().unwrap_or_else(|_| {
                println!("1");
                std::process::exit(1);
            });
        }
        if let Some(second) = iter.next() {
            b = second.parse().unwrap_or_else(|_| {
                println!("1");
                std::process::exit(1);
            });
        }
        if let Some(third) = iter.next() {
            c = third.parse().unwrap_or_else(|_| {
                println!("1");
                std::process::exit(1);
            });
        }
        if let Some(fourth) = iter.next() {
            d = fourth.parse().unwrap_or_else(|_| {
                println!("1");
                std::process::exit(1);
            });
        }

        // Ensure we have exactly four integers for A, B, C, D
        if iter.next().is_some() {
            return println!("1");
        }

        let mut x = [0; MAX_POINTS];
        let mut y = [0; MAX_POINTS];

        // Read coordinates
        for i in 0..n {
            let line = reader.lines().next().unwrap().unwrap();
            let mut coords = line.split_whitespace();
            x[i as usize] = coords.next().unwrap().parse().unwrap_or_else(|_| {
                println!("1");
                std::process::exit(1);
            });
            y[i as usize] = coords.next().unwrap().parse().unwrap_or_else(|_| {
                println!("1");
                std::process::exit(1);
            });
        }

        let mut mikakosi = 0;

        for i in 1..n {
            let naka = (x[i as usize] - x[i as usize - 1]).abs() + (y[i as usize] - y[i as usize - 1]).abs();
            let simakaze = nigeru(x[i as usize - 1], y[i as usize - 1], true, a, b, c, d) +
                           nigeru(x[i as usize], y[i as usize], false, a, b, c, d);
            mikakosi += naka.min(simakaze);
        }

        println!("{}", mikakosi);
    }
}
```
