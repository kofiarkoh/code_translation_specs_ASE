```rust
use std::io::{self, BufRead};

fn souti(a: char, b: char) -> char {
    if a == 'T' && b == 'F' {
        'F'
    } else {
        'T'
    }
}

fn main() -> io::Result<()> {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    let n_line = lines.next().ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "No input"))??;
    let n: usize = n_line.trim().parse().map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "Invalid number"))?;

    let mut p = Vec::with_capacity(n);
    
    for _ in 0..n {
        let hoge_line = lines.next().ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "Not enough input"))??;
        let hoge: String = hoge_line.trim().chars().take(3).collect(); // Read up to 3 chars
        if let Some(first_char) = hoge.chars().next() {
            p.push(first_char);
        }
    }

    let mut notomamiko = p[0]; // Start with the first character
    for i in 1..n { // Start from the second character
        notomamiko = souti(notomamiko, p[i]);
    }

    println!("{}", notomamiko);
    Ok(())
}
```
