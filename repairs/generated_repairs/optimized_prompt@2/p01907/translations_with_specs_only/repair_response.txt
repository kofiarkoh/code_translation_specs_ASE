```rust
use std::collections::HashMap;

#[derive(Debug)]
struct TreeNode {
    cost: f64, // Change cost to f64 for precision
    depth: usize,
    children: Vec<usize>,
}

struct Tree {
    nodes: HashMap<usize, TreeNode>,
}

impl Tree {
    fn new() -> Self {
        Tree {
            nodes: HashMap::new(),
        }
    }

    fn add_edge(&mut self, parent: usize, child: usize, cost: f64) { // Change cost to f64
        self.nodes.entry(parent).or_insert(TreeNode {
            cost: 0.0, // Change to f64
            depth: 0,
            children: Vec::new(),
        }).children.push(child);
        
        self.nodes.insert(child, TreeNode {
            cost,
            depth: 0,
            children: Vec::new(),
        });
    }

    fn compute_weighted_sum(&mut self, probability_factor: f64) -> f64 {
        let mut total_cost = 0.0;
        self.calculate_depths(0, 0); // Start from root node at depth 0

        for node in self.nodes.values() {
            total_cost += node.cost * (node.depth as f64) * probability_factor; // Use f64 directly
        }

        total_cost
    }

    fn calculate_depths(&mut self, node_index: usize, current_depth: usize) {
        if let Some(node) = self.nodes.get_mut(&node_index) {
            node.depth = current_depth;
            let children = node.children.clone(); // Clone children to avoid mutable borrow issues
            for child in children {
                self.calculate_depths(child, current_depth + 1);
            }
        }
    }
}

fn main() {
    let probability_factor: f64 = 0.5; // Example input
    let edges = vec![(0, 1, 10.0), (0, 2, 20.0)]; // Example input with f64

    let mut tree = Tree::new();

    // Construct the tree from edges
    for (parent, child, cost) in edges {
        tree.add_edge(parent, child, cost);
    }

    // Compute the weighted sum
    let result = tree.compute_weighted_sum(probability_factor);

    // Output the result formatted to ten decimal places
    println!("{:.10}", result);
}
```
