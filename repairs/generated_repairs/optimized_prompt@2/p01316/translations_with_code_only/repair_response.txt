```rust
use std::io::{self, BufRead};

const MAX_N: usize = 20001;
const MAX_M: usize = 256;
const MAX_C: usize = 16;

fn min(x: i32, y: i32) -> i32 {
    if x < y { x } else { y }
}

fn max(x: i32, y: i32) -> i32 {
    if x > y { x } else { y }
}

fn f(x: i32, y: i32) -> i32 {
    let diff = x - y;
    diff * diff
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut dp = [[-1; MAX_M]; MAX_N];
    let mut c = [0; MAX_C];

    for line in reader.lines() {
        let line = line.unwrap();
        let mut iter = line.split_whitespace();
        let n: i32 = iter.next().unwrap().parse().unwrap();
        let m: i32 = iter.next().unwrap().parse().unwrap();

        if n == 0 {
            break;
        }

        for i in 0..m {
            c[i as usize] = iter.next().unwrap().parse().unwrap();
        }

        dp[0][128] = 0;

        for i in 0..n {
            let x: i32 = iter.next().unwrap().parse().unwrap();
            let prev = dp[i as usize];
            let curr = &mut dp[(i + 1) as usize];

            for j in 0..MAX_M {
                if prev[j] >= 0 {
                    for k in 0..m {
                        let t = min(255, max(0, j as i32 + c[k as usize]));
                        let y = prev[j] + f(x, t);
                        curr[t as usize] = if curr[t as usize] < 0 {
                            y
                        } else {
                            min(curr[t as usize], y)
                        };
                    }
                }
            }
        }

        let mut res = i32::MAX;
        let curr = &dp[n as usize]; // Correctly reference curr here
        for i in 0..MAX_M {
            if curr[i] >= 0 {
                res = min(res, curr[i]);
            }
        }
        println!("{}", res);
    }
}
```
