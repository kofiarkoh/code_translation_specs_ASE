fn gauss_jordan(n: usize, A: &[[i32; MAX]; MAX], C: &[i32; MAX]) -> (Vec<f64>, bool) {

    let mut X = vec![0.0; n];

        for j in 0..n {
            B[i][j] = A[i][j] as f64;

        B[i][n] = C[i] as f64;
    }

    for i in 0..n {
        let mut pivot = i;
        for j in i..n {
            if B[j][i].abs() > B[pivot][i].abs() {
                pivot = j;
            }
        }
        B.swap(i, pivot);
        if B[i][i].abs() < EPS {
            return (X, false); // error
        }

        for j in 0..=n {
            B[i][j] /= B[i][i];
        }
        for j in 0..n {
            if i != j {
                for k in i + 1..=n {
                    B[j][k] -= B[j][i] * B[i][k];
                }
            }
        }

    }
    (X, true)
}
