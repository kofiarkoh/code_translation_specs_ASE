// Aizu Vol-1 0190: Eleven Puzzle
// Rust translation of the C code provided

use std::collections::HashMap;
use std::io::{self, BufRead};

const MAX: usize = 500000;
const HASHSIZ: usize = 900001;

#[derive(Clone, Copy)]
struct HashEntry {
    s: u64,
    m: i8,
}

#[derive(Clone, Copy)]
struct QueueEntry {
    s: u64,
    b1: i8,
    b2: i8,
}

struct PuzzleSolver {
    hash: Vec<HashEntry>,
    queue: Vec<QueueEntry>,
    top: usize,
    end: usize,
}

impl PuzzleSolver {
    fn new() -> Self {
        PuzzleSolver {
            hash: vec![HashEntry { s: 0, m: -1 }; HASHSIZ + 5],
            queue: vec![QueueEntry { s: 0, b1: 0, b2: 0 }; MAX + 5],
            top: 0,
            end: 0,
        }
    }

    fn lookup(&self, s: u64) -> i8 {
        let mut p = (s % HASHSIZ as u64) as usize;
        while self.hash[p].s != 0 {
            if self.hash[p].s == s {
                return self.hash[p].m;
            }
            p = (p + 1) % HASHSIZ;
        }
        -1
    }

    fn insert(&mut self, s: u64, m: i8) -> bool {
        let mut p = (s % HASHSIZ as u64) as usize;
        while self.hash[p].s != 0 {
            if self.hash[p].s == s {
                return false;
            }
            p = (p + 1) % HASHSIZ;
        }
        self.hash[p] = HashEntry { s, m };
        true
    }

    fn swap(&self, s: u64, p1: usize, sp: usize) -> u64 {
        let n1 = (s >> ((12 - p1) << 2)) & 0xf;
        let mut s = s & !(0xf << ((12 - p1) << 2));
        s |= n1 << ((12 - sp) << 2);
        s
    }

    fn sach(&mut self, s: u64, b1: i8, b2: i8, mode: i8, mut step: i32, lim: i32) -> i32 {
        let moves: [[i8; 5]; 13] = [
            [2, -1, -1, -1, -1],
            [2, 5, -1, -1, -1],
            [0, 1, 3, 6, -1],
            [2, 7, -1, -1, -1],
            [5, -1, -1, -1, -1],
            [1, 4, 6, 9, -1],
            [2, 5, 7, 10, -1],
            [3, 6, 8, 11, -1],
            [7, -1, -1, -1, -1],
            [5, 10, -1, -1, -1],
            [6, 9, 11, 12, -1],
            [7, 10, -1, -1, -1],
            [10, -1, -1, -1, -1],
        ];

        self.top = self.end;
        self.queue[self.end] = QueueEntry { s, b1, b2 };
        self.end += 1;
        self.queue[self.end] = QueueEntry { s: 0, b1: 0, b2: 0 };
        self.end += 1;

        while self.top < self.end {
            let current = self.queue[self.top];
            let s = current.s;
            let b = [current.b1, current.b2];
            self.top += 1;

            if s == 0 {
                step += 1;
                self.queue[self.end] = QueueEntry { s: 0, b1: 0, b2: 0 };
                self.end += 1;
                continue;
            }
            if step > lim {
                break;
            }

            for i in 0..2 {
                for j in 0..5 {
                    let x = moves[b[i] as usize][j];
                    if x < 0 {
                        break;
                    }
                    if x == b[0] || x == b[1] {
                        continue;
                    }
                    let s2 = self.swap(s, x as usize, b[i] as usize);
                    if mode != 0 {
                        if self.lookup(s2) >= 0 {
                            return step;
                        }
                    } else if !self.insert(s2, step as i8) {
                        continue;
                    }
                    self.queue[self.end] = QueueEntry {
                        s: s2,
                        b1: if i == 1 { b[0] } else { x },
                        b2: if i == 1 { x } else { b[1] },
                    };
                    self.end += 1;
                }
            }
        }
        step
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut solver = PuzzleSolver::new();
    let lim = 16;

    let initial_state: u64 = 0x123456789ab0;
    solver.insert(initial_state, 0);
    solver.sach(initial_state, 0, 12, 0, 1, lim);

    for line in reader.lines() {
        let line = line.unwrap();
        let mut s: u64 = 0;
        let mut b1 = -1;
        let mut b2 = -1;

        let nums: Vec<i32> = line.split_whitespace()
            .filter_map(|x| x.parse().ok())
            .collect();

        if nums.is_empty() || (nums[0] < 0) {
            return;
        }

        for (i, &p) in nums.iter().enumerate() {
            if i == 0 && p < 0 {
                return;
            }
            if p == 0 {
                if b1 < 0 {
                    b1 = i as i8;
                } else {
                    b2 = i as i8;
                }
            }
            s = (s << 4) | (p as u64);
        }

        let mut ans = solver.lookup(s);
        if ans < 0 {
            ans = solver.sach(s, b1, b2, 1, 1, 4);
            if ans > 4 {
                ans = -1;
            } else {
                ans += lim as i32; // Change to i32
            }
        }
        if ans < 0 {
            println!("NA");
        } else {
            println!("{}", ans);
        }
    }
}
