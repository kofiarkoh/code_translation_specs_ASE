```rust
use std::io::{self, BufRead};

fn main() {
    // Create a vector to hold the input integers
    let mut numbers = Vec::new();
    
    // Read from standard input
    let stdin = io::stdin();
    let handle = stdin.lock();
    
    // Read lines until EOF
    for line in handle.lines() {
        // Handle potential errors in reading lines and parsing integers
        match line {
            Ok(line) => {
                let parts: Vec<&str> = line.trim().split_whitespace().collect();
                for part in parts {
                    match part.parse::<i32>() {
                        Ok(num) => numbers.push(num),
                        Err(_) => {
                            eprintln!("Invalid input: {}", part);
                            return;
                        }
                    }
                }
            },
            Err(e) => {
                eprintln!("Error reading line: {}", e);
                return;
            }
        }
    }
    
    // Ensure we have an even number of integers
    if numbers.len() % 2 != 0 {
        eprintln!("Expected an even number of integers, got {}", numbers.len());
        return;
    }
    
    // Compute and output the differences
    for i in (0..numbers.len()).step_by(2) {
        // Subtract the second integer from the first in each pair
        let difference = numbers[i] - numbers[i + 1];
        println!("{}", difference);
    }
}
```
