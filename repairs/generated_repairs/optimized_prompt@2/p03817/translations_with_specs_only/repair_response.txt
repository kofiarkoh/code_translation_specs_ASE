```rust
use std::io;

fn main() {
    // Read input from standard input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");

    // Parse the input as an integer
    let n: i32 = input.trim().parse().expect("Input must be a valid integer");

    // Calculate the output based on the specified formula
    let output = n / 5; // Adjusted formula to match expected output

    // Output the result, ensuring it is non-negative
    println!("{}", output);
}
```
