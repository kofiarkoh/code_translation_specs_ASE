use std::io;

type Ll = i64; // Using i64 to represent long long
type KeyType = char;

fn gcd(a: i32, b: i32) -> i32 {
    if a % b != 0 {
        gcd(b, a % b)
    } else {
        b
    }
}

fn lcm(a: i32, b: i32) -> i32 {
    (a / gcd(a, b)) * b
}

/* a[0], ..., a[n-1] の数の最大公約数 */
fn ngcd(n: usize, a: &[i32]) -> i32 {
    let mut d = a[0];
    for i in 1..n {
        if d == 1 {
            break;
        }
        d = gcd(a[i], d);
    }
    d
}

fn quicksort(a: &mut [KeyType], first: usize, last: usize) {
    let mid = (first + last) / 2;
    let x = a[mid];
    let (mut i, mut j) = (first, last);
    
    loop {
        while a[i] < x {
            i += 1;
        }
        while x < a[j] {
            j = j.wrapping_sub(1); // Wrapping to handle underflow
        }
        if i >= j {
            break;
        }
        a.swap(i, j);
        i += 1;
        j = j.wrapping_sub(1); // Wrapping to handle underflow
    }
    
    if first < i - 1 {
        quicksort(a, first, i - 1);
    }
    if j + 1 < last {
        quicksort(a, j + 1, last);
    }
}

fn is_prime(x: i32) -> i32 {
    if x <= 1 {
        return 0;
    }
    for i in 2..=((x as f64).sqrt() as i32) {
        if x % i == 0 {
            return 0;
        }
    }
    1
}

fn is_substr(s: &[KeyType], t: &[KeyType], mut i: usize) -> i32 {
    if s[i] == '\0' {
        return 0;
    }

    for j in 0..t.len() {
        if s.get(i).unwrap_or(&'\0') != &t[j] {
            return 0;
        }
        i += 1;
    }
    1
}

// factor list, number of factors, natural number
fn factorize(ftr: &mut Vec<i32>, num: &mut i32, mut n: i32) {
    let mut a = 2;
    let mut cnt = 0;

    while n >= a {
        if n % a == 0 {
            n /= a;
            ftr.push(a);
            cnt += 1;
        } else {
            a += 1;
        }
    }
    *num = cnt;
}

fn nextperm(p: &mut Vec<i32>, n: usize) -> i32 {
    let mut i = n - 1;
    p[0] = 0; // Sentinel
    while p[i] >= *p.get(i + 1).unwrap_or(&i32::MIN) {
        if i == 0 {
            return 0; // Completed
        }
        i -= 1;
    }
    let mut j = n - 1; // Corrected to n - 1
    while p[i] >= p[j] {
        j -= 1;
    }
    p.swap(i, j);
    i += 1;
    j = n - 1; // Corrected to n - 1
    while i < j {
        p.swap(i, j);
        i += 1;
        j -= 1;
    }
    1 // Not completed
}

fn comb(size1: usize, _size2: usize, v: &mut Vec<Vec<i32>>) { // Prefixed _ to avoid unused variable warning
    for i in 0..size1 {
        v[i][0] = 1;
        v[i][i] = 1;
    }
    for k in 1..size1 {
        for j in 1..k {
            v[k][j] = v[k - 1][j - 1] + v[k - 1][j];
        }
    }
}

fn abs_val(a: i32, b: i32) -> i32 {
    if a > b {
        a - b
    } else {
        b - a
    }
}

}
