```rust
fn main() {
    use std::io::{self, BufRead};

    // Read dimensions M and N from standard input
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read M and N
    let first_line = lines.next().unwrap().unwrap();
    let dimensions: Vec<usize> = first_line
        .split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();
    let (m, n) = (dimensions[0], dimensions[1]);

    // Initialize the transition count
    let mut transition_count = 0;

    // Process each row of the matrix
    for _ in 0..m {
        let line = lines.next().unwrap().unwrap();
        let row: Vec<u8> = line
            .split_whitespace()
            .map(|s| s.parse().unwrap())
            .collect();

        // Ensure the row has the correct number of columns
        assert_eq!(row.len(), n);

        // Count transitions from 0 to 1 and from 1 to 0
        for j in 0..n - 1 {
            if row[j] != row[j + 1] {
                transition_count += 1;
            }
        }
    }

    // Output the total count of transitions
    println!("{}", transition_count);
}
```
