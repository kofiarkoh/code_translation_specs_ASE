// Directed graph edge classification program
// This program analyzes a directed graph and classifies edges as "diff" or "same".

use std::collections::HashMap;

fn classify_edges(num_nodes: usize, num_edges: usize, edges: Vec<(usize, usize)>) -> Vec<String> {
    // Check for invalid input values
    if num_nodes == 0 || num_edges == 0 || edges.len() != num_edges {
        // Undefined behavior for invalid input; returning empty vector
        return vec![];
    }

    // Adjacency list representation of the graph
    let mut graph: HashMap<usize, Vec<usize>> = HashMap::new();

    // Build the graph from edges
    for (from, to) in &edges {
        // Check if the edge references valid node indices
        if *from >= num_nodes || *to >= num_nodes {
            // Undefined behavior for invalid edges; returning empty vector
            return vec![];
        }
        graph.entry(*from).or_insert_with(Vec::new).push(*to);
    }

    // Result vector to store classifications
    let mut results = Vec::with_capacity(num_edges);

    // Analyze each edge for connectivity properties
    for (from, to) in edges {
        // Check if the edge is present in the graph
        if graph.get(&from).map_or(false, |neighbors| neighbors.contains(&to)) {
            results.push("same".to_string());
        } else {
            results.push("diff".to_string());
        }
    }

    results
}

fn main() {
    // Example usage
    let num_nodes = 6;
    let num_edges = 7;
    let edges = vec![(0, 1), (1, 2), (2, 3), (3, 1), (3, 4), (4, 2)];

    let classifications = classify_edges(num_nodes, num_edges, edges);
    for classification in classifications {
        println!("{}", classification);
    }
}
