```rust
use std::cmp::Ordering;

#[derive(Debug)]
struct Item {
    weight: i32,
    value: i32,
}

impl Item {
    // Calculate the value-to-weight ratio
    fn value_to_weight_ratio(&self) -> f64 {
        if self.weight == 0 {
            return f64::INFINITY; // Handle zero weight to avoid division by zero
        }
        self.value as f64 / self.weight as f64
    }
}

fn fractional_knapsack(num_items: usize, capacity: f64, items: Vec<Item>) -> f64 {
    // Filter out items with non-positive values and non-negative weights
    let mut valid_items: Vec<Item> = items
        .into_iter()
        .filter(|item| item.value > 0 && item.weight > 0) // Changed weight check to > 0
        .collect();

    // If no valid items, return 0.0
    if valid_items.is_empty() {
        return 0.0;
    }

    // Sort items by value-to-weight ratio in descending order
    valid_items.sort_by(|a, b| {
        b.value_to_weight_ratio()
            .partial_cmp(&a.value_to_weight_ratio())
            .unwrap_or(Ordering::Equal)
    });

    let mut total_value = 0.0;
    let mut remaining_capacity = capacity;

    for item in valid_items {
        if remaining_capacity <= 0.0 {
            break; // No more capacity left
        }

        if item.weight as f64 <= remaining_capacity {
            // Take the whole item
            total_value += item.value as f64;
            remaining_capacity -= item.weight as f64;
        } else {
            // Take the fractional part of the item
            total_value += item.value as f64 * (remaining_capacity / item.weight as f64);
            remaining_capacity = 0.0; // Knapsack is full
        }
    }

    total_value
}

fn main() {
    // Example usage
    let num_items = 5; // Updated to match the input
    let capacity = 50.0; // Total weight capacity of the knapsack
    let items = vec![
        Item { weight: 10, value: 60 },
        Item { weight: 20, value: 100 },
        Item { weight: 30, value: 120 },
        Item { weight: 40, value: 240 },
        Item { weight: 50, value: 300 },
    ];

    let max_value = fractional_knapsack(num_items, capacity, items);
    println!("{:.6}", max_value); // Print the maximum value with six decimal precision
}
```
