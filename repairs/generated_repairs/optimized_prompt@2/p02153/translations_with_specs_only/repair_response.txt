use std::collections::BinaryHeap;
use std::cmp::Ordering;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Cell {
    Start,
    Goal,
    Bomb,
    Obstacle,
    Normal,
}

#[derive(Debug)]
struct Grid {
    cells: Vec<Vec<Cell>>,
    height: usize,
    width: usize,
}

impl Grid {
    fn new(height: usize, width: usize, map: Vec<Vec<char>>) -> Self {
        let mut cells = vec![vec![Cell::Normal; width]; height];
        for i in 0..height {
            for j in 0..width {
                cells[i][j] = match map[i][j] {
                    'S' => Cell::Start,
                    'G' => Cell::Goal,
                    'B' => Cell::Bomb,
                    '#' => Cell::Obstacle,
                    _ => Cell::Normal,
                };
            }
        }
        Grid { cells, height, width }
    }

    fn get_start_and_goal(&self) -> Option<(usize, usize, usize, usize)> {
        let mut start = None;
        let mut goal = None;

        for i in 0..self.height {
            for j in 0..self.width {
                match self.cells[i][j] {
                    Cell::Start => start = Some((i, j)),
                    Cell::Goal => goal = Some((i, j)),
                    _ => {}
                }
            }
        }

        if let (Some(s), Some(g)) = (start, goal) {
            Some((s.0, s.1, g.0, g.1))
        } else {
            None
        }
    }
}

#[derive(Debug, PartialEq, Eq)]
struct State {
    cost: u64,
    position: (usize, usize),
}

impl Ord for State {
    fn cmp(&self, other: &Self) -> Ordering {
        other.cost.cmp(&self.cost) // Reverse order for min-heap
    }
}

impl PartialOrd for State {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

fn dijkstra(grid: &Grid, normal_cost: u64, bomb_cost: u64) -> Result<u64, &'static str> {
    let (start_x, start_y, goal_x, goal_y) = grid.get_start_and_goal()
        .ok_or("Malformed grid: missing start or goal")?;

    let mut min_time = vec![vec![u64::MAX; grid.width]; grid.height];
    min_time[start_x][start_y] = 0;

    let mut heap = BinaryHeap::new();
    heap.push(State { cost: 0, position: (start_x, start_y) });

    let directions = [(0, 1), (1, 0), (0, -1), (-1, 0)];

    while let Some(State { cost, position }) = heap.pop() {
        let (x, y) = position;

        if (x, y) == (goal_x, goal_y) {
            return Ok(cost);
        }

        for &(dx, dy) in &directions {
            let new_x = x as isize + dx;
            let new_y = y as isize + dy;

            if new_x >= 0 && new_x < grid.height as isize && new_y >= 0 && new_y < grid.width as isize {
                let new_x = new_x as usize;
                let new_y = new_y as usize;

                let new_cost = match grid.cells[new_x][new_y] {
                    Cell::Normal => cost + normal_cost,
                    Cell::Bomb => cost + bomb_cost,
                    Cell::Obstacle => continue,
                    Cell::Start => continue,
                    Cell::Goal => cost, // No additional cost to reach the goal
                };

                if new_cost < min_time[new_x][new_y] {
                    min_time[new_x][new_y] = new_cost;
                    heap.push(State { cost: new_cost, position: (new_x, new_y) });
                }
            }
        }
    }

    Err("Target is unreachable")
}

fn main() {
    let height = 5;
    let width = 5;
    let normal_cost = 1;
    let bomb_cost = 3;
    let map = vec![
        vec!['S', '.', '.', 'B', '.'],
        vec!['.', '#', '.', '#', '.'],
        vec!['.', '#', '.', 'B', '.'],
        vec!['.', '.', '.', '.', '.'],
        vec!['.', '#', 'G', '#', '.'],
    ];

    let grid = Grid::new(height, width, map);
    
    match dijkstra(&grid, normal_cost, bomb_cost) {
        Ok(time) => println!("{}", time), // Changed output to match expected format
        Err(message) => println!("{}", message),
    }
}
