```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read the first line for n
    let n: usize = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse n");
    let mut t = vec![0; n];

    // Read the next n lines for t
    for i in 0..n {
        t[i] = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse t[i]");
    }

    // Read the next line for m
    let m: usize = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse m");
    let mut p = vec![0; m];
    let mut x = vec![0; m];

    // Read the next m lines for p and x
    for i in 0..m {
        p[i] = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse p[i]");
        x[i] = lines.next().unwrap().unwrap().trim().parse().expect("Failed to parse x[i]");
    }

    // Calculate the sum of t
    let sum: i32 = t.iter().sum();

    // Output the results
    for i in 0..m {
        // Adjusting for 0-based indexing in Rust
        let index = p[i].checked_sub(1).expect("p[i] is out of bounds");
        if index < 1 || index > n as i32 {
            eprintln!("Index out of bounds: {}", index);
            continue;
        }
        let index = (index - 1) as usize; // Convert to 0-based index
        println!("{}", sum - t[index] + x[i]);
    }
}
```
