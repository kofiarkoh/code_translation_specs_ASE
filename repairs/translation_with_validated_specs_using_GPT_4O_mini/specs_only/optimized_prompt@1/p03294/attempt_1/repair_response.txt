```rust
use std::io::{self, BufRead};

fn main() {
    // Read the first line for the number of integers, N
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    // Read N, which must be a positive integer
    let n: usize = lines.next().unwrap().unwrap().parse().unwrap();

    // Initialize the sum accumulator
    let mut sum = 0;

    // Read N integers and accumulate the sum of each decremented by one
    for _ in 0..n {
        let input: i32 = lines.next().unwrap().unwrap().parse().unwrap();
        sum += input - 1; // Decrement each input by one before summing
    }

    // Output the result
    println!("{}", sum);
}
```
