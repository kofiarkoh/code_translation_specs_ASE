use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let line = line.expect("Failed to read line");
        let mut values: Vec<f64> = line.split_whitespace()
            .filter_map(|s| s.parse::<f64>().ok())
            .collect();

        // The first value indicates the number of subsequent values
        let count = values.get(0).map_or(0, |&v| v as usize);

        // If the count is zero, we terminate gracefully
        if count == 0 {
            break;
        }

        // Ensure we have the correct number of values
        if values.len() != count + 1 { // +1 for the count itself
            eprintln!("Input does not match the expected number of values");
            return; // Exit gracefully instead of panicking
        }

        // Remove the first value which is the count
        values.remove(0);

        let mut outlier_index = count - 1; // Default to the last index
        let mut found_outlier = false;

        for i in 0..count {
            let expected_value = calculate_expected_value(&values, i);
            if (expected_value - values[i]).abs() > 0.1 {
                outlier_index = i;
                found_outlier = true;
                break; // Stop at the first outlier found
            }
        }

        // Output the index of the outlier or the last index if none found
        println!("{}", outlier_index);
    }
}

// Function to calculate the expected value using Lagrange interpolation
fn calculate_expected_value(values: &[f64], exclude_index: usize) -> f64 {
    let mut expected_value = 0.0;
    let n = values.len();

    for i in 0..n {
        if i != exclude_index {
            let mut term = values[i];
            for j in 0..n {
                if j != exclude_index && j != i {
                    term *= (0.0 - values[j]) / (values[i] - values[j]);
                }
            }
            expected_value += term;
        }
    }

    expected_value
}
