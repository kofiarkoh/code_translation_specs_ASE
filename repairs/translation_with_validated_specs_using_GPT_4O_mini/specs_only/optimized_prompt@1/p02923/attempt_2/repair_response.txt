fn longest_non_increasing_subsequence(n: usize, sequence: Vec<i32>) -> i32 {
    // If n is zero, return 0 as there are no elements to evaluate.
    if n == 0 {
        return 0;
    }

    let mut max_length = 1; // At least one element means a length of 1
    let mut current_length = 1; // Start with the first element

    // Iterate through the sequence starting from the second element
    for i in 1..n {
        // Check if the current element is less than or equal to the previous one
        if sequence[i] <= sequence[i - 1] {
            current_length += 1; // Increment the current non-increasing length
        } else {
            // Update max_length if the current non-increasing length is greater
            if current_length > max_length {
                max_length = current_length;
            }
            current_length = 1; // Reset current length for a new subsequence
        }
    }

    // Final check to update max_length in case the longest subsequence ends at the last element
    if current_length > max_length {
        max_length = current_length;
    }

    max_length // Return the maximum length found
}

fn main() {
    // Example usage
    let n = 5;
    let sequence = vec![5, 4, 3, 2, 1];
    let result = longest_non_increasing_subsequence(n, sequence);
    println!("{}", result); // Output the result
}
