```rust
use std::io;

fn compute_value(a: i32, p: i32) -> i32 {
    // Compute the result based on the specified formula
    (a * 3 + p) / 2 // Integer division in Rust automatically rounds down
}

fn main() {
    // Read two integers from standard input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    
    // Split the input into parts and parse them as integers
    let mut parts = input.split_whitespace();
    let a: i32 = parts.next().expect("Expected an integer").parse().expect("Failed to parse integer");
    let p: i32 = parts.next().expect("Expected an integer").parse().expect("Failed to parse integer");
    
    // Compute the result using the specified formula
    let result = compute_value(a, p);
    
    // Output the result
    println!("{}", result);
}
```
