struct SegmentTree {
    n: usize,
    data: Vec<i32>,
    lazy: Vec<i32>,
}

impl SegmentTree {
    fn new(values: Vec<i32>) -> Self {
        let n = values.len();
        let mut data = vec![0; 4 * n]; // Allocate enough space for the segment tree
        let lazy = vec![0; 4 * n]; // Lazy propagation array
        let mut tree = SegmentTree { n, data, lazy };
        tree.build(0, 0, n - 1, &values);
        tree
    }

    fn build(&mut self, node: usize, start: usize, end: usize, values: &[i32]) {
        if start == end {
            self.data[node] = values[start]; // Leaf node
        } else {
            let mid = (start + end) / 2;
            self.build(2 * node + 1, start, mid, values);
            self.build(2 * node + 2, mid + 1, end, values);
            self.data[node] = self.data[2 * node + 1].max(self.data[2 * node + 2]); // Internal node
        }
    }

    fn update(&mut self, idx: usize, value: i32) {
        self.update_range(0, 0, self.n - 1, idx, value);
    }

    fn update_range(&mut self, node: usize, start: usize, end: usize, idx: usize, value: i32) {
        if start == end {
            self.data[node] = value; // Update the leaf node
        } else {
            let mid = (start + end) / 2;
            if idx <= mid {
                self.update_range(2 * node + 1, start, mid, idx, value);
            } else {
                self.update_range(2 * node + 2, mid + 1, end, idx, value);
            }
            self.data[node] = self.data[2 * node + 1].max(self.data[2 * node + 2]); // Update internal node
        }
    }

    fn query(&self) -> i32 {
        self.query_range(0, 0, self.n - 1)
    }

    fn query_range(&self, node: usize, start: usize, end: usize) -> i32 {
        self.data[node] // Return the maximum value in the range
    }
}

fn main() {
    let n = 5; // Example size
    let values = vec![1, 3, 5, 7, 9]; // Example initialization
    let mut segment_tree = SegmentTree::new(values);

    // Output the maximum value after initialization
    println!("{}", segment_tree.query());

    // Example queries
    segment_tree.update(2, 6); // Update index 2 to value 6
    println!("{}", segment_tree.query()); // Output the maximum value after update
}
