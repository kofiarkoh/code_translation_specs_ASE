```rust
use std::io::{self, Write};

fn main() {
    // Read three long double values from standard input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    
    // Split the input into three parts and parse them as f64 (closest to long double in Rust)
    let values: Vec<f64> = input
        .trim()
        .split_whitespace()
        .map(|s| s.parse().expect("Invalid input format"))
        .collect();

    // Ensure exactly three inputs are provided
    if values.len() != 3 {
        eprintln!("Error: Exactly three inputs are required.");
        return;
    }

    let a = values[0];
    let b = values[1];
    let c = values[2];

    // Calculate d
    let d = c - a - b;

    // Check the conditions for real and positive roots
    if d > 0.0 && d * d > 4.0 * a * b {
        println!("Yes");
    } else {
        println!("No");
    }
}
```
