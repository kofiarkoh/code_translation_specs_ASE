fn count_distinct_pairs(n: usize, sequence: Vec<i32>) -> i32 {
    // Ensure the input size matches the expected number of integers
    if sequence.len() != n {
        return 0; // This case should not happen as per the specification
    }

    let mut count = 0;
    let mut i = 0;

    while i < n - 1 {
        // Check for a pair of consecutive identical integers
        if sequence[i] == sequence[i + 1] {
            count += 1; // Increment the count for the found pair
            i += 2; // Skip the next integer
        } else {
            i += 1; // Move to the next integer
        }
    }

    count // Return the total count of distinct pairs
}
