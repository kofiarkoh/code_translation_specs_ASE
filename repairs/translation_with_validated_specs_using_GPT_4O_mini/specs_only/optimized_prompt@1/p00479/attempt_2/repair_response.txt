fn process_pairs(n: i32, k: usize, pairs: &[(i32, i32)]) -> Vec<i32> {
    // Ensure preconditions are met
    assert!(n > 0, "n must be a positive integer");
    assert!(k >= 0, "k must be a non-negative integer");
    assert!(pairs.len() == k, "Number of pairs must match k");

    let mut results = Vec::with_capacity(k);

    for &(ai, bi) in pairs {
        // Ensure each ai and bi is within the range [1, n]
        assert!(1 <= ai && ai <= n, "ai must be within the range [1, n]");
        assert!(1 <= bi && bi <= n, "bi must be within the range [1, n]");

        // Adjust ai and bi based on the specified rules
        let adjusted_ai = if n + 1 - ai < ai { n + 1 - ai } else { ai };
        let adjusted_bi = if n + 1 - bi < bi { n + 1 - bi } else { bi };

        // Compute the output based on the adjusted values
        let output = if adjusted_ai > adjusted_bi {
            (adjusted_bi - 1) % 3 + 1
        } else {
            (adjusted_ai - 1) % 3 + 1
        };

        results.push(output);
    }

    results
}

fn main() {
    // Example usage
    let n = 5;
    let k = 3;
    let pairs = [(1, 2), (3, 4), (5, 1)];
    let results = process_pairs(n, k, &pairs);
    for result in results {
        println!("{}", result);
    }
}
