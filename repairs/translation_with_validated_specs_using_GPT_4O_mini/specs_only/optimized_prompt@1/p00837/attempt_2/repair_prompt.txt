struct Library {

struct Desk {

    usage_order: VecDeque<i32>, // Tracks the order of book usage for LRU
}

impl Library {

        let desks = vec![Desk::new(c); m];
        Library { desks }
    }

    fn process_requests(&mut self, requests: Vec<Vec<i32>>) -> usize {
        let mut total_assignments = 0;
        for request in requests {
            let n = request[0]; // Number of students

                let book_id = request[i];
                total_assignments += self.assign_book(book_id);
            }
        }
        total_assignments
    }

    fn assign_book(&mut self, book_id: i32) -> usize {
        // Check if the book is already assigned
        for desk in &mut self.desks {
            if desk.books.contains_key(&book_id) {
                // Update usage order
                desk.update_usage(book_id);
                return 0; // No new assignment
            }
        }

        // Try to assign the book to a desk
        for desk in &mut self.desks {
            if desk.books.len() < desk.capacity {
                desk.add_book(book_id);
                return 1; // New assignment made
            }
        }

        // All desks are full, need to replace a book
        self.replace_book(book_id)
    }

    fn replace_book(&mut self, book_id: i32) -> usize {
        // Find the least recently used desk
        let (desk_index, _) = self.desks.iter_mut()
            .enumerate()
            .min_by_key(|(_, desk)| desk.last_used())
            .unwrap();

        self.desks[desk_index].replace_lru_book(book_id);
        1 // New assignment made
    }
}

impl Desk {
    fn new(capacity: usize) -> Self {
        Desk {
            capacity,
            books: HashMap::new(),
            usage_order: VecDeque::new(),
        }
    }

    fn add_book(&mut self, book_id: i32) {
        self.books.insert(book_id, 0); // Initial timestamp
        self.usage_order.push_back(book_id);
        self.update_usage(book_id);
    }

    fn update_usage(&mut self, book_id: i32) {
        if let Some(&mut timestamp) = self.books.get_mut(&book_id) {
            timestamp += 1; // Increment timestamp for usage
        }
        // Move the book to the back of the usage order
        if let Some(pos) = self.usage_order.iter().position(|&id| id == book_id) {

    }
