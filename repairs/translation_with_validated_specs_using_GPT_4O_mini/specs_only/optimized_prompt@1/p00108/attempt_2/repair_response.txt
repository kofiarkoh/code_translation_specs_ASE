```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let line = line.unwrap();
        let mut numbers: Vec<usize> = line.split_whitespace()
            .filter_map(|s| s.parse().ok())
            .collect();

        // The first number indicates the size of the sequence
        let n = numbers[0];
        if n == 0 {
            break; // Termination condition
        }

        // Ensure we only process the correct number of elements
        if numbers.len() != n + 1 {
            continue; // Skip invalid input
        }

        // Process the sequence
        let original_sequence = &numbers[1..=n];
        let mut transformed_sequence = original_sequence.to_vec();
        let mut transformations = 0;

        loop {
            // Create a count array to count occurrences
            let mut counts = vec![0; 101]; // Since values are between 0 and 100

            // Count occurrences of each number in the current sequence
            for &num in &transformed_sequence {
                counts[num] += 1;
            }

            // Create the new transformed sequence based on counts
            let new_transformed_sequence: Vec<usize> = original_sequence.iter()
                .map(|&num| counts[num])
                .collect();

            transformations += 1;

            // Check if the new sequence is stable (no change)
            if new_transformed_sequence == transformed_sequence {
                break; // Stable state reached
            }

            // Update the transformed sequence for the next iteration
            transformed_sequence = new_transformed_sequence;
        }

        // Output the number of transformations and the final transformed sequence
        println!("{}", transformations);
        println!("{:?}", transformed_sequence);
    }
}
```
