```rust
use std::cmp::Ordering;

#[derive(Debug, Clone)]
struct Card {
    character: char,
    value: i32,
    original_index: usize, // To check stability
}

fn quicksort(cards: &mut [Card]) {
    if cards.len() <= 1 {
        return;
    }
    let pivot_index = partition(cards);
    quicksort(&mut cards[0..pivot_index]);
    quicksort(&mut cards[pivot_index + 1..]);
}

fn partition(cards: &mut [Card]) -> usize {
    let pivot = cards[cards.len() - 1].value;
    let mut i = 0;
    for j in 0..cards.len() - 1 {
        if cards[j].value < pivot {
            cards.swap(i, j);
            i += 1;
        }
    }
    cards.swap(i, cards.len() - 1);
    i
}

fn is_stable(cards: &[Card], sorted_cards: &[Card]) -> bool {
    for (i, card) in sorted_cards.iter().enumerate() {
        if let Some(original_index) = cards.iter().position(|c| c.character == card.character && c.value == card.value) {
            if original_index != i {
                return false; // Found a card that is not in original order
            }
        }
    }
    true
}

fn main() {
    let n: usize = 5; // Example input size
    let mut cards = vec![
        Card { character: 'A', value: 3, original_index: 0 },
        Card { character: 'B', value: 1, original_index: 1 },
        Card { character: 'C', value: 2, original_index: 2 },
        Card { character: 'D', value: 3, original_index: 3 },
        Card { character: 'E', value: 1, original_index: 4 },
    ];

    // Sort the cards using quicksort
    quicksort(&mut cards);

    // Check stability
    let stable = is_stable(&cards, &cards);

    // Output results
    println!("{}", if stable { "Stable" } else { "Not stable" });
    for card in cards {
        println!("{} {}", card.character, card.value);
    }
}
```
