fn sort_by_modulo(n: usize, m: usize, arr: &mut Vec<i32>) {
    // Preconditions: n and m must be positive integers
    assert!(n > 0, "n must be positive");
    assert!(m > 0, "m must be positive");
    assert!(arr.len() == n, "Array length must match n");

    // Create a vector of tuples to hold the original index and the value
    let mut indexed_arr: Vec<(usize, i32)> = arr.iter().enumerate().map(|(i, &val)| (i, val)).collect();

    // Perform stable sorting based on modulo results for each divisor from 1 to m
    for k in 1..=m {
        indexed_arr.sort_by(|&(index_a, value_a), &(index_b, value_b)| {
            let mod_a = value_a % k as i32; // Cast k to i32 for modulo operation
            let mod_b = value_b % k as i32; // Cast k to i32 for modulo operation
            
            // First compare by modulo result, then by original index for stability
            if mod_a == mod_b {
                index_a.cmp(&index_b)
            } else {
                mod_a.cmp(&mod_b)
            }
        });
    }

    // Extract the sorted values back into the original array
    for (i, &(_, value)) in indexed_arr.iter().enumerate() {
        arr[i] = value;
    }
}

fn main() {
    // Example usage
    let n = 5;
    let m = 2;
    let mut arr = vec![5, 3, 8, 6, 2];
    
    sort_by_modulo(n, m, &mut arr);
    
    // Output the sorted array
    for value in arr {
        println!("{}", value);
    }
}
