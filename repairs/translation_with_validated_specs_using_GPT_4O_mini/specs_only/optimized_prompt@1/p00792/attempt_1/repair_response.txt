```rust
use std::collections::HashMap;

#[derive(Debug, Clone, Copy)]
enum Direction {
    Up,
    Down,
    Left,
    Right,
}

#[derive(Debug)]
struct Mirror {
    orientation: char,
    position: (i32, i32),
}

struct Point {
    position: (i32, i32),
    direction: Direction,
}

impl Point {
    fn new() -> Self {
        Point {
            position: (50, 0),
            direction: Direction::Right,
        }
    }

    fn move_point(&mut self, mirrors: &HashMap<(i32, i32), char>) {
        loop {
            match self.direction {
                Direction::Up => self.position.1 += 1,
                Direction::Down => self.position.1 -= 1,
                Direction::Left => self.position.0 -= 1,
                Direction::Right => self.position.0 += 1,
            }

            // Check for boundary conditions
            if self.position.0 < 0 || self.position.0 > 800 || self.position.1 < 0 || self.position.1 > 800 {
                break; // Stop if out of bounds
            }

            // Check for mirror interaction
            if let Some(&orientation) = mirrors.get(&self.position) {
                self.direction = match (self.direction, orientation) {
                    (Direction::Right, 'x') => Direction::Left,
                    (Direction::Left, 'x') => Direction::Right,
                    (Direction::Up, 'y') => Direction::Down,
                    (Direction::Down, 'y') => Direction::Up,
                    _ => self.direction, // No change if not aligned
                };
            }

            // Check for exit condition
            if self.position == (100, 50) && self.direction == Direction::Right {
                break; // Exit condition met
            }
        }
    }
}

fn main() {
    let n: i32 = 5; // Example input, replace with actual input handling
    if n < 0 {
        return; // Graceful termination on negative input
    }

    let mut mirrors = HashMap::new();
    // Example mirror placements, replace with actual input handling
    mirrors.insert((100, 0), 'x');
    mirrors.insert((200, 0), 'y');
    mirrors.insert((300, 0), 'x');

    let mut point = Point::new();
    point.move_point(&mirrors);

    println!("{:?}", point.position); // Output final coordinates
}
```
