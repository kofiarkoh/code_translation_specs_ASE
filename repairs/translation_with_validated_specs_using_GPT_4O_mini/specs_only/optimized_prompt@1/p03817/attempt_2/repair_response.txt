fn compute_value(n: i64) -> i64 {
    // Calculate the quotient and remainder of n divided by 11
    let quotient = n / 11;
    let remainder = n % 11;

    // Determine the output based on the value of the remainder
    match remainder {
        0 => 2 * quotient,          // If divisible by 11
        1..=6 => 2 * quotient + 1,  // Remainder between 1 and 6
        7..=10 => 2 * quotient + 2, // Remainder between 7 and 10
        _ => unreachable!(),         // This case will never occur due to the nature of modulus
    }
}

fn main() {
    // Example usage of compute_value function
    let n: i64 = 25; // Example input
    let result = compute_value(n);
    println!("The computed value for {} is {}", n, result);
}
