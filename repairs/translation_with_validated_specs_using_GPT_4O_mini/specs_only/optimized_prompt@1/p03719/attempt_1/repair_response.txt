fn main() {
    // Example input values; in a real program, these would be read from user input
    let a: i32 = 10; // Lower bound of the range
    let b: i32 = 20; // Upper bound of the range
    let c: i32 = 15; // Number to check

    // Check if c is within the inclusive range defined by a and b
    if c >= a && c <= b {
        println!("Yes"); // Output "Yes" if c is within the range
    } else {
        println!("No"); // Output "No" if c is outside the range
    }
}
