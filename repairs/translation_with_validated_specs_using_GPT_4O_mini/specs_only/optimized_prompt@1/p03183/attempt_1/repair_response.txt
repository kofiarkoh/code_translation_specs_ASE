struct Item {
    weight: usize,
    size: usize,
    value: usize,
}

fn max_value(items: &[Item], max_size: usize) -> usize {
    // Create a dynamic programming array to store the maximum values for each size
    let mut dp = vec![0; max_size + 1];

    // Iterate over each item
    for item in items {
        // Update the dp array in reverse to avoid overwriting values we still need to use
        for size in (item.size..=max_size).rev() {
            // Update the maximum value for the current size
            dp[size] = dp[size].max(dp[size - item.size] + item.value);
        }
    }

    // The maximum obtainable value is found at dp[max_size]
    dp[max_size]
}

fn main() {
    // Example usage
    let items = vec![
        Item { weight: 1, size: 5000, value: 10 },
        Item { weight: 2, size: 10000, value: 20 },
        Item { weight: 3, size: 15000, value: 30 },
    ];
    let max_size = 20000;

    let result = max_value(&items, max_size);
    println!("{}", result);
}
