            size: n,
        }
    }

    fn toggle(&mut self, x: usize, y: usize) {
        // Toggle the cell at (x, y)
        self.data[x][y] = if self.data[x][y] == '0' { '1' } else { '0' };
    }

    fn is_symmetric(&self) -> bool {
        for x in 0..self.size {
