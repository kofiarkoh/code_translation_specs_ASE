fn calculate_polygon_area(vertices: &[(i32, i32)]) -> i64 {
    let mut area = 0;
    let n = vertices.len();
    
    for i in 0..n {
        let (x1, y1) = vertices[i];
        let (x2, y2) = vertices[(i + 1) % n]; // Wrap around to the first vertex
        area += x1 as i64 * y2 as i64 - x2 as i64 * y1 as i64;
    }

}
