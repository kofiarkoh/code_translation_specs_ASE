```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let input = stdin.lock().lines().filter_map(Result::ok).collect::<Vec<String>>();

    // Parse the first integer to determine the number of subsequent integers
    let n: usize = match input.get(0) {
        Some(line) => line.parse().unwrap_or(0),
        None => {
            println!("NO COLOR");
            return;
        }
    };

    let mut numbers = Vec::new();

    // Read the integers until we encounter a zero
    for line in input.iter().skip(1) {
        let num: i32 = match line.parse() {
            Ok(n) => n,
            Err(_) => continue, // Ignore invalid integers
        };
        if num == 0 {
            break;
        }
        numbers.push(num);
    }

    // Check if we have valid numbers to process
    if numbers.len() != n {
        println!("NO COLOR");
        return;
    }

    let mut max_count = 0;
    let mut current_count = 1;
    let mut current_number = numbers[0];

    // Iterate through the numbers to find the longest contiguous segment
    for i in 1..numbers.len() {
        if numbers[i] == current_number {
            current_count += 1;
        } else {
            if current_count > max_count {
                max_count = current_count;
            }
            current_number = numbers[i];
            current_count = 1;
        }
    }
    // Check the last segment
    if current_count > max_count {
        max_count = current_count;
    }

    // Determine the threshold for output
    if max_count > n / 2 {
        println!("{}", current_number);
    } else {
        println!("NO COLOR");
    }
}
```
