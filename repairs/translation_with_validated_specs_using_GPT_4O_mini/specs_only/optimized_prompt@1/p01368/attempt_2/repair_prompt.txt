fn floyd_warshall(distances: &mut [[usize; MAX_LOCATIONS]; MAX_LOCATIONS], n: usize) {
    for k in 0..n {
        for i in 0..n {
            for j in 0..n {
                if distances[i][j] > distances[i][k] + distances[k][j] {
                    distances[i][j] = distances[i][k] + distances[k][j];
                }
            }
        }
    }
}
