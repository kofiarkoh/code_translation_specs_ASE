// Converts a string of digits into a vector of digits (in reverse order)
fn string_to_digit_array(s: &str) -> Vec<u8> {
    s.chars().rev().map(|c| c.to_digit(10).unwrap() as u8).collect()
}

// Adds two digit arrays and returns the result as a new digit array
