```rust
use std::io::{self, Write};

fn main() {
    // Read the number of cases from standard input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let num_cases: usize = input.trim().parse().expect("Input must be a positive integer");

    // Process each case
    for case_number in 1..=num_cases {
        // Read the input integer for the current case
        input.clear();
        io::stdin().read_line(&mut input).expect("Failed to read line");
        let mut value: i32 = input.trim().parse().expect("Input must be a valid integer");

        // Print the case header
        println!("Case {}:", case_number);

        // Compute and print the sequence of ten derived values
        for _ in 0..10 {
            // Perform the calculation: square the value, divide by 100, and take modulo 10000
            value = (value * value) / 100 % 10000;
            println!("{}", value);
        }
    }
}
```
