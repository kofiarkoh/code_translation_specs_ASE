use std::io::{self, BufRead};

const QMAX: usize = 150000;

#[derive(Clone, Copy)]
struct QUE {
    t: i32,
    r: i16,
    c: i16,
}

struct MinHeap {
    que: Vec<QUE>,
    qsize: usize,
}

impl MinHeap {
    fn new() -> Self {
        MinHeap {
            que: Vec::with_capacity(QMAX),
            qsize: 0,
        }
    }

    fn parent(i: usize) -> usize {
        i >> 1
    }

    fn left(i: usize) -> usize {
        i << 1
    }

    fn right(i: usize) -> usize {
        (i << 1) + 1
    }

    fn min_heapify(&mut self, i: usize) {
        let l = Self::left(i);
        let r = Self::right(i);
        let mut min = i;

        if l < self.qsize && self.que[l].t < self.que[min].t {
            min = l;
        }
        if r < self.qsize && self.que[r].t < self.que[min].t {
            min = r;
        }
        if min != i {
            self.que.swap(i, min);
            self.min_heapify(min);
        }
    }

    fn deq(&mut self) {
        if self.qsize == 0 {
            return;
        }
        self.que[0] = self.que[self.qsize - 1];
        self.qsize -= 1;
        self.min_heapify(0);
    }

    fn enq(&mut self, r: i16, c: i16, t: i32) {
        let i = self.qsize;
        self.qsize += 1;
        self.que.push(QUE { r, c, t });
        let mut i = i;

        while i > 0 && self.que[Self::parent(i)].t > self.que[i].t {
            self.que.swap(i, Self::parent(i));
            i = Self::parent(i);
        }
    }
}

fn in_number() -> i32 {
    let stdin = io::stdin();
    let mut input = stdin.lock().lines().next().unwrap().unwrap();
    let mut n = 0;
    let mut c = input.chars().next().unwrap() as u8;

    while c >= b'0' {
        n = (n << 3) + (n << 1) + (c & 0xf) as i32;
        input.remove(0);
        if input.is_empty() {
            break;
        }
        c = input.chars().next().unwrap() as u8;
    }
    n
}

fn main() {
    let mut l = [[0; 501]; 501];
    let mut level = [[0; 100001]; 2];
    let mut lsz = [0; 2];
    let mv: [(i16, i16); 4] = [(-1, 0), (0, 1), (1, 0), (0, -1)];
    let mut mk = [[0; 501]; 501];

    let mut input = io::stdin().lock().lines();
    while let Ok(r) = input.next().unwrap().unwrap().parse::<i32>() {
        if r == 0 {
            break;
        }
        for i in 0..2 {
            let w = in_number();
            let h = in_number();
            let x = in_number() - 1;
            let y = in_number() - 1;

            for r in 0..h {
                for c in 0..w {
                    l[r as usize][c as usize] = in_number();
                }
            }

            mk = [[0; 501]; 501];
            let mut heap = MinHeap::new();
            heap.qsize = 0;
            level[i][0] = 0;
            lsz[i] = 1;
            heap.enq(y, x, l[y as usize][x as usize]);
            mk[y as usize][x as usize] = 1;

            let mut max = 0;
            while heap.qsize > 0 {
                let QUE { t, r, c } = heap.que[0];
                heap.deq();
                if t > max {
                    max = t;
                }
                level[i][lsz[i]] = max;
                lsz[i] += 1;

                for j in 0..4 {
                    let r2 = r + mv[j].0;
                    let c2 = c + mv[j].1;
                    if r2 < 0 || r2 >= h || c2 < 0 || c2 >= w {
                        continue;
                    }
                    if mk[r2 as usize][c2 as usize] != 0 {
                        continue;
                    }
                    mk[r2 as usize][c2 as usize] = 1;
                    heap.enq(r2, c2, l[r2 as usize][c2 as usize]);
                }
            }
        }

        let mut ans = i32::MAX;
        for i in 0..=r {
            let j = r - i;
            if i < lsz[0] && j < lsz[1] {
                let t = level[0][i] + level[1][j];
                if t < ans {
                    ans = t;
                }
            }
        }
        println!("{}", ans);
    }
}
