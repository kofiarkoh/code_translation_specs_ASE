use std::io::{self, Write};

const MOD: i64 = 1_000_000_007;

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let k: i64 = input.trim().parse().expect("Failed to parse input");

    if k >= 50 {
        let n = 50;
        let mut a = vec![0; n as usize];
        let rem = k / 50;
        let x = 49 + rem;
        let t = k % 50;

        for i in 0..50 {
            a[i as usize] = x;
        }
        for i in 0..t {
            a[i as usize] += 50;
            for j in 0..50 {
                if j != i {
                    a[j as usize] -= 1;
                }
            }
        }
        println!("50");
        for i in 0..50 {
            print!("{} ", a[i as usize]);
        }
        io::stdout().flush().expect("Failed to flush stdout");
    } else if k >= 2 {
        let n = k;
        println!("{}", n);
        for _ in 0..n {
            print!("{} ", n);
        }
        io::stdout().flush().expect("Failed to flush stdout");
    } else if k == 0 {
        println!("4");
        println!("3 3 3 3");
    } else {
        println!("3");
        println!("1 0 3");
    }
}
