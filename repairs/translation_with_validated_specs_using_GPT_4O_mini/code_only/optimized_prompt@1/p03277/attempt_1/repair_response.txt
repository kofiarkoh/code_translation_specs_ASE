use std::io::{self, BufRead};

type Int64 = i64;

fn init_bit(bit: &mut Vec<i32>) {
    let n = bit[0];
    for i in 1..=n {
        bit[i as usize] = 0;
    }
}

fn inc(bit: &mut Vec<i32>, x: i32) {
    let n = bit[0];
    let mut idx = x;
    while idx <= n {
        bit[idx as usize] += 1;
        idx += idx & -idx; // This is a bit-level trick to navigate the Fenwick Tree
    }
}

fn get_sum(bit: &Vec<i32>, x: i32) -> i32 {
    let mut res = 0;
    let mut idx = x;
    while idx > 0 {
        res += bit[idx as usize];
        idx -= idx & -idx; // This is a bit-level trick to navigate the Fenwick Tree
    }
    res
}

fn count(m: i32, a: &Vec<i32>, n: usize) -> Int64 {
    static mut BIT: Option<Vec<i32>> = None; // Using Option to handle uninitialized state
    unsafe {
        if BIT.is_none() {
            BIT = Some(vec![0; 2 * n as usize + 2]);
            BIT.as_mut().unwrap()[0] = (2 * n + 1) as i32;
        }
        let bit = BIT.as_mut().unwrap();
        init_bit(bit);
        let mut cnt: Int64 = 0;
        let mut sum = n as i32 + 1;
        inc(bit, sum);
        for &value in a.iter() {
            sum += if value < m { -1 } else { 1 };
            cnt += get_sum(bit, sum);
            inc(bit, sum);
        }
        cnt
    }
}

fn run() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();
    
    let n: usize = lines.next().unwrap().unwrap().parse().unwrap();
    let mut a = Vec::with_capacity(n);
    for _ in 0..n {
        let value: i32 = lines.next().unwrap().unwrap().parse().unwrap();
        a.push(value);
    }
    
    let mut l = 1;
    let r = 1_000_000_001; // 1 billion + 1
    let mut r = r; // Shadowing to allow mutation
    while r - l > 1 {
        let m = (l + r) / 2;
        let cnt = count(m, &a, n);
        if 4 * cnt >= (n as Int64) * (n as Int64 + 1) {
            l = m;
        } else {
            r = m;
        }
    }
    println!("{}", l);
}

fn main() {
    run();
}
