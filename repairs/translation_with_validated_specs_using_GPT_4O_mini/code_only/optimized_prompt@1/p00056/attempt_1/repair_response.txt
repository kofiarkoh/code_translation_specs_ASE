```rust
use std::io::{self, BufRead};

const MAX: usize = 50001;

fn make_prime_table(table: &mut [u8; MAX]) {
    for i in 0..MAX {
        table[i] = 0;
    }

    for i in 2..MAX {
        if table[i] == 0 {
            let mut j = 2;
            while i * j < MAX {
                table[i * j] = 1;
                j += 1;
            }
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut ptable = [0u8; MAX];
    make_prime_table(&mut ptable);

    for line in handle.lines() {
        let n: i32 = match line {
            Ok(line) => line.trim().parse().unwrap_or(0),
            Err(_) => break,
        };

        if n == 0 {
            break;
        } else {
            let mut cnt = 0;
            for i in 2..=n / 2 {
                if ptable[i as usize] == 0 && ptable[(n - i) as usize] == 0 {
                    cnt += 1;
                }
            }
            println!("{}", cnt);
        }
    }
}
```
