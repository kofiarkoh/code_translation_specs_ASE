use std::io::{self, BufRead};

const MAX_SIZE: usize = 128;

fn dfs(i: usize, j: usize, h: usize, w: usize, str: &Vec<Vec<char>>, area_id: &mut Vec<Vec<usize>>) {
    if i + 1 < h {
        if str[i][j] == str[i + 1][j] && area_id[i + 1][j] == (i + 1) * w + j {
            area_id[i + 1][j] = area_id[i][j];
            dfs(i + 1, j, h, w, str, area_id);
        }
    }

    if i > 0 {
        if str[i][j] == str[i - 1][j] && area_id[i - 1][j] == (i - 1) * w + j {
            area_id[i - 1][j] = area_id[i][j];
            dfs(i - 1, j, h, w, str, area_id);
        }
    }

    if j + 1 < w {
        if str[i][j] == str[i][j + 1] && area_id[i][j + 1] == i * w + (j + 1) {
            area_id[i][j + 1] = area_id[i][j];
            dfs(i, j + 1, h, w, str, area_id);
        }
    }

    if j > 0 {
        if str[i][j] == str[i][j - 1] && area_id[i][j - 1] == i * w + (j - 1) {
            area_id[i][j - 1] = area_id[i][j];
            dfs(i, j - 1, h, w, str, area_id);
        }
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut input = reader.lines().map(|line| line.unwrap());

    loop {
        let first_line = input.next().unwrap();
        let mut dimensions = first_line.split_whitespace();
        let h: usize = dimensions.next().unwrap().parse().unwrap();
        let w: usize = dimensions.next().unwrap().parse().unwrap();

        if h == 0 && w == 0 {
            break;
        }

        let mut str: Vec<Vec<char>> = vec![vec![' '; MAX_SIZE]; MAX_SIZE];
        let mut area_id: Vec<Vec<usize>> = vec![vec![0; MAX_SIZE]; MAX_SIZE];
        let mut area_num = vec![0; MAX_SIZE * MAX_SIZE];
        let mut cnt = 0;

        for i in 0..h {
            let line = input.next().unwrap();
            str[i] = line.chars().collect();
            for j in 0..w {
                area_id[i][j] = i * w + j;
            }
        }

        for i in 0..h {
            for j in 0..w {
                dfs(i, j, h, w, &str, &mut area_id);
            }
        }

        for i in 0..h {
            for j in 0..w {
                area_num[area_id[i][j]] = 1;
            }
        }

        for i in 0..h {
            for j in 0..w {
                if area_num[i * w + j] == 1 {
                    cnt += 1;
                }
            }
        }

        println!("{}", cnt);
    }
}
