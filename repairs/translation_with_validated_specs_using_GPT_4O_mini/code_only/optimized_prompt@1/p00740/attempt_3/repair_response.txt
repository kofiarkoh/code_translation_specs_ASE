```rust
use std::io::{self, BufRead};

const INF: i64 = 9223372036854775807;
const MEMSET_SIZE: usize = 50;

fn in_int() -> i32 {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut input = String::new();
    handle.read_line(&mut input).unwrap();
    input.trim().parse().unwrap()
}

fn print_int(a: i32) {
    println!("{}", a);
}

fn main() {
    let mut c = [0; MEMSET_SIZE];
    
    loop {
        let n = in_int();
        let p = in_int();
        if n == 0 {
            break;
        }
        
        let mut i = 0;
        let mut w = p;
        c.fill(0); // Reset array to zero

        loop {
            if w > 0 {
                w -= 1;
                c[i % n as usize] += 1;
                if w == 0 {
                    let mut j = 0;
                    while j < n {
                        if j != (i % n as usize) && c[j as usize] > 0 {
                            break;
                        }
                        j += 1;
                    }
                    if j == n {
                        print_int(i % n as i32); // Convert usize to i32
                        break;
                    }
                }
            } else {
                w += c[i % n as usize];
                c[i % n as usize] = 0;
            }
            i += 1;
        }
    }
}
```
