```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    // Read N
    let N: usize = lines.next().unwrap().unwrap().trim().parse().unwrap();
    let mut A = vec![0; N + 1]; // Create a vector with N + 1 elements

    // Read the array A
    for i in 1..=N {
        A[i] = lines.next().unwrap().unwrap().trim().parse().unwrap();
    }

    let mut ans = 0;

    // Calculate the answer
    for i in 1..N {
        ans += A[i] / 2;
        if A[i + 1] > 1 && A[i] % 2 == 1 {
            A[i + 1] -= 1;
            ans += 1;
        }
    }
    ans += A[N] / 2;

    // Print the result
    println!("{}", ans);
    // No need for explicit flush in Rust, as println! flushes automatically
}
```
