use std::io::{self, BufRead};

const MAX_N: usize = 12;
const MAX_M: usize = 5001;
const MAX_LEN: usize = 21;
const MAX_FLAG: usize = 4096;

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let mut n: usize;
    let mut m: usize;

    while let Some(line) = lines.next() {
        let line = line.unwrap();
        let nums: Vec<usize> = line.split_whitespace()
            .filter_map(|s| s.parse().ok())
            .collect();
        n = nums[0];
        m = nums[1];

        if n == 0 && m == 0 {
            break;
        }

        let mut e: [[u8; MAX_LEN]; MAX_N] = [[0; MAX_LEN]; MAX_N];
        let mut len: [usize; MAX_N] = [0; MAX_N];
        let mut t: Vec<u8> = Vec::new();
        let mut flag: [[u8; MAX_FLAG]; MAX_M] = [[0; MAX_FLAG]; MAX_M];

        for i in 0..n {
            let line = lines.next().unwrap().unwrap();
            let bytes = line.as_bytes();
            e[i][..bytes.len()].copy_from_slice(bytes);
            len[i] = bytes.len();
        }

        for _ in 0..m {
            let line = lines.next().unwrap().unwrap();
            let bytes = line.as_bytes();
            let ln = bytes.len();
            if ln > 0 && bytes[ln - 1] < b' ' {
                t.extend_from_slice(&bytes[..ln - 1]);
            } else {
                t.extend_from_slice(bytes);
            }
        }

        let l_s = t.len();
        let loop_count = 1 << n;

        for i in 0..l_s {
            for j in 0..loop_count {
                if j != 0 && flag[i][j] == 0 {
                    continue;
                }
                let mut mask = 1;
                for k in 0..n {
                    if j & mask != 0 {
                        mask <<= 1;
                        continue;
                    }
                    let l = len[k];
                    if i + l <= l_s && &t[i..(i + l)] == &e[k][..l] {
                        let pat = j | mask;
                        flag[i + l][pat] = 1;
                    }
                    mask <<= 1;
                }
            }
        }

        let cnt = (0..=l_s).filter(|&i| flag[i][loop_count - 1] != 0).count();
        println!("{}", cnt);
    }
}
