```rust
use std::io::{self, BufRead};
use std::str::FromStr;

const MAX_NUM: usize = 100_000;
const NUM2_MAX: usize = 1 << 17;
const QUERY_MAX: usize = 100_000;
const INIT_VAL: i64 = 2_000_000_000;

#[derive(Clone, Copy)]
struct Range {
    start: usize,
    end: usize,
}

#[derive(Default, Clone)] // Added Clone to satisfy the trait bound
struct Pair {
    lazy: i64,
    val: i64,
}

fn is_overlap(r1: &Range, r2: &Range) -> bool {
    r1.end > r2.start && r2.end > r1.start
}

fn is_contain(this: &Range, box_range: &Range) -> bool {
    box_range.start <= this.start && this.end <= box_range.end
}

fn unlazy(seg: &mut [Pair], node: usize, nr: &Range) {
    let lazy = seg[node].lazy;
    if lazy == INIT_VAL {
        return;
    }
    seg[node].val = lazy; // update
    seg[node].lazy = INIT_VAL;
    if nr.end - nr.start > 1 {
        seg[node * 2].lazy = lazy / 2;
        seg[node * 2 + 1].lazy = lazy / 2;
    }
}

fn update_range(seg: &mut [Pair], x: i64, r: &Range, node: usize, nr: &Range) {
    unlazy(seg, node, nr);
    if !is_overlap(nr, r) {
        return;
    }
    if is_contain(nr, r) {
        seg[node].lazy = x * (nr.end - nr.start) as i64;
        unlazy(seg, node, nr);
        return;
    }
    let left = Range {
        start: nr.start,
        end: (nr.start + nr.end) / 2,
    };
    let right = Range {
        start: (nr.start + nr.end) / 2,
        end: nr.end,
    };
    update_range(seg, x, r, node * 2, &left);
    update_range(seg, x, r, node * 2 + 1, &right);
    seg[node].val = seg[node * 2].val + seg[node * 2 + 1].val;
}

fn get_sum(seg: &mut [Pair], r: &Range, node: usize, nr: &Range) -> i64 {
    unlazy(seg, node, nr);
    if !is_overlap(nr, r) {
        return 0;
    }
    if is_contain(nr, r) {
        return seg[node].val;
    }
    let left = Range {
        start: nr.start,
        end: (nr.start + nr.end) / 2,
    };
    let right = Range {
        start: (nr.start + nr.end) / 2,
        end: nr.end,
    };
    let mut ans = 0;
    ans += get_sum(seg, r, node * 2, &left);
    ans += get_sum(seg, r, node * 2 + 1, &right);
    ans
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines().map(|line| line.unwrap());

    let first_line: Vec<i64> = lines.next().unwrap().split_whitespace()
        .map(|s| i64::from_str(s).unwrap()).collect();
    let num = first_line[0] as usize;
    let queries = first_line[1] as usize;

    let mut num2 = 1;
    while num > num2 {
        num2 *= 2;
    }

    let mut seg = vec![Pair::default(); NUM2_MAX * 2];
    for i in 1..num2 * 2 {
        seg[i].lazy = INIT_VAL;
    }

    let whole = Range { start: 0, end: num2 };
    let mut ans = Vec::with_capacity(QUERY_MAX);
    
    for _ in 0..queries {
        let query_line: Vec<i64> = lines.next().unwrap().split_whitespace()
            .map(|s| i64::from_str(s).unwrap()).collect();
        let command_type = query_line[0];
        let start = query_line[1] as usize;
        let end = query_line[2] as usize;
        let r = Range { start, end: end + 1 };

        match command_type {
            0 => { // COMMAND_UPDATE
                let x = lines.next().unwrap().parse::<i64>().unwrap();
                update_range(&mut seg, x, &r, 1, &whole);
            }
            1 => { // COMMAND_GET_SUM
                ans.push(get_sum(&mut seg, &r, 1, &whole));
            }
            _ => {}
        }
    }

    for &result in ans.iter() {
        println!("{}", result);
    }
}
```
