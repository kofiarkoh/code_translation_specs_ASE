impl SegmentTree {
    fn new(n: usize) -> Self {
        let mut sz = 1;
        while sz < n {
            sz <<= 1;

            seg: [0; MAX_SIZE],
            sz,

    fn update(&mut self, mut k: usize, v: i64) {
        k += self.sz - 1;
        self.seg[k] = v;

            if k & 1 == 0 {
                k -= 1;

            let t = k;
            k >>= 1;

                max_val = self.seg[t];

            self.seg[k] = max_val;
        }
    }

    // get max value in [a, b)
    fn query(&self, a: usize, b: usize, mut k: usize, l: usize, r: usize) -> i64 {

    }
