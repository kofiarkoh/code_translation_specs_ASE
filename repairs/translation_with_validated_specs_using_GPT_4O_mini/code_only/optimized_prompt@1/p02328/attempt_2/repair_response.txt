```rust
use std::io::{self, BufRead};

const MAX: usize = 100001;

#[derive(Clone, Copy)]
struct Rect {
    height: i64,
    pos: usize,
}

fn large_histogram(w: usize, table: &[i64; MAX]) -> i64 {
    let mut sp = 0;
    let mut max = 0;
    let mut stack: [Rect; MAX] = [(Rect { height: 0, pos: 0 }); MAX]; // Initialize stack with default values
    let mut table_with_sentinel = [0; MAX];
    table_with_sentinel[..w].copy_from_slice(&table[..w]);
    
    for i in 0..=w {
        let r = Rect {
            height: table_with_sentinel[i],
            pos: i,
        };
        if sp == 0 || stack[sp - 1].height < r.height {
            stack[sp] = r;
            sp += 1;
        } else if stack[sp - 1].height > r.height {
            let mut target = i;
            while sp > 0 && stack[sp - 1].height >= r.height {
                let pre = stack[sp - 1];
                sp -= 1;
                let area = pre.height * (i as i64 - pre.pos as i64);
                if max < area {
                    max = area;
                }
                target = pre.pos;
            }
            let mut r = r;
            r.pos = target;
            stack[sp] = r;
            sp += 1;
        }
    }
    max
}

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();
    
    let w: i64 = lines.next().unwrap().unwrap().parse().unwrap();
    let mut table: [i64; MAX] = [0; MAX];
    
    for i in 0..w {
        table[i as usize] = lines.next().unwrap().unwrap().parse().unwrap();
    }
    
    let max = large_histogram(w as usize, &table);
    println!("{}", max);
}
```
