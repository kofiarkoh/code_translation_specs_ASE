use std::io::{self, BufRead};

const MOD: i64 = 998244353;

#[derive(Debug, Clone)]
struct Data {
    key: i64,
    id: i64,
}

fn merge_sort(data: &mut [Data]) {
    let n = data.len();
    if n <= 1 {
        return;
    }
    let mid = n / 2;
    merge_sort(&mut data[0..mid]);
    merge_sort(&mut data[mid..]);

    let mut merged = data.to_vec();
    let (mut p, mut q) = (0, mid);
    for i in 0..n {
        if p >= mid {
            merged[i] = data[q].clone();
            q += 1;
        } else if q >= n {
            merged[i] = data[p].clone();
            p += 1;
        } else {
            if data[p].key < data[q].key {
                merged[i] = data[p].clone();
                p += 1;
            } else {
                merged[i] = data[q].clone();
                q += 1;
            }
        }
    }
    data.copy_from_slice(&merged);
}

fn add_bit(bit: &mut Vec<i64>, i: usize, k: i64) {
    let mut idx = i;
    while idx < bit.len() {
        bit[idx] += k;
        idx += idx & (idx as isize).wrapping_neg() as usize; // i + (i & -i)
    }
}

fn sum_bit(bit: &Vec<i64>, r: usize) -> i64 {
    let mut sum = 0;
    let mut idx = r;
    while idx > 0 {
        sum += bit[idx];
        idx -= idx & (idx as isize).wrapping_neg() as usize; // r - (r & -r)
    }
    sum
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();
    
    let n: usize = lines.next().unwrap().unwrap().parse().unwrap();
    let mut data: Vec<Data> = Vec::with_capacity(n);
    
    for _ in 0..n {
        let line = lines.next().unwrap().unwrap();
        let mut parts = line.split_whitespace();
        let key: i64 = parts.next().unwrap().parse().unwrap();
        let id: i64 = parts.next().unwrap().parse().unwrap();
        data.push(Data { key, id });
    }

    merge_sort(&mut data);

    let mut m = 1;
    for i in 0..n {
        let tmp = data[i].key;
        let mut j = i;
        while j < n && data[j].key == tmp {
            data[j].key = data[j].id;
            data[j].id = m;
            j += 1;
        }
        i = j - 1;
        m += 1;
    }

    merge_sort(&mut data);

    let mut new_n = 1;
    for i in 0..n {
        let tmp = data[i].key;
        let mut j = i;
        while j < n && data[j].key == tmp {
            data[j].key = new_n;
            j += 1;
        }
        i = j - 1;
        new_n += 1;
    }

    let mut pow = vec![0; n + 1];
    pow[0] = 1;
    for i in 1..=n {
        pow[i] = (pow[i - 1] * 2) % MOD;
    }

    let mut bit = vec![0; n + 1];
    let mut ans = (pow[n] - 1 + MOD) % MOD * n as i64 % MOD;

    let mut i = 0;
    while i < n {
        let key = data[i].key;
        let mut j = i;
        while j < n && data[j].key == key {
            ans = (ans + pow[sum_bit(&bit, data[j].id as usize - 1)] + pow[i - sum_bit(&bit, data[j].id as usize)]) % MOD;
            j += 1;
        }
        ans = (ans + (MOD - pow[i]) * (j - i) % MOD) % MOD;

        for k in i..j {
            add_bit(&mut bit, data[k].id as usize, 1);
        }
        i = j;
    }

    for i in 1..=m {
        bit[i] = 0;
    }

    for i in (0..n).rev() {
        let key = data[i].key;
        let mut j = i;
        while j >= 0 && data[j].key == key {
            ans = (ans + pow[sum_bit(&bit, data[j].id as usize - 1)] + pow[n - i - 1 - sum_bit(&bit, data[j].id as usize)]) % MOD;
            j = j.wrapping_sub(1);
        }
        ans = (ans + (MOD - pow[n - i - 1]) * (i - j as usize) % MOD) % MOD;

        for k in (j + 1)..=i {
            add_bit(&mut bit, data[k].id as usize, 1);
        }
    }

    for i in 0..n {
        data[i].key ^= data[i].id;
        data[i].id ^= data[i].key;
        data[i].key ^= data[i].id;
    }

    merge_sort(&mut data);

    let mut i = 0;
    while i < n {
        let key = data[i].key;
        let mut j = i;
        while j < n && data[j].key == key {
            j += 1;
        }
        ans = (ans + (MOD - pow[i]) * (j - i) % MOD) % MOD;
        i = j - 1;
    }

    for i in (0..n).rev() {
        let key = data[i].key;
        let mut j = i;
        while j >= 0 && data[j].key == key {
            j = j.wrapping_sub(1);
        }
        ans = (ans + (MOD - pow[n - i - 1]) * (i - j as usize) % MOD) % MOD;
        i = j + 1;
    }

    println!("{}", ans);
}
