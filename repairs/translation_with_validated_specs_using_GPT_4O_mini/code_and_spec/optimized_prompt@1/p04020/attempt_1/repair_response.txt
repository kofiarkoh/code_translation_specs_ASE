```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();
    let mut lines = handle.lines();

    // Read the number of elements
    let n: usize = lines.next().unwrap().unwrap().parse().expect("Invalid input for N");
    
    // Ensure N is within the specified bounds
    assert!(n > 0 && n <= 100_000, "N must be a positive integer and less than or equal to 100,000");

    // Read the sequence of long long integers
    let mut a: Vec<i64> = Vec::with_capacity(n);
    for _ in 0..n {
        let value: i64 = lines.next().unwrap().unwrap().parse().expect("Invalid integer in sequence");
        a.push(value);
    }

    let mut ans: i64 = 0;

    // Calculate the total count based on the specified logic
    for i in 0..(n - 1) {
        ans += a[i] / 2;
        if a[i + 1] > 1 && a[i] % 2 == 1 {
            a[i + 1] -= 1; // Decrement the next integer if conditions are met
            ans += 1;
        }
    }
    
    // Handle the last element separately
    ans += a[n - 1] / 2;

    // Output the result
    println!("{}", ans);
}
```
