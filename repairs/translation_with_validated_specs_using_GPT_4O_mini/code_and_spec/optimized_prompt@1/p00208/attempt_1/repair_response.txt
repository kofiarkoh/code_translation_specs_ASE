```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();

    for line in handle.lines() {
        let input = line.unwrap();
        let a: i32 = input.trim().parse().unwrap(); // Assumes valid integer input

        if a == 0 {
            break; // Terminate on sentinel value
        }

        let mut n = [0; 200]; // Array to store octal digits
        let mut b = 0;

        // Convert to octal
        let mut temp = a;
        while temp > 0 {
            n[b] = temp % 8;
            temp /= 8;
            b += 1;
        }

        // Apply transformations
        for i in 0..b {
            if n[i] > 3 {
                n[i] += 1;
            }
            if n[i] > 5 {
                n[i] += 1;
            }
        }

        // Output the modified octal representation
        for i in (0..b).rev() {
            print!("{}", n[i]);
        }
        println!(); // Newline after each output
    }
}
```
