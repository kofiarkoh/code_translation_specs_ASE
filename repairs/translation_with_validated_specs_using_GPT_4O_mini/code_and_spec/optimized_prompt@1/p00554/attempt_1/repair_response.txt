use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Read n and m
    let first_line = lines.next().unwrap().unwrap();
    let mut inputs = first_line.split_whitespace();
    let n: i32 = inputs.next().unwrap().parse().unwrap();
    let m: usize = inputs.next().unwrap().parse().unwrap();

    // Preconditions
    assert!(m >= 1);
    assert!(n >= 0);

    let mut a = vec![0; m + 1]; // Using 1-based indexing
    let mut b = vec![0; m + 1];
    let mut count1 = 0;

    // Read pairs of integers
    for i in 1..=m {
        let line = lines.next().unwrap().unwrap();
        let mut pair = line.split_whitespace();
        a[i] = pair.next().unwrap().parse().unwrap();
        b[i] = pair.next().unwrap().parse().unwrap();

        // Modify arrays based on the threshold condition
        if a[i] >= n {
            a[i] = 0;
            b[i] = 0;
            count1 += 1;
        }
    }

    let mut count2 = 0;

    // Calculate cumulative sum if conditions are met
    if m - 1 > count1 {
        for i in 1..=m {
            b[i] = n - a[i];
        }

        // Sort the modified second array using a simple bubble sort
        for i in 1..m {
            for j in (i + 1)..=m {
                if b[i] > b[j] {
                    b.swap(i, j);
                }
            }
        }

        // Calculate the cumulative sum
        for i in 1..=m - count1 - 1 {
            count2 += b[i];
        }
    }

    // Output the result
    println!("{}", count2);
}
