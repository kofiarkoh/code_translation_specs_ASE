use std::collections::HashMap;
use std::io::{self, BufRead};

const HASHSIZ: usize = 10007;

#[derive(Default)]
struct Hash {
    f: [i16; 26],
}

fn insert(hash_table: &mut [Hash; HASHSIZ], f: &mut [i16; 26]) {
    f[0] += 1;
    let mut k: u64 = 0;
    for i in (0..26).step_by(2) {
        k = (k << 4) + f[i] as u64;
    }
    let mut tp = &mut hash_table[(k % HASHSIZ as u64) as usize];

    while tp.f[0] != 0 {
        if tp.f == *f {
            return;
        }
        tp = &mut hash_table[(tp as *mut _ as usize + std::mem::size_of::<Hash>()) % (HASHSIZ * std::mem::size_of::<Hash>()) as usize / std::mem::size_of::<Hash>()];
    }
    tp.f.copy_from_slice(f);
    f[0] -= 1;
}

fn lookup(hash_table: &[Hash; HASHSIZ], f: &mut [i16; 26]) -> bool {
    f[0] += 1;
    let mut k: u64 = 0;
    for i in (0..26).step_by(2) {
        k = (k << 4) + f[i] as u64;
    }
    let mut tp = &hash_table[(k % HASHSIZ as u64) as usize];

    while tp.f[0] != 0 {
        if tp.f == *f {
            f[0] -= 1;
            return true;
        }
        tp = &hash_table[(tp as *const _ as usize + std::mem::size_of::<Hash>()) % (HASHSIZ * std::mem::size_of::<Hash>()) as usize / std::mem::size_of::<Hash>()];
    }
    f[0] -= 1;
    false
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    let mut sa = String::new();
    let mut sb = String::new();

    reader.read_line(&mut sa).unwrap();
    reader.read_line(&mut sb).unwrap();

    let wa = sa.trim().len();
    let wb = sb.trim().len();
    let mut w = wa.min(wb);

    let mut hash_table: [Hash; HASHSIZ] = Default::default();
    let mut f = [0; 26];

    for k in (1..=w).rev() {
        hash_table.iter_mut().for_each(|h| h.f = [0; 26]);

        f.iter_mut().for_each(|x| *x = 0);
        for c in sa.chars().take(k) {
            f[(c as usize - 'a' as usize)] += 1;
        }
        insert(&mut hash_table, &mut f);

        let mut i = wa - k;
        let mut p = sa.chars();
        for _ in 0..i {
            f[(p.next().unwrap() as usize - 'a' as usize)] -= 1;
            f[(p.next().unwrap() as usize - 'a' as usize)] += 1;
            insert(&mut hash_table, &mut f);
        }

        f.iter_mut().for_each(|x| *x = 0);
        for c in sb.chars().take(k) {
            f[(c as usize - 'a' as usize)] += 1;
        }
        if lookup(&hash_table, &mut f) {
            println!("{}", k);
            return;
        }

        let mut i = wb - k;
        let mut q = sb.chars();
        for _ in 0..i {
            f[(q.next().unwrap() as usize - 'a' as usize)] -= 1;
            f[(q.next().unwrap() as usize - 'a' as usize)] += 1;
            if lookup(&hash_table, &mut f) {
                println!("{}", k);
                return;
            }
        }
    }
    println!("0");
}
