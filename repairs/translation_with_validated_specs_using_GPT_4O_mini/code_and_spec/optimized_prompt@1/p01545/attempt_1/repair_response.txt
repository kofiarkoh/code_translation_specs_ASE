use std::io::{self, BufRead};

struct SegmentTree {
    seg: Vec<i64>,
    sz: usize,
}

impl SegmentTree {
    fn new(size: usize) -> Self {
        let mut sz = 1;
        while sz < size {
            sz <<= 1;
        }
        SegmentTree {
            seg: vec![0; 2 * sz],
            sz,
        }
    }

    // Update the kth value to v
    fn update(&mut self, k: usize, v: i64) {
        let mut k = k + self.sz - 1;
        self.seg[k] = v;
        while k > 0 {
            if k % 2 == 0 {
                k -= 1;
            }
            let t = k;
            k >>= 1;
            let left_value = self.seg[t];
            let right_value = self.seg[t + 1];
            self.seg[k] = left_value.max(right_value);
        }
    }

    // Get max value in [a, b)
    fn query(&self, a: usize, b: usize, k: usize, l: usize, r: usize) -> i64 {
        if r <= a || b <= l {
            return 0;
        }
        if a <= l && r <= b {
            return self.seg[k];
        }
        let m = (l + r) >> 1;
        let left_max = self.query(a, b, k * 2 + 1, l, m);
        let right_max = self.query(a, b, k * 2 + 2, m, r);
        left_max.max(right_max)
    }
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    let n: usize = lines.next().unwrap().unwrap().trim().parse().unwrap();
    let mut segment_tree = SegmentTree::new(n);

    let positions: Vec<usize> = lines
        .next()
        .unwrap()
        .unwrap()
        .split_whitespace()
        .map(|x| x.parse().unwrap())
        .collect();

    for &x in &positions {
        let query_result = segment_tree.query(1, x + 1, 0, 0, segment_tree.sz);
        segment_tree.update(x, (x as i64) + query_result);
    }

    let expected_total_cost = (n * (n + 1)) / 2;
    let total_max = segment_tree.query(1, n + 1, 0, 0, segment_tree.sz);
    let ans = expected_total_cost as i64 - total_max;

    println!("{}", ans);
}
