```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();

    for line in reader.lines() {
        let line = line.unwrap();
        let numbers: Vec<f64> = line
            .split_whitespace()
            .filter_map(|s| s.parse().ok())
            .collect();

        // Ensure we have exactly six valid floating-point numbers
        if numbers.len() != 6 {
            continue; // Skip invalid input
        }

        let (a, b, c, d, e, f) = (numbers[0], numbers[1], numbers[2], numbers[3], numbers[4], numbers[5]);

        // Check precondition: a must be non-zero
        if a == 0.0 {
            continue; // Skip this set of inputs to avoid division by zero
        }

        let bai = d / a;
        let aa = a * bai;
        let bb = b * bai;
        let cc = c * bai;
        let y = (f - cc) / (e - bb);
        let x = (c - b * y) / a;

        // Output formatted to three decimal places
        println!("{:.3} {:.3}", x, y);
    }
}
```
