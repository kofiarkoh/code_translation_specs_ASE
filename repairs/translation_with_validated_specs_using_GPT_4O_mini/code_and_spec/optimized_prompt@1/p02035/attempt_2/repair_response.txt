```rust
use std::io;

const MAX_N: usize = 2010;

fn pom(a: i64, mut n: i64, m: i64) -> i64 {
    let mut x = 1;
    let mut a = a % m;
    while n > 0 {
        if n % 2 == 1 {
            x = (x * a) % m;
        }
        a = (a * a) % m;
        n /= 2;
    }
    x
}

fn invp(a: i64, p: i64) -> i64 {
    pom(a, p - 2, p)
}

struct Factorials {
    fact: Vec<i64>,
    invfact: Vec<i64>,
}

impl Factorials {
    fn new(n: usize, m: i64) -> Self {
        let mut fact = vec![0; n + 1];
        let mut invfact = vec![0; n + 1];
        fact[0] = 1;
        for i in 0..n {
            fact[i + 1] = fact[i] * (i + 1) % m; // Removed unnecessary parentheses
        }
        invfact[n] = invp(fact[n], m);
        for i in (1..=n).rev() {
            invfact[i - 1] = invfact[i] * i % m; // Removed unnecessary parentheses
        }
        Factorials { fact, invfact }
    }

    fn choose(&self, n: i64, r: i64, m: i64) -> i64 {
        if n < 0 || r < 0 || n < r {
            return 0;
        }
        (self.fact[n as usize] * self.invfact[r as usize] % m * self.invfact[(n - r) as usize] % m) % m
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let mut iter = input.split_whitespace();
    let n: i64 = iter.next().unwrap().parse().unwrap();
    let m: i64 = iter.next().unwrap().parse().unwrap();

    let factorials = Factorials::new(MAX_N, m);
    let mut ans = 0;

    for i in 1..n {
        let mut temp = 0;
        for j in 0..(n - i) {
            let aaa = (pom(pom(2, n - i - j, m) - 1, i, m) * factorials.choose(n - i, j, m) % m) % m; // Removed unnecessary parentheses
            temp += if j % 2 == 1 { -aaa } else { aaa };
        }
        ans = (ans + temp * pom(2, i * (i - 1) / 2, m) % m * pom(2, (n - i) * (n - i - 1) / 2, m) % m * factorials.choose(n, i, m) % m) % m;
    }

    println!("{}", (ans % m + m) % m);
}
```
