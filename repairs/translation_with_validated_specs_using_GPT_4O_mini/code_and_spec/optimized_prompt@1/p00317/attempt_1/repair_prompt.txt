int ins(char *s, char *r)
{
	char *ss = s;
	int n;

	do *s = gc();
	while (*s++ > ' ');
	*--s = 0;
	n = s - ss;
	while (--s >= ss) *r++ = *s;
	*r = 0;
	return n;
}

int search(char *s, int n, int idx, int part)
{
	int k, mid, low, high;

	low = 0, high = N;
	if (part) {

			mid = (low + high) >> 1;
