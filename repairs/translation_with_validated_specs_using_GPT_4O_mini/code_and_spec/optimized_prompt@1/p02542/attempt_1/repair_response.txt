use std::collections::VecDeque;
use std::cmp::Ordering;

#[derive(Clone, Copy)]
struct Data {
    key: i64,
    id: usize,
}

struct MinHeap {
    obj: Vec<Data>,
    size: usize,
}

impl MinHeap {
    fn new() -> Self {
        MinHeap {
            obj: Vec::new(),
            size: 0,
        }
    }

    fn push(&mut self, x: Data) {
        self.size += 1;
        let mut i = self.size;
        self.obj.push(x);
        while i > 1 {
            let j = i / 2;
            if self.obj[i - 1].key < self.obj[j - 1].key {
                self.obj.swap(i - 1, j - 1);
                i = j;
            } else {
                break;
            }
        }
    }

    fn pop(&mut self) -> Data {
        let output = self.obj[0];
        self.size -= 1;
        self.obj[0] = self.obj[self.size];
        self.obj.pop();
        let mut i = 1;
        let mut j = 2;
        while j <= self.size {
            if j < self.size && self.obj[j].key < self.obj[j - 1].key {
                j += 1;
            }
            if self.obj[j - 1].key < self.obj[i - 1].key {
                self.obj.swap(j - 1, i - 1);
                i = j;
                j *= 2;
            } else {
                break;
            }
        }
        output
    }
}

#[derive(Clone)]
struct Edge {
    id: usize,
    head: usize,
    tail: usize,
    flow: i64,
    cap: i64,
    cost: i64,
}

struct List {
    next: Option<Box<List>>,
    e: Edge,
}

const SUP: i64 = 1 << 60;
const INF: i64 = -SUP;

fn successive_shortest_path(N: usize, M: usize, adj: &Vec<Option<Box<List>>>, e: &mut Vec<Edge>, b: &mut Vec<i64>) -> i64 {
    let mut flag = vec![0; N + 1];
    let mut def = vec![0; N + 1];
    let mut ans = 0;
    let mut dist = vec![0; N + 1];
    let mut pi = vec![SUP; N + 1];
    let mut prev = vec![None; N + 1];
    let mut h_def = MinHeap::new();
    let mut h_dist = MinHeap::new();

    for i in 0..M {
        ans += e[i].cost * e[i].flow;
    }

    for i in 1..=N {
        def[i] = b[i];
        let mut p = &adj[i];
        while let Some(node) = p {
            def[i] += node.e.flow - e[node.e.id ^ 1].flow;
            p = node.next.as_deref();
        }
        if def[i] < 0 {
            h_def.push(Data { key: def[i], id: i });
            if pi[i] == SUP {
                pi[i] = 0;
                for j in 1..=N {
                    let mut flag_updated = false;
                    for k in 0..M {
                        if e[k].flow == e[k].cap {
                            continue;
                        }
                        let u = e[k].tail;
                        let w = e[k].head;
                        if pi[u] != SUP && pi[w] > pi[u] + e[k].cost {
                            flag_updated = true;
                            pi[w] = pi[u] + e[k].cost;
                        }
                    }
                    if !flag_updated {
                        break;
                    }
                }
                if j > N {
                    return INF;
                }
            }
        }
    }

    while h_def.size > 0 {
        let d = h_def.pop();
        let s = d.id;
        for u in 1..=N {
            flag[u] = 0;
            dist[u] = SUP;
            prev[u] = None;
        }
        dist[s] = 0;
        h_dist = MinHeap::new();
        h_dist.push(Data { key: 0, id: s });

        while h_dist.size > 0 {
            let d = h_dist.pop();
            let u = d.id;
            if flag[u] != 0 {
                continue;
            }
            flag[u] = 1;

            let mut p = &adj[u];
            while let Some(node) = p {
                let w = node.e.head;
                if node.e.flow == node.e.cap {
                    continue;
                } else if dist[w] > dist[u] + node.e.cost + pi[u] - pi[w] {
                    dist[w] = dist[u] + node.e.cost + pi[u] - pi[w];
                    prev[w] = Some(node.e.clone());
                    h_dist.push(Data { key: dist[w], id: w });
                }
                p = node.next.as_deref();
            }
        }

        let mut t = 0;
        for t_candidate in 1..=N {
            if dist[t_candidate] != SUP && def[t_candidate] > 0 {
                t = t_candidate;
                break;
            }
        }
        if t == 0 {
            return SUP;
        }

        let mut u = t;
        let min_flow = (-def[s]).min(def[t]);
        while let Some(prev_edge) = prev[u] {
            if min_flow > prev_edge.cap - prev_edge.flow {
                min_flow = prev_edge.cap - prev_edge.flow;
            }
            u = prev_edge.tail;
        }
        u = t;
        while let Some(prev_edge) = prev[u] {
            let i = prev_edge.id;
            if e[i ^ 1].flow > 0 {
                e[i ^ 1].flow -= min_flow;
                e[i].cap -= min_flow;
            } else {
                e[i].flow += min_flow;
                e[i ^ 1].cap += min_flow;
            }
            ans += e[i].cost * min_flow;
            u = prev_edge.tail;
        }
        for i in 1..=N {
            if dist[i] != SUP {
                pi[i] += dist[i];
            }
        }

        def[s] += min_flow;
        def[t] -= min_flow;
        if def[s] < 0 {
            h_def.push(Data { key: def[s], id: s });
        }
    }

    ans
}

fn main() {
    let mut input = String::new();
    std::io::stdin().read_line(&mut input).unwrap();
    let mut iter = input.split_whitespace();
    let n: usize = iter.next().unwrap().parse().unwrap();
    let m: usize = iter.next().unwrap().parse().unwrap();

    let mut grid = vec![vec![' '; 52]; 52];
    for i in 1..=n {
        let mut line = String::new();
        std::io::stdin().read_line(&mut line).unwrap();
        for (j, c) in line.trim().chars().enumerate() {
            grid[i][j + 1] = c;
        }
    }

    let mut adj: Vec<Option<Box<List>>> = vec![None; 2603];
    let mut d: Vec<List> = vec![List { next: None, e: Edge { id: 0, head: 0, tail: 0, flow: 0, cap: 0, cost: 0 } }; 30000];
    let mut e: Vec<Edge> = vec![Edge { id: 0, head: 0, tail: 0, flow: 0, cap: 0, cost: 0 }; 30000];

    let mut m_count = 0;
    let mut n_count = 1;
    let mut dist = vec![-1; 2501];
    let mut q = VecDeque::new();

    for i in (1..=n).rev() {
        for j in (1..=m).rev() {
            if grid[i][j] != 'o' {
                continue;
            }
            grid[i][j] = '.';
            dist.fill(-1);
            let start = (i - 1) * m + j;
            dist[start] = 0;
            q.push_back(start);

            while let Some(head) = q.pop_front() {
                let k = (head - 1) / m + 1;
                let l = (head - 1) % m + 1;

                if grid[k + 1][l] == '.' && dist[k * m + l] == -1 {
                    dist[k * m + l] = dist[head] + 1;
                    q.push_back(k * m + l);
                }
                if grid[k][l + 1] == '.' && dist[(k - 1) * m + l + 1] == -1 {
                    dist[(k - 1) * m + l + 1] = dist[head] + 1;
                    q.push_back((k - 1) * m + l + 1);
                }
            }

            for head in (0..q.len()).rev().take(100) {
                e[m_count * 2].id = m_count * 2;
                e[m_count * 2].tail = n * m + n_count;
                e[m_count * 2].head = q[head];
                e[m_count * 2].cap = 1;
                e[m_count * 2].cost = -dist[q[head]];
                e[m_count * 2 + 1].id = m_count * 2 + 1;
                e[m_count * 2 + 1].tail = q[head];
                e[m_count * 2 + 1].head = n * m + n_count;
                e[m_count * 2 + 1].cap = 0;
                e[m_count * 2 + 1].cost = dist[q[head]];

                d[m_count * 2].e = e[m_count * 2].clone();
                d[m_count * 2 + 1].e = e[m_count * 2 + 1].clone();
                d[m_count * 2].next = adj[n * m + n_count].take().map(Box::new);
                d[m_count * 2 + 1].next = adj[q[head]].take().map(Box::new);
                adj[n * m + n_count] = Some(Box::new(d[m_count * 2].clone()));
                adj[q[head]] = Some(Box::new(d[m_count * 2 + 1].clone()));
                m_count += 1;
            }
            n_count += 1;
        }
    }

    for i in 1..n_count {
        e[m_count * 2].id = m_count * 2;
        e[m_count * 2].tail = n * m + n_count;
        e[m_count * 2].head = n * m + i;
        e[m_count * 2].cap = 1;
        e[m_count * 2].cost = 0;
        e[m_count * 2 + 1].id = m_count * 2 + 1;
        e[m_count * 2 + 1].tail = n * m + i;
        e[m_count * 2 + 1].head = n * m + n_count;
        e[m_count * 2 + 1].cap = 0;
        e[m_count * 2 + 1].cost = 0;

        d[m_count * 2].e = e[m_count * 2].clone();
        d[m_count * 2 + 1].e = e[m_count * 2 + 1].clone();
        d[m_count * 2].next = adj[n * m + n_count].take().map(Box::new);
        d[m_count * 2 + 1].next = adj[n * m + i].take().map(Box::new);
        adj[n * m + n_count] = Some(Box::new(d[m_count * 2].clone()));
        adj[n * m + i] = Some(Box::new(d[m_count * 2 + 1].clone()));
        m_count += 1;
    }

    for i in 1..=n * m {
        e[m_count * 2].id = m_count * 2;
        e[m_count * 2].tail = i;
        e[m_count * 2].head = n * m + n_count + 1;
        e[m_count * 2].cap = 1;
        e[m_count * 2].cost = 0;
        e[m_count * 2 + 1].id = m_count * 2 + 1;
        e[m_count * 2 + 1].tail = n * m + n_count + 1;
        e[m_count * 2 + 1].head = i;
        e[m_count * 2 + 1].cap = 0;
        e[m_count * 2 + 1].cost = 0;

        d[m_count * 2].e = e[m_count * 2].clone();
        d[m_count * 2 + 1].e = e[m_count * 2 + 1].clone();
        d[m_count * 2].next = adj[i].take().map(Box::new);
        d[m_count * 2 + 1].next = adj[n * m + n_count + 1].take().map(Box::new);
        adj[i] = Some(Box::new(d[m_count * 2].clone()));
        adj[n * m + n_count + 1] = Some(Box::new(d[m_count * 2 + 1].clone()));
        m_count += 1;
    }

    let mut b = vec![0; 2603];
    b[n * m + n_count] = -(n_count - 1) as i64;
    b[n * m + n_count + 1] = (n_count - 1) as i64;

    println!("{}", -successive_shortest_path(n * m + n_count + 1, m_count * 2, &adj, &mut e, &mut b));
}
