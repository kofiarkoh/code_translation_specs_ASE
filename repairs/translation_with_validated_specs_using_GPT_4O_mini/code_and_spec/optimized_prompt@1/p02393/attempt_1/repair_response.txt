```rust
use std::io::{self, Write};

fn main() {
    // Read three integers from standard input
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    
    // Split the input string into parts and parse them as integers
    let mut numbers: Vec<i32> = input
        .split_whitespace()
        .take(3) // Ensure we only take the first three integers
        .filter_map(|s| s.parse().ok()) // Parse and filter out any invalid integers
        .collect();

    // Check if we have exactly three integers
    if numbers.len() != 3 {
        eprintln!("Error: Expected exactly three integers.");
        return;
    }

    // Sort the numbers in non-decreasing order
    numbers.sort();

    // Print the sorted numbers
    println!("{} {} {}", numbers[0], numbers[1], numbers[2]);
}
```
