```rust
use std::io::{self, Write};

fn main() {
    let mut str1 = String::new();
    let mut str2 = String::new();

    // Read the first string
    io::stdout().flush().unwrap(); // Ensure prompt is displayed before input
    io::stdin().read_line(&mut str1).expect("Failed to read input");
    // Read the second string
    io::stdin().read_line(&mut str2).expect("Failed to read input");

    // Trim whitespace and ensure strings are within the specified length
    let str1 = str1.trim();
    let str2 = str2.trim();

    // Preconditions: both strings must be non-empty and not exceed 100 characters
    if str1.is_empty() || str2.is_empty() || str1.len() > 100 || str2.len() > 100 {
        return; // In a real application, we might handle this error
    }

    // Concatenate str1 with itself
    let concatenated = format!("{}{}", str1, str1);

    // Check if str2 is a substring of the concatenated string
    if concatenated.contains(str2) {
        println!("Yes");
    } else {
        println!("No");
    }
}
```
