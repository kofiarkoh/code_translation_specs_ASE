```rust
use std::io::{self, BufRead};

const INF: i32 = 20002;

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut input = reader.lines().filter_map(Result::ok);
    
    while let Some(line) = input.next() {
        let n: usize = line.trim().parse().expect("Invalid input for number of vertices");
        let mut x = vec![0; n + 1];
        let mut y = vec![0; n + 1];

        for i in 0..n {
            let coords: Vec<i32> = input.next()
                .expect("Expected vertex coordinates")
                .trim()
                .split_whitespace()
                .map(|s| s.parse().expect("Invalid coordinate"))
                .collect();
            x[i] = coords[0];
            y[i] = coords[1];
        }
        x[n] = x[0];
        y[n] = y[0];

        let mut xmin = INF;
        let mut xmax = -INF;
        let mut ymin = INF;
        let mut ymax = -INF;

        for _ in 0..4 {
            let coords: Vec<i32> = input.next()
                .expect("Expected bounding box coordinates")
                .trim()
                .split_whitespace()
                .map(|s| s.parse().expect("Invalid coordinate"))
                .collect();
            let a = coords[0];
            let b = coords[1];

            if a < xmin { xmin = a; }
            if a > xmax { xmax = a; }
            if b < ymin { ymin = b; }
            if b > ymax { ymax = b; }
        }

        let mut win: i64 = 0;
        let mut cur: i64 = 0;

        for i in 0..n {
            if x[i] == x[i + 1] {
                cur += (x[i] as i64) * (y[i] - y[i + 1]) as i64;
                let mut a = x[i];
                if a > xmax { a = xmax; }
                if xmin > a { a = xmin; }
                let mut b = y[i];
                if b > ymax { b = ymax; }
                if ymin > b { b = ymin; }
                let mut c = y[i + 1];
                if c > ymax { c = ymax; }
                if ymin > c { c = ymin; }
                win += (a as i64) * (b - c) as i64;
            }
        }
        println!("{}", win - cur);
    }
}
```
