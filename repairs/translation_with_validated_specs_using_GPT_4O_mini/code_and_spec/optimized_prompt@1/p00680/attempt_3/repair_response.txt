use std::io::{self, BufRead};
use std::f64;

const QMAX: usize = 1000;
const EPS: f64 = 1e-8;
const INF: f64 = f64::INFINITY; // Use f64::INFINITY for a more idiomatic representation
const MAX: usize = 105;

#[derive(Clone)]
struct Edge {
    to: usize,
    rev: usize,
    cap: i32,
    cost: f64,
}

struct MinHeap {
    que: Vec<(usize, f64)>,
}

impl MinHeap {
    fn new() -> Self {
        MinHeap { que: Vec::new() }
    }

    fn min_heapify(&mut self, i: usize) {
        let mut min = i;
        let l = self.left(i);
        let r = self.right(i);

        if l < self.que.len() && self.que[l].1 < self.que[min].1 {
            min = l;
        }
        if r < self.que.len() && self.que[r].1 < self.que[min].1 {
            min = r;
        }
        if min != i {
            self.que.swap(i, min);
            self.min_heapify(min);
        }
    }

    fn deq(&mut self) -> (usize, f64) {
        let min_elem = self.que[0];
        self.que[0] = self.que.pop().unwrap();
        self.min_heapify(0);
        min_elem
    }

    fn enq(&mut self, s: usize, t: f64) {
        self.que.push((s, t));
        let mut i = self.que.len() - 1;
        while i > 0 {
            let parent = self.parent(i);
            if self.que[parent].1 > self.que[i].1 {
                self.que.swap(i, parent);
                i = parent;
            } else {
                break;
            }
        }
    }

    fn parent(&self, i: usize) -> usize {
        i >> 1
    }

    fn left(&self, i: usize) -> usize {
        i << 1
    }

    fn right(&self, i: usize) -> usize {
        (i << 1) + 1
    }
}

struct Graph {
    edges: Vec<Vec<Edge>>,
    hi: Vec<usize>,
    lim: Vec<usize>,
    potential: Vec<f64>,
    min_cost: Vec<f64>,
    prevv: Vec<usize>,
    preve: Vec<usize>,
    V: usize,
}

impl Graph {
    fn new(n: usize) -> Self {
        Graph {
            edges: vec![Vec::new(); n],
            hi: vec![0; n],
            lim: vec![0; n],
            potential: vec![0.0; n],
            min_cost: vec![INF; n],
            prevv: vec![0; n],
            preve: vec![0; n],
            V: n,
        }
    }

    fn check_lim(&mut self, a: usize) {
        if self.hi[a] < self.lim[a] {
            return;
        }
        if self.lim[a] == 0 {
            self.lim[a] = 32;
            self.hi[a] = 0;
            self.edges[a].resize(self.lim[a], Edge { to: 0, rev: 0, cap: 0, cost: 0.0 });
        } else if self.hi[a] == self.lim[a] {
            let k = self.lim[a].min(128);
            self.lim[a] += k;
            self.edges[a].resize(self.lim[a], Edge { to: 0, rev: 0, cap: 0, cost: 0.0 });
        }
    }

    fn add_edge(&mut self, from: usize, to: usize, cap: i32, cost: f64) {
        self.check_lim(from);
        self.check_lim(to);
        let f = self.hi[from];
        let t = self.hi[to];
        self.edges[from].push(Edge { to, cap, cost, rev: t });
        self.edges[to].push(Edge { to: from, cap: 0, cost: -cost, rev: f });
        self.hi[from] += 1;
        self.hi[to] += 1;
    }

    fn min_cost_flow(&mut self, S: usize, T: usize, F: &mut i32) -> f64 {
        let mut ret = 0.0;
        self.potential.fill(0.0);

        while *F > 0 {
            self.min_cost.fill(INF);
            let mut heap = MinHeap::new();
            heap.enq(S, 0.0);
            self.min_cost[S] = 0.0;

            while !heap.que.is_empty() {
                let (s, t) = heap.deq();
                if self.min_cost[s] < t {
                    continue;
                }

                for i in 0..self.hi[s] {
                    let ep = &self.edges[s][i];
                    let ns = ep.to;
                    let nt = t + ep.cost + self.potential[s] - self.potential[ns];
                    if ep.cap > 0 && self.min_cost[ns] > nt + EPS {
                        self.min_cost[ns] = nt;
                        self.prevv[ns] = s;
                        self.preve[ns] = i;
                        heap.enq(ns, nt);
                    }
                }
            }
            if self.min_cost[T] == INF {
                return -1.0;
            }
            for s in 0..self.V {
                self.potential[s] += self.min_cost[s];
            }

            let mut k = *F;
            let mut s = T;
            while s != S {
                let ep = &self.edges[self.prevv[s]][self.preve[s]];
                if k > ep.cap {
                    k = ep.cap;
                }
                s = self.prevv[s];
            }
            *F -= k;
            ret += k as f64 * self.potential[T];
            s = T;
            while s != S {
                let (prev, edge_index) = (self.prevv[s], self.preve[s]);
                {
                    let ep = &mut self.edges[prev][edge_index];
                    ep.cap -= k;
                }
                {
                    let ep = &mut self.edges[s][self.edges[prev][edge_index].rev];
                    ep.cap += k;
                }
                s = self.prevv[s];
            }
        }
        ret
    }
}

fn gauss_jordan(n: usize, A: &mut Vec<Vec<f64>>, C: &Vec<f64>, X: &mut Vec<f64>) -> bool {
    let mut B = A.clone();
    for i in 0..n {
        B[i].push(C[i]);
    }

    for i in 0..n {
        let mut pivot = i;
        for j in i..n {
            if B[j][i].abs() > B[pivot][i].abs() {
                pivot = j;
            }
        }
        B.swap(i, pivot);
        if B[i][i].abs() < EPS {
            return false; // error
        }

        for j in i + 1..=n {
            B[i][j] /= B[i][i];
        }
        for j in 0..n {
            if i != j {
                for k in i + 1..=n {
                    B[j][k] -= B[j][i] * B[i][k];
                }
            }
        }
    }
    for i in 0..n {
        X[i] = B[i][n];
    }
    true
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines().filter_map(Result::ok);
    let T: usize = lines.next().unwrap().parse().unwrap();

    for _ in 0..T {
        let n: usize = lines.next().unwrap().parse().unwrap();
        let s: usize = lines.next().unwrap().parse().unwrap();
        let t: usize = lines.next().unwrap().parse().unwrap();
        let mut F: i32 = lines.next().unwrap().parse().unwrap();

        let mut A = vec![vec![0.0; n]; n];
        let mut C = vec![0.0; n];
        for i in 0..n {
            for j in 0..n {
                A[i][j] = lines.next().unwrap().parse().unwrap();
            }
            C[i] = lines.next().unwrap().parse().unwrap();
        }

        let mut X = vec![0.0; n];
        if !gauss_jordan(n, &mut A, &C, &mut X) {
            println!("impossible");
            continue;
        }

        let mut graph = Graph::new(n);
        for i in 0..n {
            let M: usize = lines.next().unwrap().parse().unwrap();
            let mut d = vec![0; M];
            for j in 0..M {
                d[j] = lines.next().unwrap().parse().unwrap();
            }
            for j in 0..M {
                let cap: i32 = lines.next().unwrap().parse().unwrap();
                graph.add_edge(i, d[j], cap, (X[i] - X[d[j]]).abs());
            }
        }

        let ans = graph.min_cost_flow(s, t, &mut F);
        if ans < 0.0 {
            println!("impossible");
        } else {
            println!("{:.10}", ans);
        }
    }
}
