```rust
use std::io::{self, BufRead};

const SUP: i64 = 1 << 60;

#[derive(Debug)]
struct Edge {
    t: usize,
    h: usize,
    cost: i64,
}

fn main() {
    let stdin = io::stdin();
    let reader = stdin.lock();
    let mut lines = reader.lines();

    // Read N and M
    let first_line = lines.next().unwrap().unwrap();
    let mut parts = first_line.split_whitespace();
    let n: usize = parts.next().unwrap().parse().unwrap();
    let m: usize = parts.next().unwrap().parse().unwrap();

    let mut edges = Vec::with_capacity(m);
    
    // Read edges
    for _ in 0..m {
        let line = lines.next().unwrap().unwrap();
        let mut parts = line.split_whitespace();
        let u: usize = parts.next().unwrap().parse().unwrap();
        let w: usize = parts.next().unwrap().parse().unwrap();
        let c: i64 = parts.next().unwrap().parse().unwrap();
        edges.push(Edge { t: u, h: w, cost: -c });
    }

    let mut min = vec![SUP; n + 1];
    min[1] = 0; // Initial cost to reach the source node (1)

    // Bellman-Ford algorithm
    for _ in 1..n {
        for edge in &edges {
            let u = edge.t;
            let w = edge.h;
            let c = edge.cost;
            if min[u] < SUP && min[w] > min[u] + c {
                min[w] = min[u] + c;
            }
        }
    }

    let ans = min[n];

    // Check for negative weight cycles
    let mut has_negative_cycle = false;
    for k in 1..=n {
        for edge in &edges {
            let u = edge.t;
            let w = edge.h;
            let c = edge.cost;
            if min[u] < SUP && min[w] > min[u] + c {
                min[w] = min[u] + c;
                if k == n { // If we can still relax in the nth iteration
                    has_negative_cycle = true;
                }
            }
        }
    }

    if has_negative_cycle || ans == SUP {
        println!("inf");
    } else {
        println!("{}", -ans);
    }
}
```
