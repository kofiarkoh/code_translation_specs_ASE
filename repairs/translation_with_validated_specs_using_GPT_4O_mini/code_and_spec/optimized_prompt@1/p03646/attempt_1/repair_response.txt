```rust
use std::io::{self, Write};

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    
    // Parse the input as a signed integer
    let k: i64 = input.trim().parse().expect("Input is not a valid integer");

    if k >= 50 {
        let n = 50;
        let rem = k / 50;
        let base_value = 49 + rem;
        let t = k % 50;

        // Create a vector initialized with the base value
        let mut a = vec![base_value; n as usize];

        // Increment the first t elements by 50
        for i in 0..t {
            a[i as usize] += 50;
            // Decrement the rest of the elements
            for j in 0..n {
                if j != i {
                    a[j as usize] -= 1;
                }
            }
        }

        // Output the results
        println!("{}", n);
        for value in a {
            print!("{} ", value);
        }
        println!(); // Ensure a newline at the end
    } else if k >= 2 {
        let n = k;
        println!("{}", n);
        for _ in 0..n {
            print!("{} ", n);
        }
        println!(); // Ensure a newline at the end
    } else if k == 0 {
        println!("4\n3 3 3 3");
    } else {
        println!("3\n1 0 3");
    }
}
```
