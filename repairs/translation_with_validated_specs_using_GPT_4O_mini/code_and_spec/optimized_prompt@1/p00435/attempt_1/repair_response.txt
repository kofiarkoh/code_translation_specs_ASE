    let input = input.trim();

    if input.len() > 999 {
        // Undefined behavior as per the spec; we will just return.
        return;
    }

    let mut output = String::with_capacity(input.len());
    
    // Transform each character according to the specified rules.
    for c in input.chars() {
        let transformed_char = if ('D'..='Z').contains(&c) {
            // Shift uppercase letters backward by 3 positions.
            (c as u8).wrapping_sub(3) as char
        } else {
            // Shift all other characters forward by 23 positions.
            (c as u8).wrapping_add(23) as char
