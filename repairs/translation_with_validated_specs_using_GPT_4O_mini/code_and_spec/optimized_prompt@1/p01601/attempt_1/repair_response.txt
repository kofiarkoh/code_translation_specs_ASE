```rust
use std::io;

fn rev(n: i32) -> i32 {
    let mut out = 0;
    let mut num = n;

    while num > 0 {
        out = out * 10 + (num % 10);
        num /= 10;
    }

    out
}

fn nearest_palindrome(n: i32) -> i32 {
    let mut i = 0;

    loop {
        if n - i >= 0 {
            if rev(n - i) == n - i {
                return n - i;
            }
        }
        if rev(n + i) == n + i {
            return n + i;
        }
        i += 1;
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let n: i32 = input.trim().parse().expect("Please enter a valid integer");

    // Preconditions check: n must be non-negative
    if n < 0 {
        panic!("Input must be a non-negative integer");
    }

    let result = nearest_palindrome(n);
    println!("{}", result);
}
```
