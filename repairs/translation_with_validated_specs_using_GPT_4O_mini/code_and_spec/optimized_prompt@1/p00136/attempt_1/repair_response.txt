```rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let handle = stdin.lock();

    // Read the number of height measurements
    let mut lines = handle.lines();
    let n: usize = lines.next().unwrap().unwrap().parse().unwrap();

    // Initialize counts for each height range
    let mut counts = [0; 6];

    // Read each height measurement and categorize it
    for _ in 0..n {
        let h: f64 = lines.next().unwrap().unwrap().parse().unwrap();
        if h < 165.0 {
            counts[0] += 1; // Range 1
        } else if h < 170.0 {
            counts[1] += 1; // Range 2
        } else if h < 175.0 {
            counts[2] += 1; // Range 3
        } else if h < 180.0 {
            counts[3] += 1; // Range 4
        } else if h < 185.0 {
            counts[4] += 1; // Range 5
        } else {
            counts[5] += 1; // Range 6
        }
    }

    // Output the counts as asterisks
    for (i, &count) in counts.iter().enumerate() {
        print!("{}:", i + 1);
        for _ in 0..count {
            print!("*");
        }
        println!(); // Newline after each range
    }
}
```
