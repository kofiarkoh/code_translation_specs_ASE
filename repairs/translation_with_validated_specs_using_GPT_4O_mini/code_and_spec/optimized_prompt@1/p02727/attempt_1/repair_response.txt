```rust
use std::io::{self, BufRead};
use std::cmp::Ordering;

fn mizuhasu(x: &i32, y: &i32) -> Ordering {
    y.cmp(x) // Sort in descending order
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let stdin = io::stdin();
    let mut input = stdin.lock().lines();

    // Read X, Y, A, B, C
    let first_line = input.next().ok_or("Failed to read input")??;
    let mut values: Vec<i32> = first_line.split_whitespace()
        .map(|s| s.parse().unwrap())
        .collect();

    if values.len() != 5 {
        return Err("Expected exactly five integers".into());
    }

    let (x, y, a, b, c) = (values[0], values[1], values[2], values[3], values[4]);

    // Read first array
    let mut p: Vec<i32> = input.next().ok_or("Failed to read input")??.split_whitespace()
        .take(a as usize)
        .map(|s| s.parse().unwrap())
        .collect();

    // Read second array
    let mut q: Vec<i32> = input.next().ok_or("Failed to read input")??.split_whitespace()
        .take(b as usize)
        .map(|s| s.parse().unwrap())
        .collect();

    // Read third array
    let mut r: Vec<i32> = input.next().ok_or("Failed to read input")??.split_whitespace()
        .take(c as usize)
        .map(|s| s.parse().unwrap())
        .collect();

    // Sort arrays in descending order
    p.sort_by(mizuhasu);
    q.sort_by(mizuhasu);
    r.sort_by(mizuhasu);

    let mut asumi = (x - 1) as isize; // Index for first array
    let mut iguti = (y - 1) as isize; // Index for second array

    for &value in &r {
        if asumi < 0 && iguti < 0 {
            break;
        } else if asumi < 0 || (iguti >= 0 && p[asumi as usize] > q[iguti as usize]) {
            if value > q[iguti as usize] {
                q[iguti as usize] = value;
                iguti -= 1;
            } else {
                break;
            }
        } else { // asumi >= 0 && (iguti < 0 || p[asumi] <= q[iguti])
            if value > p[asumi as usize] {
                p[asumi as usize] = value;
                asumi -= 1;
            } else {
                break;
            }
        }
    }

    // Calculate the sum of the first X elements from p and first Y elements from q
    let kitaeri: i64 = p.iter().take(x as usize).sum::<i32>() as i64
        + q.iter().take(y as usize).sum::<i32>() as i64;

    println!("{}", kitaeri);
    Ok(())
}
```
