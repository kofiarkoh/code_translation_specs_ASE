void game_init(void) {
	int x,y;
	for(y=0;y<height;y++) {
		for(x=0;x<width;x++) {
			if(map[y][x]=='S') {
				px=x;py=y;

				gx=x;gy=y;

	gameover=0;
}

void moveto(int to) {
	int old_px=px;
	int i;
	int add;
	if(to<1 || to>width)return;
	to--;
	add=(to<px)?-1:1;
	if(map[py][to]!='.')return;

	for(i=px;i!=to;i+=add) {
		if(map[py+1][i]=='.')break;

}
