```rust
use std::io::{self, Write};

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).expect("Failed to read line");
    let counter: usize = input.trim().parse().expect("Input must be a positive integer");

    for case_number in 0..counter {
        input.clear();
        io::stdin().read_line(&mut input).expect("Failed to read line");
        let mut x: i32 = input.trim().parse().expect("Input must be a valid integer");

        const ITERATIONS: usize = 10;
        println!("Case {}:", case_number + 1);
        
        for _ in 0..ITERATIONS {
            x = (x * x) / 100 % 10000; // Perform the calculations as specified
            println!("{}", x);
        }
    }
}
```
