```rust
use std::io::{self, Read};

type Ll = i64;

fn main() {
    // Read all input; C uses fgets twice, but the effective behavior is just parsing n then numbers.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let bytes = input.as_bytes();
    let mut pos: usize = 0;

    // C's atoi on the first line: skip leading whitespace, optional sign, then digits.
    while pos < bytes.len() && bytes[pos].is_ascii_whitespace() {
        pos += 1;
    }
    let mut sign: i64 = 1;
    if pos < bytes.len() && (bytes[pos] == b'+' || bytes[pos] == b'-') {
        if bytes[pos] == b'-' {
            sign = -1;
        }
        pos += 1;
    }
    let mut n_i64: i64 = 0;
    while pos < bytes.len() && bytes[pos].is_ascii_digit() {
        n_i64 = n_i64 * 10 + (bytes[pos] - b'0') as i64;
        pos += 1;
    }
    let n: usize = (n_i64 * sign) as usize;

    // Advance to the start of the next line (mimic the second fgets).
    while pos < bytes.len() && bytes[pos] != b'\n' {
        pos += 1;
    }
    if pos < bytes.len() && bytes[pos] == b'\n' {
        pos += 1;
    }

    // getlong() as in C:
    // while (*p >= '0') n = n*10 + (*p & 0xf); p++; return n;
    // Important: in the original, it assumes there is a delimiter after each number.
    // Here we guard bounds to avoid panics on the last number / EOF.
    let mut getlong = |p: &mut usize| -> Ll {
        let mut v: Ll = 0;
        while *p < bytes.len() && bytes[*p] >= b'0' {
            v = (v << 3) + (v << 1) + ((bytes[*p] & 0x0f) as Ll);
            *p += 1;
        }
        if *p < bytes.len() {
            *p += 1; // skip one delimiter if present
        }
        v
    };

    if n == 0 {
        // The C program would access dp[0][n-1] (UB). For AOJ inputs n>0.
        // Keep a defined output to avoid crashing on malformed input.
        print!("0\n");
        return;
    }

    let mut w = vec![0i64; n];
    w[0] = getlong(&mut pos);
    for i in 1..n {
        w[i] = w[i - 1] + getlong(&mut pos);
    }

    // dp initialized with memset 0x33 => each byte 0x33, i64 becomes 0x3333333333333333
    const INF: i64 = 0x3333_3333_3333_3333u64 as i64;
    let mut dp = vec![vec![INF; n]; n];
    let mut k = vec![vec![0usize; n]; n];

    for i in 0..n {
        dp[i][i] = 0;
        k[i][i] = i;
    }

    // This is Knuth optimization; the C loops rely on k[i+1][j] being valid.
    // Therefore, a must be <= n-1 (interval length), not <= n.
    for a in 1..n {
        for i in 0..(n - a) {
            let j = i + a;
            let b_start = k[i][j - 1];
            let b_end = k[i + 1][j];
            for b in b_start..=b_end {
                let mut c = dp[i][b] + dp[b + 1][j] + w[j];
                if i > 0 {
                    c -= w[i - 1];
                }
                if dp[i][j] > c {
                    dp[i][j] = c;
                    k[i][j] = b;
                }
            }
        }
    }

    print!("{}\n", dp[0][n - 1]);
}
```
