```rust
use std::io::{self, Read};

const MAX: usize = 100;
const PGMAX: usize = 1001;
const WDMAX: usize = 31;

#[derive(Clone)]
struct Index {
    pgnum: [i32; PGMAX],
    tm: i32,
    str_: String,
}

fn main() {
    // C: index_t idx[MAX]; (uninitialized in C, but all fields are written before read
    // except pgnum which is only ever compared to 1; we must zero it to match intent.)
    let mut idx: Vec<Index> = (0..MAX)
        .map(|_| Index {
            pgnum: [0; PGMAX],
            tm: 0,
            str_: String::new(),
        })
        .collect();

    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut rgnum: usize = 0;

    // C loop: while (scanf("%s %d", tmp_str, &tmp_pgnum) != EOF)
    // Note: scanf returns number of items assigned (0,1,2) or EOF (-1).
    // The C code continues for return values 0 or 1 (since != EOF), which can lead to
    // using stale tmp_str/tmp_pgnum. We emulate this behavior by keeping previous values
    // and continuing until we truly hit end-of-input.
    let mut tmp_str = String::new();
    let mut tmp_pgnum: i32 = 0;

    loop {
        let next_s = it.next();
        if next_s.is_none() {
            break; // EOF
        }
        let next_pg = it.next();

        // Update tmp_str if present (scanf would write it only if %s succeeds).
        if let Some(s) = next_s {
            // C "%s" into char[WDMAX] can overflow; we approximate by truncating to WDMAX-1.
            tmp_str = s.chars().take(WDMAX - 1).collect();
        }

        // Update tmp_pgnum only if parse succeeds (scanf would assign only on success).
        if let Some(pg_s) = next_pg {
            if let Ok(v) = pg_s.parse::<i32>() {
                tmp_pgnum = v;
            } else {
                // scanf would return 1 here (only %s assigned) and continue; keep old tmp_pgnum.
            }
        } else {
            // Missing second token: scanf would return 1 and continue; keep old tmp_pgnum.
        }

        // Now execute the body using current tmp_str/tmp_pgnum (possibly stale), like C.
        let mut flag = true;

        for i in 0..rgnum {
            if idx[i].str_ == tmp_str {
                // C has UB if tmp_pgnum is out of bounds; avoid panic and just skip the write.
                if (0..PGMAX as i32).contains(&tmp_pgnum) {
                    idx[i].pgnum[tmp_pgnum as usize] = 1;
                }
                idx[i].tm = idx[i].tm.wrapping_add(1);
                flag = false;
                break;
            }
        }

        if flag {
            if rgnum >= MAX {
                // C would write out of bounds; stop to avoid memory corruption.
                break;
            }
            if (0..PGMAX as i32).contains(&tmp_pgnum) {
                idx[rgnum].pgnum[tmp_pgnum as usize] = 1;
            }
            idx[rgnum].tm = idx[rgnum].tm.wrapping_add(1);
            idx[rgnum].str_ = tmp_str.clone();
            rgnum += 1;
        }
    }

    // C uses a nested-loop swap sort; stable/unstable doesn't matter because equal keys
    // are identical strings. Use Rust sort for same final ordering.
    idx[..rgnum].sort_by(|a, b| a.str_.cmp(&b.str_));

    // Output: for each word, print word then list pages where pgnum[j]==1.
    // C prints exactly tm numbers (with spaces) and stops after printing the (tm-1)th index.
    // If tm exceeds the number of set bits, C would print fewer numbers and no trailing newline
    // for that entry; we preserve that behavior.
    let mut out = String::new();
    for i in 0..rgnum {
        out.push_str(&idx[i].str_);
        out.push('\n');

        let mut k: i32 = 0;
        for j in 0..PGMAX {
            if idx[i].pgnum[j] == 1 {
                if k == idx[i].tm.wrapping_sub(1) {
                    out.push_str(&format!("{j}\n"));
                    break;
                } else {
                    out.push_str(&format!("{j} "));
                    k = k.wrapping_add(1);
                }
            }
        }
    }

    print!("{out}");
}
```
