```rust
use std::io::{self, Read};

fn main() {
    // C code:
    // m[1<<18];*p;main(n,a,b){
    //   for(scanf("%d%*d",&n);~scanf("%d%d",&a,&b);p-m&&*p?n=0:++*p)
    //     p=m+(a-1?b-n?0:a:b);
    //   puts(n?"IMPOSSIBLE":"POSSIBLE");
    // }
    //
    // Key semantic fix vs previous attempt:
    // The first scanf is "%d%*d": it reads n and discards ONE MORE integer.
    // Many inputs only provide a single integer on the first line; in C, scanf
    // would then return 1 (not EOF), leaving the next integer to be read by the
    // loop's scanf("%d%d"). We must NOT unconditionally discard a token; only
    // discard if it actually exists in the input stream at that point.
    //
    // We implement a small scanf-like parser that can:
    // - read an i32 if present
    // - attempt to discard an i32, but only if present (matching scanf's behavior)

    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let bytes = s.as_bytes();
    let mut i = 0usize;

    fn skip_ws(bytes: &[u8], i: &mut usize) {
        while *i < bytes.len() && bytes[*i].is_ascii_whitespace() {
            *i += 1;
        }
    }

    fn read_i32(bytes: &[u8], i: &mut usize) -> Option<i32> {
        skip_ws(bytes, i);
        if *i >= bytes.len() {
            return None;
        }
        let mut sign = 1i32;
        if bytes[*i] == b'+' {
            *i += 1;
        } else if bytes[*i] == b'-' {
            sign = -1;
            *i += 1;
        }
        if *i >= bytes.len() || !bytes[*i].is_ascii_digit() {
            return None;
        }
        let mut val: i32 = 0;
        while *i < bytes.len() && bytes[*i].is_ascii_digit() {
            let d = (bytes[*i] - b'0') as i32;
            // C's scanf into int has undefined behavior on overflow; we use wrapping
            // to avoid panics and approximate typical two's-complement behavior.
            val = val.wrapping_mul(10).wrapping_add(d);
            *i += 1;
        }
        Some(val.wrapping_mul(sign))
    }

    // scanf("%d%*d",&n)
    let mut n: i32 = match read_i32(bytes, &mut i) {
        Some(v) => v,
        None => 0, // if missing, C would leave n uninitialized; choose 0 to stay defined.
    };
    // %*d: discard one int ONLY if it can be read
    let _ = read_i32(bytes, &mut i);

    let mut m = vec![0i32; 1 << 18];
    let mut p_idx: usize = 0;

    // for (; ~scanf("%d%d",&a,&b); post)
    loop {
        let a = match read_i32(bytes, &mut i) {
            Some(v) => v,
            None => break,
        };
        let b = match read_i32(bytes, &mut i) {
            Some(v) => v,
            None => break,
        };

        // p = m + (a-1 ? (b-n ? 0 : a) : b);
        let offset: i32 = if a.wrapping_sub(1) != 0 {
            if b.wrapping_sub(n) != 0 {
                0
            } else {
                a
            }
        } else {
            b
        };

        // In the original C, out-of-bounds would be UB; contest inputs should avoid it.
        // We'll clamp by treating OOB as immediate "POSSIBLE" (n=0) and keep p at 0.
        if offset < 0 || (offset as usize) >= m.len() {
            n = 0;
            p_idx = 0;
        } else {
            p_idx = offset as usize;
        }

        // post: p-m&&*p ? n=0 : ++*p
        if p_idx != 0 && m[p_idx] != 0 {
            n = 0;
        } else {
            m[p_idx] = m[p_idx].wrapping_add(1);
        }
    }

    if n != 0 {
        print!("IMPOSSIBLE\n");
    } else {
        print!("POSSIBLE\n");
    }
}
```
