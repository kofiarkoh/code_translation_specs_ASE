```rust
use std::io::{self, Read};

const MAX: usize = 602;

#[derive(Clone, Copy)]
struct T {
    used: i32,
    m: i32,
    mx: i32,
    xl: i32,
    xr: i32,
    par: i32,
    ch: i32,
    sib: i32,
}

impl Default for T {
    fn default() -> Self {
        T {
            used: 0,
            m: 0,
            mx: 0,
            xl: 0,
            xr: 0,
            par: 0,
            ch: 0,
            sib: 0,
        }
    }
}

/* UNION-FIND */
fn find_set(p: &mut [i32], x: i32) -> i32 {
    let xi = x as usize;
    if p[xi] != x {
        let root = find_set(p, p[xi]);
        p[xi] = root;
    }
    p[xi]
}

fn link(p: &mut [i32], rank: &mut [i32], x: i32, y: i32) {
    let xi = x as usize;
    let yi = y as usize;
    if rank[xi] > rank[yi] {
        p[yi] = x;
    } else {
        p[xi] = y;
        if rank[xi] == rank[yi] {
            rank[yi] = rank[yi] + 1;
        }
    }
}

fn union_set(p: &mut [i32], rank: &mut [i32], x: i32, y: i32) {
    let rx = find_set(p, x);
    let ry = find_set(p, y);
    link(p, rank, rx, ry);
}

fn mktree(
    r: i32,
    c: i32,
    this: i32,
    w: i32,
    h: i32,
    map: &[Vec<u8>],
    rel: &Vec<Vec<i32>>,
    tbl: &mut [T],
    tr: &[i32],
) {
    if r == h - 1 {
        if map[r as usize][c as usize] == b'.' {
            return;
        }
        let t = &mut tbl[this as usize];
        if t.xl < 0 {
            t.xl = c;
        } else if c < t.xl {
            t.xl = c;
        }
        if c > t.xr {
            t.xr = c;
        }
        return;
    }
    if map[(r + 1) as usize][c as usize] == b'.' {
        return;
    }

    let k = rel[(r + 1) as usize][c as usize] as usize;
    let pidx = tr[k];
    if pidx == this {
        return;
    }

    {
        let t = &mut tbl[this as usize];
        if t.xl < 0 {
            t.xl = c;
        } else if c < t.xl {
            t.xl = c;
        }
        if c > t.xr {
            t.xr = c;
        }
        if t.par != 0 {
            return;
        }
    }

    let p = pidx;
    let mut s = tbl[p as usize].ch;
    if s == 0 {
        tbl[p as usize].ch = this;
    } else {
        while tbl[s as usize].sib != 0 {
            s = tbl[s as usize].sib;
        }
        tbl[s as usize].sib = this;
    }
    tbl[this as usize].par = p;

    let _ = w; // kept to mirror signature/intent; unused like in C.
}

fn check(this: i32, tbl: &mut [T]) -> bool {
    if tbl[this as usize].used != 0 {
        return true;
    }
    let ch = tbl[this as usize].ch;
    if ch != 0 && !check(ch, tbl) {
        return false;
    }

    let mut sib = tbl[this as usize].sib;
    while sib != 0 {
        if !check(sib, tbl) {
            return false;
        }
        sib = tbl[sib as usize].sib;
    }

    // C uses double: x = 0.5 + (double)mx / m;
    // m is positive for any real node; keep same behavior.
    let m = tbl[this as usize].m as f64;
    let mx = tbl[this as usize].mx as f64;
    let x = 0.5f64 + mx / m;

    let xl = tbl[this as usize].xl as f64;
    let xr = tbl[this as usize].xr as f64;
    if x <= xl || (xr + 1.0) <= x {
        return false;
    }

    let par = tbl[this as usize].par;
    tbl[par as usize].m += tbl[this as usize].m;
    tbl[par as usize].mx += tbl[this as usize].mx;
    tbl[this as usize].used = 1;
    true
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mv: [(i32, i32); 4] = [(-1, 0), (0, 1), (1, 0), (0, -1)];

    let mut out = String::new();

    loop {
        let w_opt = it.next();
        if w_opt.is_none() {
            break;
        }
        let w: i32 = w_opt.unwrap().parse().unwrap();
        let h: i32 = it.next().unwrap().parse().unwrap();
        if w <= 0 {
            break;
        }

        let mut p = vec![0i32; MAX];
        let mut rank = vec![0i32; MAX];

        let mut map: Vec<Vec<u8>> = vec![vec![b'.'; w as usize]; h as usize];
        let mut rel: Vec<Vec<i32>> = vec![vec![0i32; w as usize]; h as usize];

        let mut idx: i32 = 0;
        for r in 0..h {
            let s = it.next().unwrap().as_bytes().to_vec();
            // C reads into fixed buffer; assume input line length == w.
            for c in 0..w {
                map[r as usize][c as usize] = s[c as usize];
                p[idx as usize] = idx;
                rel[r as usize][c as usize] = idx;
                idx += 1;
            }
        }

        for r in 0..h {
            for c in 0..w {
                if map[r as usize][c as usize] == b'.' {
                    continue;
                }
                for (dr, dc) in mv.iter() {
                    let r2 = r + dr;
                    let c2 = c + dc;
                    if 0 <= r2 && r2 < h && 0 <= c2 && c2 < w {
                        if map[r2 as usize][c2 as usize] == map[r as usize][c as usize] {
                            let a = rel[r as usize][c as usize];
                            let b = rel[r2 as usize][c2 as usize];
                            union_set(&mut p, &mut rank, a, b);
                        }
                    }
                }
            }
        }

        let mut tbl = vec![T::default(); 100];
        let mut tr = vec![-1i32; MAX];
        let mut sz: i32 = 1;

        for r in (0..h).rev() {
            for c in 0..w {
                if map[r as usize][c as usize] == b'.' {
                    continue;
                }
                let k0 = rel[r as usize][c as usize];
                let k = find_set(&mut p, k0);
                rel[r as usize][c as usize] = k;

                let ku = k as usize;
                let i: i32;
                if tr[ku] < 0 {
                    i = sz;
                    tr[ku] = sz;
                    sz += 1;
                    tbl[i as usize].xl = -1; // matches C initialization for new component
                } else {
                    i = tr[ku];
                }

                tbl[i as usize].m += 1;
                tbl[i as usize].mx += c;

                mktree(r, c, i, w, h, &map, &rel, &mut tbl, &tr);
            }
        }
        tbl[1].par = 1;

        if check(1, &mut tbl) {
            out.push_str("STABLE\n");
        } else {
            out.push_str("UNSTABLE\n");
        }
    }

    print!("{out}");
}
```
