```rust
use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;

fn main() {
    // Read entire stdin, parse two integers like scanf("%d %d", ...)
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return, // If no input, do nothing (closest to failing scanf and then UB in C).
    };
    let m: i32 = it.next().unwrap().parse().unwrap();

    // dp[302][302] = {} in C => zero-initialized.
    // Use i64 to match C's long long.
    let mut dp = vec![vec![0i64; 302]; 302];

    // for (k = 1, dp[1][1] = 1; k <= M; k++)
    dp[1][1] = 1;
    for k in 1..=m {
        // for (i = k; i >= 1; i--)
        let mut i = k;
        while i >= 1 {
            // for (j = i; j >= 1; j--)
            let mut j = i;
            while j >= 1 {
                let ii = i as usize;
                let jj = j as usize;

                // tmp = dp[i][j] % Mod; dp[i][j] = 0;
                // In C, % on negative keeps the sign of the dividend; Rust matches that.
                let tmp = dp[ii][jj] % MOD;
                dp[ii][jj] = 0;

                // dp[i+1][j] += tmp * (N - i);
                // dp[i][i]   += tmp * j;
                // dp[i][j]   += tmp * (i - j);
                //
                // Use wrapping arithmetic to emulate potential signed overflow UB in C
                // as closely as possible to typical two's-complement wraparound.
                let ni = (n - i) as i64;
                let jv = j as i64;
                let ij = (i - j) as i64;

                dp[(i + 1) as usize][jj] = dp[(i + 1) as usize][jj]
                    .wrapping_add(tmp.wrapping_mul(ni));
                dp[ii][ii] = dp[ii][ii].wrapping_add(tmp.wrapping_mul(jv));
                dp[ii][jj] = dp[ii][jj].wrapping_add(tmp.wrapping_mul(ij));

                j -= 1;
            }
            i -= 1;
        }
    }

    // printf("%lld\n", dp[N][N] % Mod);
    // Note: if n is out of bounds (>=302 or negative), C would have UB.
    // Here we assume inputs keep indices in range, as the C code implicitly does.
    let ans = dp[n as usize][n as usize] % MOD;
    print!("{ans}\n");
    // stdout is line-buffered in many environments; explicit flush not required for correctness here.
}
```
