// Emulate C gets() into a destination buffer that represents a C string.
// Copies up to dest.len()-1 bytes and always NUL-terminates.
fn gets_into(dest: &mut [u8], lines: &mut std::str::SplitTerminator<'_, char>) -> bool {
    let Some(mut s) = lines.next() else { return false };
    if s.ends_with('\r') {
        s = &s[..s.len() - 1];
    }
    let bytes = s.as_bytes();
    let max = dest.len().saturating_sub(1);
    let n = bytes.len().min(max);
    dest[..n].copy_from_slice(&bytes[..n]);
    if !dest.is_empty() {
        dest[n] = 0;
        for b in &mut dest[n + 1..] {
            *b = 0;
        }
    }
    true
}

fn parse_int_from_cbuf(buf: &[u8]) -> i32 {
    let end = buf.iter().position(|&b| b == 0).unwrap_or(buf.len());
    let s = std::str::from_utf8(&buf[..end]).unwrap_or("");
    s.split_whitespace()
        .next()
        .and_then(|tok| tok.parse::<i32>().ok())
        .unwrap_or(0)
}
