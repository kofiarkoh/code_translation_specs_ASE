```rust
use std::io::{self, Read};

const QMAX: usize = 150000;

#[derive(Clone, Copy, Default)]
struct Que {
    t: i32,
    r: i16,
    c: i16,
}

struct MinHeap {
    a: [Que; QMAX],
    size: usize,
}

impl MinHeap {
    fn new() -> Self {
        Self {
            a: [Que::default(); QMAX],
            size: 0,
        }
    }

    #[inline]
    fn parent(i: usize) -> usize {
        i >> 1
    }
    #[inline]
    fn left(i: usize) -> usize {
        i << 1
    }
    #[inline]
    fn right(i: usize) -> usize {
        (i << 1) + 1
    }

    fn min_heapify(&mut self, mut i: usize) {
        loop {
            let l = Self::left(i);
            let r = Self::right(i);
            let mut min = i;

            if l < self.size && self.a[l].t < self.a[i].t {
                min = l;
            }
            if r < self.size && self.a[r].t < self.a[min].t {
                min = r;
            }
            if min == i {
                break;
            }
            self.a.swap(i, min);
            i = min;
        }
    }

    fn deq(&mut self) {
        // C code assumes size > 0 when called.
        self.size -= 1;
        self.a[0] = self.a[self.size];
        self.min_heapify(0);
    }

    fn enq(&mut self, r: i32, c: i32, t: i32) {
        let mut i = self.size;
        self.size += 1;
        self.a[i].r = r as i16;
        self.a[i].c = c as i16;
        self.a[i].t = t;

        while i > 0 {
            let p = Self::parent(i);
            if self.a[p].t <= self.a[i].t {
                break;
            }
            self.a.swap(i, p);
            i = p;
        }
    }
}

struct Scanner {
    buf: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            buf: input.into_bytes(),
            idx: 0,
        }
    }

    // Emulates the C in(): reads an unsigned integer by consuming bytes and
    // stopping when the next byte is < '0'. It does not skip non-digits first;
    // it assumes the first read is a digit (as in the original).
    fn in_i32(&mut self) -> i32 {
        let mut n: i32 = 0;
        if self.idx >= self.buf.len() {
            return 0;
        }
        let mut c = self.buf[self.idx];
        self.idx += 1;
        loop {
            n = n
                .wrapping_shl(3)
                .wrapping_add(n.wrapping_shl(1))
                .wrapping_add((c & 0x0f) as i32);
            if self.idx >= self.buf.len() {
                break;
            }
            c = self.buf[self.idx];
            self.idx += 1;
            if c < b'0' {
                break;
            }
        }
        n
    }
}

fn main() {
    let mut sc = Scanner::new();

    // Fixed-size arrays to mirror the C globals.
    let mut l = [[0i32; 501]; 501];
    let mut mk = [[0u8; 501]; 501];
    let mut level = [[0i32; 100001]; 2];
    let mut lsz = [0usize; 2];
    let mv = [(-1i32, 0i32), (0, 1), (1, 0), (0, -1)];

    let mut heap = MinHeap::new();

    loop {
        let r_budget = sc.in_i32();
        if r_budget == 0 {
            break;
        }
        let r_budget_usize = r_budget as usize;

        for side in 0..2 {
            let w = sc.in_i32() as usize;
            let h = sc.in_i32() as usize;
            let x = (sc.in_i32() - 1) as i32;
            let y = (sc.in_i32() - 1) as i32;

            for rr in 0..h {
                for cc in 0..w {
                    l[rr][cc] = sc.in_i32();
                }
            }

            // memset(mk, 0, sizeof(mk))
            for rr in 0..=500 {
                mk[rr].fill(0);
            }

            heap.size = 0;
            level[side][0] = 0;
            let mut maxv: i32 = 0;
            lsz[side] = 1;

            heap.enq(y, x, l[y as usize][x as usize]);
            mk[y as usize][x as usize] = 1;

            while heap.size != 0 {
                let t = heap.a[0].t;
                let rr = heap.a[0].r as i32;
                let cc = heap.a[0].c as i32;
                heap.deq();

                if t > maxv {
                    maxv = t;
                }
                // C writes then increments lsz
                let idx = lsz[side];
                if idx < 100001 {
                    level[side][idx] = maxv;
                }
                lsz[side] += 1;

                for &(dr, dc) in &mv {
                    let r2 = rr + dr;
                    let c2 = cc + dc;
                    if r2 < 0 || r2 >= h as i32 || c2 < 0 || c2 >= w as i32 {
                        continue;
                    }
                    let ur2 = r2 as usize;
                    let uc2 = c2 as usize;
                    if mk[ur2][uc2] != 0 {
                        continue;
                    }
                    // Same room is never enqueued twice.
                    mk[ur2][uc2] = 1;
                    heap.enq(r2, c2, l[ur2][uc2]);
                }
            }
        }

        let mut ans: i32 = 0x7ffffff; // matches C literal
        for i in 0..=r_budget_usize {
            let j = r_budget_usize - i;
            if i < lsz[0] && j < lsz[1] {
                let t = level[0][i].wrapping_add(level[1][j]);
                if t < ans {
                    ans = t;
                }
            }
        }

        print!("{ans}\n");
    }
}
```
