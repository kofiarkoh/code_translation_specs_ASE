```rust
use std::io::{self, Read};

const FACT8: u64 = 40320;
const N_OF_PRINCES: usize = 8;

// Global-like state (kept in a struct to stay safe/idiomatic).
struct Solver {
    n: usize,
    result: [i32; 70],
    memo: [[[[u32; 3]; 3]; 9]; 70], // memo[pos][princes][sitdown][zero_sitdown]
}

impl Solver {
    fn new(n: usize) -> Self {
        Self {
            n,
            result: [0; 70],
            memo: [[[[0u32; 3]; 3]; 9]; 70],
        }
    }

    #[allow(dead_code)]
    fn dump(&self, _f: i32) {
        // DEBUG-only in C; no output in normal builds to preserve I/O behavior.
    }

    fn calc_odd_1(&mut self, pos: usize, princes: usize, sitdown: usize, zero_sitdown: usize) -> u32 {
        let cached = self.memo[pos][princes][sitdown][zero_sitdown];
        if cached != 0 {
            return cached;
        }

        let ret: u32;
        if princes == 0 {
            self.dump(0);
            ret = 1;
        } else if pos == self.n - 1 && princes > 1 {
            ret = 0;
        } else if pos == self.n - 1 && princes == 1 {
            if zero_sitdown == 0 && sitdown == 0 {
                self.dump(1);
                ret = 1;
            } else {
                ret = 0;
            }
        } else {
            let mut acc: u32 = 0;
            if sitdown == 0 {
                self.result[pos] = 1;
                acc = acc.wrapping_add(self.calc_odd_1(pos + 1, princes - 1, 1, zero_sitdown));
            }
            self.result[pos] = 0;
            acc = acc.wrapping_add(self.calc_odd_1(pos + 1, princes, 0, zero_sitdown));
            ret = acc;
        }

        self.memo[pos][princes][sitdown][zero_sitdown] = ret;
        ret
    }

    fn calc_odd(&mut self) -> u32 {
        let mut ret: u32 = 0;

        self.result[0] = 1;
        ret = ret.wrapping_add(self.calc_odd_1(1, N_OF_PRINCES - 1, 1, 1));
        self.result[0] = 0;
        ret = ret.wrapping_add(self.calc_odd_1(1, N_OF_PRINCES, 0, 0));

        ret
    }

    fn calc_even_1(&mut self, pos: usize, princes: usize, sitdown: usize, zero_sitdown: usize) -> u32 {
        let cached = self.memo[pos][princes][sitdown][zero_sitdown];
        if cached != 0 {
            return cached;
        }

        let ret: u32;
        if princes == 0 {
            self.dump(0);
            ret = 1;
        } else if pos == self.n / 2 - 1 && princes > 1 {
            ret = 0;
        } else if pos == self.n / 2 - 1 && princes == 1 {
            if zero_sitdown == 0 && sitdown == 0 {
                self.dump(2);
                ret = 2;
            } else if (zero_sitdown == 2 || zero_sitdown == 0) && (sitdown == 1 || sitdown == 0) {
                self.dump(3);
                ret = 1;
            } else if (zero_sitdown == 1 || zero_sitdown == 0) && (sitdown == 2 || sitdown == 0) {
                self.dump(4);
                ret = 1;
            } else {
                ret = 0;
            }
        } else {
            let mut acc: u32 = 0;

            if sitdown == 0 || sitdown == 2 {
                self.result[pos] = 1;
                self.result[pos + self.n / 2] = 0;
                acc = acc.wrapping_add(self.calc_even_1(pos + 1, princes - 1, 1, zero_sitdown));
            }
            if sitdown == 0 || sitdown == 1 {
                self.result[pos] = 0;
                self.result[pos + self.n / 2] = 1;
                acc = acc.wrapping_add(self.calc_even_1(pos + 1, princes - 1, 2, zero_sitdown));
            }

            self.result[pos] = 0;
            self.result[pos + self.n / 2] = 0;
            acc = acc.wrapping_add(self.calc_even_1(pos + 1, princes, 0, zero_sitdown));

            ret = acc;
        }

        self.memo[pos][princes][sitdown][zero_sitdown] = ret;
        ret
    }

    fn calc_even(&mut self) -> u32 {
        let mut ret: u32 = 0;

        self.result[0] = 1;
        self.result[self.n / 2] = 0;
        ret = ret.wrapping_add(self.calc_even_1(1, N_OF_PRINCES - 1, 1, 1)); // zenhan

        self.result[0] = 0;
        self.result[self.n / 2] = 1;
        ret = ret.wrapping_add(self.calc_even_1(1, N_OF_PRINCES - 1, 2, 2)); // kouhan

        self.result[0] = 0;
        self.result[self.n / 2] = 0;
        ret = ret.wrapping_add(self.calc_even_1(1, N_OF_PRINCES, 0, 0)); // none

        ret
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    while let Some(tok) = it.next() {
        let n_i32: i32 = match tok.parse() {
            Ok(v) => v,
            Err(_) => continue,
        };
        if n_i32 == 0 {
            break;
        }

        // C code assumes N is a valid positive int for the problem.
        // If negative appears, casting to usize would be nonsensical; we mimic
        // "best effort" by treating it as 0-sized and producing 0.
        let n: usize = if n_i32 > 0 { n_i32 as usize } else { 0 };

        let ret: u64;
        if n <= 16 {
            ret = 0;
        } else {
            let mut solver = Solver::new(n);
            if (n & 1) == 1 {
                ret = FACT8 * (solver.calc_odd() as u64);
            } else {
                ret = FACT8 * (solver.calc_even() as u64);
            }
        }

        out.push_str(&format!("{ret}\n"));
    }

    print!("{out}");
}
```
