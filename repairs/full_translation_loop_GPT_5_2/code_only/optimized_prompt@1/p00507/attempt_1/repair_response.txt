use std::io::{self, Read};

fn main() {
    // Read entire stdin, parse as i32 like scanf("%d")
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };
    let n_usize = if n < 0 { 0 } else { n as usize };

    // A has fixed size 10,000,000 in C; we only need first N elements.
    let mut a_vec: Vec<i32> = Vec::with_capacity(n_usize);
    for _ in 0..n_usize {
        if let Some(s) = it.next() {
            a_vec.push(s.parse::<i32>().unwrap());
        } else {
            a_vec.push(0);
        }
    }

    // In the C code, R is int[3] initialized with 9999999999, which overflows
    // a 32-bit signed int. On typical two's-complement systems this wraps to:
    // 9999999999 mod 2^32 = 1410065407.
    // We emulate that wrap explicitly.
    let init_r: i32 = 9_999_999_999u64 as u32 as i32;
    let mut r: [i32; 3] = [init_r, init_r, init_r];

    for i in 0..n_usize {
        for j in 0..n_usize {
            if i != j {
                // for(k=10; A[j]%k!=A[j]; k*=10);
                // This loop can overflow k in C (undefined), but for typical inputs
                // it finds the smallest power of 10 greater than |A[j]| (for non-negative).
                // We emulate i32 wrapping on k*=10 and on %.
                let mut k: i32 = 10;
                loop {
                    let aj = a_vec[j];
                    if aj.wrapping_rem(k) == aj {
                        break;
                    }
                    k = k.wrapping_mul(10);
                }

                // a = A[i]*k + A[j]; with int overflow/wraparound
                let mut a: i32 = a_vec[i]
                    .wrapping_mul(k)
                    .wrapping_add(a_vec[j]);

                // for(k=0;k<3;k++) if(R[k]>a) swap via XOR trick
                for kk in 0..3 {
                    if r[kk] > a {
                        // XOR-swap sequence: a^=R; R^=a; a^=R;
                        // Use wrapping ops to match C int behavior.
                        a = a.wrapping_bitxor(r[kk]);
                        r[kk] = r[kk].wrapping_bitxor(a);
                        a = a.wrapping_bitxor(r[kk]);
                    }
                }
            }
        }
    }

    // printf("%d\n", R[2]);
    print!("{}\n", r[2]);
}
