```rust
use std::io::{self, Read};

const M: i64 = 1_000_000_007;

fn mbk(mut a: i64, mut n: i32, m: i64) -> i64 {
    // Mirrors the recursive C version: fast pow mod, with n as int.
    let mut res: i64 = 1;
    a %= m;
    while n != 0 {
        if (n & 1) != 0 {
            res = (res * a) % m;
        }
        a = (a * a) % m;
        n >>= 1;
    }
    res
}

fn inv(a: i64, m: i64) -> i64 {
    // C: inv(int a,int m){return mbk((long long)a,m-2,m);}
    // Here m is prime in the code (1e9+7), but we follow the same computation.
    mbk(a.rem_euclid(m), (m - 2) as i32, m)
}

struct Solver {
    ta: Vec<i32>,
    nt: Vec<i32>,
    to: Vec<i32>,
    t_s: Vec<i32>,
    tu: Vec<i32>,
    ans: Vec<i64>,
    tans: Vec<i64>,
}

impl Solver {
    fn new(n: usize) -> Self {
        // Sizes follow the C arrays: ta[200010], nt/to/tS[400010], tu/ANS/tans[200020]
        // We allocate based on n but keep the same upper bounds logic: indices go up to 2*(n-1)-1.
        let ta_len = n + 5;
        let edge_len = 2 * (n.saturating_sub(1)) + 5;
        let node_len = n + 5;
        Self {
            ta: vec![0; ta_len],
            nt: vec![0; edge_len],
            to: vec![0; edge_len],
            t_s: vec![0; edge_len], // in C tS is sized 400010; only node indices are used.
            tu: vec![0; node_len],
            ans: vec![0; node_len],
            tans: vec![0; node_len],
        }
    }

    // sが親でtから帰ってきた時の処理
    fn fc(&mut self, s: usize, t: usize) {
        self.tans[s] = (self.tans[s] * self.tans[t]) % M;
    }
    // fcの逆処理
    fn ic(&mut self, s: usize, t: usize) {
        self.tans[s] = (self.tans[s] * inv(self.tans[t], M)) % M;
    }
    // すべての子の処理が終わった時の処理
    fn fp(&mut self, s: usize) {
        let ts = self.t_s[s] as i64;
        self.tans[s] = (self.tans[s] * inv(ts, M)) % M;
    }
    // fpの逆処理
    fn ip(&mut self, s: usize) {
        self.tans[s] = (self.tans[s] * (self.t_s[s] as i64)) % M;
    }

    fn tdp(&mut self, s: usize) {
        self.tu[s] = 1;
        let mut i = self.ta[s];
        while i != -1 {
            let idx = i as usize;
            let v = self.to[idx] as usize;
            if self.tu[v] != 0 {
                i = self.nt[idx];
                continue;
            }
            self.tdp(v);
            self.fc(s, v);
            self.t_s[s] += self.t_s[v];
            i = self.nt[idx];
        }
        self.fp(s);
    }

    fn chrt(&mut self, s: usize, t: usize) {
        self.ip(s);
        self.ic(s, t);
        self.ip(t);

        // These are int operations in C; keep i32.
        self.t_s[s] -= self.t_s[t];
        self.t_s[t] += self.t_s[s];

        self.fp(s);
        self.fc(t, s);
        self.fp(t);
    }

    fn atdp(&mut self, s: usize) {
        self.tu[s] = 2;
        self.ans[s] = self.tans[s];
        let mut i = self.ta[s];
        while i != -1 {
            let idx = i as usize;
            let v = self.to[idx] as usize;
            if self.tu[v] == 2 {
                i = self.nt[idx];
                continue;
            }
            self.chrt(s, v);
            self.atdp(v);
            self.chrt(v, s);
            i = self.nt[idx];
        }
    }

    fn atdp_entry(&mut self, v: usize, a: &[i32], b: &[i32]) {
        let i_root: usize = 1;

        // Initialize arrays exactly like C:
        // for(i=0;i<v;i++)ta[i+I]=-1;
        for i in 0..v {
            self.ta[i + i_root] = -1;
        }
        // for(i=0;i<v;i++)tS[i+I]=1;
        for i in 0..v {
            self.t_s[i + i_root] = 1;
        }
        // for(i=0;i<v;i++)tans[i+I]=1;
        for i in 0..v {
            self.tans[i + i_root] = 1;
        }

        // Build adjacency in the same packed-edge style.
        // for(i=0;i<v-1;i++){
        //   nt[i]=ta[to[i+v-1]=a[i]];
        //   nt[i+v-1]=ta[to[i]=b[i]];
        //   ta[a[i]]=i;
        //   ta[b[i]]=i+v-1;
        // }
        for i in 0..(v - 1) {
            let ai = a[i] as usize;
            let bi = b[i] as usize;

            let e1 = i;
            let e2 = i + (v - 1);

            self.to[e2] = a[i];
            self.nt[e1] = self.ta[ai];

            self.to[e1] = b[i];
            self.nt[e2] = self.ta[bi];

            self.ta[ai] = e1 as i32;
            self.ta[bi] = e2 as i32;
        }

        self.tdp(i_root);
        self.atdp(i_root);
    }
}

fn main() {
    // Read all input, parse as integers.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let mut a = vec![0i32; n.saturating_sub(1)];
    let mut b = vec![0i32; n.saturating_sub(1)];
    for i in 0..n.saturating_sub(1) {
        a[i] = it.next().unwrap().parse().unwrap();
        b[i] = it.next().unwrap().parse().unwrap();
    }

    let mut p: i64 = 1;
    for i in 1..=n as i64 {
        p = (p * i) % M;
    }

    let mut solver = Solver::new(n);
    solver.atdp_entry(n, &a, &b);

    let mut out = String::new();
    for i in 0..n {
        let val = (solver.ans[i + 1] * p) % M;
        out.push_str(&format!("{val}\n"));
    }
    print!("{out}");
}
```
