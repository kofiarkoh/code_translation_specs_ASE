```rust
use std::cmp::Ordering;
use std::io::{self, Read};

const MAX: usize = 105;

#[derive(Clone, Copy)]
struct T {
    a: [i8; 52],
}

static mut MX: [T; 10000] = [T { a: [0; 52] }; 10000];
static mut SZ: usize = 0;

const MX_ZERO: usize = 0;
const MX_INF: usize = 1;

fn mx_new() -> usize {
    unsafe {
        let id = SZ;
        SZ += 1;
        id
    }
}

fn mx_clear(idx: usize) {
    unsafe {
        MX[idx].a = [0; 52];
    }
}

fn mx_add1(a: usize, b: usize) {
    unsafe {
        for i in 0..52 {
            // C signed char arithmetic wraps on typical 2's complement; emulate with wrapping.
            MX[a].a[i] = MX[a].a[i].wrapping_add(MX[b].a[i]);
        }
    }
}

fn mx_add2(s: usize, a: usize, b: usize) {
    unsafe {
        for i in 0..52 {
            MX[s].a[i] = MX[a].a[i].wrapping_add(MX[b].a[i]);
        }
    }
}

fn mx_sub1(a: usize, b: usize) {
    unsafe {
        for i in 0..52 {
            MX[a].a[i] = MX[a].a[i].wrapping_sub(MX[b].a[i]);
        }
    }
}

#[allow(dead_code)]
fn mx_sub2(d: usize, a: usize, b: usize) {
    unsafe {
        for i in 0..52 {
            MX[d].a[i] = MX[a].a[i].wrapping_sub(MX[b].a[i]);
        }
    }
}

fn mx_mul2(m: usize, a: usize, k: i32) {
    unsafe {
        for i in 0..52 {
            // In C: signed char * int -> int, then assigned back to signed char (implementation-defined
            // if out of range). We emulate typical 2's complement truncation by wrapping to i8.
            let v = (MX[a].a[i] as i32).wrapping_mul(k);
            MX[m].a[i] = v as i8; // truncates low 8 bits, matching common C behavior.
        }
    }
}

fn mx_cpy(a: usize, b: usize) {
    unsafe {
        MX[a] = MX[b];
    }
}

fn mx_cmp(a: usize, b: usize) -> i32 {
    unsafe {
        for i in 0..52 {
            let k = (MX[a].a[i] as i32) - (MX[b].a[i] as i32);
            if k != 0 {
                return k;
            }
        }
        0
    }
}

#[derive(Clone, Copy)]
struct Edge {
    to: usize,
    rev: usize,
    cap: i32,
    cost: usize, // index into MX
}

static mut EDGE: [[Edge; MAX]; MAX] = [[Edge {
    to: 0,
    rev: 0,
    cap: 0,
    cost: 0,
}; MAX]; MAX];
static mut HI: [usize; MAX] = [0; MAX];

static mut V: usize = 0;
static mut DIST: [usize; MAX] = [0; MAX];
static mut PREVV: [usize; MAX] = [0; MAX];
static mut PREVE: [usize; MAX] = [0; MAX];

fn add_edge(from: usize, to: usize, cap: i32, cost: usize, rcost: usize) {
    unsafe {
        let f = HI[from];
        HI[from] += 1;
        let t = HI[to];
        HI[to] += 1;

        EDGE[from][f] = Edge {
            to,
            cap,
            cost,
            rev: t,
        };
        EDGE[to][t] = Edge {
            to: from,
            cap: 0,
            cost: rcost,
            rev: f,
        };
    }
}

fn min_cost_flow(s: usize, t: usize, mut f: i32) -> usize {
    let tmp = mx_new();
    unsafe {
        for i in 0..V {
            DIST[i] = mx_new();
        }
    }
    let ret = mx_new();

    while f > 0 {
        unsafe {
            for v in 0..V {
                mx_cpy(DIST[v], MX_INF);
            }
        }
        mx_clear(unsafe { DIST[s] });

        loop {
            let mut update = false;
            unsafe {
                for v in 0..V {
                    if mx_cmp(DIST[v], MX_INF) == 0 {
                        continue;
                    }
                    for i in 0..HI[v] {
                        let e = EDGE[v][i];
                        if e.cap <= 0 {
                            continue;
                        }
                        let nv = e.to;
                        mx_add2(tmp, DIST[v], e.cost);
                        if mx_cmp(DIST[nv], tmp) > 0 {
                            mx_cpy(DIST[nv], tmp);
                            PREVV[nv] = v;
                            PREVE[nv] = i; // note: C uses preve[e->to] = i; same as preve[nv]
                            update = true;
                        }
                    }
                }
            }
            if !update {
                break;
            }
        }

        let mut d = f;
        let mut v = t;
        while v != s {
            let pv = unsafe { PREVV[v] };
            let pe = unsafe { PREVE[v] };
            let e = unsafe { EDGE[pv][pe] };
            if d > e.cap {
                d = e.cap;
            }
            v = pv;
        }
        f -= d;

        mx_mul2(tmp, unsafe { DIST[t] }, d);
        mx_add1(ret, tmp);

        let mut v2 = t;
        while v2 != s {
            let pv = unsafe { PREVV[v2] };
            let pe = unsafe { PREVE[v2] };
            unsafe {
                let rev = EDGE[pv][pe].rev;
                EDGE[pv][pe].cap -= d;
                EDGE[v2][rev].cap += d;
            }
            v2 = pv;
        }
    }

    ret
}

fn main() {
    // Read entire stdin; C uses fgets/atoi and then reads N lines.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap_or(0),
        None => 0,
    };

    // map is only used for first N chars of each line; split_whitespace is fine for AOJ-style input.
    let mut map: Vec<Vec<u8>> = Vec::with_capacity(n);
    for _ in 0..n {
        let s = it.next().unwrap_or("");
        map.push(s.as_bytes().to_vec());
    }

    unsafe {
        // Initialize mx_inf to 60 in all 52 positions.
        for i in 0..52 {
            MX[MX_INF].a[i] = 60i8;
        }
        SZ = 2;
        for i in 0..MAX {
            HI[i] = 0;
        }
    }

    let source = n << 1;
    let sink = source + 1;
    unsafe {
        V = sink + 1;
    }

    for i in 0..n {
        add_edge(source, i, 1, mx_new(), mx_new());
        for j in 0..n {
            let c = map[i][j];
            let a = mx_new();
            let b = mx_new();
            unsafe {
                if c < b'a' {
                    let idx = (c - b'A') as usize;
                    MX[a].a[idx] = -1;
                    MX[b].a[idx] = 1;
                } else {
                    let idx = 26 + (c - b'a') as usize;
                    MX[a].a[idx] = -1;
                    MX[b].a[idx] = 1;
                }
            }
            add_edge(i, n + j, 1, a, b);
        }
        add_edge(n + i, sink, 1, mx_new(), mx_new());
    }

    let a_idx = min_cost_flow(source, sink, n as i32);

    let mut out = String::new();
    unsafe {
        for i in 0..26 {
            let mut b = -(MX[a_idx].a[i] as i32);
            while b != 0 {
                out.push((b'A' + i as u8) as char);
                b -= 1;
            }
        }
        for i in 0..26 {
            let mut b = -(MX[a_idx].a[i + 26] as i32);
            while b != 0 {
                out.push((b'a' + i as u8) as char);
                b -= 1;
            }
        }
    }
    out.push('\n');
    print!("{out}");
}
```
