```rust
use std::io::{self, Read};

const MAX: usize = 50001;

fn make_prime_table(table: &mut [i32; MAX]) -> i32 {
    for i in 0..MAX {
        table[i] = 0;
    }

    for i in 2..MAX {
        if table[i] == 0 {
            let mut j: usize = 2;
            while i * j < MAX {
                table[i * j] = 1;
                j += 1;
            }
        }
    }
    0
}

fn main() {
    // Read entire stdin, then parse ints similarly to repeated scanf("%d", &n).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut ptable = [0i32; MAX];
    make_prime_table(&mut ptable);

    let mut out = String::new();

    for tok in input.split_whitespace() {
        // C's scanf("%d") reads into int; emulate by parsing i32.
        // If parsing fails, stop (closest to scanf failing).
        let n: i32 = match tok.parse() {
            Ok(v) => v,
            Err(_) => break,
        };

        if n == 0 {
            break;
        }

        let mut cnt: i32 = 0;

        // C loop: for (i = 2; i <= n/2; i++)
        // Note: integer division truncates toward zero in both C (for non-negative)
        // and Rust; for negative n, the loop won't run (same effect).
        let half = n / 2;
        let mut i: i32 = 2;
        while i <= half {
            // The original C code assumes indices are within [0, MAX).
            // We preserve behavior for valid inputs; for out-of-range, we avoid panic
            // by treating as non-prime (i.e., not counted), which is a conservative
            // approximation since C would have undefined behavior.
            let a = i;
            let b = n - i;

            if a >= 0 && b >= 0 {
                let au = a as usize;
                let bu = b as usize;
                if au < MAX && bu < MAX && ptable[au] == 0 && ptable[bu] == 0 {
                    cnt += 1;
                }
            }

            i += 1;
        }

        out.push_str(&format!("{}\n", cnt));
    }

    print!("{out}");
}
```
