```rust
use std::f64;
use std::io::{self, Read};

const ONE_SIXTH: f64 = 1.0 / 6.0;

fn myabs(x: i32) -> i32 {
    if x >= 0 { x } else { -x }
}

fn solve(
    rest: usize,
    turn: i32,
    n: usize,
    sq: &[u8; 100],
    memo: &mut [[f64; 101]; 101],
) -> f64 {
    // In the original C, memo is initialized to DBL_MAX and used as "uncomputed" sentinel.
    if memo[rest][turn as usize] != f64::MAX {
        return memo[rest][turn as usize];
    }

    let ret = if rest == 0 {
        1.0
    } else if turn <= 0 {
        0.0
    } else if turn == 1 && rest <= 6 {
        ONE_SIXTH
    } else {
        let mut acc = 0.0;
        for i in 1..=6 {
            let next_pos_i32 = myabs(rest as i32 - i);
            let next_pos = next_pos_i32 as usize;

            // C indexes sq[N-next_pos]; N and next_pos are within 0..=100 by construction.
            let idx = n - next_pos;
            match sq[idx] {
                0 => {
                    acc += ONE_SIXTH * solve(next_pos, turn - 1, n, sq, memo);
                }
                b'L' => {
                    acc += ONE_SIXTH * solve(next_pos, turn - 2, n, sq, memo);
                }
                b'B' => {
                    acc += ONE_SIXTH * solve(n, turn - 1, n, sq, memo);
                }
                _ => {
                    // Unreachable given how sq is filled (0, 'L', 'B'), but keep C-like behavior.
                    acc += ONE_SIXTH * solve(next_pos, turn - 1, n, sq, memo);
                }
            }
        }
        acc
    };

    memo[rest][turn as usize] = ret;
    ret
}

fn main() {
    // Read all input and parse as integers like scanf("%d").
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: i32 = n_opt.unwrap().parse().unwrap();
        let t: i32 = it.next().unwrap().parse().unwrap();
        let lcnt: i32 = it.next().unwrap().parse().unwrap();
        let bcnt: i32 = it.next().unwrap().parse().unwrap();

        if n == 0 && t == 0 && lcnt == 0 && bcnt == 0 {
            break;
        }

        let n_usize = n as usize;

        let mut sq = [0u8; 100];
        let mut memo = [[f64::MAX; 101]; 101];

        for _ in 0..lcnt {
            let lpos: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;
            if lpos < sq.len() {
                sq[lpos] = b'L';
            }
        }
        for _ in 0..bcnt {
            let bpos: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;
            if bpos < sq.len() {
                sq[bpos] = b'B';
            }
        }

        let ret = solve(n_usize, t, n_usize, &sq, &mut memo);
        // Match C's printf("%lf\n", ret);
        println!("{:.6}", ret);
    }
}
```
