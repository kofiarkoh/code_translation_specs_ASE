use std::io::{self, Read};

fn main() {
    // Read two "long integers" from stdin. Interpret as signed 64-bit.
    // Preconditions say: first is non-negative, second is positive; otherwise undefined.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let a: i64 = it.next().unwrap().parse().unwrap(); // "first input"
    let b: i64 = it.next().unwrap().parse().unwrap(); // "second input"

    // Narrowest reasonable interpretation consistent with the spec:
    // - Factorize the second input (b) into prime factors with multiplicity.
    // - Let count be the total number of prime factors (with multiplicity).
    // - Determine an exponent k = count / a (integer division), and compute:
    //     product = Π p^k over distinct prime factors p of b
    // - If no valid prime factors meet conditions, output 0.
    //
    // Corner cases:
    // - If a == 0, "count / a" is undefined; spec says behavior undefined, so we do not guard.
    // - If b == 1, there are no prime factors => output 0.
    // - Overflow behavior is not specified; use wrapping arithmetic to emulate typical C "long"
    //   overflow behavior as closely as possible.
    let n = b;

    if n <= 1 {
        println!("0");
        return;
    }

    // Prime factorization of n (positive) with multiplicity count, and distinct primes list.
    let mut x = n as u64;
    let mut distinct_primes: Vec<u64> = Vec::new();
    let mut total_count: u64 = 0;

    // Factor out 2
    if x % 2 == 0 {
        distinct_primes.push(2);
        while x % 2 == 0 {
            x /= 2;
            total_count = total_count.wrapping_add(1);
        }
    }

    // Factor odd primes up to sqrt(x)
    let mut p: u64 = 3;
    while p <= x / p {
        if x % p == 0 {
            distinct_primes.push(p);
            while x % p == 0 {
                x /= p;
                total_count = total_count.wrapping_add(1);
            }
        }
        p = p.wrapping_add(2);
    }

    // Remaining prime
    if x > 1 {
        distinct_primes.push(x);
        total_count = total_count.wrapping_add(1);
    }

    // Determine exponent k = total_count / a
    // a is i64; precondition says non-negative. Use u64 conversion.
    let a_u = a as u64;
    let k = total_count / a_u;

    if distinct_primes.is_empty() || k == 0 {
        println!("0");
        return;
    }

    // Compute product = Π p^k with wrapping i64 arithmetic.
    let mut product: i64 = 1;
    for &prime in &distinct_primes {
        let mut pow: i64 = 1;
        let base: i64 = prime as i64;
        for _ in 0..k {
            pow = pow.wrapping_mul(base);
        }
        product = product.wrapping_mul(pow);
    }

    // Invariant says non-negative; however wrapping may produce negative.
    // Spec does not define overflow; we keep wrapping result as the observable output.
    println!("{}", product);
}
