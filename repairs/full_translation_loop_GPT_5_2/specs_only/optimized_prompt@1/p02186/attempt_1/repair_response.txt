```rust
use std::io::{self, Read};

#[derive(Clone, Copy, Debug)]
struct Node {
    max: i64,
    add: i64, // lazy add to be applied to this segment
}

struct LazySegTree {
    n: usize,      // number of leaves (power of two)
    size: usize,   // original length
    tree: Vec<Node>,
}

impl LazySegTree {
    fn new_from(values: &[i64]) -> Self {
        let size = values.len();
        let mut n = 1usize;
        while n < size {
            n <<= 1;
        }
        let mut tree = vec![Node { max: 0, add: 0 }; 2 * n];

        // Build leaves
        for i in 0..size {
            tree[n + i].max = values[i];
        }
        // Remaining leaves are 0 by default (narrowest reasonable choice)

        // Build internal nodes
        for idx in (1..n).rev() {
            tree[idx].max = tree[idx * 2].max.max(tree[idx * 2 + 1].max);
        }

        Self { n, size, tree }
    }

    fn apply(&mut self, idx: usize, delta: i64) {
        self.tree[idx].max = self.tree[idx].max.wrapping_add(delta);
        self.tree[idx].add = self.tree[idx].add.wrapping_add(delta);
    }

    fn push(&mut self, idx: usize) {
        let add = self.tree[idx].add;
        if add != 0 {
            self.apply(idx * 2, add);
            self.apply(idx * 2 + 1, add);
            self.tree[idx].add = 0;
        }
    }

    fn pull(&mut self, idx: usize) {
        self.tree[idx].max = self.tree[idx * 2].max.max(self.tree[idx * 2 + 1].max);
    }

    fn range_add(&mut self, l: usize, r: usize, delta: i64) {
        // Adds delta to [l, r) (0-based, half-open)
        self.range_add_rec(1, 0, self.n, l, r, delta);
    }

    fn range_add_rec(&mut self, idx: usize, seg_l: usize, seg_r: usize, ql: usize, qr: usize, delta: i64) {
        if ql >= seg_r || qr <= seg_l {
            return;
        }
        if ql <= seg_l && seg_r <= qr {
            self.apply(idx, delta);
            return;
        }
        self.push(idx);
        let mid = (seg_l + seg_r) / 2;
        self.range_add_rec(idx * 2, seg_l, mid, ql, qr, delta);
        self.range_add_rec(idx * 2 + 1, mid, seg_r, ql, qr, delta);
        self.pull(idx);
    }

    fn range_max(&mut self, l: usize, r: usize) -> i64 {
        // Max over [l, r)
        self.range_max_rec(1, 0, self.n, l, r)
    }

    fn range_max_rec(&mut self, idx: usize, seg_l: usize, seg_r: usize, ql: usize, qr: usize) -> i64 {
        if ql >= seg_r || qr <= seg_l {
            // Identity for max; use i64::MIN to avoid inventing behavior.
            return i64::MIN;
        }
        if ql <= seg_l && seg_r <= qr {
            return self.tree[idx].max;
        }
        self.push(idx);
        let mid = (seg_l + seg_r) / 2;
        let a = self.range_max_rec(idx * 2, seg_l, mid, ql, qr);
        let b = self.range_max_rec(idx * 2 + 1, mid, seg_r, ql, qr);
        a.max(b)
    }
}

fn main() {
    // Input format inferred narrowly from the specification:
    // n q
    // a0 a1 ... a(n-1)
    // then q lines: i t
    //
    // Output: maximum value in range [0, n+1] after initialization and after each update.
    // Since the underlying array has length n, we interpret [0, n+1] as [0, n) (narrowest safe interpretation).
    // (If the original intended structure had n+2 elements, the spec does not provide those values.)
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let q: usize = it.next().unwrap().parse().unwrap();

    let mut a = vec![0i64; n];
    for i in 0..n {
        a[i] = it.next().unwrap().parse().unwrap();
    }

    // Initialize with cumulative sums as specified.
    let mut pref = vec![0i64; n];
    let mut acc: i64 = 0;
    for i in 0..n {
        acc = acc.wrapping_add(a[i]);
        pref[i] = acc;
    }

    let mut st = LazySegTree::new_from(&pref);

    let mut out = String::new();
    // After initialization
    let init_max = st.range_max(0, n);
    out.push_str(&format!("{init_max}\n"));

    // For each query: update index i to new value t.
    // Since the tree stores prefix sums, changing a[i] by delta affects all prefixes j >= i.
    for _ in 0..q {
        let i: usize = it.next().unwrap().parse().unwrap();
        let t: i64 = it.next().unwrap().parse().unwrap();

        let old = a[i];
        a[i] = t;
        let delta = t.wrapping_sub(old);

        // Apply delta to pref[i..n)
        st.range_add(i, n, delta);

        let mx = st.range_max(0, n);
        out.push_str(&format!("{mx}\n"));
    }

    print!("{out}");
}
```
