fn exact_max_disjoint(groups: &[Vec<usize>], n: usize) -> (usize, Vec<usize>) {
    // Represent each group as a bitset over up to 128 names.
    // If n > 128, caller won't use this.
    let mut masks: Vec<(u128, usize)> = Vec::with_capacity(groups.len());
    for (i, g) in groups.iter().enumerate() {
        let mut m: u128 = 0;
        for &id in g {
            if id < 128 {
                m |= 1u128 << id;
            } else {
                // If any id is out of range, treat as non-representable; skip exact.
                // Caller ensures n <= 128, so this should not happen.
            }
        }
        masks.push((m, i));
    }

    // Sort by descending size to improve pruning.
    masks.sort_by_key(|(m, _)| std::cmp::Reverse(m.count_ones()));

    let mut best_count = 0usize;
    let mut best_sel: Vec<usize> = Vec::new();
    let mut cur_sel: Vec<usize> = Vec::new();

    // Precompute an optimistic upper bound: remaining groups count.
    fn dfs(
        idx: usize,
        used: u128,
        masks: &[(u128, usize)],
        cur_sel: &mut Vec<usize>,
        best_count: &mut usize,
        best_sel: &mut Vec<usize>,
    ) {
        // Upper bound: even if we take all remaining groups, can we beat best?
        let remaining = masks.len().saturating_sub(idx);
        if cur_sel.len() + remaining <= *best_count {
            return;
        }

        if idx == masks.len() {
            if cur_sel.len() > *best_count {
                *best_count = cur_sel.len();
                *best_sel = cur_sel.clone();
            }
            return;
        }

        let (m, original_idx) = masks[idx];

        // Option 1: take it if disjoint.
        if used & m == 0 {
            cur_sel.push(original_idx);
            dfs(idx + 1, used | m, masks, cur_sel, best_count, best_sel);
            cur_sel.pop();
        }

        // Option 2: skip it.
        dfs(idx + 1, used, masks, cur_sel, best_count, best_sel);
    }

    dfs(0, 0, &masks, &mut cur_sel, &mut best_count, &mut best_sel);

    // Restore output order as encountered in input (ascending group index).
    best_sel.sort_unstable();
    (best_count, best_sel)
}

fn greedy_max_disjoint(groups: &[Vec<usize>], _n: usize) -> (usize, Vec<usize>) {
    // Greedy heuristic: pick smallest groups first to maximize count.
    // This does not guarantee optimality, but is used only when exact is infeasible.
    let mut order: Vec<usize> = (0..groups.len()).collect();
    order.sort_by_key(|&i| groups[i].len());

    let mut used: HashSet<usize> = HashSet::new();
    let mut sel: Vec<usize> = Vec::new();

    'outer: for i in order {
        for &id in &groups[i] {
            if used.contains(&id) {
                continue 'outer;
            }
        }
        for &id in &groups[i] {
            used.insert(id);
        }
        sel.push(i);
    }

    sel.sort_unstable();
    (sel.len(), sel)
}
