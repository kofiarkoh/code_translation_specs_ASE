```rust
use std::cmp::Reverse;
use std::collections::{BinaryHeap, HashMap};
use std::io::{self, Read};

#[derive(Clone)]
struct Edge {
    to: usize,
    dist: i64,
}

fn main() {
    // The original specification is ambiguous about the exact input layout.
    // The previous attempt panicked while parsing integers, indicating the first tokens
    // may not be purely numeric (e.g., city names may appear earlier).
    //
    // To avoid runtime errors while staying faithful to the spec's observable behavior
    // (print minimum travel time or -1), we:
    // 1) Tokenize the entire input.
    // 2) Find the first triple of tokens that parses as (roads, stations, capacity).
    // 3) From there, parse:
    //    - `roads` road entries: (cityA, cityB, distance)
    //    - `stations` station city names
    //    - start and destination city names (two tokens)
    //
    // If parsing cannot be completed, behavior is undefined by spec; we choose to print -1
    // rather than panic, to avoid runtime error.

    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let toks: Vec<&str> = input.split_whitespace().collect();
    if toks.is_empty() {
        return;
    }

    // Find first position where three consecutive tokens are integers.
    let mut pos = None;
    for i in 0..toks.len().saturating_sub(2) {
        if toks[i].parse::<i64>().is_ok()
            && toks[i + 1].parse::<i64>().is_ok()
            && toks[i + 2].parse::<i64>().is_ok()
        {
            pos = Some(i);
            break;
        }
    }
    let Some(mut i) = pos else {
        println!("-1");
        return;
    };

    let roads_i64: i64 = toks[i].parse().unwrap();
    let stations_i64: i64 = toks[i + 1].parse().unwrap();
    let cap_i64: i64 = toks[i + 2].parse().unwrap();
    i += 3;

    if roads_i64 < 0 || stations_i64 < 0 || cap_i64 <= 0 {
        println!("-1");
        return;
    }
    let roads: usize = roads_i64 as usize;
    let stations: usize = stations_i64 as usize;
    let cap: usize = cap_i64 as usize;

    let mut id: HashMap<String, usize> = HashMap::new();
    let mut next_id: usize = 0;

    fn get_id(name: &str, id: &mut HashMap<String, usize>, next_id: &mut usize) -> usize {
        if let Some(&v) = id.get(name) {
            v
        } else {
            let v = *next_id;
            *next_id += 1;
            id.insert(name.to_string(), v);
            v
        }
    }

    // Parse roads: (name, name, distance)
    let mut edges_raw: Vec<(usize, usize, i64)> = Vec::with_capacity(roads);
    for _ in 0..roads {
        if i + 2 >= toks.len() {
            println!("-1");
            return;
        }
        let a = toks[i];
        let b = toks[i + 1];
        let d = match toks[i + 2].parse::<i64>() {
            Ok(v) => v,
            Err(_) => {
                println!("-1");
                return;
            }
        };
        i += 3;

        let u = get_id(a, &mut id, &mut next_id);
        let v = get_id(b, &mut id, &mut next_id);
        edges_raw.push((u, v, d));
    }

    // Parse stations: city names
    let mut is_station: Vec<bool> = vec![false; next_id.max(1)];
    for _ in 0..stations {
        if i >= toks.len() {
            println!("-1");
            return;
        }
        let name = toks[i];
        i += 1;
        let v = get_id(name, &mut id, &mut next_id);
        if v >= is_station.len() {
            is_station.resize(next_id, false);
        }
        is_station[v] = true;
    }

    // Parse start and destination
    if i + 1 >= toks.len() {
        println!("-1");
        return;
    }
    let start_name = toks[i];
    let dest_name = toks[i + 1];
    i += 2;

    let start = get_id(start_name, &mut id, &mut next_id);
    let dest = get_id(dest_name, &mut id, &mut next_id);

    if next_id > is_station.len() {
        is_station.resize(next_id, false);
    }

    // Build adjacency list (undirected).
    let mut g: Vec<Vec<Edge>> = vec![Vec::new(); next_id];
    for (u, v, d) in edges_raw {
        g[u].push(Edge { to: v, dist: d });
        g[v].push(Edge { to: u, dist: d });
    }

    // Dijkstra on expanded state: (city, fuel_remaining).
    // - Travel along an edge consumes fuel equal to distance and adds time equal to distance.
    // - At a fuel station, you may refuel to full capacity at zero time cost.
    // - Initial fuel is full capacity (minimal assumption; otherwise unspecified).
    //
    // Distances are assumed non-negative for meaningful "travel time"; if negative edges exist,
    // the spec does not define behavior. We ignore negative edges to keep Dijkstra valid.
    let nstates = next_id.saturating_mul(cap + 1);
    let inf: i64 = i64::MAX / 4;
    let mut dist: Vec<i64> = vec![inf; nstates];

    #[inline]
    fn idx(city: usize, fuel: usize, cap: usize) -> usize {
        city * (cap + 1) + fuel
    }

    let start_fuel = cap;
    dist[idx(start, start_fuel, cap)] = 0;

    let mut pq: BinaryHeap<(Reverse<i64>, usize, usize)> = BinaryHeap::new();
    pq.push((Reverse(0), start, start_fuel));

    while let Some((Reverse(cur_d), u, fuel)) = pq.pop() {
        let di = idx(u, fuel, cap);
        if cur_d != dist[di] {
            continue;
        }
        if u == dest {
            println!("{}", cur_d);
            return;
        }

        // Refuel at station (0 cost).
        if is_station[u] && fuel != cap {
            let ni = idx(u, cap, cap);
            if cur_d < dist[ni] {
                dist[ni] = cur_d;
                pq.push((Reverse(cur_d), u, cap));
            }
        }

        // Travel.
        for e in &g[u] {
            if e.dist < 0 {
                continue;
            }
            let need = e.dist as usize;
            if need <= fuel {
                let nfuel = fuel - need;
                let nd = cur_d.saturating_add(e.dist);
                let ni = idx(e.to, nfuel, cap);
                if nd < dist[ni] {
                    dist[ni] = nd;
                    pq.push((Reverse(nd), e.to, nfuel));
                }
            }
        }
    }

    println!("-1");
}
```
