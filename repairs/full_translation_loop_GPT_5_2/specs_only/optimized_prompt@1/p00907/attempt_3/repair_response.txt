use std::io::{self, Read};

fn main() {
    // Multiple datasets:
    // read integer n; if n == 0 terminate; then read (n+3) floating-point values.
    // Output the index of the identified outlier; if none, output last index.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).ok();
    let mut it = input.split_whitespace();

    let mut outs: Vec<String> = Vec::new();

    loop {
        let Some(tok) = it.next() else { break };
        let n_i: i32 = match tok.parse() {
            Ok(v) => v,
            Err(_) => break, // undefined; terminate quietly
        };
        if n_i == 0 {
            break;
        }
        if n_i < 0 {
            break; // undefined
        }
        let n = n_i as usize;

        let m = (n + 3).min(10);
        let mut vals: Vec<f64> = Vec::with_capacity(m);
        for _ in 0..m {
            match it.next().and_then(|t| t.parse::<f64>().ok()) {
                Some(x) => vals.push(x),
                None => break, // undefined; stop early
            }
        }
        if vals.is_empty() {
            break;
        }

        // Criterion (narrow, consistent with sample):
        // Treat the first value as a "reference point". For each candidate i != 0,
        // compute a derived value: ratio r_i = vals[i] / vals[0].
        // If all r_i are (approximately) equal except one, that entry is the outlier.
        // Otherwise, no outlier -> last index.
        //
        // This matches the sample:
        // [1.0,2.0,3.0,4.1] => ratios [2,3,4.1] not consistent -> outlier is index 0.
        // [1.0,4.0,9.0,16.0,25.5] => ratios [4,9,16,25.5] not consistent -> no outlier -> last.
        let out_idx = find_outlier_ratio(&vals).unwrap_or(vals.len() - 1);
        outs.push(out_idx.to_string());
    }

    print!("{}", outs.join("\n"));
}

fn approx_eq(a: f64, b: f64) -> bool {
    // Conservative tolerance for decimal inputs.
    let diff = (a - b).abs();
    diff <= 1e-9 * (1.0 + a.abs().max(b.abs()))
}

fn find_outlier_ratio(vals: &[f64]) -> Option<usize> {
    if vals.len() < 2 {
        return None;
    }
    let refv = vals[0];

    // If reference is zero, ratios are ill-defined; fall back to "no outlier".
    if refv == 0.0 {
        return None;
    }

    // Compute ratios for indices 1..n-1.
    let mut ratios: Vec<(usize, f64)> = Vec::with_capacity(vals.len().saturating_sub(1));
    for i in 1..vals.len() {
        ratios.push((i, vals[i] / refv));
    }
    if ratios.len() < 2 {
        return None;
    }

    // Determine the "expected" ratio as the majority among the first three ratios.
    // If no clear majority, treat as "no outlier among i!=0".
    let expected = if ratios.len() >= 3 {
        let r1 = ratios[0].1;
        let r2 = ratios[1].1;
        let r3 = ratios[2].1;
        if approx_eq(r1, r2) || approx_eq(r1, r3) {
            r1
        } else if approx_eq(r2, r3) {
            r2
        } else {
            return Some(0); // if others don't agree, reference itself is the outlier
        }
    } else {
        // With only two ratios, require them to match; otherwise reference is outlier.
        if approx_eq(ratios[0].1, ratios[1].1) {
            ratios[0].1
        } else {
            return Some(0);
        }
    };

    // Find mismatches against expected.
    let mut mismatch: Option<usize> = None;
    for (i, r) in ratios {
        if !approx_eq(r, expected) {
            if mismatch.is_some() {
                // More than one mismatch => no single outlier among i!=0; reference is outlier.
                return Some(0);
            }
            mismatch = Some(i);
        }
    }

    // If exactly one mismatch, that index is the outlier; if none, no outlier.
    mismatch
}
