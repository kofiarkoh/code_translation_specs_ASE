use std::io::{self, Read};

fn transform(seq: &[u32]) -> Vec<u32> {
    // Narrowest reasonable interpretation of "transforming each sequence":
    // Replace the sequence by the multiset of frequency counts of values 0..=100,
    // listed in non-increasing order, excluding zeros.
    //
    // This is a common "stable state" process: once the multiset of counts equals
    // the previous sequence, further transformations do not change it.
    let mut freq = [0u32; 101];
    for &x in seq {
        // Preconditions say 0..=100; behavior undefined otherwise, so we don't clamp.
        freq[x as usize] += 1;
    }

    let mut counts: Vec<u32> = freq.iter().copied().filter(|&c| c != 0).collect();
    counts.sort_unstable_by(|a, b| b.cmp(a)); // non-increasing
    counts
}
