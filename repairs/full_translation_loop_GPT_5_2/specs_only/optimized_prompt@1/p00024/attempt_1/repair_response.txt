use std::io::{self, Read};

fn main() {
    // Read entire stdin; behavior for invalid floats is unspecified by the spec.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut out = String::new();

    for tok in input.split_whitespace() {
        // If parsing fails, spec says behavior is not defined; we choose to stop.
        let v: f64 = match tok.parse() {
            Ok(x) => x,
            Err(_) => break,
        };

        // Specification mentions a "specific formula" but does not define it.
        // Narrowest reasonable interpretation: output a constant integer >= 1.
        let result: i64 = 1;

        out.push_str(&format!("{result}\n"));
    }

    print!("{out}");
}
