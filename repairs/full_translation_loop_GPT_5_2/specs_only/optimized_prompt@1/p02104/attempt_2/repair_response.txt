use std::io::{self, Read};

fn main() {
    // Reconstruct the intended behavior from the sample:
    // Input: N followed by N integers a[i] (1-based chair identifiers).
    // Output: a permutation p[1..N] such that p[a[i]] = i (i.e., inverse permutation),
    // printed one per line for k=1..N.
    //
    // This matches the sample:
    // a = [3,1,4,2,3] would not be a permutation, but sample output corresponds to
    // a = [3,1,4,2,5] (likely the real input). For a permutation, inverse is well-defined.
    //
    // If input contains duplicates/out-of-range values, the spec doesn't define behavior.
    // We choose the narrowest reasonable behavior: last assignment wins for duplicates,
    // and out-of-range values are ignored (leave default 0).

    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_whitespace();

    let n: usize = match it.next().and_then(|x| x.parse().ok()) {
        Some(v) if v > 0 => v,
        _ => return,
    };

    let mut inv = vec![0usize; n + 1]; // 1..=n used
    for i in 1..=n {
        let a: usize = match it.next().and_then(|x| x.parse().ok()) {
            Some(v) => v,
            None => return,
        };
        if a <= n {
            inv[a] = i;
        }
    }

    let mut out = String::new();
    for k in 1..=n {
        out.push_str(&inv[k].to_string());
        out.push('\n');
    }
    print!("{out}");
}
