```rust
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut outputs: Vec<String> = Vec::new();

    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let n: usize = match n_opt.unwrap().parse() {
            Ok(v) => v,
            Err(_) => break,
        };

        if n == 0 {
            break; // termination condition
        }

        let a = match it.next() {
            Some(s) => s,
            None => break,
        };
        let b = match it.next() {
            Some(s) => s,
            None => break,
        };

        // Behavior is undefined if lengths are not as specified; we assume valid input per spec.
        // Also assume digit-only strings per spec.
        if a == b {
            outputs.push("0".to_string());
            continue;
        }

        // DP over positions with two possible "carry" states:
        // carry=0: no pending +1 applied to current digit from previous rotation.
        // carry=1: current digit is effectively (a[i] + 1) mod 10 due to previous step.
        //
        // At each position, choose minimal rotations to make current digit match target.
        // Rotations are interpreted as incrementing a digit (mod 10); applying k increments
        // to position i also causes a carry of 1 to position i+1 iff k is odd.
        //
        // This matches the invariant that per-digit differences are in 0..=9 and yields
        // a non-negative minimal total.
        let bytes_a = a.as_bytes();
        let bytes_b = b.as_bytes();

        let inf: i32 = i32::MAX / 4;
        let mut dp0: i32 = 0;
        let mut dp1: i32 = inf;

        for i in 0..n {
            let da = (bytes_a[i] - b'0') as i32;
            let db = (bytes_b[i] - b'0') as i32;

            let mut ndp0 = inf;
            let mut ndp1 = inf;

            for (carry, cur_cost) in [(0, dp0), (1, dp1)] {
                if cur_cost >= inf {
                    continue;
                }
                let cur_digit = (da + carry) % 10;
                let diff = (db - cur_digit + 10) % 10; // 0..9 increments needed

                // Choose k = diff (minimal) or k = diff + 10 (same digit, toggles parity).
                // Since cost is minimized, only these two are relevant.
                for add10 in [0, 10] {
                    let k = diff + add10;
                    let next_carry = (k & 1) as i32;
                    let next_cost = cur_cost + k;
                    if next_carry == 0 {
                        if next_cost < ndp0 {
                            ndp0 = next_cost;
                        }
                    } else if next_cost < ndp1 {
                        ndp1 = next_cost;
                    }
                }
            }

            dp0 = ndp0;
            dp1 = ndp1;
        }

        // After the last digit, any remaining carry would affect a non-existent digit,
        // so only carry=0 is a valid completed transformation.
        outputs.push(dp0.to_string());
    }

    print!("{}", outputs.join("\n"));
}
```
