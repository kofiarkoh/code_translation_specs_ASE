```rust
use std::io::{self, Read};

fn main() {
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_whitespace();

    let n: usize = it.next().unwrap().parse().unwrap();

    // Semantics consistent with the spec and the failing sample:
    // - Start with sum of floor(a[i]/2) over all i.
    // - For each adjacent pair (a[i], a[i+1]), if both are odd, subtract 1.
    //
    // This is an "adjustment based on the parity of adjacent integers" and matches
    // the sample where adding would overshoot by 1.
    let mut prev: i64 = it.next().unwrap().parse().unwrap();
    let mut acc: i128 = (prev / 2) as i128;

    for _ in 1..n {
        let cur: i64 = it.next().unwrap().parse().unwrap();
        acc += (cur / 2) as i128;

        if (prev & 1) != 0 && (cur & 1) != 0 {
            acc -= 1;
        }

        prev = cur;
    }

    println!("{}", acc);
}
```
