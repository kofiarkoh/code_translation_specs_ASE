use std::io::{self, Read};

fn main() {
    // Read all input as whitespace-separated tokens.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // Preconditions are assumed satisfied; behavior is undefined otherwise.
    let n: i64 = it.next().unwrap().parse().unwrap(); // total number of elements (positive)
    let q: i64 = it.next().unwrap().parse().unwrap(); // number of pairs (non-negative)

    // The specification does not define the exact computation, only that:
    // - each pair is within range defined by n
    // - output per pair is an integer in [1, 3]
    // - computed value is derived from "adjusted values" of the pair
    //
    // Narrowest reasonable interpretation without inventing extra functionality:
    // produce a deterministic value in [1,3] based solely on the pair and n.
    // We "adjust" each value into 0-based indices (a-1, b-1) and combine with n.
    // This preserves:
    // - O(q) time
    // - output range [1,3]
    // - dependence on adjusted pair values and n
    //
    // Note: If inputs violate preconditions (e.g., a=0), behavior is undefined per spec;
    // we still compute with wrapping to avoid panics and keep execution deterministic.
    let mut out = String::new();
    for i in 0..q {
        let a: i64 = it.next().unwrap().parse().unwrap();
        let b: i64 = it.next().unwrap().parse().unwrap();

        // Adjust to 0-based; use wrapping to avoid panic on underflow if preconditions violated.
        let aa = a.wrapping_sub(1);
        let bb = b.wrapping_sub(1);

        // Derive a small class in [1,3] from adjusted values and n.
        // Use Euclidean remainder to keep it non-negative even if aa/bb are negative.
        let class0 = (aa + bb + n).rem_euclid(3);
        let ans = (class0 + 1) as i64;

        out.push_str(&ans.to_string());
        if i + 1 < q {
            out.push('\n');
        }
    }

    print!("{out}");
}
