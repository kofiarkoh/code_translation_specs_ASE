use std::io::{self, Read};

fn next_permutation(a: &mut [usize]) -> bool {
    // Lexicographic next_permutation; returns false if already last permutation.
    if a.len() < 2 {
        return false;
    }
    let mut i = a.len() - 2;
    loop {
        if a[i] < a[i + 1] {
            break;
        }
        if i == 0 {
            a.reverse();
            return false;
        }
        i -= 1;
    }
    let mut j = a.len() - 1;
    while a[j] <= a[i] {
        j -= 1;
    }
    a.swap(i, j);
    a[i + 1..].reverse();
    true
}

fn main() {
    // Read entire stdin.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    // Parse integers.
    let nums: Vec<i64> = input
        .split_whitespace()
        .map(|s| s.parse::<i64>())
        .collect::<Result<Vec<_>, _>>()
        .unwrap_or_else(|_| Vec::new());

    // Error handling: must be exactly six pairs of integers (12 integers total).
    if nums.len() != 12 {
        // "terminates with a return value indicating an error"
        std::process::exit(1);
    }

    // Preconditions say non-negative; spec does not define behavior otherwise.
    // We keep values as-is but treat them as dimensions.
    let mut rects = [(0i64, 0i64); 6];
    for i in 0..6 {
        rects[i] = (nums[2 * i], nums[2 * i + 1]);
    }

    // Evaluate all permutations and orientations.
    // The specification does not define the exact "connection criteria".
    // Narrowest reasonable interpretation consistent with "specific connection lengths are equal":
    // there exists a permutation and orientation such that all six components share a common
    // connection length (i.e., at least one side length is equal across all six after choosing
    // orientation per component).
    let mut perm: Vec<usize> = (0..6).collect();
    loop {
        // For a fixed permutation, try all 2^6 orientation choices.
        for mask in 0u32..(1u32 << 6) {
            let mut common: Option<i64> = None;
            let mut ok = true;

            for (pos, &idx) in perm.iter().enumerate() {
                let (h, w) = rects[idx];
                let (a, b) = if ((mask >> pos) & 1) == 0 { (h, w) } else { (w, h) };

                // Choose the "connection length" for this component as the first dimension (a)
                // after orientation; require all to match.
                match common {
                    None => common = Some(a),
                    Some(c) => {
                        if a != c {
                            ok = false;
                            break;
                        }
                    }
                }

                // Ensure dimensions are not altered (invariant): we never mutate rects.
                let _ = b; // b is unused in this minimal criterion.
            }

            if ok {
                print!("yes");
                return;
            }
        }

        if !next_permutation(&mut perm) {
            break;
        }
    }

    print!("no");
}
