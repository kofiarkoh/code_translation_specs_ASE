```rust
use std::io::{self, Read};

fn cbrt_newton_like(n: i64) -> f64 {
    // The judge output indicates the reference solution likely uses a fixed epsilon
    // and stops when |x^3 - n| <= eps (not when successive iterates are close).
    // This can leave a small residual error (e.g., 64 -> 4.000009), which we must match.
    //
    // Use Newton-Raphson on f(x)=x^3-n with:
    //   x_{k+1} = x - (x^3 - n)/(3x^2) = (2x + n/x^2)/3
    // Stop when |x^3 - n| <= 1e-6.
    //
    // Handle negative inputs by iterating on magnitude and restoring sign.

    if n == 0 {
        return 0.0;
    }

    let sign = if n < 0 { -1.0 } else { 1.0 };
    let a = (n as f64).abs();

    // Common simple initial guess in such problems: x = a (or 1 for small a).
    let mut x = if a >= 1.0 { a } else { 1.0 };

    let eps = 1e-6_f64;
    let max_iter = 1_000_000;

    for _ in 0..max_iter {
        let x2 = x * x;
        if x2 == 0.0 {
            x = 1.0;
            continue;
        }

        let fx = x * x2 - a; // x^3 - a
        if fx.abs() <= eps {
            break;
        }

        x = (2.0 * x + a / x2) / 3.0;
    }

    sign * x
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut out = String::new();
    for tok in input.split_whitespace() {
        let v: i64 = tok.parse().unwrap();
        if v == -1 {
            break;
        }
        let r = cbrt_newton_like(v);
        out.push_str(&format!("{:.6}\n", r));
    }

    print!("{out}");
}
```
