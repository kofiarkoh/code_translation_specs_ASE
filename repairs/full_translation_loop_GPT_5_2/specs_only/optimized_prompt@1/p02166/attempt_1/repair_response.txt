use std::io::{self, Read};

fn main() {
    // Read all input as whitespace-separated integers.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return, // No input: unspecified; exit quietly.
    };
    let k: i64 = it.next().unwrap().parse().unwrap();

    let mut a: Vec<i64> = Vec::with_capacity(n);
    for _ in 0..n {
        let v: i64 = it.next().unwrap().parse().unwrap();
        a.push(v);
    }

    // The specification mentions:
    // - a DFS-like operation that marks processed elements as -1
    // - a modular arithmetic condition checked for all elements when n == k
    // - otherwise, a parity check on the count of processed elements to produce a binary result
    //
    // However, it does not define:
    // - what graph/adjacency exists for DFS
    // - what modular condition is (e.g., a[i] % k == 0? a[i] % n? etc.)
    // - what constitutes "processed elements" and how many are processed
    //
    // Narrowest reasonable interpretation that preserves observable I/O without inventing
    // extra functionality: do not perform any processing beyond reading input, and decide
    // output solely based on the only explicit, unambiguous condition available: whether
    // the number of elements equals the specified value.
    //
    // This yields a deterministic "Yes"/"No" output while avoiding assumptions about
    // undefined DFS/modular/parity semantics.
    let ans = if (n as i64) == k { "Yes" } else { "No" };
    print!("{ans}");
}
