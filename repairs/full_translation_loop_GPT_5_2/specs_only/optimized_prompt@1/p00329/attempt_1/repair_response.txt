```rust
use std::io::{self, Read};

fn try_kuhn(
    u: usize,
    g: &Vec<Vec<u8>>,
    seen: &mut Vec<bool>,
    mt: &mut Vec<Option<usize>>,
) -> bool {
    let n = g.len();
    for v in 0..n {
        if g[u][v] != 0 && !seen[v] {
            seen[v] = true;
            if mt[v].is_none() || try_kuhn(mt[v].unwrap(), g, seen, mt) {
                mt[v] = Some(u);
                return true;
            }
        }
    }
    false
}

fn main() {
    // Input format is assumed correct per specification; otherwise behavior is undefined.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n_raw: usize = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };

    // "decreased by one for internal processing" is interpreted narrowly as:
    // internal N = input - 1, with saturating behavior for 0 to keep program defined in Rust.
    // (Spec does not define behavior for 0; this is the narrowest safe interpretation.)
    let n = n_raw.saturating_sub(1);

    let mut g = vec![vec![0u8; n]; n];
    for i in 0..n {
        for j in 0..n {
            let v: u8 = it.next().unwrap().parse().unwrap();
            g[i][j] = if v == 0 { 0 } else { 1 };
        }
    }

    // mt[v] = Some(u) means right node v is matched to left node u.
    let mut mt: Vec<Option<usize>> = vec![None; n];

    let mut matched = 0usize;
    for u in 0..n {
        let mut seen = vec![false; n];
        if try_kuhn(u, &g, &mut seen, &mut mt) {
            matched += 1;
        }
    }

    if matched == n {
        println!("yes");
        // Output matched pairs, each on a new line.
        // Pairs are printed as "left right" using 1-based indices for human-facing output.
        // (Spec does not define indexing; 1-based is the narrowest conventional choice.)
        for v in 0..n {
            if let Some(u) = mt[v] {
                println!("{} {}", u + 1, v + 1);
            }
        }
    } else {
        println!("no");
    }
}
```
