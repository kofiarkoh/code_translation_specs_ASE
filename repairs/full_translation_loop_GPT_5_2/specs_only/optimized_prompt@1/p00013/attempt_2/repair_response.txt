```rust
use std::io::{self, Read};

fn main() {
    // The sample behavior indicates LIFO (stack): the most recently stored car exits first.
    // Capacity is 10; behavior is undefined if exceeded, so we do not add extra checks.
    let mut stack: Vec<i32> = Vec::with_capacity(10);

    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut out = String::new();

    for tok in input.split_whitespace() {
        let x: i32 = tok.parse().unwrap();
        if x == 0 {
            if let Some(exited) = stack.pop() {
                out.push_str(&exited.to_string());
                out.push('\n');
            }
        } else {
            stack.push(x);
        }
    }

    print!("{out}");
}
```
