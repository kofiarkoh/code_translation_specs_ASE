```rust
use std::io::{self, Read};

const MOD: u64 = 1_000_000_007;

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // Preconditions: valid input; undefined otherwise.
    let n: usize = it.next().unwrap().parse().unwrap();
    let target: usize = it.next().unwrap().parse().unwrap();

    let mut elems: Vec<usize> = Vec::with_capacity(n);
    for _ in 0..n {
        elems.push(it.next().unwrap().parse().unwrap());
    }

    // Interpret "select elements from a given list" as: each list element can be used at most once.
    // "Considering the order of selection" means different permutations of the same chosen subset count
    // as distinct ways.
    //
    // DP over sums and number of chosen elements:
    // dp[k][s] = number of ways to pick k distinct elements (from those processed so far) whose sum is s,
    //            counting orderings (i.e., sequences) of length k.
    //
    // Transition when adding a new element x:
    // - either not use it
    // - or append it to an existing ordered sequence of length k-1, which can be done in k positions,
    //   hence multiply by k.
    let mut dp = vec![vec![0u64; target + 1]; n + 1];
    dp[0][0] = 1;

    for &x in &elems {
        // Iterate k and s descending to ensure each element is used at most once.
        for k in (1..=n).rev() {
            for s in (x..=target).rev() {
                let add = (dp[k - 1][s - x] * (k as u64)) % MOD;
                dp[k][s] += add;
                if dp[k][s] >= MOD {
                    dp[k][s] %= MOD;
                }
            }
        }
    }

    let mut ans = 0u64;
    for k in 0..=n {
        ans += dp[k][target];
        if ans >= MOD {
            ans %= MOD;
        }
    }

    println!("{}", ans % MOD);
}
```
