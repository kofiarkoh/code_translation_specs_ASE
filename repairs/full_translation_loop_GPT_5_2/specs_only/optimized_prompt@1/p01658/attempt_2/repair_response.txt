fn main() {

    // - For each row, count the number of pairs (i, j) with i < j such that a[i] == 1 and a[j] == 0.

    let n: usize = it.next().unwrap().parse().unwrap();

    let mut total: i64 = 0;

    for _ in 0..m {
        let mut ones_so_far: i64 = 0;
        for _ in 0..n {
            let v: i32 = it.next().unwrap().parse().unwrap();
            if v == 1 {
                ones_so_far += 1;
            } else {
                // v is expected to be 0/1; behavior is undefined otherwise per spec.
                total += ones_so_far;
            }
        }
    }

    print!("{total}");
}
