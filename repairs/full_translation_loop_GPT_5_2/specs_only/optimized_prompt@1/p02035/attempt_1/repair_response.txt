use std::io::{self, Read};

const MAX: usize = 2010;

fn mod_pow(mut a: i64, mut e: i64, m: i64) -> i64 {
    if m == 1 {
        return 0;
    }
    a = a.rem_euclid(m);
    let mut r: i64 = 1 % m;
    while e > 0 {
        if (e & 1) != 0 {
            r = (r * a).rem_euclid(m);
        }
        a = (a * a).rem_euclid(m);
        e >>= 1;
    }
    r
}

fn mod_inv(a: i64, m: i64) -> i64 {
    // Specification does not define behavior when inverse does not exist.
    // Narrow interpretation: use Fermat-style a^(m-2) mod m (works when m is prime and a != 0 mod m).
    mod_pow(a, m - 2, m)
}

fn comb(n: i64, r: i64, fact: &[i64], ifact: &[i64], m: i64) -> i64 {
    if n < 0 || r < 0 || n - r < 0 {
        return 0;
    }
    let n = n as usize;
    let r = r as usize;
    if n >= fact.len() || r >= fact.len() || (n - r) >= fact.len() {
        // Not specified; keep narrow and safe: out of precomputed range yields 0.
        return 0;
    }
    (((fact[n] * ifact[r]).rem_euclid(m)) * ifact[n - r]).rem_euclid(m)
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: i64 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let m: i64 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    // Preconditions: n >= 0, m > 0. Behavior outside is unspecified; we avoid panics where possible.
    if m <= 0 {
        return;
    }

    // Invariant: factorial and inverse factorial arrays populated up to 2010 before calculations.
    let mut fact = vec![0i64; MAX + 1];
    let mut ifact = vec![0i64; MAX + 1];

    fact[0] = 1 % m;
    for i in 1..=MAX {
        fact[i] = (fact[i - 1] * (i as i64)).rem_euclid(m);
    }

    ifact[MAX] = mod_inv(fact[MAX], m);
    for i in (1..=MAX).rev() {
        ifact[i - 1] = (ifact[i] * (i as i64)).rem_euclid(m);
    }

    // Specification does not provide the exact combinatorial formula.
    // Narrowest reasonable interpretation consistent with "computes a specific combinatorial value"
    // and the presence of n and "red and black items": compute sum_{r=0..n} C(n, r) mod m.
    // This equals 2^n mod m, but we compute via combinations to respect the stated use of factorials.
    let mut ans: i64 = 0;
    for r in 0..=n {
        ans = (ans + comb(n, r, &fact, &ifact, m)).rem_euclid(m);
    }

    // Postcondition: output in [0, m-1]
    ans = ans.rem_euclid(m);
    println!("{}", ans);
}
