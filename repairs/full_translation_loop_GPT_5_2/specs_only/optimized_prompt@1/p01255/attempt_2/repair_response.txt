use std::cmp::{max, min};
use std::io::{self, Read};

#[derive(Clone, Copy, Debug)]
struct Point {
    x: i64,
    y: i64,
}

#[derive(Clone, Copy, Debug)]
struct Segment {
    a: Point,
    b: Point,
}

fn cross(ax: i64, ay: i64, bx: i64, by: i64) -> i128 {
    (ax as i128) * (by as i128) - (ay as i128) * (bx as i128)
}

fn orient(a: Point, b: Point, c: Point) -> i128 {
    cross(b.x - a.x, b.y - a.y, c.x - a.x, c.y - a.y)
}

fn on_segment(a: Point, b: Point, p: Point) -> bool {
    min(a.x, b.x) <= p.x
        && p.x <= max(a.x, b.x)
        && min(a.y, b.y) <= p.y
        && p.y <= max(a.y, b.y)
}

fn segments_intersect(s1: Segment, s2: Segment) -> bool {
    // Closed-segment intersection (touching counts as intersection).
    let p1 = s1.a;
    let q1 = s1.b;
    let p2 = s2.a;
    let q2 = s2.b;

    let o1 = orient(p1, q1, p2);
    let o2 = orient(p1, q1, q2);
    let o3 = orient(p2, q2, p1);
    let o4 = orient(p2, q2, q1);

    if (o1 > 0 && o2 < 0 || o1 < 0 && o2 > 0) && (o3 > 0 && o4 < 0 || o3 < 0 && o4 > 0) {
        return true;
    }
    if o1 == 0 && on_segment(p1, q1, p2) {
        return true;
    }
    if o2 == 0 && on_segment(p1, q1, q2) {
        return true;
    }
    if o3 == 0 && on_segment(p2, q2, p1) {
        return true;
    }
    if o4 == 0 && on_segment(p2, q2, q1) {
        return true;
    }
    false
}

fn chromatic_number_exact(adj: &[u64], n: usize) -> usize {
    // Exact chromatic number via branch-and-bound coloring on bitsets.
    // n is expected to be small (problem-defined maximum); if it exceeds 64,
    // behavior is unspecified by the spec; we handle it by terminating.
    if n == 0 {
        return 0;
    }
    if n > 64 {
        std::process::exit(1);
    }

    // Order vertices by descending degree to improve pruning.
    let mut order: Vec<usize> = (0..n).collect();
    order.sort_by_key(|&v| std::cmp::Reverse((adj[v].count_ones() as usize, v)));

    // Remap adjacency to ordered indices.
    let mut pos = vec![0usize; n];
    for (i, &v) in order.iter().enumerate() {
        pos[v] = i;
    }
    let mut adj2 = vec![0u64; n];
    for (i, &v) in order.iter().enumerate() {
        let mut m = 0u64;
        let mut bits = adj[v];
        while bits != 0 {
            let u = bits.trailing_zeros() as usize;
            bits &= bits - 1;
            m |= 1u64 << pos[u];
        }
        adj2[i] = m;
    }

    // Greedy coloring to get an initial upper bound.
    let mut ub = {
        let mut color_masks: Vec<u64> = Vec::new();
        for v in 0..n {
            let mut placed = false;
            for cm in color_masks.iter_mut() {
                if (*cm & adj2[v]) == 0 {
                    *cm |= 1u64 << v;
                    placed = true;
                    break;
                }
            }
            if !placed {
                color_masks.push(1u64 << v);
            }
        }
        color_masks.len()
    };

    // Lower bound: size of maximum clique is a lower bound, but computing it
    // exactly is extra; we use 1 as minimal and rely on pruning.
    let mut best = ub;

    // Recursive search: assign colors to vertices in fixed order.
    fn dfs(
        idx: usize,
        n: usize,
        adj: &[u64],
        color_masks: &mut Vec<u64>,
        best: &mut usize,
    ) {
        if idx == n {
            *best = (*best).min(color_masks.len());
            return;
        }
        if color_masks.len() >= *best {
            return;
        }

        let v_mask = 1u64 << idx;
        let neigh = adj[idx];

        // Try existing colors first.
        for c in 0..color_masks.len() {
            if (color_masks[c] & neigh) == 0 {
                color_masks[c] |= v_mask;
                dfs(idx + 1, n, adj, color_masks, best);
                color_masks[c] &= !v_mask;
                if color_masks.len() >= *best {
                    return;
                }
            }
        }

        // Try new color.
        if color_masks.len() + 1 < *best {
            color_masks.push(v_mask);
            dfs(idx + 1, n, adj, color_masks, best);
            color_masks.pop();
        }
    }

    let mut color_masks: Vec<u64> = Vec::new();
    dfs(0, n, &adj2, &mut color_masks, &mut best);
    best
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();
    let mut first = true;

    loop {
        let Some(tok) = it.next() else { break };
        let n: usize = match tok.parse() {
            Ok(v) => v,
            Err(_) => break,
        };

        // Many problems of this style use 0 as a terminator; the spec is silent.
        // To avoid panics on such inputs and to match common observable behavior,
        // we treat n==0 as end-of-input (no output for the terminator record).
        if n == 0 {
            break;
        }

        let mut segs = Vec::with_capacity(n);
        for _ in 0..n {
            // If input is truncated/malformed, spec allows undefined behavior;
            // we choose to stop processing without panicking.
            let (Some(x1), Some(y1), Some(x2), Some(y2)) =
                (it.next(), it.next(), it.next(), it.next())
            else {
                return;
            };
            let x1: i64 = match x1.parse() {
                Ok(v) => v,
                Err(_) => return,
            };
            let y1: i64 = match y1.parse() {
                Ok(v) => v,
                Err(_) => return,
            };
            let x2: i64 = match x2.parse() {
                Ok(v) => v,
                Err(_) => return,
            };
            let y2: i64 = match y2.parse() {
                Ok(v) => v,
                Err(_) => return,
            };
            segs.push(Segment {
                a: Point { x: x1, y: y1 },
                b: Point { x: x2, y: y2 },
            });
        }

        if n > 64 {
            std::process::exit(1);
        }

        let mut adj = vec![0u64; n];
        for i in 0..n {
            for j in (i + 1)..n {
                if segments_intersect(segs[i], segs[j]) {
                    adj[i] |= 1u64 << j;
                    adj[j] |= 1u64 << i;
                }
            }
        }

        let ans = chromatic_number_exact(&adj, n);

        if !first {
            out.push('\n');
        }
        first = false;
        out.push_str(&ans.to_string());
    }

    print!("{out}");
}
