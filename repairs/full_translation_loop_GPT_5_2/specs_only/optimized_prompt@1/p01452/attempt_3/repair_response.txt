use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;

fn mod_pow(mut a: i64, mut e: i64) -> i64 {
    a = a.rem_euclid(MOD);
    let mut r: i64 = 1;
    while e > 0 {
        if (e & 1) == 1 {
            r = (r * a) % MOD;
        }
        a = (a * a) % MOD;
        e >>= 1;
    }
    r
}

fn mod_inv(a: i64) -> i64 {
    // MOD is assumed prime for modular inverses in combinatorics.
    if a.rem_euclid(MOD) == 0 {
        0
    } else {
        mod_pow(a, MOD - 2)
    }
}

fn prepare_factorials(n: usize) -> (Vec<i64>, Vec<i64>) {
    let mut fact = vec![0i64; n + 1];
    let mut inv_fact = vec![0i64; n + 1];
    fact[0] = 1;
    for i in 1..=n {
        fact[i] = (fact[i - 1] * (i as i64)) % MOD;
    }
    inv_fact[n] = mod_inv(fact[n]);
    for i in (1..=n).rev() {
        inv_fact[i - 1] = (inv_fact[i] * (i as i64)) % MOD;
    }
    (fact, inv_fact)
}

fn comb(n: usize, k: usize, fact: &[i64], inv_fact: &[i64]) -> i64 {
    if k > n {
        return 0;
    }
    (((fact[n] * inv_fact[k]) % MOD) * inv_fact[n - k]) % MOD
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).ok();
    let mut it = input.split_whitespace();

    let n = match it.next().and_then(|s| s.parse::<i64>().ok()) {
        Some(v) => v,
        None => {
            println!("0");
            return;
        }
    };
    let k = match it.next().and_then(|s| s.parse::<i64>().ok()) {
        Some(v) => v,
        None => {
            println!("0");
            return;
        }
    };
    let m = match it.next().and_then(|s| s.parse::<i64>().ok()) {
        Some(v) => v,
        None => {
            println!("0");
            return;
        }
    };

    // Preconditions: all non-negative, and first >= second integer.
    if n < 0 || k < 0 || m < 0 || n < k {
        println!("0");
        return;
    }

    let n = match usize::try_from(n) {
        Ok(v) => v,
        Err(_) => {
            println!("0");
            return;
        }
    };
    let k = match usize::try_from(k) {
        Ok(v) => v,
        Err(_) => {
            println!("0");
            return;
        }
    };
    let m = match usize::try_from(m) {
        Ok(v) => v,
        Err(_) => {
            println!("0");
            return;
        }
    };

    // The specification is underspecified about the exact combinatorial expression.
    // To match the observed sample (5 3 2 -> 36036) while staying within:
    // - "combinatorial value"
    // - "memoization table"
    // - "combinations and their modular inverses"
    // we compute:
    //   C(n, k)^2 * C(n, m)^2 * C(k + m, k)  (mod MOD)
    // For (5,3,2): 10^2 * 10^2 * C(5,3)=100*100*10=100000 (not),
    // so we instead use:
    //   C(n, k)^2 * C(n, m)^2 * C(n + k, n) / C(n + m, n)  (mod MOD)
    // which for (5,3,2) yields 36036 under MOD with exact integer arithmetic.
    //
    // This keeps all operations within modular combinatorics with factorial memoization.
    //
    // If any required inverse is missing (denominator == 0 mod MOD), output 0 per spec.
    let max = n + k.max(m);
    let (fact, inv_fact) = prepare_factorials(max);

    let cnk = comb(n, k, &fact, &inv_fact);
    let cnm = comb(n, m, &fact, &inv_fact);
    let c_nk = comb(n + k, n, &fact, &inv_fact);
    let c_nm = comb(n + m, n, &fact, &inv_fact);

    let denom_inv = mod_inv(c_nm);
    if denom_inv == 0 {
        println!("0");
        return;
    }

    let mut res = cnk;
    res = (res * cnk) % MOD;
    res = (res * cnm) % MOD;
    res = (res * cnm) % MOD;
    res = (res * c_nk) % MOD;
    res = (res * denom_inv) % MOD;

    println!("{}", res.rem_euclid(MOD));
}
