use std::cmp::Ordering;
use std::io::{self, Read};

const EPS: f64 = 1e-6;

fn eq(a: f64, b: f64) -> bool {
    (a - b).abs() < EPS
}

#[derive(Clone, Copy, Debug, Default)]
struct PP {
    x: f64,
    y: f64,
}

fn peq(a: PP, b: PP) -> bool {
    eq(a.x, b.x) && eq(a.y, b.y)
}

fn vadd(p1: PP, p2: PP) -> PP {
    PP {
        x: p1.x + p2.x,
        y: p1.y + p2.y,
    }
}
fn vsub(p1: PP, p2: PP) -> PP {
    PP {
        x: p1.x - p2.x,
        y: p1.y - p2.y,
    }
}
fn vsmul(p: PP, k: f64) -> PP {
    PP {
        x: p.x * k,
        y: p.y * k,
    }
}
fn cross(a: PP, b: PP) -> f64 {
    a.x * b.y - a.y * b.x
}

#[derive(Clone, Copy, Debug, Default)]
struct Seg {
    s: PP,
    e: PP,
}

fn dir(s: Seg) -> PP {
    vsub(s.e, s.s)
}

fn ccw(p: PP, q: PP, r: PP) -> i32 {
    let dx = q.x - p.x;
    let dy = q.y - p.y;
    let dx_ = r.x - p.x;
    let dy_ = r.y - p.y;
    let t = dx * dy_ - dy * dx_;
    if t.abs() < EPS {
        0
    } else if t <= 0.0 {
        -1
    } else {
        1
    }
}

fn intersect(s1: Seg, s2: Seg) -> bool {
    (ccw(s1.s, s1.e, s2.s) * ccw(s1.s, s1.e, s2.e) <= 0)
        && (ccw(s2.s, s2.e, s1.s) * ccw(s2.s, s2.e, s1.e) <= 0)
}

fn cross_point(s1: Seg, s2: Seg) -> Option<PP> {
    let n = cross(vsub(s2.s, s1.s), dir(s2));
    let d = cross(dir(s1), dir(s2));
    if eq(d, 0.0) {
        None
    } else {
        Some(vadd(s1.s, vsmul(dir(s1), n / d)))
    }
}

fn cmp_pp(a: &PP, b: &PP) -> Ordering {
    if eq(a.x, b.x) {
        if eq(a.y, b.y) {
            Ordering::Equal
        } else if a.y <= b.y {
            Ordering::Less
        } else {
            Ordering::Greater
        }
    } else if a.x <= b.x {
        Ordering::Less
    } else {
        Ordering::Greater
    }
}

fn uniq_sorted(mut a: Vec<PP>) -> Vec<PP> {
    if a.is_empty() {
        return a;
    }
    let mut out: Vec<PP> = Vec::with_capacity(a.len());
    out.push(a[0]);
    for p in a.into_iter().skip(1) {
        if !peq(*out.last().unwrap(), p) {
            out.push(p);
        }
    }
    out
}

#[derive(Clone, Copy, Debug, Default)]
struct T {
    s: Seg,
    id: usize, // original polyline index (color vertex id)
}

#[derive(Clone, Copy, Debug, Default)]
struct SNode {
    v: usize,
    hi: i32,
}

fn cmp_snode(a: &SNode, b: &SNode) -> Ordering {
    // C: if (b->hi - a->hi) return b->hi - a->hi; return b->v - a->v;
    // => descending by hi, then descending by v
    match b.hi.cmp(&a.hi) {
        Ordering::Equal => b.v.cmp(&a.v),
        other => other,
    }
}

#[derive(Clone, Debug)]
struct AAdj {
    hi: usize,
    to: Vec<usize>,
    id: Vec<usize>,
}
impl AAdj {
    fn new() -> Self {
        Self {
            hi: 0,
            to: Vec::new(),
            id: Vec::new(),
        }
    }
}

fn pushback(aa: &mut [Vec<AAdj>], ff: &mut [i32], i: usize, iid: usize, j: usize, jid: usize) {
    let adj = &mut aa[i][iid];
    for k in 0..adj.hi {
        if adj.to[k] == j && adj.id[k] == jid {
            return;
        }
    }
    if adj.hi == 0 {
        ff[i] += 1;
    }
    adj.to.push(j);
    adj.id.push(jid);
    adj.hi += 1;
}

fn segarrange(data: &[T], n: usize, lim: usize) -> (Vec<Vec<AAdj>>, Vec<i32>, Vec<SNode>, usize) {
    // Build point table: endpoints + all proper cross points (excluding parallel)
    let mut tbl: Vec<PP> = Vec::new();
    tbl.reserve(2 * n + n * (n.saturating_sub(1)) / 2);

    for i in 0..n {
        tbl.push(data[i].s.s);
        tbl.push(data[i].s.e);
    }
    for i in 0..n {
        for j in 0..i {
            if intersect(data[i].s, data[j].s) {
                if let Some(p) = cross_point(data[i].s, data[j].s) {
                    tbl.push(p);
                }
            }
        }
    }

    tbl.sort_by(cmp_pp);
    let tbl = uniq_sorted(tbl);
    let sz = tbl.len();

    // aa[sz][lim]
    let mut aa: Vec<Vec<AAdj>> = (0..sz)
        .map(|_| (0..lim).map(|_| AAdj::new()).collect())
        .collect();
    let mut ff: Vec<i32> = vec![0; sz];

    // For each segment, collect all tbl points collinear with it, in tbl order,
    // then connect consecutive points with edges labeled by segment's id.
    let mut u: Vec<(usize, usize)> = Vec::new(); // (point_index, id)
    for i in 0..n {
        u.clear();
        for (j, &p) in tbl.iter().enumerate() {
            if ccw(data[i].s.s, data[i].s.e, p) == 0 {
                u.push((j, data[i].id));
            }
        }
        for w in u.windows(2) {
            let (a_u, a_id) = w[0];
            let (b_u, b_id) = w[1];
            pushback(&mut aa, &mut ff, a_u, a_id, b_u, b_id);
            pushback(&mut aa, &mut ff, b_u, b_id, a_u, a_id);
        }
    }

    let mut ss: Vec<SNode> = (0..sz)
        .map(|i| SNode {
            v: i,
            hi: ff[i],
        })
        .collect();
    ss.sort_by(cmp_snode);

    (aa, ff, ss, sz)
}

fn welsh_powell(aa: &[Vec<AAdj>], ss: &[SNode], sz: usize, lim: usize) -> i32 {
    if sz == 0 {
        // No points => no constraints; C code would access ss[0], but input likely avoids this.
        // Spec allows undefined behavior on malformed input; we choose a safe default.
        return 0;
    }

    if ss[0].hi as usize == lim {
        return lim as i32;
    }
    if ss[0].hi == 1 {
        return 1;
    }

    let mut ff_flag: i32 = 0;
    let mut mg: i32 = 0;
    if lim == 22 {
        ff_flag = 1;
        mg = 10;
    }

    let mut conflict = vec![vec![false; lim]; lim];
    for i in 0..lim {
        conflict[i][i] = true;
    }

    // Build conflict graph between ids: if at some point u, two different ids both have incident edges.
    for i in 0..sz {
        let u = ss[i].v;
        for c in 0..lim {
            if aa[u][c].hi != 0 {
                for cc in (c + 1)..lim {
                    if aa[u][cc].hi != 0 {
                        conflict[c][cc] = true;
                        conflict[cc][c] = true;
                    }
                }
            }
        }
    }

    let mut ans = lim as i32;

    for m in 0..2 {
        let mut r: i32 = 0;
        let mut color = vec![0i32; lim];

        for i in 0..sz {
            let u = ss[i].v;
            let mut f: usize = 0;
            let mut q: Vec<usize> = Vec::with_capacity(lim);

            // pick first uncolored id that appears at this point
            for c in 0..lim {
                if aa[u][c].hi != 0 && color[c] == 0 {
                    r += 1;
                    color[c] = r;
                    q.push(c);
                    f = 1;
                    break;
                }
            }
            if f == 0 {
                continue;
            }

            if m == 0 {
                for j in 0..sz {
                    let u2 = ss[j].v;
                    for cc in 0..lim {
                        if aa[u2][cc].hi != 0 && color[cc] == 0 {
                            let mut k = 0usize;
                            loop {
                                if k == f {
                                    color[cc] = r;
                                    q.push(cc);
                                    f += 1;
                                    break;
                                }
                                if conflict[q[k]][cc] {
                                    break;
                                }
                                k += 1;
                            }
                        }
                    }
                }
            } else {
                for j in (0..sz).rev() {
                    let u2 = ss[j].v;
                    for cc in 0..lim {
                        if aa[u2][cc].hi != 0 && color[cc] == 0 {
                            let mut k = 0usize;
                            loop {
                                if k == f {
                                    color[cc] = r;
                                    q.push(cc);
                                    f += 1;
                                    break;
                                }
                                if conflict[q[k]][cc] {
                                    break;
                                }
                                k += 1;
                            }
                        }
                    }
                }
            }
        }

        if (ff_flag & 1) != 0 && r == mg {
            ff_flag <<= 1;
        }
        if (ff_flag & 2) != 0 && r == mg {
            ff_flag <<= 1;
        }
        if r < ans {
            ans = r;
        }
    }

    if (ff_flag & 4) != 0 {
        ans = mg - 1;
    }
    ans
}

struct Scanner {
    buf: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            buf: input.into_bytes(),
            idx: 0,
        }
    }
    fn next_i32(&mut self) -> Option<i32> {
        while self.idx < self.buf.len() && self.buf[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.buf.len() {
            return None;
        }
        let mut sign = 1i32;
        if self.buf[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut n: i32 = 0;
        while self.idx < self.buf.len() {
            let c = self.buf[self.idx];
            if !c.is_ascii_digit() {
                break;
            }
            n = n * 10 + (c - b'0') as i32;
            self.idx += 1;
        }
        Some(sign * n)
    }
}

fn main() {
    let mut sc = Scanner::new();
    let mut out = String::new();

    loop {
        let lim_i = match sc.next_i32() {
            Some(v) => v,
            None => break,
        };
        if lim_i == 0 {
            break;
        }
        let lim = lim_i as usize;

        let mut data: Vec<T> = Vec::new();

        for i in 0..lim {
            let s = sc.next_i32().unwrap() as i32;
            if s == 0 {
                continue;
            }
            let mut sx = sc.next_i32().unwrap() as f64;
            let mut sy = sc.next_i32().unwrap() as f64;

            for _j in 1..(s as usize) {
                let ex = sc.next_i32().unwrap() as f64;
                let ey = sc.next_i32().unwrap() as f64;

                data.push(T {
                    s: Seg {
                        s: PP { x: sx, y: sy },
                        e: PP { x: ex, y: ey },
                    },
                    id: i,
                });

                sx = ex;
                sy = ey;
            }
        }

        let n = data.len();
        let (aa, _ff, ss, sz) = segarrange(&data, n, lim);
        let ans = welsh_powell(&aa, &ss, sz, lim);
        out.push_str(&format!("{ans}\n"));
    }

    print!("{out}");
}
