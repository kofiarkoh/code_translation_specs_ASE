#define HASHSIZ 300017
typedef struct { int b; int v; } HASH;
HASH hash[HASHSIZ+2], *hashend = hash + HASHSIZ;

int lookup(int b)
{
	HASH *p = hash + b % HASHSIZ;
	while (p->b) {
		if (p->b == b) return p->v;
		if (++p == hashend) p = hash;
	}
	return 0;
}

void insert(int b)
{
	HASH *p = hash + b % HASHSIZ;

		if (p->b == b) { p->v++; return; }
		if (++p == hashend) p = hash;
	}
	p->b = b, p->v = 1;
}
