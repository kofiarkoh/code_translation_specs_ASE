```rust
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Node {
    parent: i32,
    left: i32,
    right: i32,
}

fn get_depth(r: usize, d: i32, tree: &[Node], depth: &mut [i32]) {
    depth[r] = d;

    // Note: This follows the C code's traversal order exactly:
    // first traverse "right" with same depth, then "left" with depth+1.
    // This is unusual for a left-child/right-sibling representation, but we preserve semantics.
    let right = tree[r].right;
    if right != -1 {
        get_depth(right as usize, d, tree, depth);
    }
    let left = tree[r].left;
    if left != -1 {
        get_depth(left as usize, d + 1, tree, depth);
    }
}

fn print_tree(i: usize, tree: &[Node], depth: &[i32]) -> String {
    let mut out = String::new();
    out.push_str(&format!(
        "node {}: parent = {}, depth = {}, ",
        i, tree[i].parent, depth[i]
    ));

    if tree[i].parent == -1 {
        out.push_str("root, [");
    } else if tree[i].left == -1 {
        out.push_str("leaf, [");
    } else {
        out.push_str("internal node, [");
    }

    let mut m = tree[i].left;
    let mut j = 0;
    while m != -1 {
        if j != 0 {
            out.push_str(", ");
        }
        out.push_str(&format!("{}", m));
        m = tree[m as usize].right;
        j += 1;
    }
    out.push_str("]\n");
    out
}

fn main() {
    // Read all input
    let mut s = String::new();
    io::stdin().read_to_string(&mut s).unwrap();
    let mut it = s.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    // Initialize tree
    let mut tree = vec![
        Node {
            parent: -1,
            left: -1,
            right: -1
        };
        n
    ];
    let mut depth = vec![0i32; n];

    // Build tree (left-child / right-sibling)
    for _ in 0..n {
        let id: usize = it.next().unwrap().parse().unwrap();
        let num: usize = it.next().unwrap().parse().unwrap();

        let mut l: i32 = -1;
        for j in 0..num {
            let p: i32 = it.next().unwrap().parse().unwrap();
            if j == 0 {
                tree[id].left = p;
            } else {
                tree[l as usize].right = p;
            }
            tree[p as usize].parent = id as i32;
            l = p;
        }
    }

    // Find root (last node with parent == -1, matching C behavior)
    let mut root: usize = 0;
    for i in 0..n {
        if tree[i].parent == -1 {
            root = i;
        }
    }

    // Compute depths
    if n > 0 {
        get_depth(root, 0, &tree, &mut depth);
    }

    // Output
    let mut out = String::new();
    for i in 0..n {
        out.push_str(&print_tree(i, &tree, &depth));
    }
    print!("{}", out);
}
```
