int in()
{
	int n = 0;
	int c = getchar_unlocked();
	do n = 10*n + (c & 0xf), c = getchar_unlocked();
	while (c >= '0');
	return n;
}

int main()
{
	int n, i, j, l, t;

	n = in();
	for (i = 0; i < n; i++) x[i] = in(), y[i] = in();

	for (i = 0; i < n; i++) dp[i][i] = 0;
