lli extgcd(lli a,lli b,lli *x,lli *y){
  lli d=a;

  if(b!=0){
    d=extgcd(b,a%b,y,x);

    *y-=(a/b)*(*x);
  }

}

lli mod_inv(lli a,lli m){
  lli x,y;

  extgcd(a,m,&x,&y);

  return (m+x%m)%m;
}
