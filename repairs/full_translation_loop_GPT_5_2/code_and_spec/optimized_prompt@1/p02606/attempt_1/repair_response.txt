```rust
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let l: i32 = it.next().unwrap().parse().unwrap();
    let r: i32 = it.next().unwrap().parse().unwrap();
    let d: i32 = it.next().unwrap().parse().unwrap();

    // Matches C semantics: integer division truncates toward zero.
    // Per spec, d is positive and l <= r; if d == 0 behavior is undefined (same as C).
    let ans: i32 = r / d - (l - 1) / d;

    print!("{ans}\n");
}
```
