```rust
use std::io::{self, Read};

const P: i64 = 998_244_353;

fn gcd(mut a: i64, mut b: i64) -> i64 {
    // Mirrors the C recursive gcd, but iterative.
    // Note: In C, gcd can return negative if `a` is negative and `b` becomes 0.
    // We preserve that behavior (no abs), because the spec allows any integers.
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn main() {
    // Read all input (C scanf-style, whitespace separated).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = it.next().unwrap().parse::<i64>().unwrap() as usize;

    let mut a: Vec<i64> = Vec::with_capacity(n);
    for _ in 0..n {
        let v = it.next().unwrap().parse::<i64>().unwrap();
        a.push(v);
    }

    // C qsort with compare that effectively sorts ascending.
    a.sort();

    let mut ans: i64 = 1;
    for (i, &val) in a.iter().enumerate() {
        let g = gcd(val, i as i64);
        // In C, `% p` can yield negative if `ans * g` is negative.
        // The spec requires a non-negative output < P, so we normalize.
        ans = (ans * g) % P;
        if ans < 0 {
            ans += P;
        }
    }

    print!("{ans}\n");
}
```
