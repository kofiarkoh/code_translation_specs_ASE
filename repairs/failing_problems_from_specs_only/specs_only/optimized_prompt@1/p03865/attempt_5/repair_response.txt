use std::io::{self, Read, Write};

fn main() {
    let mut buf = vec![0u8; 114_514];

    // Single read from stdin.
    let l: usize = match io::stdin().read(&mut buf) {
        Ok(n) => n,
        Err(_) => return, // unspecified on read failure
    };

    // Spec: undefined for L < 2; keep behavior unspecified by exiting.
    if l < 2 {
        return;
    }

    // To match the observed sample, treat the input as a raw byte stream but
    // ignore a single trailing '\n' if present (common in judge inputs).
    // This is the narrowest adjustment consistent with the sample behavior.
    let mut l_eff = l;
    if l_eff > 0 && buf[l_eff - 1] == b'\n' {
        l_eff -= 1;
    }
    if l_eff < 2 {
        return;
    }

    // Decision rule (with L = l_eff):
    // n = L - 1
    // eq = (buf[0] == buf[L-2])
    // cond = ((n + eq) & 1)
    let n = l_eff - 1;
    let eq: usize = (buf[0] == buf[l_eff - 2]) as usize;
    let cond = (n + eq) & 1;

    let mut stdout = io::stdout().lock();
    if cond != 0 {
        let _ = stdout.write_all(b"First\n");
    } else {
        let _ = stdout.write_all(b"Second\n");
    }
}
