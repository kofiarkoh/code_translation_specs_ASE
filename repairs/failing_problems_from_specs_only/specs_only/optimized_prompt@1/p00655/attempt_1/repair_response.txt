```rust
use std::collections::VecDeque;
use std::io::{self, Read};

#[derive(Default)]
struct FH {
    stack: Vec<i32>,
    // Prefix-extrema histories (strictly improving sequences).
    min_hist: Vec<i32>,
    max_hist: Vec<i32>,
}

impl FH {
    fn clear(&mut self) {
        self.stack.clear();
        self.min_hist.clear();
        self.max_hist.clear();
    }

    fn push(&mut self, x: i32) {
        self.stack.push(x);

        if self.min_hist.last().map_or(true, |&m| x < m) {
            self.min_hist.push(x);
        }
        if self.max_hist.last().map_or(true, |&m| x > m) {
            self.max_hist.push(x);
        }
    }

    fn pop(&mut self) -> i32 {
        let x = self.stack.pop().unwrap();

        if self.min_hist.last().map_or(false, |&m| m == x) {
            self.min_hist.pop();
        }
        if self.max_hist.last().map_or(false, |&m| m == x) {
            self.max_hist.pop();
        }
        x
    }

    fn len(&self) -> usize {
        self.stack.len()
    }

    // k-th most recent (1-based from end).
    fn query_min_k(&self, k: usize) -> i32 {
        let idx = self.min_hist.len() - k;
        self.min_hist[idx]
    }

    fn query_max_k(&self, k: usize) -> i32 {
        let idx = self.max_hist.len() - k;
        self.max_hist[idx]
    }
}

#[derive(Default)]
struct SH {
    q: VecDeque<i32>,
    // Monotone sequences stored from front to back.
    // For min: nondecreasing; for max: nonincreasing.
    min_mono: VecDeque<i32>,
    max_mono: VecDeque<i32>,
}

impl SH {
    fn clear(&mut self) {
        self.q.clear();
        self.min_mono.clear();
        self.max_mono.clear();
    }

    fn push_back(&mut self, x: i32) {
        self.q.push_back(x);

        while let Some(&back) = self.min_mono.back() {
            if back > x {
                self.min_mono.pop_back();
            } else {
                break;
            }
        }
        self.min_mono.push_back(x);

        while let Some(&back) = self.max_mono.back() {
            if back < x {
                self.max_mono.pop_back();
            } else {
                break;
            }
        }
        self.max_mono.push_back(x);
    }

    fn pop_front(&mut self) -> i32 {
        let x = self.q.pop_front().unwrap();

        if self.min_mono.front().map_or(false, |&v| v == x) {
            self.min_mono.pop_front();
        }
        if self.max_mono.front().map_or(false, |&v| v == x) {
            self.max_mono.pop_front();
        }
        x
    }

    fn len(&self) -> usize {
        self.q.len()
    }

    // k-th position (1-based) in the maintained monotone sequence from front to back.
    fn query_min_k(&self, k: usize) -> i32 {
        self.min_mono[k - 1]
    }

    fn query_max_k(&self, k: usize) -> i32 {
        self.max_mono[k - 1]
    }
}

fn rebalance_if_needed(fh: &mut FH, sh: &mut SH) {
    // Only moves from SH -> FH, and only when FH becomes smaller than SH.
    if fh.len() < sh.len() {
        let x = sh.pop_front();
        fh.push(x);
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    let mut fh = FH::default();
    let mut sh = SH::default();

    loop {
        let Some(qs) = it.next() else { break };
        let q: usize = qs.parse().unwrap();
        if q == 0 {
            break;
        }

        fh.clear();
        sh.clear();

        for _ in 0..q {
            let op: i32 = it.next().unwrap().parse().unwrap();
            match op {
                0 => {
                    let x: i32 = it.next().unwrap().parse().unwrap();
                    sh.push_back(x);
                    rebalance_if_needed(&mut fh, &mut sh);
                }
                1 => {
                    let v = fh.pop();
                    rebalance_if_needed(&mut fh, &mut sh);
                    out.push_str(&format!("{v}\n"));
                }
                2 => {
                    let v = fh.query_min_k(1);
                    out.push_str(&format!("{v}\n"));
                }
                3 => {
                    let v = sh.query_min_k(1);
                    out.push_str(&format!("{v}\n"));
                }
                4 => {
                    let k: usize = it.next().unwrap().parse().unwrap();
                    let v = fh.query_min_k(k);
                    out.push_str(&format!("{v}\n"));
                }
                5 => {
                    let k: usize = it.next().unwrap().parse().unwrap();
                    let v = sh.query_min_k(k);
                    out.push_str(&format!("{v}\n"));
                }
                6 => {
                    let v = fh.query_max_k(1);
                    out.push_str(&format!("{v}\n"));
                }
                7 => {
                    let v = sh.query_max_k(1);
                    out.push_str(&format!("{v}\n"));
                }
                8 => {
                    let k: usize = it.next().unwrap().parse().unwrap();
                    let v = fh.query_max_k(k);
                    out.push_str(&format!("{v}\n"));
                }
                9 => {
                    let k: usize = it.next().unwrap().parse().unwrap();
                    let v = sh.query_max_k(k);
                    out.push_str(&format!("{v}\n"));
                }
                _ => {
                    // Undefined behavior per spec for malformed input/opcodes.
                    // We still unwrap/parse above; reaching here is unexpected.
                    unreachable!();
                }
            }
        }

        out.push_str("end\n");
    }

    print!("{out}");
}
```
