```rust
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Pair {
    u: i64,
    c: i64,
}

#[derive(Clone, Copy)]
enum State {
    Active { rem_u: i64 },
    Charging { rem_c: i64 },
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut outs: Vec<String> = Vec::new();

    loop {
        let Some(ns) = it.next() else { break };
        let n_in: i64 = ns.parse().unwrap();
        let lim: i64 = match it.next() {
            Some(x) => x.parse().unwrap(),
            None => break,
        };

        if n_in <= 0 {
            break;
        }
        let n = n_in as usize;

        let mut cycles: Vec<Vec<Pair>> = Vec::with_capacity(n);
        for _ in 0..n {
            let mut v: Vec<Pair> = Vec::new();
            loop {
                let u: i64 = it.next().unwrap().parse().unwrap();
                if u <= 0 {
                    break;
                }
                let c: i64 = it.next().unwrap().parse().unwrap();
                v.push(Pair { u, c });
            }
            cycles.push(v);
        }

        if lim <= 0 {
            outs.push("0".to_string());
            continue;
        }

        let mut idx_in_cycle = vec![0usize; n];
        let mut state: Vec<State> = Vec::with_capacity(n);
        for i in 0..n {
            let p = cycles[i][0];
            state.push(State::Active { rem_u: p.u });
        }

        // Fixed-size circular buffer queue with overwrite-on-wrap semantics and no full detection.
        const QCAP: usize = 102;
        let mut qbuf: [usize; QCAP] = [0; QCAP];
        let mut qhead: usize = 0;
        let mut qtail: usize = 0;

        let mut misses: i128 = 0;

        for _ in 0..(lim as usize) {
            // Step A: advance active devices; enqueue those that enter charging.
            let mut just_entered = vec![false; n];
            for i in 0..n {
                if let State::Active { rem_u } = state[i] {
                    let new_rem = rem_u - 1;
                    if new_rem == 0 {
                        let p = cycles[i][idx_in_cycle[i]];
                        state[i] = State::Charging { rem_c: p.c };
                        just_entered[i] = true;

                        qbuf[qtail % QCAP] = i;
                        qtail = qtail.wrapping_add(1);
                    } else {
                        state[i] = State::Active { rem_u: new_rem };
                    }
                }
            }

            if qhead == qtail {
                continue;
            }

            // Step B (as implied by samples): for each time unit with non-empty queue,
            // every device that is in charging after step A contributes 1 miss unless it is
            // the queue-front device. (No exclusion for "just entered" devices.)
            let front = qbuf[qhead % QCAP];

            let mut charging_cnt: i128 = 0;
            for i in 0..n {
                if matches!(state[i], State::Charging { .. }) {
                    charging_cnt += 1;
                }
            }

            // Exclude the front device from misses if it is charging; otherwise exclude nothing.
            if matches!(state[front], State::Charging { .. }) {
                if charging_cnt > 0 {
                    misses += charging_cnt - 1;
                }
            } else {
                misses += charging_cnt;
            }

            // Step C: apply one unit of charging service to the queue front.
            let dev = front;
            if let State::Charging { rem_c } = state[dev] {
                let new_rem_c = rem_c - 1;
                if new_rem_c == 0 {
                    qhead = qhead.wrapping_add(1);

                    let clen = cycles[dev].len();
                    let next_idx = (idx_in_cycle[dev] + 1) % clen;
                    idx_in_cycle[dev] = next_idx;
                    let p = cycles[dev][next_idx];
                    state[dev] = State::Active { rem_u: p.u };
                } else {
                    state[dev] = State::Charging { rem_c: new_rem_c };
                }
            } else {
                // Overwrite-on-wrap may corrupt queue contents; proceed without error.
                // No service applied if the front isn't charging.
                let _ = just_entered; // keep variable used in this loop per semantics above
            }
        }

        outs.push(misses.to_string());
    }

    print!("{}", outs.join("\n"));
}
```
