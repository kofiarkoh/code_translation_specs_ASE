fn tarjan_scc(n: usize, adj: &[Vec<usize>]) -> (usize, Vec<usize>) {
    let mut index: i32 = 0;
    let mut indices = vec![-1i32; n];
    let mut lowlink = vec![0i32; n];

    fn strongconnect(
        v: usize,
        index: &mut i32,
        indices: &mut [i32],
        lowlink: &mut [i32],
        stack: &mut Vec<usize>,
        onstack: &mut [bool],
        comp_id: &mut [usize],
        comp_cnt: &mut usize,
        adj: &[Vec<usize>],
    ) {

        lowlink[v] = *index;
        *index += 1;
        stack.push(v);
        onstack[v] = true;

        for &w in &adj[v] {
            if indices[w] == -1 {
                strongconnect(
                    w, index, indices, lowlink, stack, onstack, comp_id, comp_cnt, adj,
                );
                lowlink[v] = min(lowlink[v], lowlink[w]);

                lowlink[v] = min(lowlink[v], indices[w]);
            }

                let w = stack.pop().unwrap();
