```rust
use std::cmp::min;
use std::collections::{HashMap, VecDeque};
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Edge {
    u: usize,
    v: usize,
}

fn tarjan_scc(n: usize, adj: &[Vec<usize>]) -> (usize, Vec<usize>) {
    let mut index: i32 = 0;
    let mut indices = vec![-1i32; n];
    let mut lowlink = vec![0i32; n];
    let mut onstack = vec![false; n];
    let mut stack: Vec<usize> = Vec::new();

    let mut comp_id = vec![usize::MAX; n];
    let mut comp_cnt: usize = 0;

    fn strongconnect(
        v: usize,
        index: &mut i32,
        indices: &mut [i32],
        lowlink: &mut [i32],
        stack: &mut Vec<usize>,
        onstack: &mut [bool],
        comp_id: &mut [usize],
        comp_cnt: &mut usize,
        adj: &[Vec<usize>],
    ) {
        indices[v] = *index;
        lowlink[v] = *index;
        *index += 1;
        stack.push(v);
        onstack[v] = true;

        for &w in &adj[v] {
            if indices[w] == -1 {
                strongconnect(
                    w, index, indices, lowlink, stack, onstack, comp_id, comp_cnt, adj,
                );
                lowlink[v] = min(lowlink[v], lowlink[w]);
            } else if onstack[w] {
                lowlink[v] = min(lowlink[v], indices[w]);
            }
        }

        if lowlink[v] == indices[v] {
            loop {
                let w = stack.pop().unwrap();
                onstack[w] = false;
                comp_id[w] = *comp_cnt;
                if w == v {
                    break;
                }
            }
            *comp_cnt += 1;
        }
    }

    for v in 0..n {
        if indices[v] == -1 {
            strongconnect(
                v,
                &mut index,
                &mut indices,
                &mut lowlink,
                &mut stack,
                &mut onstack,
                &mut comp_id,
                &mut comp_cnt,
                adj,
            );
        }
    }

    (comp_cnt, comp_id)
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let m: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let mut edges: Vec<Edge> = Vec::with_capacity(m);
    let mut adj: Vec<Vec<usize>> = vec![Vec::new(); n];

    for _ in 0..m {
        let u1: usize = it.next().unwrap().parse().unwrap();
        let v1: usize = it.next().unwrap().parse().unwrap();
        let u = u1 - 1;
        let v = v1 - 1;
        edges.push(Edge { u, v });
        adj[u].push(v);
    }

    let (scc_cnt, scc_of) = tarjan_scc(n, &adj);

    let mut is_diff = vec![false; m];

    // Build intra-SCC adjacency with edge indices (to exclude the tested edge).
    // Layout: intra_adj[scc][u] -> list of (v, edge_id)
    let mut intra_adj: Vec<Vec<Vec<(usize, usize)>>> = vec![vec![Vec::new(); n]; scc_cnt];
    for (i, e) in edges.iter().enumerate() {
        let su = scc_of[e.u];
        let sv = scc_of[e.v];
        if su == sv {
            intra_adj[su][e.u].push((e.v, i));
        }
    }

    // Intra-SCC classification:
    // diff iff there exists a path u=>v within SCC that does not use the edge being classified.
    for (i, e) in edges.iter().enumerate() {
        let s = scc_of[e.u];
        if s != scc_of[e.v] {
            continue;
        }

        let mut q = VecDeque::new();
        let mut vis = vec![false; n];
        vis[e.u] = true;
        q.push_back(e.u);

        let mut found = false;
        while let Some(x) = q.pop_front() {
            if x == e.v {
                found = true;
                break;
            }
            for &(to, eid) in &intra_adj[s][x] {
                if eid == i {
                    continue;
                }
                if !vis[to] {
                    vis[to] = true;
                    q.push_back(to);
                }
            }
        }

        if found {
            is_diff[i] = true;
        }
    }

    // Inter-SCC processing:
    // Condition (1): multiple original edges induce the same ordered SCC pair (S,T) => all diff.
    // Condition (2): there exists a condensation path from S to T of length >= 2 => all diff.
    let mut pair_to_edges: HashMap<(usize, usize), Vec<usize>> = HashMap::new();
    for (i, e) in edges.iter().enumerate() {
        let su = scc_of[e.u];
        let sv = scc_of[e.v];
        if su != sv {
            pair_to_edges.entry((su, sv)).or_default().push(i);
        }
    }

    for idxs in pair_to_edges.values() {
        if idxs.len() > 1 {
            for &eid in idxs {
                is_diff[eid] = true;
            }
        }
    }

    // Build condensation adjacency (unique arcs).
    let mut cond_adj: Vec<Vec<usize>> = vec![Vec::new(); scc_cnt];
    let mut indeg = vec![0usize; scc_cnt];
    {
        let mut seen: HashMap<(usize, usize), ()> = HashMap::new();
        for &(s, t) in pair_to_edges.keys() {
            if seen.insert((s, t), ()).is_none() {
                cond_adj[s].push(t);
                indeg[t] += 1;
            }
        }
    }

    // Topological order of condensation DAG.
    let mut dq = VecDeque::new();
    for i in 0..scc_cnt {
        if indeg[i] == 0 {
            dq.push_back(i);
        }
    }
    let mut topo: Vec<usize> = Vec::with_capacity(scc_cnt);
    while let Some(x) = dq.pop_front() {
        topo.push(x);
        for &y in &cond_adj[x] {
            indeg[y] -= 1;
            if indeg[y] == 0 {
                dq.push_back(y);
            }
        }
    }

    // Reachability bitsets on condensation DAG:
    // reach[s] contains all nodes reachable from s via >=1 arcs.
    let words = (scc_cnt + 63) / 64;
    let mut reach: Vec<Vec<u64>> = vec![vec![0u64; words]; scc_cnt];
    for &s in topo.iter().rev() {
        for &t in &cond_adj[s] {
            reach[s][t / 64] |= 1u64 << (t % 64);
            for w in 0..words {
                reach[s][w] |= reach[t][w];
            }
        }
    }

    // Condition (2) per spec: there exists a directed path from S to T that uses
    // at least one intermediate SCC (length >= 2).
    //
    // This is true iff there exists some X != T such that S=>X (>=1) and X=>T (>=1).
    // Note: X may be an immediate neighbor of S or further away; restricting X to neighbors
    // is incorrect.
    for (&(s, t), idxs) in pair_to_edges.iter() {
        let mut has_longer = false;

        // Iterate all possible intermediates X by scanning reach[s] bitset.
        for x in 0..scc_cnt {
            if x == t {
                continue;
            }
            let bit_sx = (reach[s][x / 64] >> (x % 64)) & 1;
            if bit_sx == 0 {
                continue;
            }
            let bit_xt = (reach[x][t / 64] >> (t % 64)) & 1;
            if bit_xt == 1 {
                has_longer = true;
                break;
            }
        }

        if has_longer {
            for &eid in idxs {
                is_diff[eid] = true;
            }
        }
    }

    let mut out = String::new();
    for i in 0..m {
        if is_diff[i] {
            out.push_str("diff\n");
        } else {
            out.push_str("same\n");
        }
    }
    print!("{out}");
}
```
