struct Edge {
    u: usize,
    v: usize,
}

fn tarjan_scc(n: usize, adj: &[Vec<usize>]) -> (usize, Vec<usize>) {
    let mut index: i32 = 0;
    let mut indices = vec![-1i32; n];
    let mut lowlink = vec![0i32; n];
    let mut onstack = vec![false; n];
    let mut stack: Vec<usize> = Vec::new();

    let mut comp_id = vec![usize::MAX; n];
    let mut comp_cnt: usize = 0;

    fn strongconnect(
        v: usize,
        index: &mut i32,
        indices: &mut [i32],
        lowlink: &mut [i32],
        stack: &mut Vec<usize>,
        onstack: &mut [bool],
        comp_id: &mut [usize],
        comp_cnt: &mut usize,
        adj: &[Vec<usize>],
    ) {
        indices[v] = *index;
        lowlink[v] = *index;
        *index += 1;
        stack.push(v);
        onstack[v] = true;

        for &w in &adj[v] {
            if indices[w] == -1 {
                strongconnect(
                    w, index, indices, lowlink, stack, onstack, comp_id, comp_cnt, adj,
                );
                lowlink[v] = min(lowlink[v], lowlink[w]);
            } else if onstack[w] {
                lowlink[v] = min(lowlink[v], indices[w]);
            }
        }

        if lowlink[v] == indices[v] {
            loop {
                let w = stack.pop().unwrap();
                onstack[w] = false;
                comp_id[w] = *comp_cnt;
                if w == v {
                    break;
                }
            }
            *comp_cnt += 1;
        }
    }

    for v in 0..n {
        if indices[v] == -1 {
            strongconnect(
                v,
                &mut index,
                &mut indices,
                &mut lowlink,
                &mut stack,
                &mut onstack,
                &mut comp_id,
                &mut comp_cnt,
                adj,
            );
        }
    }

    (comp_cnt, comp_id)
}
