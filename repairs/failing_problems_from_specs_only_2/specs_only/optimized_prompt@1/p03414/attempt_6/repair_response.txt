```rust
use std::collections::{HashMap, VecDeque};
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Edge {
    u: usize,
    v: usize,
}

fn kosaraju_scc(n: usize, adj: &[Vec<usize>]) -> (usize, Vec<usize>) {
    let mut radj: Vec<Vec<usize>> = vec![Vec::new(); n];
    for u in 0..n {
        for &v in &adj[u] {
            radj[v].push(u);
        }
    }

    let mut vis = vec![false; n];
    let mut order: Vec<usize> = Vec::with_capacity(n);

    for start in 0..n {
        if vis[start] {
            continue;
        }
        let mut stack: Vec<(usize, usize)> = Vec::new();
        stack.push((start, 0));
        vis[start] = true;

        while let Some((u, i)) = stack.pop() {
            if i < adj[u].len() {
                stack.push((u, i + 1));
                let v = adj[u][i];
                if !vis[v] {
                    vis[v] = true;
                    stack.push((v, 0));
                }
            } else {
                order.push(u);
            }
        }
    }

    let mut comp = vec![usize::MAX; n];
    let mut comp_cnt = 0usize;

    for &start in order.iter().rev() {
        if comp[start] != usize::MAX {
            continue;
        }
        let mut q: VecDeque<usize> = VecDeque::new();
        q.push_back(start);
        comp[start] = comp_cnt;

        while let Some(u) = q.pop_front() {
            for &v in &radj[u] {
                if comp[v] == usize::MAX {
                    comp[v] = comp_cnt;
                    q.push_back(v);
                }
            }
        }
        comp_cnt += 1;
    }

    (comp_cnt, comp)
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let m: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let mut edges: Vec<Edge> = Vec::with_capacity(m);
    let mut adj: Vec<Vec<usize>> = vec![Vec::new(); n];

    for _ in 0..m {
        let u1: usize = it.next().unwrap().parse().unwrap();
        let v1: usize = it.next().unwrap().parse().unwrap();
        let u = u1 - 1;
        let v = v1 - 1;
        edges.push(Edge { u, v });
        adj[u].push(v);
    }

    let (k, comp) = kosaraju_scc(n, &adj);

    let mut comp_vertices: Vec<Vec<usize>> = vec![Vec::new(); k];
    for v in 0..n {
        comp_vertices[comp[v]].push(v);
    }

    let mut is_diff = vec![false; m];

    // Build condensation DAG and SCC-pair -> edge indices.
    let mut pair_to_edges: HashMap<(usize, usize), Vec<usize>> = HashMap::new();
    let mut dag_adj: Vec<Vec<usize>> = vec![Vec::new(); k];
    let mut dag_has: HashMap<(usize, usize), ()> = HashMap::new();

    for (i, e) in edges.iter().enumerate() {
        let s = comp[e.u];
        let t = comp[e.v];
        if s != t {
            pair_to_edges.entry((s, t)).or_default().push(i);
            if !dag_has.contains_key(&(s, t)) {
                dag_has.insert((s, t), ());
                dag_adj[s].push(t);
            }
        }
    }

    // A1: parallel SCC-arc duplication
    for idxs in pair_to_edges.values() {
        if idxs.len() > 1 {
            for &ei in idxs {
                is_diff[ei] = true;
            }
        }
    }

    // A2: longer SCC-level route exists.
    //
    // The observed sample expects edges inside a nontrivial SCC to be classified by rule B,
    // and inter-SCC edges by A. The repeated "all same" indicates SCCs were computed correctly
    // but rule B must treat "alternative path" as requiring at least one edge (non-empty path),
    // i.e., empty path does NOT count when u==v. Also, for rule A2, we must detect any path
    // from S to T that does not use the direct arc S->T (which implies length >= 2 in a DAG).
    //
    // Implement A2 by BFS from S in condensation DAG while forbidding the arc S->T anywhere it appears.
    // (In a DAG, any alternative path necessarily has length >= 2.)
    let mut targets_by_source: Vec<Vec<usize>> = vec![Vec::new(); k];
    for (&(s, t), _) in pair_to_edges.iter() {
        targets_by_source[s].push(t);
    }
    for s in 0..k {
        targets_by_source[s].sort_unstable();
        targets_by_source[s].dedup();
    }

    let mut vis_scc = vec![false; k];
    let mut qs: VecDeque<usize> = VecDeque::new();

    for s in 0..k {
        if targets_by_source[s].is_empty() {
            continue;
        }
        for &t in &targets_by_source[s] {
            vis_scc.fill(false);
            qs.clear();

            vis_scc[s] = true;
            qs.push_back(s);

            let mut alt = false;
            while let Some(u) = qs.pop_front() {
                for &nx in &dag_adj[u] {
                    if u == s && nx == t {
                        // Forbid using the direct arc S->T.
                        continue;
                    }
                    if !vis_scc[nx] {
                        vis_scc[nx] = true;
                        if nx == t {
                            alt = true;
                            qs.clear();
                            break;
                        }
                        qs.push_back(nx);
                    }
                }
            }

            if alt {
                if let Some(idxs) = pair_to_edges.get(&(s, t)) {
                    for &ei in idxs {
                        is_diff[ei] = true;
                    }
                }
            }
        }
    }

    // Intra-SCC: BFS per edge excluding that edge.
    let mut scc_intra_edges: Vec<Vec<usize>> = vec![Vec::new(); k];
    for (i, e) in edges.iter().enumerate() {
        if comp[e.u] == comp[e.v] {
            scc_intra_edges[comp[e.u]].push(i);
        }
    }

    let mut out_edge_ids: Vec<Vec<usize>> = vec![Vec::new(); n];
    for (i, e) in edges.iter().enumerate() {
        out_edge_ids[e.u].push(i);
    }

    let mut in_scc = vec![false; n];
    let mut vis_v = vec![false; n];
    let mut qv: VecDeque<usize> = VecDeque::new();

    for scc_id in 0..k {
        let intra = &scc_intra_edges[scc_id];
        if intra.is_empty() {
            continue;
        }

        for &v in &comp_vertices[scc_id] {
            in_scc[v] = true;
        }

        for &eid in intra {
            let e = edges[eid];
            let src = e.u;
            let dst = e.v;

            for &v in &comp_vertices[scc_id] {
                vis_v[v] = false;
            }
            qv.clear();

            // Rule B requires an alternative directed path from src to dst within the SCC
            // that does not use the edge under test. Interpret "path" as non-empty (>=1 edge).
            // Therefore, do NOT accept src==dst immediately; we must find a cycle/path with edges.
            vis_v[src] = true;
            qv.push_back(src);

            let mut found = false;
            while let Some(u) = qv.pop_front() {
                for &oeid in &out_edge_ids[u] {
                    if oeid == eid {
                        continue;
                    }
                    let to = edges[oeid].v;
                    if !in_scc[to] {
                        continue;
                    }
                    if to == dst {
                        found = true;
                        qv.clear();
                        break;
                    }
                    if !vis_v[to] {
                        vis_v[to] = true;
                        qv.push_back(to);
                    }
                }
            }

            if found {
                is_diff[eid] = true;
            }
        }

        for &v in &comp_vertices[scc_id] {
            in_scc[v] = false;
        }
    }

    let mut out = String::new();
    for i in 0..m {
        out.push_str(if is_diff[i] { "diff\n" } else { "same\n" });
    }
    print!("{out}");
}
```
