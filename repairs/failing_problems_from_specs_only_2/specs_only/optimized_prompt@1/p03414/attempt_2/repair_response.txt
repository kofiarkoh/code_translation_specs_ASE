```rust
use std::collections::{HashMap, VecDeque};
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Edge {
    u: usize,
    v: usize,
}

fn tarjan_scc(n: usize, adj: &[Vec<usize>]) -> (usize, Vec<usize>) {
    let mut index: i32 = 0;
    let mut stack: Vec<usize> = Vec::new();
    let mut on_stack = vec![false; n];
    let mut idx = vec![-1i32; n];
    let mut low = vec![0i32; n];
    let mut comp = vec![usize::MAX; n];
    let mut comp_cnt: usize = 0;

    fn dfs(
        v: usize,
        adj: &[Vec<usize>],
        index: &mut i32,
        stack: &mut Vec<usize>,
        on_stack: &mut [bool],
        idx: &mut [i32],
        low: &mut [i32],
        comp: &mut [usize],
        comp_cnt: &mut usize,
    ) {
        idx[v] = *index;
        low[v] = *index;
        *index += 1;
        stack.push(v);
        on_stack[v] = true;

        for &to in &adj[v] {
            if idx[to] == -1 {
                dfs(to, adj, index, stack, on_stack, idx, low, comp, comp_cnt);
                low[v] = low[v].min(low[to]);
            } else if on_stack[to] {
                low[v] = low[v].min(idx[to]);
            }
        }

        if low[v] == idx[v] {
            loop {
                let w = stack.pop().unwrap();
                on_stack[w] = false;
                comp[w] = *comp_cnt;
                if w == v {
                    break;
                }
            }
            *comp_cnt += 1;
        }
    }

    for v in 0..n {
        if idx[v] == -1 {
            dfs(
                v,
                adj,
                &mut index,
                &mut stack,
                &mut on_stack,
                &mut idx,
                &mut low,
                &mut comp,
                &mut comp_cnt,
            );
        }
    }

    (comp_cnt, comp)
}

fn topo_order_dag(k: usize, dag_adj: &[Vec<usize>]) -> Vec<usize> {
    let mut indeg = vec![0usize; k];
    for u in 0..k {
        for &v in &dag_adj[u] {
            indeg[v] += 1;
        }
    }
    let mut q = VecDeque::new();
    for i in 0..k {
        if indeg[i] == 0 {
            q.push_back(i);
        }
    }
    let mut order = Vec::with_capacity(k);
    while let Some(u) = q.pop_front() {
        order.push(u);
        for &v in &dag_adj[u] {
            indeg[v] -= 1;
            if indeg[v] == 0 {
                q.push_back(v);
            }
        }
    }
    order
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let m: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let mut edges: Vec<Edge> = Vec::with_capacity(m);
    let mut adj: Vec<Vec<usize>> = vec![Vec::new(); n];

    for _ in 0..m {
        let u1: usize = it.next().unwrap().parse().unwrap();
        let v1: usize = it.next().unwrap().parse().unwrap();
        let u = u1 - 1;
        let v = v1 - 1;
        edges.push(Edge { u, v });
        adj[u].push(v);
    }

    let (k, comp) = tarjan_scc(n, &adj);

    let mut comp_vertices: Vec<Vec<usize>> = vec![Vec::new(); k];
    for v in 0..n {
        comp_vertices[comp[v]].push(v);
    }

    let mut is_diff = vec![false; m];

    // Inter-SCC processing
    let mut pair_to_edges: HashMap<(usize, usize), Vec<usize>> = HashMap::new();
    let mut dag_adj: Vec<Vec<usize>> = vec![Vec::new(); k];
    let mut dag_has: HashMap<(usize, usize), ()> = HashMap::new();

    for (i, e) in edges.iter().enumerate() {
        let s = comp[e.u];
        let t = comp[e.v];
        if s != t {
            pair_to_edges.entry((s, t)).or_default().push(i);
            if !dag_has.contains_key(&(s, t)) {
                dag_has.insert((s, t), ());
                dag_adj[s].push(t);
            }
        }
    }

    // A1: parallel SCC-arc duplication
    for idxs in pair_to_edges.values() {
        if idxs.len() > 1 {
            for &ei in idxs {
                is_diff[ei] = true;
            }
        }
    }

    // A2: longer SCC-level route exists (path length >= 2)
    // Important: we must detect reachability via >=2 arcs, i.e., existence of an intermediate SCC X
    // such that S -> X and X => T (including X -> T directly).
    let words = (k + 63) / 64;
    let mut reach: Vec<Vec<u64>> = vec![vec![0u64; words]; k];
    for u in 0..k {
        for &v in &dag_adj[u] {
            reach[u][v / 64] |= 1u64 << (v % 64);
        }
    }
    let order = topo_order_dag(k, &dag_adj);
    for &u in order.iter().rev() {
        for &v in &dag_adj[u] {
            for w in 0..words {
                reach[u][w] |= reach[v][w];
            }
        }
    }

    for (&(s, t), idxs) in pair_to_edges.iter() {
        let mut longer = false;
        for &x in &dag_adj[s] {
            if x == t {
                continue;
            }
            // Need X => T, allowing the direct arc X->T.
            let bit = (reach[x][t / 64] >> (t % 64)) & 1;
            if bit != 0 {
                longer = true;
                break;
            }
        }
        if longer {
            for &ei in idxs {
                is_diff[ei] = true;
            }
        }
    }

    // Intra-SCC processing: BFS per edge excluding that edge.
    let mut scc_intra_edges: Vec<Vec<usize>> = vec![Vec::new(); k];
    for (i, e) in edges.iter().enumerate() {
        if comp[e.u] == comp[e.v] {
            scc_intra_edges[comp[e.u]].push(i);
        }
    }

    let mut out_edge_ids: Vec<Vec<usize>> = vec![Vec::new(); n];
    for (i, e) in edges.iter().enumerate() {
        out_edge_ids[e.u].push(i);
    }

    let mut in_scc = vec![false; n];
    let mut vis = vec![false; n];
    let mut q: VecDeque<usize> = VecDeque::new();

    for scc_id in 0..k {
        let intra = &scc_intra_edges[scc_id];
        if intra.is_empty() {
            continue;
        }

        // Mark membership for this SCC
        for &v in &comp_vertices[scc_id] {
            in_scc[v] = true;
        }

        for &eid in intra {
            let e = edges[eid];
            let src = e.u;
            let dst = e.v;

            // Reset visited only for vertices in this SCC
            for &v in &comp_vertices[scc_id] {
                vis[v] = false;
            }
            q.clear();

            vis[src] = true;
            q.push_back(src);

            let mut found = false;
            while let Some(u) = q.pop_front() {
                // Spec: path from u to v within SCC not using the edge under test.
                // For src==dst, the empty path qualifies (uses no edges), so it's always found.
                if u == dst {
                    found = true;
                    break;
                }
                for &oeid in &out_edge_ids[u] {
                    if oeid == eid {
                        continue;
                    }
                    let to = edges[oeid].v;
                    if !in_scc[to] {
                        continue;
                    }
                    if !vis[to] {
                        vis[to] = true;
                        q.push_back(to);
                    }
                }
            }

            if found {
                is_diff[eid] = true;
            }
        }

        // Unmark membership
        for &v in &comp_vertices[scc_id] {
            in_scc[v] = false;
        }
    }

    let mut out = String::new();
    for i in 0..m {
        if is_diff[i] {
            out.push_str("diff\n");
        } else {
            out.push_str("same\n");
        }
    }
    print!("{out}");
}
```
