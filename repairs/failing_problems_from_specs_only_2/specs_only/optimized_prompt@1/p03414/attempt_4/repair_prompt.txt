fn topo_order_dag(k: usize, dag_adj: &[Vec<usize>]) -> Vec<usize> {
    let mut indeg = vec![0usize; k];
    for u in 0..k {
        for &v in &dag_adj[u] {
            indeg[v] += 1;
        }
    }
    let mut q = VecDeque::new();
    for i in 0..k {
        if indeg[i] == 0 {
            q.push_back(i);
        }
    }
    let mut order = Vec::with_capacity(k);
    while let Some(u) = q.pop_front() {
        order.push(u);
        for &v in &dag_adj[u] {
            indeg[v] -= 1;
            if indeg[v] == 0 {
                q.push_back(v);
            }
        }
    }

}
