```rust
use std::io::{self, Read};

const PGMAX: usize = 1001; // scan indices 0..=1000

#[derive(Clone)]
struct Entry {
    word: String,
    pg: [u8; PGMAX], // present iff == 1
    tm: i32,         // incremented once per input record for this word
}

fn main() {
    // Read all input and parse as repeated (%s, %d) pairs until EOF/parse failure.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut entries: Vec<Entry> = Vec::new();

    loop {
        let w = match it.next() {
            Some(s) => s.to_string(),
            None => break,
        };
        let p_tok = match it.next() {
            Some(s) => s,
            None => break,
        };
        let p_i32 = match p_tok.parse::<i32>() {
            Ok(v) => v,
            Err(_) => break,
        };

        // Spec: out-of-range page numbers are UB in C. Here we avoid panics and skip.
        let p = match usize::try_from(p_i32).ok().filter(|&u| u < PGMAX) {
            Some(u) => u,
            None => continue,
        };

        if let Some(e) = entries.iter_mut().find(|e| e.word == w) {
            e.pg[p] = 1;
            e.tm = e.tm.wrapping_add(1);
        } else {
            let mut e = Entry {
                word: w,
                pg: [0u8; PGMAX],
                tm: 0,
            };
            e.pg[p] = 1;
            e.tm = e.tm.wrapping_add(1);
            entries.push(e);
        }
    }

    // Sort ascending lexicographic order (strcmp-like on bytes).
    entries.sort_by(|a, b| a.word.as_bytes().cmp(b.word.as_bytes()));

    let mut out = String::new();

    for e in &entries {
        // Print the word line.
        out.push_str(&e.word);
        out.push('\n');

        // Then print page numbers on the next line(s) according to the "tm last" rule.
        // The C program's behavior can omit the terminating newline if tm is larger
        // than the number of present pages encountered during the scan.
        let mut printed: i32 = 0;
        for j in 0..PGMAX {
            if e.pg[j] == 1 {
                printed = printed.wrapping_add(1);
                if printed == e.tm {
                    out.push_str(&j.to_string());
                    out.push('\n');
                    break;
                } else {
                    out.push_str(&j.to_string());
                    out.push(' ');
                }
            }
        }
    }

    // Write output.
    print!("{out}");
}
```
