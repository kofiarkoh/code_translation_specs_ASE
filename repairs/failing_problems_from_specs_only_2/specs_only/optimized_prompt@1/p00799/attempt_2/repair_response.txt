```rust
use std::io::{self, Read};

const QCAP: usize = 102;

#[derive(Clone, Copy, Debug)]
struct Pair {
    u: i64,
    c: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Mode {
    Active,
    Charging,
}

#[derive(Clone, Debug)]
struct Device {
    cycle: Vec<Pair>,
    idx: usize, // current cycle index
    mode: Mode,
    rem: i64, // remaining time in current mode
}

#[derive(Clone, Debug)]
struct RingQueue {
    buf: [usize; QCAP],
    head: usize, // front position
    tail: usize, // next write position
}

impl RingQueue {
    fn new() -> Self {
        Self {
            buf: [0usize; QCAP],
            head: 0,
            tail: 0,
        }
    }

    fn is_empty(&self) -> bool {
        self.head == self.tail
    }

    fn push(&mut self, v: usize) {
        // Fixed-size circular buffer with overwrite on wraparound and no full detection.
        self.buf[self.tail] = v;
        self.tail = (self.tail + 1) % QCAP;
        // If tail laps head, older entries are overwritten; we intentionally do not adjust head.
    }

    fn front(&self) -> usize {
        self.buf[self.head]
    }

    fn pop_front(&mut self) {
        self.head = (self.head + 1) % QCAP;
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut outs: Vec<String> = Vec::new();

    loop {
        let Some(n_tok) = it.next() else { break };
        let n: i64 = n_tok.parse().unwrap();
        let lim: i64 = match it.next() {
            Some(x) => x.parse().unwrap(),
            None => break,
        };

        if n <= 0 {
            break;
        }

        let n_usize = n as usize;
        let mut devices: Vec<Device> = Vec::with_capacity(n_usize);

        for _ in 0..n_usize {
            let mut cycle: Vec<Pair> = Vec::new();
            loop {
                let u: i64 = it.next().unwrap().parse().unwrap();
                if u <= 0 {
                    break;
                }
                let c: i64 = it.next().unwrap().parse().unwrap();
                cycle.push(Pair { u, c });
            }
            let first = cycle[0];
            devices.push(Device {
                cycle,
                idx: 0,
                mode: Mode::Active,
                rem: first.u,
            });
        }

        if lim <= 0 {
            outs.push("0".to_string());
            continue;
        }

        let mut q = RingQueue::new();
        let mut misses: i128 = 0;

        // Key semantic fix:
        // Misses are counted among devices that were already charging *before* step (A),
        // excluding the queue-front device for that time unit.
        // Devices that enter charging during step (A) do not count as misses in that same unit.
        for _ in 0..lim {
            // Snapshot of "already charging before step (A)".
            let mut was_charging: Vec<bool> = Vec::with_capacity(devices.len());
            for d in &devices {
                was_charging.push(d.mode == Mode::Charging);
            }

            // Step A: advance active devices and enqueue new chargers.
            for i in 0..devices.len() {
                if devices[i].mode == Mode::Active {
                    devices[i].rem -= 1;
                    if devices[i].rem == 0 {
                        devices[i].mode = Mode::Charging;
                        devices[i].rem = devices[i].cycle[devices[i].idx].c;
                        q.push(i);
                    }
                }
            }

            // Step B: count misses among devices that were already charging before step (A).
            if q.is_empty() {
                continue; // no misses and no charging progress
            }
            let front = q.front();
            for i in 0..devices.len() {
                if was_charging[i] && i != front {
                    misses += 1;
                }
            }

            // Step C: apply one unit of charging service to the queue front.
            let fi = front;
            devices[fi].rem -= 1;
            if devices[fi].rem == 0 {
                q.pop_front();
                devices[fi].mode = Mode::Active;
                devices[fi].idx = (devices[fi].idx + 1) % devices[fi].cycle.len();
                devices[fi].rem = devices[fi].cycle[devices[fi].idx].u;
            }
        }

        outs.push(misses.to_string());
    }

    print!("{}", outs.join("\n"));
}
```
